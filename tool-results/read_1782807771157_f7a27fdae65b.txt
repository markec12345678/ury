    96→    96→function loadUIPrefs(): Partial<UIPreferences> {
    97→    97→  if (typeof window === 'undefined') return {};
    98→    98→  try {
    99→    99→    const raw = localStorage.getItem(UI_PREFS_KEY);
   100→   100→    if (!raw) return {};
   101→   101→    return JSON.parse(raw);
   102→   102→  } catch {
   103→   103→    return {};
   104→   104→  }
   105→   105→}
   106→   106→
   107→   107→function saveUIPrefs(prefs: Partial<UIPreferences>): void {
   108→   108→  if (typeof window === 'undefined') return;
   109→   109→  try {
   110→   110→    localStorage.setItem(UI_PREFS_KEY, JSON.stringify(prefs));
   111→   111→  } catch {
   112→   112→    // ignore storage errors
   113→   113→  }
   114→   114→}
