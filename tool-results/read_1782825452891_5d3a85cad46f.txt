   123→// ── UI Preference Persistence ────────────────────────────
   124→const UI_PREFS_KEY = 'ury_ui_prefs';
   125→
   126→interface UIPreferences {
   127→  darkMode: boolean;
   128→  activeTab: string;
   129→}
   130→
   131→function loadUIPrefs(): Partial<UIPreferences> {
   132→  if (typeof window === 'undefined') return {};
   133→  try {
   134→    const raw = localStorage.getItem(UI_PREFS_KEY);
   135→    if (!raw) return {};
   136→    return JSON.parse(raw);
   137→  } catch {
   138→    return {};
   139→  }
   140→}
   141→
   142→function saveUIPrefs(prefs: Partial<UIPreferences>): void {
   143→  if (typeof window === 'undefined') return;
   144→  try {
   145→    localStorage.setItem(UI_PREFS_KEY, JSON.stringify(prefs));
   146→  } catch {
   147→    // ignore storage errors
   148→  }
   149→}

   487→    refreshData: async () => {

   491→      set({ isRefreshing: true });

   493→      try {

   534→            : [];
   535→          set({ kotCards: transformedKOTs });
   536→        }
   537→
   538→        // ── Process Tables ──────────────────────────────
   539→        if (tablesResult.status === 'fulfilled' && tablesResult.value?.data) {
   540→          const transformedTables: TableData[] = Array.isArray(tablesResult.value.data)
   541→            ? tablesResult.value.data.map((t: Record<string, unknown>) => ({
   542→                id: Number(t.name || t.id || 0),
   543→                room: String(t.room || ''),
   544→                status: (t.status as TableData['status']) || 'free',

   551→            : [];

   553→        }
   554→
   555→        // ── Process Invoices → Recent Orders + KPIs ─────
   556→        if (invoicesResult.status === 'fulfilled' && invoicesResult.value?.data) {
   557→          const invoices = Array.isArray(invoicesResult.value.data)
   558→            ? invoicesResult.value.data
   559→            : [];
   560→
   561→          // Recent orders
   562→          const transformedOrders: RecentOrder[] = invoices.map((inv: Record<string, unknown>) => ({

   569→          }));
   570→          set({ recentOrders: transformedOrders });
   571→
   572→          // Compute KPIs from invoices
   573→          const totalSales = invoices.reduce((sum: number, inv: Record<string, unknown>) => sum + Number(inv.grand_total || 0), 0);
   574→          const totalOrders = invoices.length;
   575→          const occupiedTables = invoices.filter((inv: Record<string, unknown>) => inv.type === 'Dine-in').length;
   576→          const allTables = get().tables;
   577→          set({
   578→            kpis: {
   579→              dailySales: totalSales,
   580→              totalOrders,
   581→              avgBill: totalOrders > 0 ? Math.round(totalSales / totalOrders) : 0,
   582→              occupiedTables,
   583→              totalTables: allTables.length || 32,
   584→            },
   585→          });
   586→
   587→          // Compute hourly sales from invoices
   588→          const hourMap = new Map<string, { dineIn: number; takeaway: number }>();
   589→          // Initialize all hours from 11:00 to 22:00
   590→          for (let h = 11; h <= 22; h++) {
   591→            const key = `${String(h).padStart(2, '0')}:00`;
   592→            hourMap.set(key, { dineIn: 0, takeaway: 0 });
   593→          }
   594→          // Aggregate invoice totals by hour
   595→          invoices.forEach((inv: Record<string, unknown>) => {
   596→            const timeStr = String(inv.posting_time || '').slice(0, 5);
   597→            if (!timeStr) return;
   598→            const hour = timeStr.split(':')[0];
   599→            const key = `${hour}:00`;
