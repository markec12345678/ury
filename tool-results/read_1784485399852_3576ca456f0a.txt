   175→   175→    aggregator_id=None,
   176→   176→    room=None
   177→   177→):
   178→   178→    frappe.only_for("Restaurant Manager", "Restaurant User", "Cashier")
   179→   179→    # Validate pos_profile belongs to user's branch (BE-R36-002)
   180→   180→    pos_profile_branch = frappe.db.get_value("POS Profile", pos_profile, "branch")
   181→   181→    user_branch = getBranch()
   182→   182→    if pos_profile_branch != user_branch:
   183→   183→        frappe.throw(_("POS Profile does not belong to your branch"), frappe.PermissionError)
   184→   184→    user_role = frappe.get_roles()
   185→   185→    billing_roles = frappe.get_all(
   186→   186→        "POS Profile Role",
   187→   187→        filters={"parent": pos_profile},
   188→   188→        fields=["role"],
   189→   189→        pluck="role",
   190→   190→    )
   191→   191→    billing_user = bool(set(user_role).intersection(billing_roles))
   192→   192→
   193→   193→    # Check if the last invoice was already billed
   194→   194→    if (
   195→   195→        last_invoice
   196→   196→        and frappe.db.get_value("POS Invoice", last_invoice, "invoice_printed") == 1
   197→   197→        and (not billing_user)
   198→   198→    ):
   199→   199→        frappe.msgprint(
   200→   200→            title=_("Invoice Already Billed"),
   201→   201→            indicator="red",
   202→   202→            msg=_("This order has already been billed. Please reload the page."),
   203→   203→        )
   204→   204→        return {"status": "Failure"}
   205→   205→
   206→   206→    # R41-FIX: Use _get_order_invoice_doc to get the document object, not a dict.
   207→   207→    # The public get_order_invoice returns a dict for frontend safety (BE-R36-003),
   208→   208→    # but sync_order needs the document to call .save() etc.
   209→   209→    invoice = _get_order_invoice_doc(table, invoice, order_type)
   210→   210→
   211→   211→    # R41-FIX: Validate last_invoice belongs to user's branch
   212→   212→    if last_invoice:
   213→   213→        last_inv_branch = frappe.db.get_value("POS Invoice", last_invoice, "branch")
   214→   214→        if last_inv_branch and last_inv_branch != user_branch:
   215→   215→            frappe.throw(_("Invoice does not belong to your branch"), frappe.PermissionError)
   216→   216→
   217→   217→    if last_invoice and last_modified_time:
   218→   218→        lastModifiedTime = invoice.modified
   219→   219→        from datetime import datetime
   220→   220→
   221→   221→        if isinstance(last_modified_time, str):
   222→   222→            try:
   223→   223→                last_modified_time = datetime.strptime(
   224→   224→                    last_modified_time, "%Y-%m-%d %H:%M:%S.%f"
   225→   225→                )
   226→   226→            except ValueError:
   227→   227→                last_modified_time = datetime.strptime(
   228→   228→                    last_modified_time, "%Y-%m-%d %H:%M:%S"
   229→   229→                )
   230→   230→        if isinstance(lastModifiedTime, str):
   231→   231→            try:
   232→   232→                lastModifiedTime = datetime.strptime(
   233→   233→                    lastModifiedTime, "%Y-%m-%d %H:%M:%S.%f"
   234→   234→                )
   235→   235→            except ValueError:
   236→   236→                lastModifiedTime = datetime.strptime(
   237→   237→                    lastModifiedTime, "%Y-%m-%d %H:%M:%S"
   238→   238→                )
   239→   239→        if lastModifiedTime != last_modified_time:
   240→   240→            frappe.msgprint(
   241→   241→                title=_("Order has been modified"),
   242→   242→                indicator="red",
   243→   243→                msg=_("This order has been modified. Please reload the page to retrieve the latest edits."),
   244→   244→            )
   245→   245→            return {"status": "Failure"}
   246→   246→    else:
   247→   247→        if invoice.name and invoice.invoice_printed == 0 and not billing_user:
   248→   248→            frappe.msgprint(
   249→   249→                title=_("Table Occupied"),
   250→   250→                indicator="red",
   251→   251→                msg=_("{0} is already occupied. Please refresh the page.").format(table),
   252→   252→            )
   253→   253→            return {"status": "Failure"}
   254→   254→
   255→   255→    if not customer:
   256→   256→        frappe.throw(_("Please enter valid customer details"))
   257→   257→    else:
   258→   258→        invoice.customer = customer
   259→   259→
   260→   260→    if order_type:
   261→   261→        invoice.order_type = order_type
   262→   262→
   263→   263→    invoice.mobile_number = frappe.db.get_value("Customer", customer, "mobile_number")
   264→   264→    if comments:
   265→   265→        invoice.custom_comments = comments
   266→   266→    invoice.no_of_pax = no_of_pax
   267→   267→    invoice.pos_profile = pos_profile
   268→   268→    invoice.cashier = cashier
   269→   269→    invoice.waiter = waiter
   270→   270→    invoice.custom_aggregator_id = aggregator_id
   271→   271→    invoice.custom_restaurant_room =room
   272→   272→    invoice.restaurant_table = table
   273→   273→    
   274→   274→    if order_type == "Aggregators":
   275→   275→        price_list = frappe.db.get_value("Aggregator Settings",{"customer": customer, "parent": invoice.branch, "parenttype": "Branch"},"price_list",)
   276→   276→        
   277→   277→        if not price_list:
   278→   278→            frappe.throw(_("Price list for customer {0} in branch {1} not found in Aggregator Settings.").format(customer, invoice.branch))
   279→   279→    else:
   280→   280→        price_list = invoice.selling_price_list
   281→   281→
   282→   282→    # dummy payment
   283→   283→    if invoice.invoice_created == 0:
   284→   284→        invoice.append(
   285→   285→            "payments",
   286→   286→            dict(mode_of_payment=mode_of_payment, amount=invoice.grand_total),
   287→   287→        )
   288→   288→        invoice.invoice_created = 1
   289→   289→
   290→   290→    past_item = []
   291→   291→    for item in invoice.items:
   292→   292→        previous_item = {
   293→   293→            "item_code": item.item_code,
   294→   294→            "item_name": item.item_name,
   295→   295→            "qty": item.qty,
   296→   296→            "comments": "",
   297→   297→        }
   298→   298→        past_item.append(previous_item)
   299→   299→        
   300→   300→
   301→   301→    # Conditional checking for 'items' type:
   302→   302→    # - 'ury': JSON passed, hence using isinstance
   303→   303→    # - 'ury_pos': Already formatted list, hence using else
   304→   304→    if isinstance(items, str):
   305→   305→        items = json.loads(items)
   306→   306→    invoice.items = []
   307→   307→    
   308→   308→    menu = frappe.db.get_value("URY Menu", {"branch": invoice.branch}, "name")
   309→   309→
   310→   310→    # Batch-fetch courses, prices, and cost_center to avoid N+1 queries
   311→   311→    item_codes = [d.get("item") for d in items]
   312→   312→    item_codes_unique = list(set(item_codes))
   313→   313→
   314→   314→    courses = {}
   315→   315→    if item_codes_unique:
   316→   316→        course_rows = frappe.db.get_all(
   317→   317→            "URY Menu Item",
   318→   318→            filters={"item": ("in", item_codes_unique), "parent": menu},
   319→   319→            fields=["item", "course"],
   320→   320→        )
   321→   321→        courses = {r.item: r.course for r in course_rows}
   322→   322→
   323→   323→    prices = {}
   324→   324→    if item_codes_unique:
   325→   325→        price_rows = frappe.db.get_all(
   326→   326→            "Item Price",

   348→   348→                rate=rate,

   353→   353→        )

   381→   381→    invoice.db_set("owner", frappe.session.user)

   384→   384→        "customer": invoice.customer,
   385→   385→        "grand_total": invoice.grand_total,
   386→   386→        "modified": invoice.modified,
   387→   387→        "invoice_printed": invoice.invoice_printed,
   388→   388→        "restaurant_table": invoice.restaurant_table,
   389→   389→    }

   443→   443→            _("Please set an active menu for Restaurant {0}").format(restaurant)

   574→   574→        except Exception as e:
