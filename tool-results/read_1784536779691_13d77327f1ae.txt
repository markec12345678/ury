   238→// R41-FIX: Add cancel() method to debounce return value so pending timers

   240→function debounce(fn, delay) {
   241→    let timer = null;
   242→    const debounced = function(...args) {
   243→        clearTimeout(timer);
   244→        timer = setTimeout(() => fn.apply(this, args), delay);
   245→    };
   246→    debounced.cancel = () => { clearTimeout(timer); timer = null; };
   247→    return debounced;

   835→    calculateTimeRemaining(targetTime) {
   836→      if (!targetTime || !targetTime.includes(":")) return '— : —';
   837→      const currentTime = new Date();
   838→      // R39-FIX: Guard against malformed time strings with fewer than 3 parts.
   839→      // Previously, "HH:MM" without seconds would set targetSeconds=undefined,
   840→      // which Date() treats as NaN → Invalid Date → NaN propagation.
   841→      const parts = targetTime.split(":");
   842→      const targetHours = parseInt(parts[0], 10);
   843→      const targetMinutes = parseInt(parts[1], 10);
   844→      const targetSeconds = parseInt(parts[2] || '0', 10);
   845→      if (isNaN(targetHours) || isNaN(targetMinutes) || isNaN(targetSeconds)) return '— : —';
   846→      let targetDate = new Date(
   847→        currentTime.getFullYear(),

   855→      if (targetDate > currentTime) {

   860→      const hoursRemaining = Math.floor(timeDifference / 3600000);

   866→    // Without this, a transient server error leaves the KDS showing stale
   867→    // data indefinitely — the only recovery was a manual page reload.
   868→    fetchKOTWithRetry(maxRetries = 3, initialDelay = 2000) {
   869→      // R41-FIX: Clear any pending retry timer before starting a new retry chain.
   870→      // Without this, calling fetchKOTWithRetry() while a previous retry timer is
   871→      // pending would leave the old timer running, potentially causing duplicate
   872→      // fetchKOT calls when both timers fire.
   873→      if (this._fetchRetryTimer) { clearTimeout(this._fetchRetryTimer); this._fetchRetryTimer = null; }
   874→      // R42-FIX: Bump generation counter so any still-running retry chain from
   875→      // a previous fetchKOTWithRetry call knows it's been superseded. Without
   876→      // this, clearing the timer orphans the old promise — its resolve/reject
   877→      // callbacks are never called, leaking the promise and its closures.
   878→      const generation = ++this._fetchGeneration;
   879→      const attempt = (retriesLeft, delay) => {
   880→        // If a newer fetchKOTWithRetry has started, abort this chain
   881→        if (generation !== this._fetchGeneration) {
   882→          return Promise.reject(new Error('Superseded by newer fetchKOTWithRetry call'));
   883→        }
   884→        return this.fetchKOT().catch(err => {
   885→          if (retriesLeft <= 0 || !this._isMounted || generation !== this._fetchGeneration) throw err;
   886→          if (import.meta.env?.DEV) console.warn(`fetchKOT failed, retrying in ${delay}ms...`, err);
   887→          return new Promise((resolve, reject) => {
   888→            this._fetchRetryTimer = setTimeout(() => {
   889→              if (!this._isMounted || generation !== this._fetchGeneration) { reject(err); return; }
   890→              attempt(retriesLeft - 1, Math.min(delay * 2, 30000)).then(resolve).catch(reject);
   891→            }, delay);
   892→          });
   893→        });
   894→      };
   895→      return attempt(maxRetries, initialDelay);
   896→    },

  1434→    visibleKots() {

  1458→            (first || modal).focus();

  1461→      } else {
  1462→        // R48-FIX (M3): Restore focus to the element that was active before modal opened
  1463→        if (this._preModalFocus && typeof this._preModalFocus.focus === 'function') {

  1471→    // R41-FIX: Vue reuses component instances when navigating between routes
  1472→    // that use the same component (Home). Without this watcher, navigating
