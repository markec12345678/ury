   319→   319→// ---- Period Helpers ----
   320→   320→
   321→   321→function getPreviousPeriodDates(
   322→   322→  period: ReportPeriod,
   323→   323→  customFromDate?: string | null,
   324→   324→  customToDate?: string | null
   325→   325→): { prevFrom: string; prevTo: string } {
   326→   326→  const today = new Date();
   327→   327→
   328→   328→  // If custom date range, calculate the previous equivalent period
   329→   329→  if (customFromDate && customToDate) {
   330→   330→    const from = new Date(customFromDate);
   331→   331→    const to = new Date(customToDate);
   332→   332→    const diffMs = to.getTime() - from.getTime();
   333→   333→    const prevTo = new Date(from.getTime() - 1); // Day before from
   334→   334→    const prevFrom = new Date(prevTo.getTime() - diffMs);
   335→   335→    return {
   336→   336→      prevFrom: prevFrom.toISOString().split('T')[0],
   337→   337→      prevTo: prevTo.toISOString().split('T')[0],
   338→   338→    };
   339→   339→  }
   340→   340→
   341→   341→  switch (period) {
   342→   342→    case 'daily':
   343→   343→    case 'yesterday': {
   344→   344→      // Previous day
   345→   345→      const prev = new Date(today);
   346→   346→      prev.setDate(prev.getDate() - 1);
   347→   347→      const dateStr = prev.toISOString().split('T')[0];
   348→   348→      return { prevFrom: dateStr, prevTo: dateStr };
   349→   349→    }
   350→   350→    case 'weekly': {
   351→   351→      // Previous week (same day range)
   352→   352→      const prevWeekStart = new Date(today);
   353→   353→      prevWeekStart.setDate(prevWeekStart.getDate() - 7 - prevWeekStart.getDay());
   354→   354→      const prevWeekEnd = new Date(prevWeekStart);
   355→   355→      prevWeekEnd.setDate(prevWeekEnd.getDate() + 6);
   356→   356→      return {
   357→   357→        prevFrom: prevWeekStart.toISOString().split('T')[0],
   358→   358→        prevTo: prevWeekEnd.toISOString().split('T')[0],
   359→   359→      };
   360→   360→    }
   361→   361→    case 'monthly': {
   362→   362→      // Previous month
   363→   363→      const prevMonth = new Date(today.getFullYear(), today.getMonth() - 1, 1);
   364→   364→      const prevMonthEnd = new Date(today.getFullYear(), today.getMonth(), 0);
   365→   365→      return {
   366→   366→        prevFrom: prevMonth.toISOString().split('T')[0],
   367→   367→        prevTo: prevMonthEnd.toISOString().split('T')[0],
   368→   368→      };
   369→   369→    }
   370→   370→    case 'last_month': {
   371→   371→      // Two months ago
   372→   372→      const twoMonthsAgo = new Date(today.getFullYear(), today.getMonth() - 2, 1);
   373→   373→      const twoMonthsAgoEnd = new Date(today.getFullYear(), today.getMonth() - 1, 0);
   374→   374→      return {
   375→   375→        prevFrom: twoMonthsAgo.toISOString().split('T')[0],
   376→   376→        prevTo: twoMonthsAgoEnd.toISOString().split('T')[0],
   377→   377→      };
   378→   378→    }
   379→   379→    case 'last_7_days': {
   380→   380→      const prevEnd = new Date(today);
   381→   381→      prevEnd.setDate(prevEnd.getDate() - 8);
   382→   382→      const prevStart = new Date(prevEnd);
   383→   383→      prevStart.setDate(prevStart.getDate() - 6);
   384→   384→      return {
   385→   385→        prevFrom: prevStart.toISOString().split('T')[0],
   386→   386→        prevTo: prevEnd.toISOString().split('T')[0],
   387→   387→      };
   388→   388→    }
   389→   389→    case 'last_30_days': {
   390→   390→      const prevEnd = new Date(today);
   391→   391→      prevEnd.setDate(prevEnd.getDate() - 31);
   392→   392→      const prevStart = new Date(prevEnd);
   393→   393→      prevStart.setDate(prevStart.getDate() - 29);
   394→   394→      return {
   395→   395→        prevFrom: prevStart.toISOString().split('T')[0],
   396→   396→        prevTo: prevEnd.toISOString().split('T')[0],
   397→   397→      };
   398→   398→    }
   399→   399→    default: {
   400→   400→      const prev = new Date(today);
   401→   401→      prev.setDate(prev.getDate() - 1);
   402→   402→      const dateStr = prev.toISOString().split('T')[0];
   403→   403→      return { prevFrom: dateStr, prevTo: dateStr };
   404→   404→    }
   405→   405→  }
   406→   406→}

   410→   410→function escapeCsvField(field: string | number): string {
   411→   411→  const str = String(field);
   412→   412→  if (str.startsWith('=') || str.startsWith('+') || str.startsWith('-') || str.startsWith('@') || str.startsWith('\t') || str.startsWith('\r')) {
   413→   413→    return `"'${str.replace(/"/g, '""')}"`;
   414→   414→  }
   415→   415→  if (str.includes(',') || str.includes('"') || str.includes('\n')) {
   416→   416→    return `"${str.replace(/"/g, '""')}"`;
   417→   417→  }
   418→   418→  return str;
   419→   419→}
   420→   420→
   421→   421→function csvRow(...fields: (string | number)[]): string {
   422→   422→  return fields.map(escapeCsvField).join(',');
   423→   423→}

   425→   425→function downloadCsv(content: string, filename: string): void {
   426→   426→  const BOM = '\uFEFF'; // UTF-8 BOM for Excel compatibility
   427→   427→  const blob = new Blob([BOM + content], { type: 'text/csv;charset=utf-8;' });
   428→   428→  const url = URL.createObjectURL(blob);
   429→   429→  const link = document.createElement('a');
   430→   430→  link.setAttribute('href', url);
   431→   431→  link.setAttribute('download', filename);
   432→   432→  document.body.appendChild(link);
   433→   433→  link.click();
   434→   434→  document.body.removeChild(link);
   435→   435→  URL.revokeObjectURL(url);
   436→   436→}

   438→   438→function generateSalesCsv(report: SalesReport): string {
   439→   439→  const lines: string[] = [];
   440→   440→
   441→   441→  // Header
   442→   442→  lines.push(csvRow('URY Restaurant - Sales Report'));
   443→   443→  lines.push(csvRow('Period', `${report.from_date} to ${report.to_date}`));
   444→   444→  lines.push(csvRow('Generated', new Date().toLocaleString()));
   445→   445→  lines.push('');
   446→   446→
   447→   447→  // Summary
   448→   448→  lines.push(csvRow('--- Summary ---'));
   449→   449→  lines.push(csvRow('Metric', 'Value'));
   450→   450→  lines.push(csvRow('Total Revenue', report.summary.total_revenue.toFixed(2)));
   451→   451→  lines.push(csvRow('Total Orders', report.summary.total_orders));
   452→   452→  lines.push(csvRow('Avg Order Value', report.summary.avg_order_value.toFixed(2)));
   453→   453→  lines.push(csvRow('Net Revenue', report.summary.net_revenue.toFixed(2)));
   454→   454→  lines.push(csvRow('Total Tax', report.summary.total_tax.toFixed(2)));
   455→   455→  lines.push(csvRow('Unique Customers', report.summary.unique_customers));
   456→   456→  lines.push('');
   457→   457→
   458→   458→  // Item Sales
   459→   459→  if (report.item_sales.length > 0) {
   460→   460→    lines.push(csvRow('--- Item-wise Sales ---'));
   461→   461→    lines.push(csvRow('Item', 'Qty', 'Avg Rate', 'Total Amount'));
   462→   462→    report.item_sales.forEach((item) => {
   463→   463→      lines.push(csvRow(item.item_name, item.total_qty, item.avg_rate.toFixed(2), item.total_amount.toFixed(2)));
   464→   464→    });
   465→   465→    lines.push('');
   466→   466→  }
   467→   467→
   468→   468→  // Order Type Sales
   469→   469→  if (report.order_type_sales.length > 0) {
   470→   470→    lines.push(csvRow('--- Order Type Breakdown ---'));
   471→   471→    lines.push(csvRow('Order Type', 'Orders', 'Revenue'));
   472→   472→    report.order_type_sales.forEach((ot) => {
   473→   473→      lines.push(csvRow(ot.order_type, ot.order_count, ot.revenue.toFixed(2)));
   474→   474→    });
