   123→// ── UI Preference Persistence ────────────────────────────
   124→const UI_PREFS_KEY = 'ury_ui_prefs';
   125→
   126→interface UIPreferences {
   127→  darkMode: boolean;
   128→  activeTab: string;
   129→}
   130→
   131→function loadUIPrefs(): Partial<UIPreferences> {
   132→  if (typeof window === 'undefined') return {};
   133→  try {
   134→    const raw = localStorage.getItem(UI_PREFS_KEY);
   135→    if (!raw) return {};
   136→    return JSON.parse(raw);
   137→  } catch {
   138→    return {};
   139→  }
   140→}
   141→
   142→function saveUIPrefs(prefs: Partial<UIPreferences>): void {
   143→  if (typeof window === 'undefined') return;
   144→  try {
   145→    localStorage.setItem(UI_PREFS_KEY, JSON.stringify(prefs));
   146→  } catch {
   147→    // ignore storage errors
   148→  }
   149→}

   488→    refreshData: async () => {
   489→      const { isConnected } = get();
   490→      if (!isConnected) return;
   491→
   492→      set({ isRefreshing: true });
   493→
   494→      try {
   495→        const client = getFrappeClient();
   496→
   497→        // Run all fetches in parallel for better performance
   498→        const [
   499→          kotResult,
   500→          tablesResult,
   501→          invoicesResult,
   502→          roomsResult,

   534→              }))
   535→            : [];
   536→          set({ kotCards: transformedKOTs });
   537→        }
   538→
   539→        // ── Process Tables ──────────────────────────────
   540→        if (tablesResult.status === 'fulfilled' && tablesResult.value?.data) {
   541→          const transformedTables: TableData[] = Array.isArray(tablesResult.value.data)
   542→            ? tablesResult.value.data.map((t: Record<string, unknown>) => ({
   543→                id: Number(t.name || t.id || 0),
   544→                room: String(t.room || ''),
   545→                status: (t.status as TableData['status']) || 'free',

   550→                customer: t.customer ? String(t.customer) : undefined,
   551→              }))
   552→            : [];
   553→          set({ tables: transformedTables });
   554→        }
   555→
   556→        // ── Process Invoices → Recent Orders + KPIs ─────
   557→        if (invoicesResult.status === 'fulfilled' && invoicesResult.value?.data) {
   558→          const invoices = Array.isArray(invoicesResult.value.data)
   559→            ? invoicesResult.value.data
   560→            : [];
   561→
   562→          // Recent orders
   563→          const transformedOrders: RecentOrder[] = invoices.map((inv: Record<string, unknown>) => ({
   564→            invoice: String(inv.name || ''),
   565→            customer: String(inv.customer || ''),
   566→            type: (inv.type === 'Dine-in' ? 'Dine-in' : inv.type === 'Takeaway' ? 'Takeaway' : 'Delivery') as RecentOrder['type'],
   567→            amount: Number(inv.grand_total || 0),
   568→            status: (inv.status as RecentOrder['status']) || 'Paid',

   570→          }));
   571→          set({ recentOrders: transformedOrders });
   572→
   573→          // Compute KPIs from invoices
   574→          const totalSales = invoices.reduce((sum: number, inv: Record<string, unknown>) => sum + Number(inv.grand_total || 0), 0);
   575→          const totalOrders = invoices.length;
   576→          const occupiedTables = invoices.filter((inv: Record<string, unknown>) => inv.type === 'Dine-in').length;
   577→          const allTables = get().tables;
   578→          set({
   579→            kpis: {
   580→              dailySales: totalSales,
   581→              totalOrders,
   582→              avgBill: totalOrders > 0 ? Math.round(totalSales / totalOrders) : 0,
   583→              occupiedTables,
   584→              totalTables: allTables.length || 32,
   585→            },
   586→          });
   587→
   588→          // Compute hourly sales from invoices
   589→          const hourMap = new Map<string, { dineIn: number; takeaway: number }>();
   590→          // Initialize all hours from 11:00 to 22:00
   591→          for (let h = 11; h <= 22; h++) {
   592→            const key = `${String(h).padStart(2, '0')}:00`;
   593→            hourMap.set(key, { dineIn: 0, takeaway: 0 });
   594→          }
   595→          // Aggregate invoice totals by hour
   596→          invoices.forEach((inv: Record<string, unknown>) => {
   597→            const timeStr = String(inv.posting_time || '').slice(0, 5);
   598→            if (!timeStr) return;
   599→            const hour = timeStr.split(':')[0];
