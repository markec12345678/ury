   321→function getPreviousPeriodDates(
   322→  period: ReportPeriod,
   323→  customFromDate?: string | null,
   324→  customToDate?: string | null
   325→): { prevFrom: string; prevTo: string } {
   326→  const today = new Date();
   327→
   328→  // If custom date range, calculate the previous equivalent period
   329→  if (customFromDate && customToDate) {
   330→    const from = new Date(customFromDate);
   331→    const to = new Date(customToDate);
   332→    const diffMs = to.getTime() - from.getTime();
   333→    const prevTo = new Date(from.getTime() - 1); // Day before from
   334→    const prevFrom = new Date(prevTo.getTime() - diffMs);
   335→    return {
   336→      prevFrom: prevFrom.toISOString().split('T')[0],
   337→      prevTo: prevTo.toISOString().split('T')[0],
   338→    };
   339→  }
   340→
   341→  switch (period) {
   342→    case 'daily':
   343→    case 'yesterday': {
   344→      // Previous day
   345→      const prev = new Date(today);
   346→      prev.setDate(prev.getDate() - 1);
   347→      const dateStr = prev.toISOString().split('T')[0];
   348→      return { prevFrom: dateStr, prevTo: dateStr };
   349→    }
   350→    case 'weekly': {
   351→      // Previous week (same day range)
   352→      const prevWeekStart = new Date(today);
   353→      prevWeekStart.setDate(prevWeekStart.getDate() - 7 - prevWeekStart.getDay());
   354→      const prevWeekEnd = new Date(prevWeekStart);
   355→      prevWeekEnd.setDate(prevWeekEnd.getDate() + 6);
   356→      return {
   357→        prevFrom: prevWeekStart.toISOString().split('T')[0],
   358→        prevTo: prevWeekEnd.toISOString().split('T')[0],
   359→      };
   360→    }
   361→    case 'monthly': {
   362→      // Previous month
   363→      const prevMonth = new Date(today.getFullYear(), today.getMonth() - 1, 1);
   364→      const prevMonthEnd = new Date(today.getFullYear(), today.getMonth(), 0);
   365→      return {
   366→        prevFrom: prevMonth.toISOString().split('T')[0],
   367→        prevTo: prevMonthEnd.toISOString().split('T')[0],
   368→      };
   369→    }
   370→    case 'last_month': {
   371→      // Two months ago
   372→      const twoMonthsAgo = new Date(today.getFullYear(), today.getMonth() - 2, 1);
   373→      const twoMonthsAgoEnd = new Date(today.getFullYear(), today.getMonth() - 1, 0);
   374→      return {
   375→        prevFrom: twoMonthsAgo.toISOString().split('T')[0],
   376→        prevTo: twoMonthsAgoEnd.toISOString().split('T')[0],
   377→      };
   378→    }
   379→    case 'last_7_days': {
   380→      const prevEnd = new Date(today);
   381→      prevEnd.setDate(prevEnd.getDate() - 8);
   382→      const prevStart = new Date(prevEnd);
   383→      prevStart.setDate(prevStart.getDate() - 6);
   384→      return {
   385→        prevFrom: prevStart.toISOString().split('T')[0],
   386→        prevTo: prevEnd.toISOString().split('T')[0],
   387→      };
   388→    }
   389→    case 'last_30_days': {
   390→      const prevEnd = new Date(today);
   391→      prevEnd.setDate(prevEnd.getDate() - 31);
   392→      const prevStart = new Date(prevEnd);
   393→      prevStart.setDate(prevStart.getDate() - 29);
   394→      return {
   395→        prevFrom: prevStart.toISOString().split('T')[0],
   396→        prevTo: prevEnd.toISOString().split('T')[0],
   397→      };
   398→    }
   399→    default: {
   400→      const prev = new Date(today);
   401→      prev.setDate(prev.getDate() - 1);
   402→      const dateStr = prev.toISOString().split('T')[0];
   403→      return { prevFrom: dateStr, prevTo: dateStr };
   404→    }
   405→  }
   406→}

   410→function escapeCsvField(field: string | number): string {
   411→  const str = String(field);
   412→  if (str.startsWith('=') || str.startsWith('+') || str.startsWith('-') || str.startsWith('@') || str.startsWith('\t') || str.startsWith('\r')) {
   413→    return `"'${str.replace(/"/g, '""')}"`;
   414→  }
   415→  if (str.includes(',') || str.includes('"') || str.includes('\n')) {
   416→    return `"${str.replace(/"/g, '""')}"`;
   417→  }
   418→  return str;
   419→}

   421→function csvRow(...fields: (string | number)[]): string {
   422→  return fields.map(escapeCsvField).join(',');
   423→}

   425→function downloadCsv(content: string, filename: string): void {
   426→  const BOM = '\uFEFF'; // UTF-8 BOM for Excel compatibility
   427→  const blob = new Blob([BOM + content], { type: 'text/csv;charset=utf-8;' });
   428→  const url = URL.createObjectURL(blob);
   429→  const link = document.createElement('a');
   430→  link.setAttribute('href', url);
   431→  link.setAttribute('download', filename);
   432→  document.body.appendChild(link);
   433→  link.click();
   434→  document.body.removeChild(link);
   435→  URL.revokeObjectURL(url);
   436→}
