   198→   198→        pluck="role",
   199→   199→    )

   201→   201→
   202→   202→    # Check if the last invoice was already billed

   204→   204→        last_invoice
   205→   205→        and frappe.db.get_value("POS Invoice", last_invoice, "invoice_printed") == 1
   206→   206→        and (not billing_user)

   212→   212→        )

   214→   214→
   215→   215→    # R41-FIX: Use _get_order_invoice_doc to get the document object, not a dict.
   216→   216→    # The public get_order_invoice returns a dict for frontend safety (BE-R36-003),
   217→   217→    # but sync_order needs the document to call .save() etc.

   219→   219→
   220→   220→    # R41-FIX: Validate last_invoice belongs to user's branch

   235→   235→                    last_modified_time, "%Y-%m-%d %H:%M:%S"
   236→   236→                )
   237→   237→        if isinstance(lastModifiedTime, str):
   238→   238→            try:
   239→   239→                lastModifiedTime = datetime.strptime(
   240→   240→                    lastModifiedTime, "%Y-%m-%d %H:%M:%S.%f"
   241→   241→                )
   242→   242→            except ValueError:
   243→   243→                lastModifiedTime = datetime.strptime(
   244→   244→                    lastModifiedTime, "%Y-%m-%d %H:%M:%S"
   245→   245→                )
   246→   246→        if lastModifiedTime != last_modified_time:
   247→   247→            frappe.msgprint(
   248→   248→                title=_("Order has been modified"),
   249→   249→                indicator="red",
   250→   250→                msg=_("This order has been modified. Please reload the page to retrieve the latest edits."),
   251→   251→            )
   252→   252→            return {"status": "Failure"}
   253→   253→    else:
   254→   254→        if invoice.name and invoice.invoice_printed == 0 and not billing_user:
   255→   255→            frappe.msgprint(
   256→   256→                title=_("Table Occupied"),
   257→   257→                indicator="red",
   258→   258→                msg=_("{0} is already occupied. Please refresh the page.").format(table),
   259→   259→            )
   260→   260→            return {"status": "Failure"}
   261→   261→
   262→   262→    if not customer:

   267→   267→    if order_type:
   268→   268→        invoice.order_type = order_type

   274→   274→    invoice.pos_profile = pos_profile

   277→   277→        if role_user[1] and not frappe.db.exists("User", {"name": role_user[1], "enabled": 1}):
   278→   278→            frappe.throw(_("Invalid {0}: {1}").format(role_user[0], role_user[1]))
   279→   279→
   280→   280→    invoice.cashier = cashier
   281→   281→    invoice.waiter = waiter
   282→   282→    invoice.custom_aggregator_id = aggregator_id
   283→   283→    invoice.custom_restaurant_room =room

   285→   285→

   292→   292→        price_list = invoice.selling_price_list

   294→   294→    # dummy payment

   300→   300→    if invoice.invoice_created == 0:
   301→   301→        invoice.append(

   317→   317→

   390→   390→    except Exception as e:

   411→   411→        "customer": invoice.customer,
   412→   412→        "grand_total": invoice.grand_total,
   413→   413→        "modified": invoice.modified,
