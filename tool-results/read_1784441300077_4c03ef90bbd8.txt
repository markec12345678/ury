   491→    updateTimeRemaining() {
   492→      this.kot.forEach((kot) => {
   493→        kot.timeRemaining = this.calculateTimeRemaining(kot.time);
   494→
   495→        const timeRemaining = kot.timeRemaining.split(":");
   496→        const minutes =
   497→          parseInt(timeRemaining[0], 10) * 60 + parseInt(timeRemaining[1], 10);
   498→
   499→        // R36-FIX: Handle NaN from invalid time format — treat as elapsed time exceeded
   500→        const validMinutes = isNaN(minutes) ? Infinity : minutes;
   501→
   502→        if (
   503→          validMinutes === Number(this.kot_alert_time) &&
   504→          kot.type !== "Cancelled" &&
   505→          kot.type !== "Partially cancelled" &&
   506→          !this.notifiedKots.has(kot.name)
   507→        ) {
   508→          this.notifiedKots.add(kot.name);
   509→          this.orderDelayNotify(kot);
   510→        }
   511→        if (validMinutes >= this.kot_alert_time) {
   512→          kot.timecolor = "text-[#DC0000]";
   513→        } else {
   514→          kot.timecolor = "text-black";
   515→        }
   516→      });

   518→    calculateTimeRemaining(targetTime) {
   519→      if (!targetTime || !targetTime.includes(":")) return '— : —';
   520→      const currentTime = new Date();
   521→      const [targetHours, targetMinutes, targetSeconds] = targetTime.split(":");
   522→      let targetDate = new Date(
   523→        currentTime.getFullYear(),
   524→        currentTime.getMonth(),
   525→        currentTime.getDate(),
   526→        targetHours,
   527→        targetMinutes,
   528→        targetSeconds
   529→      );
   530→      // Handle cross-midnight KOTs: if target is in the future, it was created yesterday
   531→      if (targetDate > currentTime) {
   532→        targetDate = new Date(targetDate.getTime() - 86400000);
   533→      }
   534→
   535→      const timeDifference = Math.max(0, currentTime - targetDate);
   536→      const hoursRemaining = Math.floor(timeDifference / 3600000);
   537→      const minutesRemaining = Math.floor((timeDifference % 3600000) / 60000);
   538→
   539→      return `${hoursRemaining} : ${String(minutesRemaining).padStart(2, '0')}`;
   540→    },

   600→    handleTransitionEnd() {
   601→      if (!this.isOnline) {
   602→        // Reset the status message after transition end
   603→        this.setStatusMessage("");
   604→      }
   605→    },
   606→  },
   607→  created() {
   608→    // R36-FIX: Initialize as non-reactive instance properties (no Proxy overhead)
   609→    this._isMounted = false;
   610→    this._fetchInProgress = null;
   611→    this.notifiedKots = new Set();
   612→    // API client as non-reactive instance property (avoids Proxy overhead)
   613→    this.call = markRaw(frappe.call());
   614→  },
   615→  mounted() {
   616→    this._isMounted = true;
   617→    window.addEventListener("online", this.handleOnline);
   618→    window.addEventListener("offline", this.handleOffline);
   619→    document.addEventListener("click", this.hideAudioAlertMessage);
   620→    // R36-FIX: Use route params instead of fragile URL parsing
   621→    const production = this.$route?.params?.production || '';
   622→    const decodedProduction = decodeURIComponent(production);
   623→    this.production = decodedProduction;
   624→
   625→    const debouncedMasonry = debounce(() => { this.masonryLoading(); }, 150);
   626→    this._resizeHandler = debouncedMasonry;
   627→    window.addEventListener("resize", this._resizeHandler);
   628→    this.masonryLoading();
   629→
   630→    // Initialize socket in mounted() so re-mount gets a fresh connection
   631→    this._socketInitPromise = initializeSocket();
   632→
   633→    // Wait for both socket init and auth before attaching listeners
   634→    Promise.all([this._socketInitPromise, this.auth()])
   635→      .then(([sock]) => {
   636→        // R36-FIX: Guard against post-unmount execution
   637→        if (!this._isMounted) return;
   638→        this._socket = sock;
   639→        if (this._socket) this._socket.on('connect_error', (err) => {
   640→          if (!this._isMounted) return;
   641→          console.error("Socket connection error:", err);
   642→          this.setStatusMessage("Connection error. Retrying...");
   643→        });
   644→        if (this._socket) this._socket.on('disconnect', (reason) => {
   645→          if (!this._isMounted) return;
   646→          console.warn("Socket disconnected:", reason);
   647→          this.setStatusMessage("Connection lost. Reconnecting...");
   648→        });
   649→        if (this._socket) this._socket.on('connect', () => {
   650→          if (!this._isMounted) return;
   651→          this.setStatusMessage("Reconnected");
   652→          this.hideStatusMessageAfterDelay();
   653→        });
   654→
   655→        return this.fetchKOT();
   656→      })
   657→      .then(() => {
   658→        if (!this._isMounted) return;
   659→        if (this.audio_alert === 1) {
   660→          this.showAudioAlertMessage = true;
   661→        }
   662→        this.socketHandler = (doc) => {
   663→          if (!this._isMounted) return;
   664→          try {
   665→            if (this.audio_alert === 1) {
   666→              this.playAlertSound(doc.audio_file);
   667→            }
   668→            // R36-FIX: Namespace localStorage key per production station to avoid cross-tab collision
   669→            let kottime = localStorage.getItem("kot_time_" + this.production);
   670→            if (doc.last_kot_time !== kottime) {
   671→              // Full refresh needed — skip intermediate mutations
   672→              this.fetchKOT().then(() => { this.masonryLoading(); }).catch((e) => { console.error("KOT fetch failed:", e); });
   673→            } else {
   674→              // R36-FIX: Guard against missing doc.kot to prevent TypeError crash
   675→              if (!doc.kot) {
   676→                this.fetchKOT().then(() => { this.masonryLoading(); }).catch((e) => { console.error("KOT fetch failed:", e); });
   677→                return;
   678→              }
   679→              // Incremental update — deduplicate to avoid duplicate cards
   680→              const existingIndex = this.kot.findIndex(k => k.name === doc.kot.name);
   681→              if (existingIndex !== -1) {
   682→                // R36-FIX: Preserve strikethrough state before Object.assign overwrites kot_items
   683→                const strikeMap = new Map(
   684→                  (this.kot[existingIndex].kot_items || []).map(i => [i.name, i.striked])
   685→                );
   686→                Object.assign(this.kot[existingIndex], { timecolor: 'text-black', timeRemaining: '— : —', ...doc.kot });
   687→                // Restore strikethrough state after Object.assign
   688→                if (this.kot[existingIndex].kot_items) {
   689→                  this.kot[existingIndex].kot_items.forEach(i => {
   690→                    if (strikeMap.has(i.name)) i.striked = strikeMap.get(i.name);
   691→                  });
   692→                }
   693→              } else {
   694→                const newKot = { isRotated: false, showDiv: false, timecolor: 'text-black', timeRemaining: '— : —', ...doc.kot };
   695→                this.kot.unshift(newKot);
   696→              }
   697→              this.updateQtyColorTable();
   698→              this.updateTimeRemaining();
   699→              this.masonryLoading();
   700→            }
   701→            if (this._cancelTimeout) clearTimeout(this._cancelTimeout);
   702→            this._cancelTimeout = setTimeout(() => {
   703→              if (!this._isMounted) return;
   704→              if (doc.kot && doc.kot.type === "Cancelled") {
   705→                this.fetchKOT().then(() => {
   706→                  this.masonryLoading();
   707→                }).catch((e) => { console.error("KOT fetch failed:", e); });
   708→              }
   709→            }, 1500);
   710→            if (doc.kot) {
   711→              try {
   712→                localStorage.setItem("kot_time_" + this.production, doc.kot.time);
   713→              } catch (e) {
   714→                if (import.meta.env?.DEV) console.error('localStorage write failed:', e);
   715→              }
   716→            }
   717→          } catch (err) {
   718→            if (import.meta.env?.DEV) console.error("Socket handler error:", err);
   719→          }
   720→        };
   721→        if (this._socket) this._socket.on(this.kot_channel, this.socketHandler);
   722→      })
   723→      .catch((error) => {
   724→        console.error("Initialization or authentication error:", error);
   725→        if (this._isMounted) this.showModal = true;
   726→      });
   727→    this.timer = setInterval(this.updateTimeRemaining, 60000);
   728→  },
   729→  beforeUnmount() {
   730→    this._isMounted = false;
   731→    window.removeEventListener("online", this.handleOnline);
   732→    window.removeEventListener("offline", this.handleOffline);
   733→    document.removeEventListener("click", this.hideAudioAlertMessage);
   734→    window.removeEventListener("resize", this._resizeHandler);
   735→    if (this.socketHandler && this._socket) {
   736→      this._socket.off(this.kot_channel, this.socketHandler);
   737→    }
   738→    if (this._socket) {
   739→      this._socket.off('connect_error');
   740→      this._socket.off('disconnect');
   741→      this._socket.off('connect');
   742→      this._socket.disconnect();
   743→    }
   744→    this._socketInitPromise = null;
   745→    if (this._cancelTimeout) clearTimeout(this._cancelTimeout);
   746→    if (this._statusTimeout) clearTimeout(this._statusTimeout);
   747→    if (this.timer) clearInterval(this.timer);
   748→    if (this._alertAudio) { this._alertAudio.pause(); this._alertAudio = null; }
   749→  },
   750→  computed: {
   751→    sortedKotItems() {
   752→      return (kot) => {
   753→        if (!kot._sortedItems || kot._sortKey !== (kot.kot_items || []).length) {
   754→          kot._sortKey = (kot.kot_items || []).length;
   755→          kot._sortedItems = [...(kot.kot_items || [])].sort((a, b) => a.serve_priority - b.serve_priority);
   756→        }
   757→        return kot._sortedItems;
   758→      };
   759→    },
   760→    visibleKots() {
   761→      return this.kot.filter(kot => !kot.showDiv && kot.production === this.production);
   762→    },
   763→  },
   764→};
