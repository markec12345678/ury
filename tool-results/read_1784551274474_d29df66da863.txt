   793→   793→   793→    allowed_mops = frappe.get_all("POS Profile Payment", filters={"parent": pos_profile}, pluck="mode_of_payment")

   823→   823→   823→    invoice.customer = customer
   824→   824→   824→    invoice.pos_profile = pos_profile

   827→   827→   827→
   828→   828→   828→    # H-07: Validate that payment amounts match the grand total (prevent negative invoice from discount)
