   229→// R41-FIX: Add cancel() method to debounce return value so pending timers

   231→function debounce(fn, delay) {
   232→    let timer = null;
   233→    const debounced = function(...args) {
   234→        clearTimeout(timer);
   235→        timer = setTimeout(() => fn.apply(this, args), delay);
   236→    };
   237→    debounced.cancel = () => { clearTimeout(timer); timer = null; };
   238→    return debounced;
   239→}

   666→    calculateTimeRemaining(targetTime) {
   667→      if (!targetTime || !targetTime.includes(":")) return '— : —';
   668→      const currentTime = new Date();
   669→      // R39-FIX: Guard against malformed time strings with fewer than 3 parts.
   670→      // Previously, "HH:MM" without seconds would set targetSeconds=undefined,
   671→      // which Date() treats as NaN → Invalid Date → NaN propagation.
   672→      const parts = targetTime.split(":");
   673→      const targetHours = parseInt(parts[0], 10);
   674→      const targetMinutes = parseInt(parts[1], 10);
   675→      const targetSeconds = parseInt(parts[2] || '0', 10);
   676→      if (isNaN(targetHours) || isNaN(targetMinutes) || isNaN(targetSeconds)) return '— : —';
   677→      let targetDate = new Date(
   678→        currentTime.getFullYear(),
   679→        currentTime.getMonth(),
   680→        currentTime.getDate(),
   681→        targetHours,
   682→        targetMinutes,
   683→        targetSeconds
   684→      );
   685→      // Handle cross-midnight KOTs: if target is in the future, it was created yesterday
   686→      if (targetDate > currentTime) {
   687→        targetDate = new Date(targetDate.getTime() - 86400000);
   688→      }
   689→
   690→      const timeDifference = Math.max(0, currentTime - targetDate);
   691→      const hoursRemaining = Math.floor(timeDifference / 3600000);
   692→      const minutesRemaining = Math.floor((timeDifference % 3600000) / 60000);
   693→
   694→      return `${hoursRemaining} : ${String(minutesRemaining).padStart(2, '0')}`;
   695→    },

   697→    // Without this, a transient server error leaves the KDS showing stale
   698→    // data indefinitely — the only recovery was a manual page reload.
   699→    fetchKOTWithRetry(maxRetries = 3, initialDelay = 2000) {
   700→      // R41-FIX: Clear any pending retry timer before starting a new retry chain.
   701→      // Without this, calling fetchKOTWithRetry() while a previous retry timer is
   702→      // pending would leave the old timer running, potentially causing duplicate
   703→      // fetchKOT calls when both timers fire.
   704→      if (this._fetchRetryTimer) { clearTimeout(this._fetchRetryTimer); this._fetchRetryTimer = null; }
   705→      const attempt = (retriesLeft, delay) => {
   706→        return this.fetchKOT().catch(err => {
   707→          if (retriesLeft <= 0 || !this._isMounted) throw err;
   708→          if (import.meta.env?.DEV) console.warn(`fetchKOT failed, retrying in ${delay}ms...`, err);
   709→          return new Promise((resolve, reject) => {
   710→            this._fetchRetryTimer = setTimeout(() => {
   711→              if (!this._isMounted) { reject(err); return; }
   712→              attempt(retriesLeft - 1, Math.min(delay * 2, 30000)).then(resolve).catch(reject);
   713→            }, delay);
   714→          });
   715→        });
   716→      };
   717→      return attempt(maxRetries, initialDelay);

  1022→              }, 1500);
  1023→            }
  1024→            // R41-FIX: Guard against storing null/undefined as string "null"/"undefined"
  1025→            // in localStorage. These corrupted sentinel values would cause the
  1026→            // next socket event to always take the full-refresh path.
  1027→            if (doc.kot.time != null) {
  1028→              try {
  1029→                localStorage.setItem("kot_time_" + this.production, doc.kot.time);
  1030→              } catch (e) {
  1031→                if (import.meta.env?.DEV) console.error('localStorage write failed:', e);
  1032→              }
  1033→            }
  1034→          } catch (err) {
  1035→            if (import.meta.env?.DEV) console.error("Socket handler error:", err);
  1036→          }
  1037→        };
  1038→        if (this._socket) this._socket.on(this.kot_channel, this.socketHandler);
  1039→      })
  1040→      .catch((error) => {
  1041→        console.error("Initialization or authentication error:", error);
  1042→        // R40-FIX: this._socket is now stored early, so beforeUnmount will
  1043→        // disconnect it. No need for manual cleanup here.
  1044→        if (this._isMounted) {
  1045→          // R41-FIX: Differentiate auth errors from network/server errors.
  1046→          // Auth failure → "Not Permitted" with login prompt.
  1047→          // Network/server failure → "Connection Error" with retry guidance.
  1048→          const isAuthError = error && (
  1049→            error.httpStatus === 401 ||
  1050→            error.httpStatus === 403 ||
  1051→            /auth/i.test(String(error.message || ''))
  1052→          );
  1053→          if (isAuthError) {
  1054→            this.modalTitle = "Not Permitted";
  1055→            this.modalMessage = "Log in to access this page.";
  1056→          } else {
  1057→            this.modalTitle = "Connection Error";
  1058→            this.modalMessage = "Unable to reach the server. Please check your connection and try again.";
  1059→          }
  1060→          this.showModal = true;
  1061→        }
  1062→      });
  1063→    this.timer = setInterval(this.updateTimeRemaining, 60000);
  1064→  },
  1065→  beforeUnmount() {

  1067→    window.removeEventListener("online", this.handleOnline);
  1068→    window.removeEventListener("offline", this.handleOffline);

  1074→    }

  1077→      this._socket.off('disconnect');
  1078→      this._socket.off('connect');
  1079→      this._socket.disconnect();
  1080→    }
  1081→    this._socketInitPromise = null;
  1082→    if (this._cancelTimeout) clearTimeout(this._cancelTimeout);
  1083→    if (this._statusTimeout) clearTimeout(this._statusTimeout);
  1084→    if (this._socketRetryTimer) clearTimeout(this._socketRetryTimer);
  1085→    if (this._fetchRetryTimer) clearTimeout(this._fetchRetryTimer);
  1086→    // R41-FIX: Cancel pending debounced masonry calls on unmount.
  1087→    // Without this, a debounced masonryLayout could fire after unmount,
  1088→    // attempting DOM operations on a detached element tree.
  1089→    if (this._debouncedMasonryLayout) this._debouncedMasonryLayout.cancel();
  1090→    if (this._resizeHandler) this._resizeHandler.cancel?.();
  1091→    if (this.timer) clearInterval(this.timer);
  1092→    if (this._alertAudio) { this._alertAudio.pause(); this._alertAudio = null; }
  1093→    // R39-FIX: Destroy non-reactive masonry instance on unmount
  1094→    if (this._masonry) {
  1095→      this._masonry.destroy();
  1096→      this._masonry = null;
  1097→    }
  1098→  },
  1099→  computed: {
  1100→    // R39-FIX: sortedKotItems moved from computed (which returned a function) to
  1101→    // a method. Previously, the computed returned a new function object every time
  1102→    // its reactive dependencies (this.kot) changed, which happened every minute
  1103→    // via updateTimeRemaining. Each new function object was unnecessary churn.
  1104→    // As a method, the function identity is stable and the manual _sortedItemsCache
  1105→    // still provides memoization. The template call `sortedKotItems(kot)` works
  1106→    // identically for both computed-returning-function and method.
  1107→    visibleKots() {
  1108→      // R39-FIX: Removed !kot.showDiv filter — showDiv was always false (dead code).
  1109→      // When production is empty (root route "/"), show ALL KOTs instead of filtering
  1110→      // by empty string which would exclude KOTs that have a station assigned.
  1111→      if (!this.production) {
  1112→        return this.kot;
  1113→      }
  1114→      return this.kot.filter(kot => kot.production === this.production);
  1115→    },
  1116→  },
  1117→  watch: {
  1118→    // R41-FIX: Vue reuses component instances when navigating between routes
  1119→    // that use the same component (Home). Without this watcher, navigating
  1120→    // from /station/kitchen to /station/bar would NOT re-mount the KOT
  1121→    // component, so this.production would still be "kitchen" and the KDS
  1122→    // would filter for the wrong station.
  1123→    '$route.params.production'(newVal) {
  1124→      const decoded = decodeURIComponent(newVal || '');
  1125→      if (decoded === this.production) return;
  1126→      this.production = decoded;
  1127→      // Re-register socket handler on the new channel
  1128→      if (this.kot_channel && this._socket && this.socketHandler) {
  1129→        this._socket.off(this.kot_channel, this.socketHandler);
  1130→      }
  1131→      // fetchKOT will set the new kot_channel and re-register the handler
  1132→      this.fetchKOTWithRetry().then(() => {
  1133→        this.masonryLoading();
  1134→      }).catch(() => {});
  1135→    },
  1136→  },
