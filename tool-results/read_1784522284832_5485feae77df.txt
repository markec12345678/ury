   742→   742→    // Without this, a transient server error leaves the KDS showing stale
   743→   743→    // data indefinitely — the only recovery was a manual page reload.
   744→   744→    fetchKOTWithRetry(maxRetries = 3, initialDelay = 2000) {
   745→   745→      // R41-FIX: Clear any pending retry timer before starting a new retry chain.
   746→   746→      // Without this, calling fetchKOTWithRetry() while a previous retry timer is
   747→   747→      // pending would leave the old timer running, potentially causing duplicate
   748→   748→      // fetchKOT calls when both timers fire.
   749→   749→      if (this._fetchRetryTimer) { clearTimeout(this._fetchRetryTimer); this._fetchRetryTimer = null; }
   750→   750→      // R42-FIX: Bump generation counter so any still-running retry chain from
   751→   751→      // a previous fetchKOTWithRetry call knows it's been superseded. Without
   752→   752→      // this, clearing the timer orphans the old promise — its resolve/reject
   753→   753→      // callbacks are never called, leaking the promise and its closures.
   754→   754→      const generation = ++this._fetchGeneration;
   755→   755→      const attempt = (retriesLeft, delay) => {
   756→   756→        // If a newer fetchKOTWithRetry has started, abort this chain
   757→   757→        if (generation !== this._fetchGeneration) {
   758→   758→          return Promise.reject(new Error('Superseded by newer fetchKOTWithRetry call'));
   759→   759→        }
   760→   760→        return this.fetchKOT().catch(err => {
   761→   761→          if (retriesLeft <= 0 || !this._isMounted || generation !== this._fetchGeneration) throw err;
   762→   762→          if (import.meta.env?.DEV) console.warn(`fetchKOT failed, retrying in ${delay}ms...`, err);
   763→   763→          return new Promise((resolve, reject) => {
   764→   764→            this._fetchRetryTimer = setTimeout(() => {
   765→   765→              if (!this._isMounted || generation !== this._fetchGeneration) { reject(err); return; }
   766→   766→              attempt(retriesLeft - 1, Math.min(delay * 2, 30000)).then(resolve).catch(reject);
   767→   767→            }, delay);
   768→   768→          });
   769→   769→        });
   770→   770→      };
   771→   771→      return attempt(maxRetries, initialDelay);
   772→   772→    },
