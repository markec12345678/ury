     1→'use client';
     2→
     3→import { useState, useMemo } from 'react';
     4→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
     5→import { Button } from '@/components/ui/button';
     6→import { Badge } from '@/components/ui/badge';
     7→import { Input } from '@/components/ui/input';
     8→import { Label } from '@/components/ui/label';
     9→import { Switch } from '@/components/ui/switch';
    10→import { Separator } from '@/components/ui/separator';
    11→import {
    12→  Dialog,
    13→  DialogContent,
    14→  DialogDescription,
    15→  DialogFooter,
    16→  DialogHeader,
    17→  DialogTitle,
    18→  DialogTrigger,
    19→} from '@/components/ui/dialog';
    20→import {
    21→  AlertDialog,
    22→  AlertDialogAction,
    23→  AlertDialogCancel,
    24→  AlertDialogContent,
    25→  AlertDialogDescription,
    26→  AlertDialogFooter,
    27→  AlertDialogHeader,
    28→  AlertDialogTitle,
    29→} from '@/components/ui/alert-dialog';
    30→import {
    31→  Select,
    32→  SelectContent,
    33→  SelectItem,
    34→  SelectTrigger,
    35→  SelectValue,
    36→} from '@/components/ui/select';
    37→import {
    38→  Table,
    39→  TableBody,
    40→  TableCell,
    41→  TableHead,
    42→  TableHeader,
    43→  TableRow,
    44→} from '@/components/ui/table';
    45→import { Textarea } from '@/components/ui/textarea';
    46→import {
    47→  BookOpen,
    48→  Search,
    49→  Leaf,
    50→  Plus,
    51→  Pencil,
    52→  Trash2,
    53→  X,
    54→  Tag,
    55→  GripVertical,
    56→  ArrowUpDown,
    57→  Eye,
    58→  EyeOff,
    59→  SlidersHorizontal,
    60→} from 'lucide-react';
    61→import { useURYStore } from '@/lib/ury-store';
    62→import type { MenuItem, MenuCourse, MenuFormData, CourseFormData } from '@/lib/ury-types';
    63→
    64→export function MenuManagementTab() {
    65→  const {
    66→    menuCourses,
    67→    menuItems,
    68→    currency,
    69→    addMenuItem,
    70→    updateMenuItem,
    71→    deleteMenuItem,
    72→    toggleMenuItemAvailability,
    73→    addMenuCourse,
    74→    updateMenuCourse,
    75→    deleteMenuCourse,
    76→  } = useURYStore();
    77→
    78→  const [searchQuery, setSearchQuery] = useState('');
    79→  const [activeCourse, setActiveCourse] = useState<string>('all');
    80→  const [viewMode, setViewMode] = useState<'grid' | 'table'>('table');
    81→
    82→  // Menu Item Dialog state
    83→  const [menuItemDialogOpen, setMenuItemDialogOpen] = useState(false);
    84→  const [editingItem, setEditingItem] = useState<MenuItem | null>(null);
    85→  const [menuItemForm, setMenuItemForm] = useState<MenuFormData>({
    86→    name: '',
    87→    nameLocal: '',
    88→    course: '',
    89→    price: 0,
    90→    description: '',
    91→    isVeg: true,
    92→    isAvailable: true,
    93→    tags: [],
    94→  });
    95→  const [tagInput, setTagInput] = useState('');
    96→
    97→  // Course Dialog state
    98→  const [courseDialogOpen, setCourseDialogOpen] = useState(false);
    99→  const [editingCourse, setEditingCourse] = useState<MenuCourse | null>(null);
   100→  const [courseForm, setCourseForm] = useState<CourseFormData>({
   101→    name: '',
   102→    priority: 0,
   103→  });
   104→
   105→  // Delete confirmation
   106→  const [deleteTarget, setDeleteTarget] = useState<{ type: 'item' | 'course'; id: string; name: string } | null>(null);
   107→
   108→  // Filter items
   109→  const filteredItems = useMemo(() => {
   110→    let items = menuItems;
   111→    if (activeCourse !== 'all') {
   112→      items = items.filter((item) => item.course === activeCourse);
   113→    }
   114→    if (searchQuery.trim()) {
   115→      const q = searchQuery.toLowerCase();
   116→      items = items.filter(
   117→        (item) =>
   118→          item.name.toLowerCase().includes(q) ||
   119→          item.courseName.toLowerCase().includes(q) ||
   120→          item.description?.toLowerCase().includes(q) ||
   121→          item.tags?.some((t) => t.toLowerCase().includes(q))
   122→      );
   123→    }
   124→    return items;
   125→  }, [menuItems, activeCourse, searchQuery]);
   126→
   127→  // Stats
   128→  const totalItems = menuItems.length;
   129→  const availableItems = menuItems.filter((i) => i.isAvailable).length;
   130→  const vegItems = menuItems.filter((i) => i.isVeg).length;
   131→  const avgPrice = totalItems > 0
   132→    ? Math.round(menuItems.reduce((sum, i) => sum + i.price, 0) / totalItems)
   133→    : 0;
   134→
   135→  // ── Menu Item CRUD handlers ─────────────────────────────
   136→
   137→  const openCreateItemDialog = () => {
   138→    setEditingItem(null);
   139→    setMenuItemForm({
   140→      name: '',
   141→      nameLocal: '',
   142→      course: menuCourses[0]?.id || '',
   143→      price: 0,
   144→      description: '',
   145→      isVeg: true,
   146→      isAvailable: true,
   147→      tags: [],
   148→    });
   149→    setTagInput('');
   150→    setMenuItemDialogOpen(true);
   151→  };
   152→
   153→  const openEditItemDialog = (item: MenuItem) => {
   154→    setEditingItem(item);
   155→    setMenuItemForm({
   156→      name: item.name,
   157→      nameLocal: item.nameHi || '',
   158→      course: item.course,
   159→      price: item.price,
   160→      description: item.description || '',
   161→      isVeg: item.isVeg,
   162→      isAvailable: item.isAvailable,
   163→      tags: item.tags || [],
   164→    });
   165→    setTagInput('');
   166→    setMenuItemDialogOpen(true);
   167→  };
   168→
   169→  const handleSaveMenuItem = () => {
   170→    if (!menuItemForm.name.trim() || menuItemForm.price <= 0) return;
   171→    if (editingItem) {
   172→      updateMenuItem(editingItem.id, menuItemForm);
   173→    } else {
   174→      addMenuItem(menuItemForm);
   175→    }
   176→    setMenuItemDialogOpen(false);
   177→  };
   178→
   179→  const handleAddTag = () => {
   180→    const tag = tagInput.trim().toLowerCase();
   181→    if (tag && !menuItemForm.tags?.includes(tag)) {
   182→      setMenuItemForm((prev) => ({
   183→        ...prev,
   184→        tags: [...(prev.tags || []), tag],
   185→      }));
   186→    }
   187→    setTagInput('');
   188→  };
   189→
   190→  const handleRemoveTag = (tag: string) => {
   191→    setMenuItemForm((prev) => ({
   192→      ...prev,
   193→      tags: prev.tags?.filter((t) => t !== tag) || [],
   194→    }));
   195→  };
   196→
   197→  // ── Course CRUD handlers ────────────────────────────────
   198→
   199→  const openCreateCourseDialog = () => {
   200→    setEditingCourse(null);
   201→    setCourseForm({
   202→      name: '',
   203→      priority: menuCourses.length + 1,
   204→    });
   205→    setCourseDialogOpen(true);
   206→  };
   207→
   208→  const openEditCourseDialog = (course: MenuCourse) => {
   209→    setEditingCourse(course);
   210→    setCourseForm({
   211→      name: course.name,
   212→      priority: course.priority,
   213→    });
   214→    setCourseDialogOpen(true);
   215→  };
   216→
   217→  const handleSaveCourse = () => {
   218→    if (!courseForm.name.trim()) return;
   219→    if (editingCourse) {
   220→      updateMenuCourse(editingCourse.id, courseForm);
   221→    } else {
   222→      addMenuCourse(courseForm);
   223→    }
   224→    setCourseDialogOpen(false);
   225→  };
   226→
   227→  return (
   228→    <div className="space-y-4">
   229→      {/* Header Stats */}
   230→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   231→        <Card className="bg-emerald-50 border-emerald-200">
   232→          <CardContent className="p-3 flex items-center gap-3">
   233→            <BookOpen className="h-5 w-5 text-emerald-600" />
   234→            <div>
   235→              <p className="text-xs text-emerald-600 font-medium">Skupaj artiklov</p>
   236→              <p className="text-xl font-bold text-emerald-800">{totalItems}</p>
   237→            </div>
   238→          </CardContent>
   239→        </Card>
   240→        <Card className="bg-green-50 border-green-200">
   241→          <CardContent className="p-3 flex items-center gap-3">
   242→            <Leaf className="h-5 w-5 text-green-600" />
   243→            <div>
   244→              <p className="text-xs text-green-600 font-medium">Vegetarijansko</p>
   245→              <p className="text-xl font-bold text-green-800">{vegItems}</p>
   246→            </div>
   247→          </CardContent>
   248→        </Card>
   249→        <Card className="bg-blue-50 border-blue-200">
   250→          <CardContent className="p-3 flex items-center gap-3">
   251→            <Tag className="h-5 w-5 text-blue-600" />
   252→            <div>
   253→              <p className="text-xs text-blue-600 font-medium">Dosegljivo</p>
   254→              <p className="text-xl font-bold text-blue-800">{availableItems}/{totalItems}</p>
   255→            </div>
   256→          </CardContent>
   257→        </Card>
   258→        <Card className="bg-amber-50 border-amber-200">
   259→          <CardContent className="p-3 flex items-center gap-3">
   260→            <SlidersHorizontal className="h-5 w-5 text-amber-600" />
   261→            <div>
   262→              <p className="text-xs text-amber-600 font-medium">Povprečna cena</p>
   263→              <p className="text-xl font-bold text-amber-800">{currency}{avgPrice}</p>
   264→            </div>
   265→          </CardContent>
   266→        </Card>
   267→      </div>
   268→
   269→      {/* Action Bar */}
   270→      <div className="flex flex-wrap items-center gap-3">
   271→        <div className="relative flex-1 min-w-[200px]">
   272→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   273→          <Input
   274→            placeholder="Išči artikle, kategorije, oznake..."
   275→            value={searchQuery}
   276→            onChange={(e) => setSearchQuery(e.target.value)}
   277→            className="pl-9"
   278→          />
   279→          {searchQuery && (
   280→            <Button variant="ghost" size="icon" className="absolute right-1 top-1/2 -translate-y-1/2 h-7 w-7" onClick={() => setSearchQuery('')}>
   281→              <X className="h-3.5 w-3.5" />
   282→            </Button>
   283→          )}
   284→        </div>
   285→        <Button size="sm" className="bg-emerald-600 hover:bg-emerald-700" onClick={openCreateItemDialog}>
   286→          <Plus className="h-4 w-4 mr-1" />
   287→          Nov artikel
   288→        </Button>
   289→        <Button variant="outline" size="sm" onClick={openCreateCourseDialog}>
   290→          <Plus className="h-4 w-4 mr-1" />
   291→          Nova kategorija
   292→        </Button>
   293→        <div className="flex items-center gap-1 border rounded-md p-0.5">
   294→          <Button variant={viewMode === 'table' ? 'default' : 'ghost'} size="sm" className="h-7 px-2" onClick={() => setViewMode('table')}>
   295→            <ArrowUpDown className="h-3.5 w-3.5" />
   296→          </Button>
   297→          <Button variant={viewMode === 'grid' ? 'default' : 'ghost'} size="sm" className="h-7 px-2" onClick={() => setViewMode('grid')}>
   298→            <GripVertical className="h-3.5 w-3.5" />
   299→          </Button>
   300→        </div>
   301→      </div>
   302→
   303→      {/* Course Filters */}
   304→      <div className="flex flex-wrap gap-2">
   305→        <Button
   306→          variant={activeCourse === 'all' ? 'default' : 'outline'}
   307→          size="sm"
   308→          className={activeCourse === 'all' ? 'bg-emerald-600 hover:bg-emerald-700' : ''}
   309→          onClick={() => setActiveCourse('all')}
   310→        >
   311→          Vse ({menuItems.length})
   312→        </Button>
   313→        {menuCourses.map((course) => {
   314→          const count = menuItems.filter((i) => i.course === course.id).length;
   315→          return (
   316→            <Button
   317→              key={course.id}
   318→              variant={activeCourse === course.id ? 'default' : 'outline'}
   319→              size="sm"
   320→              className={activeCourse === course.id ? 'bg-emerald-600 hover:bg-emerald-700' : ''}
   321→              onClick={() => setActiveCourse(course.id)}
   322→            >
   323→              {course.name} ({count})
   324→            </Button>
   325→          );
   326→        })}
   327→      </div>
   328→
   329→      {/* Course Management Section */}
   330→      <Card>
   331→        <CardHeader className="pb-2">
   332→          <div className="flex items-center justify-between">
   333→            <CardTitle className="text-base font-semibold">Kategorije jedilnika</CardTitle>
   334→            <Badge variant="secondary" className="text-xs">{menuCourses.length} kategorij</Badge>
   335→          </div>
   336→        </CardHeader>
   337→        <CardContent>
   338→          <div className="flex flex-wrap gap-2">
   339→            {menuCourses.map((course) => (
   340→              <div
   341→                key={course.id}
   342→                className="flex items-center gap-2 px-3 py-1.5 rounded-lg border bg-muted/50 text-sm"
   343→              >
   344→                <span className="text-muted-foreground text-xs">#{course.priority}</span>
   345→                <span className="font-medium">{course.name}</span>
   346→                <Badge variant="outline" className="text-[10px] h-4 px-1">{course.itemCount} artiklov</Badge>
   347→                <Button variant="ghost" size="icon" className="h-5 w-5" onClick={() => openEditCourseDialog(course)}>
   348→                  <Pencil className="h-3 w-3" />
   349→                </Button>
   350→                <Button variant="ghost" size="icon" className="h-5 w-5 text-red-500 hover:text-red-700" onClick={() => setDeleteTarget({ type: 'course', id: course.id, name: course.name })}>
   351→                  <Trash2 className="h-3 w-3" />
   352→                </Button>
   353→              </div>
   354→            ))}
   355→          </div>
   356→        </CardContent>
   357→      </Card>
   358→
   359→      {/* Results count */}
   360→      <div className="text-sm text-muted-foreground">
   361→        Prikazano {filteredItems.length} od {menuItems.length} artiklov
   362→        {searchQuery && ` za "${searchQuery}"`}
   363→      </div>
   364→
   365→      {/* Items Display */}
   366→      {viewMode === 'table' ? (
   367→        <Card>
   368→          <CardContent className="p-0">
   369→            <div className="overflow-x-auto">
   370→              <Table>
   371→                <TableHeader>
   372→                  <TableRow>
   373→                    <TableHead className="w-10">Veg</TableHead>
   374→                    <TableHead>Ime</TableHead>
   375→                    <TableHead>Kategorija</TableHead>
   376→                    <TableHead className="text-right">Cena</TableHead>
   377→                    <TableHead>Status</TableHead>
   378→                    <TableHead>Oznake</TableHead>
   379→                    <TableHead className="text-right">Dejanja</TableHead>
   380→                  </TableRow>
   381→                </TableHeader>
   382→                <TableBody>
   383→                  {filteredItems.map((item) => (
   384→                    <TableRow key={item.id} className={!item.isAvailable ? 'opacity-50' : ''}>
   385→                      <TableCell>
   386→                        <span className={`inline-block w-3 h-3 rounded-sm border ${
   387→                          item.isVeg ? 'bg-green-500 border-green-600' : 'bg-red-500 border-red-600'
   388→                        }`} />
   389→                      </TableCell>
   390→                      <TableCell>
   391→                        <div>
   392→                          <p className="font-medium text-sm">{item.name}</p>
   393→                          {item.nameHi && <p className="text-xs text-muted-foreground">{item.nameHi}</p>}
   394→                          {item.description && <p className="text-xs text-muted-foreground line-clamp-1 mt-0.5">{item.description}</p>}
   395→                        </div>
   396→                      </TableCell>
   397→                      <TableCell className="text-sm">{item.courseName}</TableCell>
   398→                      <TableCell className="text-right font-mono font-medium text-emerald-600">
   399→                        {currency}{item.price.toLocaleString('en-IN')}
   400→                      </TableCell>
   401→                      <TableCell>
   402→                        <Badge
   403→                          variant="outline"
   404→                          className={`text-[10px] ${
   405→                            item.isAvailable
   406→                              ? 'border-emerald-300 text-emerald-700'
   407→                              : 'border-red-300 text-red-700 bg-red-50'
   408→                          }`}
   409→                        >
   410→                          {item.isAvailable ? 'Na voljo' : 'Ni na voljo'}
   411→                        </Badge>
   412→                      </TableCell>
   413→                      <TableCell>
   414→                        <div className="flex flex-wrap gap-1">
   415→                          {item.tags?.map((tag) => (
   416→                            <Badge key={tag} variant="outline" className="text-[10px] px-1.5 py-0 h-4">
   417→                              {tag}
   418→                            </Badge>
   419→                          ))}
   420→                        </div>
   421→                      </TableCell>
   422→                      <TableCell className="text-right">
   423→                        <div className="flex items-center justify-end gap-1">
   424→                          <Button
   425→                            variant="ghost"
   426→                            size="icon"
   427→                            className="h-7 w-7"
   428→                            onClick={() => toggleMenuItemAvailability(item.id)}
   429→                            title={item.isAvailable ? 'Onemogoči' : 'Omogoči'}
   430→                          >
   431→                            {item.isAvailable ? <Eye className="h-3.5 w-3.5" /> : <EyeOff className="h-3.5 w-3.5 text-red-500" />}
   432→                          </Button>
   433→                          <Button variant="ghost" size="icon" className="h-7 w-7" onClick={() => openEditItemDialog(item)}>
   434→                            <Pencil className="h-3.5 w-3.5" />
   435→                          </Button>
   436→                          <Button
   437→                            variant="ghost"
   438→                            size="icon"
   439→                            className="h-7 w-7 text-red-500 hover:text-red-700"
   440→                            onClick={() => setDeleteTarget({ type: 'item', id: item.id, name: item.name })}
   441→                          >
   442→                            <Trash2 className="h-3.5 w-3.5" />
   443→                          </Button>
   444→                        </div>
   445→                      </TableCell>
   446→                    </TableRow>
   447→                  ))}
   448→                </TableBody>
   449→              </Table>
   450→            </div>
   451→          </CardContent>
   452→        </Card>
   453→      ) : (
   454→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-3">
   455→          {filteredItems.map((item) => (
   456→            <Card key={item.id} className={`transition-all hover:shadow-md ${!item.isAvailable ? 'opacity-60' : ''}`}>
   457→              <CardContent className="p-4 space-y-3">
   458→                <div className="flex items-start justify-between gap-2">
   459→                  <div className="flex-1 min-w-0">
   460→                    <div className="flex items-center gap-2">
   461→                      <span className={`w-3 h-3 rounded-sm border shrink-0 ${
   462→                        item.isVeg ? 'bg-green-500 border-green-600' : 'bg-red-500 border-red-600'
   463→                      }`} />
   464→                      <h3 className="font-semibold text-sm truncate">{item.name}</h3>
   465→                    </div>
   466→                    <p className="text-xs text-muted-foreground mt-0.5">{item.courseName}</p>
   467→                  </div>
   468→                  <span className="text-base font-bold text-emerald-600 shrink-0">
   469→                    {currency}{item.price.toLocaleString('en-IN')}
   470→                  </span>
   471→                </div>
   472→                {item.description && (
   473→                  <p className="text-xs text-muted-foreground line-clamp-2">{item.description}</p>
   474→                )}
   475→                {item.tags && item.tags.length > 0 && (
   476→                  <div className="flex flex-wrap gap-1">
   477→                    {item.tags.map((tag) => (
   478→                      <Badge key={tag} variant="outline" className="text-[10px] px-1.5 py-0 h-4">{tag}</Badge>
   479→                    ))}
   480→                  </div>
   481→                )}
   482→                <div className="flex items-center justify-between">
   483→                  <Badge
   484→                    variant="outline"
   485→                    className={`text-[10px] ${
   486→                      item.isAvailable ? 'border-emerald-300 text-emerald-700' : 'border-red-300 text-red-700 bg-red-50'
   487→                    }`}
   488→                  >
   489→                    {item.isAvailable ? 'Na voljo' : 'Ni na voljo'}
   490→                  </Badge>
   491→                  <div className="flex items-center gap-1">
   492→                    <Button variant="ghost" size="icon" className="h-6 w-6" onClick={() => toggleMenuItemAvailability(item.id)}>
   493→                      {item.isAvailable ? <Eye className="h-3 w-3" /> : <EyeOff className="h-3 w-3 text-red-500" />}
   494→                    </Button>
   495→                    <Button variant="ghost" size="icon" className="h-6 w-6" onClick={() => openEditItemDialog(item)}>
   496→                      <Pencil className="h-3 w-3" />
   497→                    </Button>
   498→                    <Button variant="ghost" size="icon" className="h-6 w-6 text-red-500" onClick={() => setDeleteTarget({ type: 'item', id: item.id, name: item.name })}>
   499→                      <Trash2 className="h-3 w-3" />
   500→                    </Button>
   501→                  </div>
   502→                </div>
   503→              </CardContent>
   504→            </Card>
   505→          ))}
   506→        </div>
   507→      )}
   508→
   509→      {filteredItems.length === 0 && (
   510→        <div className="py-12 text-center text-muted-foreground">
   511→          <BookOpen className="h-12 w-12 mx-auto mb-3 opacity-30" />
   512→          <p>Ni zadetkov</p>
   513→          <p className="text-sm">Poskusite spremeniti iskalni niz ali filtre</p>
   514→        </div>
   515→      )}
   516→
   517→      {/* ── Menu Item Dialog ──────────────────────────────── */}
   518→      <Dialog open={menuItemDialogOpen} onOpenChange={setMenuItemDialogOpen}>
   519→        <DialogContent className="sm:max-w-[525px]">
   520→          <DialogHeader>
   521→            <DialogTitle>{editingItem ? 'Uredi artikel' : 'Nov artikel'}</DialogTitle>
   522→            <DialogDescription>
   523→              {editingItem ? 'Posodobite podatke artikla v jedilniku.' : 'Dodajte nov artikel v jedilnik.'}
   524→            </DialogDescription>
   525→          </DialogHeader>
   526→          <div className="grid gap-4 py-4">
   527→            <div className="grid grid-cols-2 gap-4">
   528→              <div className="space-y-2">
   529→                <Label htmlFor="item-name">Ime artikla *</Label>
   530→                <Input
   531→                  id="item-name"
   532→                  placeholder="npr. Butter Chicken"
   533→                  value={menuItemForm.name}
   534→                  onChange={(e) => setMenuItemForm((prev) => ({ ...prev, name: e.target.value }))}
   535→                />
   536→              </div>
   537→              <div className="space-y-2">
