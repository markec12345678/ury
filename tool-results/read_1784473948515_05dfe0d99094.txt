   229→function debounce(fn, delay) {
   230→    let timer = null;
   231→    return function(...args) {
   232→        clearTimeout(timer);
   233→        timer = setTimeout(() => fn.apply(this, args), delay);
   234→    };

   486→    updateQtyColorTable() {

   489→      // single pass, then look up from the map. This replaces N individual
   490→      // localStorage.getItem() calls with one iteration + O(1) map lookups.
   491→      const strikeMap = {};
   492→      try {
   493→        for (let i = 0; i < localStorage.length; i++) {

   495→          if (key.endsWith('_strike')) {
   496→            strikeMap[key] = localStorage.getItem(key);

   505→          kot,
   506→          kot.restaurant_table,
   507→          kot.type,
   508→          kot.table_takeaway
   509→        );

   511→        // R37-FIX: Null-check kot_items before forEach

   514→          const key = `${kot.name}_${kotitem.name}_strike`;
   515→          const savedState = strikeMap[key];

   517→            try {
   518→              kotitem.striked = JSON.parse(savedState);
   519→            } catch (e) {

   543→      this._sortedItemsCache.set(kot.name, { key, items: sorted });

   566→
   567→    updateTimeRemaining() {
   568→      this.kot.forEach((kot) => {
   569→        kot.timeRemaining = this.calculateTimeRemaining(kot.time);
   570→
   571→        const timeRemaining = kot.timeRemaining.split(":");
   572→        const minutes =
   573→          parseInt(timeRemaining[0], 10) * 60 + parseInt(timeRemaining[1], 10);
   574→
   575→        // R36-FIX: Handle NaN from invalid time format — treat as elapsed time exceeded
   576→        const validMinutes = isNaN(minutes) ? Infinity : minutes;
   577→
   578→        if (
   579→          validMinutes === Number(this.kot_alert_time) &&
   580→          kot.type !== "Cancelled" &&
   581→          kot.type !== "Partially cancelled" &&
   582→          !this.notifiedKots.has(kot.name)
   583→        ) {
   584→          this.notifiedKots.add(kot.name);
   585→          this.orderDelayNotify(kot);
   586→        }
   587→        // R38-FIX: Guard against empty/falsy kot_alert_time. An empty string
   588→        // coerces to 0 in >= comparison, making ALL KOTs show red time.
   589→        const alertThreshold = Number(this.kot_alert_time);
   590→        if (alertThreshold > 0 && validMinutes >= alertThreshold) {
   591→          kot.timecolor = "text-[#DC0000]";
   592→        } else {
   593→          kot.timecolor = "text-black";
   594→        }
   595→      });
   596→    },
   597→    calculateTimeRemaining(targetTime) {
   598→      if (!targetTime || !targetTime.includes(":")) return '— : —';
   599→      const currentTime = new Date();
   600→      // R39-FIX: Guard against malformed time strings with fewer than 3 parts.
   601→      // Previously, "HH:MM" without seconds would set targetSeconds=undefined,
   602→      // which Date() treats as NaN → Invalid Date → NaN propagation.
   603→      const parts = targetTime.split(":");
   604→      const targetHours = parseInt(parts[0], 10);
   605→      const targetMinutes = parseInt(parts[1], 10);
   606→      const targetSeconds = parseInt(parts[2] || '0', 10);
   607→      if (isNaN(targetHours) || isNaN(targetMinutes) || isNaN(targetSeconds)) return '— : —';
   608→      let targetDate = new Date(
   609→        currentTime.getFullYear(),
   610→        currentTime.getMonth(),
   611→        currentTime.getDate(),
   612→        targetHours,
   613→        targetMinutes,
   614→        targetSeconds
   615→      );
   616→      // Handle cross-midnight KOTs: if target is in the future, it was created yesterday
   617→      if (targetDate > currentTime) {
   618→        targetDate = new Date(targetDate.getTime() - 86400000);
   619→      }
   620→
   621→      const timeDifference = Math.max(0, currentTime - targetDate);
   622→      const hoursRemaining = Math.floor(timeDifference / 3600000);
   623→      const minutesRemaining = Math.floor((timeDifference % 3600000) / 60000);
   624→
   625→      return `${hoursRemaining} : ${String(minutesRemaining).padStart(2, '0')}`;
   626→    },
   627→    fetchkotwithmasonry() {
   628→      return this.fetchKOT().then(() => {
   629→        this.masonryLoading();
   630→      }).catch((e) => { console.error("KOT fetch failed:", e); });
   631→    },
   632→    redirectToLogin() {
   633→      // R38-FIX: Use Vue Router instead of window.location.href to avoid full page reload
   634→      // and preserve app state. The previous approach caused a hard navigation that
   635→      // lost all Vue state and required a full re-initialization.
   636→      this.$router.push({ name: 'Login', query: { route: this.$route.path } });
   637→    },
   638→    masonryLoading(forceRecreate = false) {
   639→      // R39-FIX: Use non-reactive _masonry instead of reactive masonry from data()
   640→      if (this._masonry && !forceRecreate) {
   641→        this.$nextTick(() => {
   642→          if (this._masonry) {
   643→            this._masonry.reloadItems?.();
   644→            this._masonry.layout();
   645→          }
   646→        });
   647→        return;
   648→      }
   649→      if (this._masonry) {
   650→        this._masonry.destroy();
   651→        this._masonry = null;
   652→      }
   653→      this.$nextTick(() => {
   654→        if (!this.$el) return;
   655→        const grid = this.$el.querySelector(".grid");
   656→        if (!grid) return;
   657→        this._masonry = markRaw(new Masonry(grid, {
   658→          itemSelector: ".masonry-item",
   659→          gutter: 28,
   660→        }));
   661→        this._masonry.layout();
   662→      });
   663→    },
   664→    hideAudioAlertMessage() {
   665→      this.showAudioAlertMessage = false;
   666→    },
   667→    handleOnline() {
   668→      // R37-FIX: Guard against post-unmount execution
   669→      if (!this._isMounted) return;
   670→      this.isOnline = true;
   671→      this.setStatusMessage("You are online");
   672→      this.hideStatusMessageAfterDelay();
   673→      this.fetchKOT().then(() => {
   674→        if (!this._isMounted) return;
   675→        this.masonryLoading();
   676→      }).catch((e) => { console.error("KOT fetch failed:", e); });
   677→    },
   678→    handleOffline() {
   679→      // R37-FIX: Guard against post-unmount execution
   680→      if (!this._isMounted) return;
   681→      this.isOnline = false;
   682→      this.setStatusMessage("You are Offline");
   683→    },
   684→    setStatusMessage(message) {
   685→      this.statusMessage = message;
   686→    },
   687→    hideStatusMessageAfterDelay() {
   688→      if (this._statusTimeout) clearTimeout(this._statusTimeout);
   689→      this._statusTimeout = setTimeout(() => {
   690→        // R37-FIX: Guard against mutating unmounted component
   691→        if (!this._isMounted) return;
   692→        this.statusMessage = "";
   693→      }, 3000);
   694→    },
   695→    handleTransitionEnd() {
   696→      if (!this.isOnline) {
   697→        // Reset the status message after transition end
   698→        this.setStatusMessage("");
   699→      }
   700→    },
   701→  },
   702→  created() {
   703→    // R36-FIX: Initialize as non-reactive instance properties (no Proxy overhead)
   704→    this._isMounted = false;
   705→    this._fetchInProgress = null;
   706→    this.notifiedKots = new Set();
   707→    // API client as non-reactive instance property (avoids Proxy overhead)
   708→    this.call = markRaw(frappe.call());
   709→    // R37-FIX: socketHandler as non-reactive to avoid unnecessary Proxy overhead
   710→    this.socketHandler = null;
   711→    // R37-FIX: sortedItems cache map — avoids mutating reactive kot objects in computed
   712→    this._sortedItemsCache = new Map();
   713→    // R39-FIX: masonry as non-reactive instance property — Masonry objects are large
   714→    // and should not be tracked by Vue's reactivity system. Previously in data(),
   715→    // it triggered unnecessary Proxy overhead and could cause cascading re-renders
   716→    // when the masonry reference changed during reloadItems/layout calls.
   717→    this._masonry = null;
   718→    // R39-FIX: Track socket init retry timer for cleanup
   719→    this._socketRetryTimer = null;
   720→  },
   721→  mounted() {
   722→    this._isMounted = true;
   723→    window.addEventListener("online", this.handleOnline);
   724→    window.addEventListener("offline", this.handleOffline);
   725→    document.addEventListener("click", this.hideAudioAlertMessage);
   726→    // R36-FIX: Use route params instead of fragile URL parsing
   727→    const production = this.$route?.params?.production || '';
   728→    const decodedProduction = decodeURIComponent(production);
   729→    this.production = decodedProduction;
   730→
   731→    const debouncedMasonry = debounce(() => { this.masonryLoading(); }, 150);
   732→    this._resizeHandler = debouncedMasonry;
   733→    window.addEventListener("resize", this._resizeHandler);
   734→    this.masonryLoading();
   735→
   736→    // R39-FIX: Handle bfcache restore — when the browser restores this page from
   737→    // back-forward cache, the socket may be stale and timers may not fire.
   738→    // Re-initialize socket and re-fetch KOT data on pageshow with persisted=true.
   739→    this._pageshowHandler = (event) => {
   740→      if (event.persisted) {
   741→        // Page was restored from bfcache — re-fetch data and re-init socket
   742→        this.fetchKOT().then(() => { this.masonryLoading(); }).catch(() => {});
   743→        if (this._socket && !this._socket.connected) {
   744→          this._socket.connect();
   745→        }
   746→      }
   747→    };
   748→    window.addEventListener('pageshow', this._pageshowHandler);
   749→
   750→    // R39-FIX: Socket init with retry — if fetchSiteName fails (network issue,
   751→    // server down), retry with exponential backoff instead of giving up forever.
   752→    // Without this, a transient site-name fetch failure would leave the KDS
   753→    // without real-time updates until a full page reload.
   754→    const initSocketWithRetry = (retries = 0) => {
   755→      return initializeSocket().then(sock => {
   756→        if (sock) return sock;
   757→        // initializeSocket returned null (siteName fetch failed) — retry
   758→        const delay = Math.min(2000 * Math.pow(2, retries), 30000);
   759→        if (!this._isMounted) return null;
   760→        return new Promise(resolve => {
   761→          this._socketRetryTimer = setTimeout(() => {
   762→            if (!this._isMounted) { resolve(null); return; }
   763→            resolve(initSocketWithRetry(retries + 1));
   764→          }, delay);
   765→        });
   766→      });
   767→    };
   768→    this._socketInitPromise = initSocketWithRetry();
   769→
   770→    // Wait for both socket init and auth before attaching listeners
   771→    Promise.all([this._socketInitPromise, this.auth()])
   772→      .then(([sock]) => {
   773→        // R36-FIX: Guard against post-unmount execution
   774→        if (!this._isMounted) return;
   775→        this._socket = sock;
   776→        if (this._socket) this._socket.on('connect_error', (err) => {
   777→          if (!this._isMounted) return;
   778→          console.error("Socket connection error:", err);
   779→          this.setStatusMessage("Connection error. Retrying...");
   780→        });
   781→        if (this._socket) this._socket.on('disconnect', (reason) => {
   782→          if (!this._isMounted) return;
   783→          console.warn("Socket disconnected:", reason);
   784→          this.setStatusMessage("Connection lost. Reconnecting...");
   785→        });
   786→        if (this._socket) this._socket.on('connect', () => {
   787→          if (!this._isMounted) return;
   788→          this.setStatusMessage("Reconnected");
   789→          this.hideStatusMessageAfterDelay();
   790→          // R38-FIX: Re-fetch KOT data after reconnect to sync any missed updates
   791→          // during disconnection period. Without this, the UI could show stale KOTs.
   792→          this.fetchKOT().then(() => { this.masonryLoading(); }).catch(() => {});
   793→        });
   794→
   795→        return this.fetchKOT();
   796→      })
   797→      .then(() => {
   798→        if (!this._isMounted) return;
   799→        if (this.audio_alert === 1) {
   800→          this.showAudioAlertMessage = true;
   801→        }
   802→        this.socketHandler = (doc) => {
   803→          if (!this._isMounted) return;
   804→          // R39-FIX: Guard against null/undefined doc from malformed socket messages
   805→          if (!doc) return;
   806→          try {
   807→            if (this.audio_alert === 1) {
   808→              this.playAlertSound(doc.audio_file);
   809→            }
   810→            // R36-FIX: Namespace localStorage key per production station to avoid cross-tab collision
   811→            // R39-FIX: Normalize null kottime from localStorage — if getItem returns null
   812→            // (no previous value stored) and doc.last_kot_time is also null, they'd be
   813→            // equal and fall through to the incremental path incorrectly. Treat null
   814→            // localStorage value as a signal to do a full refresh.
   815→            let kottime = localStorage.getItem("kot_time_" + this.production);
   816→            if (doc.last_kot_time !== kottime || kottime === null) {
   817→              // Full refresh needed — skip intermediate mutations
   818→              this.fetchKOT().then(() => { this.masonryLoading(); }).catch((e) => { console.error("KOT fetch failed:", e); });
   819→            } else {
   820→              // R36-FIX: Guard against missing doc.kot to prevent TypeError crash
   821→              if (!doc.kot) {
   822→                this.fetchKOT().then(() => { this.masonryLoading(); }).catch((e) => { console.error("KOT fetch failed:", e); });
   823→                return;
   824→              }
   825→              // Incremental update — deduplicate to avoid duplicate cards
   826→              const existingIndex = this.kot.findIndex(k => k.name === doc.kot.name);
   827→              if (existingIndex !== -1) {
   828→                // R36-FIX: Preserve strikethrough state before Object.assign overwrites kot_items
   829→                const strikeMap = new Map(
   830→                  (this.kot[existingIndex].kot_items || []).map(i => [i.name, i.striked])
   831→                );
   832→                Object.assign(this.kot[existingIndex], { timecolor: 'text-black', timeRemaining: '— : —', ...doc.kot });
   833→                // Restore strikethrough state after Object.assign
   834→                if (this.kot[existingIndex].kot_items) {
   835→                  this.kot[existingIndex].kot_items.forEach(i => {
   836→                    if (strikeMap.has(i.name)) i.striked = strikeMap.get(i.name);
   837→                  });
   838→                }
   839→                // R37-FIX: Invalidate sorted cache for this KOT since items may have changed
   840→                this._sortedItemsCache.delete(doc.kot.name);
   841→              } else {
   842→                const newKot = { isRotated: false, showDiv: false, timecolor: 'text-black', timeRemaining: '— : —', ...doc.kot };
   843→                this.kot.unshift(newKot);
   844→              }
   845→              this.updateQtyColorTable();
   846→              this.updateTimeRemaining();
   847→              this.masonryLoading();
   848→            }
   849→            if (this._cancelTimeout) clearTimeout(this._cancelTimeout);
   850→            // R39-FIX: Only schedule cancel re-fetch if the KOT is actually a cancellation.
   851→            // Previously, _cancelTimeout was set for ALL socket events, causing an
   852→            // unnecessary re-fetch 1.5s after every non-cancel update. Now we only
   853→            // set the timeout when doc.kot.type is "Cancelled", matching the check inside.
   854→            if (doc.kot && doc.kot.type === "Cancelled") {
   855→              this._cancelTimeout = setTimeout(() => {
   856→                if (!this._isMounted) return;
   857→                this.fetchKOT().then(() => {
   858→                  this.masonryLoading();
   859→                }).catch((e) => { console.error("KOT fetch failed:", e); });
   860→              }, 1500);
   861→            }
   862→            if (doc.kot) {
   863→              try {
   864→                localStorage.setItem("kot_time_" + this.production, doc.kot.time);
   865→              } catch (e) {
   866→                if (import.meta.env?.DEV) console.error('localStorage write failed:', e);
   867→              }
   868→            }
   869→          } catch (err) {
   870→            if (import.meta.env?.DEV) console.error("Socket handler error:", err);
   871→          }
   872→        };
   873→        if (this._socket) this._socket.on(this.kot_channel, this.socketHandler);
   874→      })
   875→      .catch((error) => {
   876→        console.error("Initialization or authentication error:", error);
   877→        if (this._isMounted) this.showModal = true;
   878→      });
   879→    this.timer = setInterval(this.updateTimeRemaining, 60000);
   880→  },
   881→  beforeUnmount() {
   882→    this._isMounted = false;
   883→    window.removeEventListener("online", this.handleOnline);
   884→    window.removeEventListener("offline", this.handleOffline);
   885→    document.removeEventListener("click", this.hideAudioAlertMessage);
   886→    window.removeEventListener("resize", this._resizeHandler);
   887→    if (this._pageshowHandler) window.removeEventListener('pageshow', this._pageshowHandler);
   888→    if (this.socketHandler && this._socket) {
   889→      this._socket.off(this.kot_channel, this.socketHandler);
   890→    }
   891→    if (this._socket) {
   892→      this._socket.off('connect_error');
   893→      this._socket.off('disconnect');
   894→      this._socket.off('connect');
   895→      this._socket.disconnect();
   896→    }
   897→    this._socketInitPromise = null;
   898→    if (this._cancelTimeout) clearTimeout(this._cancelTimeout);
   899→    if (this._statusTimeout) clearTimeout(this._statusTimeout);
   900→    if (this._socketRetryTimer) clearTimeout(this._socketRetryTimer);
   901→    if (this.timer) clearInterval(this.timer);
   902→    if (this._alertAudio) { this._alertAudio.pause(); this._alertAudio = null; }
   903→    // R39-FIX: Destroy non-reactive masonry instance on unmount
   904→    if (this._masonry) {
   905→      this._masonry.destroy();
   906→      this._masonry = null;
   907→    }
   908→  },
   909→  computed: {
   910→    // R39-FIX: sortedKotItems moved from computed (which returned a function) to
   911→    // a method. Previously, the computed returned a new function object every time
   912→    // its reactive dependencies (this.kot) changed, which happened every minute
   913→    // via updateTimeRemaining. Each new function object was unnecessary churn.
   914→    // As a method, the function identity is stable and the manual _sortedItemsCache
   915→    // still provides memoization. The template call `sortedKotItems(kot)` works
   916→    // identically for both computed-returning-function and method.
   917→    visibleKots() {
   918→      // R39-FIX: When production is empty (root route "/"), show ALL KOTs instead
   919→      // of filtering by empty string which would show nothing. The root route has
   920→      // no :production param, so this.production is ''. Filtering by kot.production === ''
   921→      // would exclude every KOT that has a station assigned, resulting in an empty display.
   922→      if (!this.production) {
   923→        return this.kot.filter(kot => !kot.showDiv);
   924→      }
   925→      return this.kot.filter(kot => !kot.showDiv && kot.production === this.production);
   926→    },
   927→  },
   928→};
