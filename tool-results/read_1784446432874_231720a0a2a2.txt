   229→function debounce(fn, delay) {
   230→    let timer = null;
   231→    return function(...args) {
   232→        clearTimeout(timer);
   233→        timer = setTimeout(() => fn.apply(this, args), delay);
   234→    };

   545→    calculateTimeRemaining(targetTime) {
   546→      if (!targetTime || !targetTime.includes(":")) return '— : —';
   547→      const currentTime = new Date();
   548→      const [targetHours, targetMinutes, targetSeconds] = targetTime.split(":");
   549→      let targetDate = new Date(
   550→        currentTime.getFullYear(),
   551→        currentTime.getMonth(),
   552→        currentTime.getDate(),
   553→        targetHours,
   554→        targetMinutes,
   555→        targetSeconds
   556→      );
   557→      // Handle cross-midnight KOTs: if target is in the future, it was created yesterday
   558→      if (targetDate > currentTime) {
   559→        targetDate = new Date(targetDate.getTime() - 86400000);
   560→      }
   561→
   562→      const timeDifference = Math.max(0, currentTime - targetDate);
   563→      const hoursRemaining = Math.floor(timeDifference / 3600000);
   564→      const minutesRemaining = Math.floor((timeDifference % 3600000) / 60000);
   565→
   566→      return `${hoursRemaining} : ${String(minutesRemaining).padStart(2, '0')}`;

   579→      if (this.masonry && !forceRecreate) {

   581→          if (this.masonry) {
   582→            this.masonry.reloadItems?.();
   583→            this.masonry.layout();
   584→          }
   585→        });
   586→        return;
   587→      }
   588→      if (this.masonry) {
   589→        this.masonry.destroy();
   590→        this.masonry = null;
   591→      }
   592→      this.$nextTick(() => {
   593→        if (!this.$el) return;
   594→        const grid = this.$el.querySelector(".grid");
   595→        if (!grid) return;
   596→        this.masonry = markRaw(new Masonry(grid, {
   597→          itemSelector: ".masonry-item",
   598→          gutter: 28,
   599→        }));
   600→        this.masonry.layout();
   601→      });
   602→    },
   603→    hideAudioAlertMessage() {
   604→      this.showAudioAlertMessage = false;
   605→    },
   606→    handleOnline() {
   607→      // R37-FIX: Guard against post-unmount execution
   608→      if (!this._isMounted) return;
   609→      this.isOnline = true;
   610→      this.setStatusMessage("You are online");
   611→      this.hideStatusMessageAfterDelay();
   612→      this.fetchKOT().then(() => {
   613→        if (!this._isMounted) return;
   614→        this.masonryLoading();
   615→      }).catch((e) => { console.error("KOT fetch failed:", e); });
   616→    },
   617→    handleOffline() {
   618→      // R37-FIX: Guard against post-unmount execution
   619→      if (!this._isMounted) return;
   620→      this.isOnline = false;
   621→      this.setStatusMessage("You are Offline");
   622→    },
   623→    setStatusMessage(message) {
   624→      this.statusMessage = message;
   625→    },
   626→    hideStatusMessageAfterDelay() {
   627→      if (this._statusTimeout) clearTimeout(this._statusTimeout);
   628→      this._statusTimeout = setTimeout(() => {
   629→        // R37-FIX: Guard against mutating unmounted component
   630→        if (!this._isMounted) return;
   631→        this.statusMessage = "";
   632→      }, 3000);
   633→    },
   634→    handleTransitionEnd() {
   635→      if (!this.isOnline) {
   636→        // Reset the status message after transition end
   637→        this.setStatusMessage("");
   638→      }
   639→    },
   640→  },
   641→  created() {
   642→    // R36-FIX: Initialize as non-reactive instance properties (no Proxy overhead)
   643→    this._isMounted = false;
   644→    this._fetchInProgress = null;
   645→    this.notifiedKots = new Set();
   646→    // API client as non-reactive instance property (avoids Proxy overhead)
   647→    this.call = markRaw(frappe.call());
   648→    // R37-FIX: socketHandler as non-reactive to avoid unnecessary Proxy overhead
   649→    this.socketHandler = null;
   650→    // R37-FIX: sortedItems cache map — avoids mutating reactive kot objects in computed
   651→    this._sortedItemsCache = new Map();
   652→  },
   653→  mounted() {
   654→    this._isMounted = true;
   655→    window.addEventListener("online", this.handleOnline);
   656→    window.addEventListener("offline", this.handleOffline);
   657→    document.addEventListener("click", this.hideAudioAlertMessage);
   658→    // R36-FIX: Use route params instead of fragile URL parsing
   659→    const production = this.$route?.params?.production || '';
   660→    const decodedProduction = decodeURIComponent(production);
   661→    this.production = decodedProduction;
   662→
   663→    const debouncedMasonry = debounce(() => { this.masonryLoading(); }, 150);
   664→    this._resizeHandler = debouncedMasonry;
   665→    window.addEventListener("resize", this._resizeHandler);
   666→    this.masonryLoading();
   667→
   668→    // Initialize socket in mounted() so re-mount gets a fresh connection
   669→    this._socketInitPromise = initializeSocket();
   670→
   671→    // Wait for both socket init and auth before attaching listeners
   672→    Promise.all([this._socketInitPromise, this.auth()])
   673→      .then(([sock]) => {
   674→        // R36-FIX: Guard against post-unmount execution
   675→        if (!this._isMounted) return;
   676→        this._socket = sock;
   677→        if (this._socket) this._socket.on('connect_error', (err) => {
   678→          if (!this._isMounted) return;
   679→          console.error("Socket connection error:", err);
   680→          this.setStatusMessage("Connection error. Retrying...");
   681→        });
   682→        if (this._socket) this._socket.on('disconnect', (reason) => {
   683→          if (!this._isMounted) return;
   684→          console.warn("Socket disconnected:", reason);
   685→          this.setStatusMessage("Connection lost. Reconnecting...");
   686→        });
   687→        if (this._socket) this._socket.on('connect', () => {
   688→          if (!this._isMounted) return;
   689→          this.setStatusMessage("Reconnected");
   690→          this.hideStatusMessageAfterDelay();
   691→        });
   692→
   693→        return this.fetchKOT();
   694→      })
   695→      .then(() => {
   696→        if (!this._isMounted) return;
   697→        if (this.audio_alert === 1) {
   698→          this.showAudioAlertMessage = true;
   699→        }
   700→        this.socketHandler = (doc) => {
   701→          if (!this._isMounted) return;
   702→          try {
   703→            if (this.audio_alert === 1) {
   704→              this.playAlertSound(doc.audio_file);
   705→            }
   706→            // R36-FIX: Namespace localStorage key per production station to avoid cross-tab collision
   707→            let kottime = localStorage.getItem("kot_time_" + this.production);
   708→            if (doc.last_kot_time !== kottime) {
   709→              // Full refresh needed — skip intermediate mutations
   710→              this.fetchKOT().then(() => { this.masonryLoading(); }).catch((e) => { console.error("KOT fetch failed:", e); });
   711→            } else {
   712→              // R36-FIX: Guard against missing doc.kot to prevent TypeError crash
   713→              if (!doc.kot) {
   714→                this.fetchKOT().then(() => { this.masonryLoading(); }).catch((e) => { console.error("KOT fetch failed:", e); });
   715→                return;
   716→              }
   717→              // Incremental update — deduplicate to avoid duplicate cards
   718→              const existingIndex = this.kot.findIndex(k => k.name === doc.kot.name);
   719→              if (existingIndex !== -1) {
   720→                // R36-FIX: Preserve strikethrough state before Object.assign overwrites kot_items
   721→                const strikeMap = new Map(
   722→                  (this.kot[existingIndex].kot_items || []).map(i => [i.name, i.striked])
   723→                );
   724→                Object.assign(this.kot[existingIndex], { timecolor: 'text-black', timeRemaining: '— : —', ...doc.kot });
   725→                // Restore strikethrough state after Object.assign
   726→                if (this.kot[existingIndex].kot_items) {
   727→                  this.kot[existingIndex].kot_items.forEach(i => {
   728→                    if (strikeMap.has(i.name)) i.striked = strikeMap.get(i.name);
   729→                  });
   730→                }
   731→                // R37-FIX: Invalidate sorted cache for this KOT since items may have changed
   732→                this._sortedItemsCache.delete(doc.kot.name);
   733→              } else {
   734→                const newKot = { isRotated: false, showDiv: false, timecolor: 'text-black', timeRemaining: '— : —', ...doc.kot };
   735→                this.kot.unshift(newKot);
   736→              }
   737→              this.updateQtyColorTable();
   738→              this.updateTimeRemaining();
   739→              this.masonryLoading();
   740→            }
   741→            if (this._cancelTimeout) clearTimeout(this._cancelTimeout);
   742→            this._cancelTimeout = setTimeout(() => {
   743→              if (!this._isMounted) return;
   744→              if (doc.kot && doc.kot.type === "Cancelled") {
   745→                this.fetchKOT().then(() => {
   746→                  this.masonryLoading();
   747→                }).catch((e) => { console.error("KOT fetch failed:", e); });
   748→              }
   749→            }, 1500);
   750→            if (doc.kot) {
   751→              try {
   752→                localStorage.setItem("kot_time_" + this.production, doc.kot.time);
   753→              } catch (e) {
   754→                if (import.meta.env?.DEV) console.error('localStorage write failed:', e);
   755→              }
   756→            }
   757→          } catch (err) {
   758→            if (import.meta.env?.DEV) console.error("Socket handler error:", err);
   759→          }
   760→        };
   761→        if (this._socket) this._socket.on(this.kot_channel, this.socketHandler);
   762→      })
   763→      .catch((error) => {
   764→        console.error("Initialization or authentication error:", error);
   765→        if (this._isMounted) this.showModal = true;
   766→      });
   767→    this.timer = setInterval(this.updateTimeRemaining, 60000);
   768→  },
   769→  beforeUnmount() {
   770→    this._isMounted = false;
   771→    window.removeEventListener("online", this.handleOnline);
   772→    window.removeEventListener("offline", this.handleOffline);
   773→    document.removeEventListener("click", this.hideAudioAlertMessage);
   774→    window.removeEventListener("resize", this._resizeHandler);
   775→    if (this.socketHandler && this._socket) {
   776→      this._socket.off(this.kot_channel, this.socketHandler);
   777→    }
   778→    if (this._socket) {
   779→      this._socket.off('connect_error');
   780→      this._socket.off('disconnect');
   781→      this._socket.off('connect');
   782→      this._socket.disconnect();
   783→    }
   784→    this._socketInitPromise = null;
   785→    if (this._cancelTimeout) clearTimeout(this._cancelTimeout);
   786→    if (this._statusTimeout) clearTimeout(this._statusTimeout);
   787→    if (this.timer) clearInterval(this.timer);
   788→    if (this._alertAudio) { this._alertAudio.pause(); this._alertAudio = null; }
   789→  },
   790→  computed: {
   791→    sortedKotItems() {
   792→      // R37-FIX: Pure computed — no side effects on reactive kot objects.
   793→      // Uses instance-level _sortedItemsCache Map (non-reactive) instead of
   794→      // mutating kot._sortedItems / kot._sortKey which caused reactivity issues
   795→      // and stale cache after Object.assign overwrites kot_items in socket handler.
   796→      return (kot) => {
   797→        const items = kot.kot_items || [];
   798→        const key = `${kot.name}:${items.length}:${items.map(i => i.name).join(',')}`;
   799→        const cached = this._sortedItemsCache.get(kot.name);
   800→        if (cached && cached.key === key) {
   801→          return cached.items;
   802→        }
   803→        const sorted = [...items].sort((a, b) => (a.serve_priority || 0) - (b.serve_priority || 0));
   804→        this._sortedItemsCache.set(kot.name, { key, items: sorted });
   805→        return sorted;
   806→      };
   807→    },
   808→    visibleKots() {
   809→      return this.kot.filter(kot => !kot.showDiv && kot.production === this.production);
   810→    },
   811→  },
   812→};
