   125→// ── Dark mode aware tooltip style helper ─────────────────
   126→function useIsDark() {
   127→  const [isDark, setIsDark] = useState(false);
   128→  useEffect(() => {
   129→    const mq = window.matchMedia('(prefers-color-scheme: dark)');
   130→    setIsDark(mq.matches || document.documentElement.classList.contains('dark'));
   131→    const obs = new MutationObserver(() => {
   132→      setIsDark(document.documentElement.classList.contains('dark'));
   133→    });
   134→    obs.observe(document.documentElement, { attributes: true, attributeFilter: ['class'] });
   135→    return () => obs.disconnect();
   136→  }, []);
   137→  return isDark;
   138→}

   150→// ── Circular Progress Component ──────────────────────────
   151→function CircularTimer({ seconds, total }: { seconds: number; total: number }) {
   152→  const radius = 18;
   153→  const circumference = 2 * Math.PI * radius;
   154→  const progress = seconds / total;
   155→  const offset = circumference * (1 - progress);
   156→
   157→  return (
   158→    <div className="relative inline-flex items-center justify-center">
   159→      <svg width="44" height="44" className="-rotate-90">
   160→        <circle
   161→          cx="22"
   162→          cy="22"
   163→          r={radius}
   164→          fill="none"
   165→          stroke="currentColor"
   166→          strokeWidth="3"
   167→          className="text-muted-foreground/20"
   168→        />
   169→        <circle
   170→          cx="22"
   171→          cy="22"
   172→          r={radius}
   173→          fill="none"
   174→          stroke="currentColor"
   175→          strokeWidth="3"
   176→          strokeDasharray={circumference}
   177→          strokeDashoffset={offset}
   178→          strokeLinecap="round"
   179→          className="text-emerald-500 transition-all duration-1000"
   180→        />
   181→      </svg>
   182→      <span className="absolute text-[10px] font-bold tabular-nums">{seconds}s</span>
   183→    </div>
   184→  );
   185→}

   187→// ── Gauge Component ──────────────────────────────────────
   188→function WaitTimeGauge({ value, max, label, color }: { value: number; max: number; label: string; color: string }) {
   189→  const percentage = Math.min((value / max) * 100, 100);
   190→  const isOver = value > max * 0.8;
   191→
   192→  return (
   193→    <div className="space-y-2">
   194→      <div className="flex items-center justify-between">
   195→        <span className="text-xs text-muted-foreground">{label}</span>
   196→        <span className={`text-sm font-bold ${isOver ? 'text-red-500' : 'text-emerald-500'}`}>
   197→          {value} min
   198→        </span>
   199→      </div>
   200→      <div className="relative h-3 w-full rounded-full bg-muted overflow-hidden">
   201→        <div
   202→          className="absolute left-0 top-0 h-full rounded-full transition-all duration-700 ease-out"
   203→          style={{
   204→            width: `${percentage}%`,
   205→            backgroundColor: isOver ? '#ef4444' : color,
   206→          }}
   207→        />
   208→        {/* Target marker */}
   209→        <div
   210→          className="absolute top-0 h-full w-0.5 bg-foreground/50"
   211→          style={{ left: `${(max * 0.8 / max) * 100}%` }}
   212→        />
   213→      </div>
   214→    </div>
   215→  );
   216→}
