    21→def _get_order_invoice_doc(table=None, invoiceNo=None, order_type=None, is_payment=None):
    22→    """Internal: returns POS Invoice document object for server-side manipulation.
    23→    Used by sync_order and make_invoice which need the doc, not a dict."""
    24→    # Verify user has access to the table's room (BE-R36-017)
    25→    if table:
    26→        from ury.ury_pos.api import _get_user_branch_rooms
    27→        table_room = frappe.db.get_value("URY Table", table, "restaurant_room")
    28→        if table_room:
    29→            user_rooms = [r.room for r in _get_user_branch_rooms() if r.room]
    30→            if user_rooms and table_room not in user_rooms:
    31→                frappe.throw(_("You do not have access to this table's room"), frappe.PermissionError)
    32→
    33→    if table:
    34→        # Lock the table row to prevent concurrent invoice creation (BE-R36-001)
    35→        frappe.db.sql(
    36→            "SELECT name FROM `tabURY Table` WHERE name = %s FOR UPDATE",
    37→            (table,), as_dict=True
    38→        )
    39→        
    40→        if is_payment == "Payments":
    41→            invoice_name = frappe.get_value(
    42→                "POS Invoice", dict(restaurant_table=table, docstatus=0, name=invoiceNo)
    43→            )
    44→            
    45→        else:
    46→            if invoiceNo:
    47→                invoice_name = frappe.get_value(
    48→                    "POS Invoice",
    49→                    dict(restaurant_table=table, docstatus=0, name=invoiceNo),
    50→                )
    51→               
    52→            else:
    53→                invoice_name = frappe.get_value(
    54→                    "POS Invoice",
    55→                    dict(restaurant_table=table, docstatus=0, invoice_printed=0),
    56→                )
    57→                
    58→        branch, menu_name, restaurant = get_restaurant_and_menu_name(table)
    59→
    60→        if invoice_name:
    61→            invoice = frappe.get_doc("POS Invoice", invoice_name)
    62→
    63→        else:
    64→            invoice = frappe.new_doc("POS Invoice")
    65→
    66→            invoice.naming_series = frappe.db.get_value(
    67→                "URY Restaurant", restaurant, "invoice_series_prefix"
    68→            )
    69→
    70→            invoice.is_pos = 1
    71→            invoice.update_stock = 1
    72→            invoice.restaurant = restaurant
    73→            invoice.branch = branch
    74→
    75→            is_take_away = frappe.db.get_value("URY Table", table, "is_take_away")
    76→            if is_take_away == 1:
    77→                invoice.order_type = "Take Away"
    78→            else:
    79→                invoice.order_type= "Dine In"
    80→
    81→        invoice.taxes_and_charges = frappe.db.get_value(
    82→            "URY Restaurant", restaurant, "default_tax_template"
    83→        )
    84→
    85→        invoice.selling_price_list = frappe.db.get_value(
    86→            "Price List", dict(restaurant_menu=menu_name, enabled=1)
    87→        )
    88→
    89→    else:
    90→
    91→        invoice_name = frappe.get_value(
    92→                "POS Invoice", dict(docstatus=0, name=invoiceNo)
    93→            )
    94→            
    95→        if invoice_name:
    96→            invoice = frappe.get_doc("POS Invoice", invoice_name)
    97→            # R44-FIX: Validate that the loaded invoice belongs to the user's branch
    98→            user_branch = _get_user_branch()
    99→            if invoice.branch and invoice.branch != user_branch:
   100→                frappe.throw(_("You do not have access to invoices from another branch"), frappe.PermissionError)
   101→            
   102→
   103→        else:
   104→            invoice = frappe.new_doc("POS Invoice")
   105→            invoice.is_pos = 1
   106→            invoice.update_stock = 1
   107→        
   108→        branch = _get_user_branch()
   109→        restaurant = frappe.db.get_value("URY Restaurant", {"branch": branch}, "name")
   110→
   111→        menu=get_menu_name(order_type)
   112→
   113→        if (order_type == "Aggregators" and frappe.db.get_value("Branch", branch, "custom_no_taxes") == 0) or order_type != "Aggregators":
   114→            invoice.taxes_and_charges = frappe.db.get_value("URY Restaurant", restaurant, "default_tax_template")
   115→        
   116→        invoice.selling_price_list = frappe.db.get_value(
   117→            "Price List", dict(restaurant_menu=menu, enabled=1)
   118→        )
   119→        # R43-FIX: Set branch on non-table invoices to prevent branchless invoices
   120→        invoice.branch = branch
   121→
   122→    return invoice

   168→@frappe.whitelist()
   169→def sync_order(
   170→    items,
   171→    cashier,
   172→    owner,
   173→    mode_of_payment,
   174→    customer,
   175→    no_of_pax,
   176→    last_invoice,
   177→    waiter,
   178→    pos_profile,
   179→    last_modified_time=None,
   180→    table=None,
   181→    invoice=None,
   182→    comments=None,
   183→    order_type=None,
   184→    aggregator_id=None,
   185→    room=None
   186→):
   187→    frappe.only_for("Restaurant Manager", "Restaurant User", "Cashier")
   188→    # Validate pos_profile belongs to user's branch (BE-R36-002)
   189→    pos_profile_branch = frappe.db.get_value("POS Profile", pos_profile, "branch")
   190→    user_branch = _get_user_branch()
   191→    if pos_profile_branch != user_branch:
   192→        frappe.throw(_("POS Profile does not belong to your branch"), frappe.PermissionError)
   193→    user_role = frappe.get_roles()
   194→    billing_roles = frappe.get_all(
   195→        "POS Profile Role",
   196→        filters={"parent": pos_profile},
   197→        fields=["role"],
   198→        pluck="role",
   199→    )
   200→    billing_user = bool(set(user_role).intersection(billing_roles))
   201→
   202→    # Check if the last invoice was already billed
   203→    if (
   204→        last_invoice
   205→        and frappe.db.get_value("POS Invoice", last_invoice, "invoice_printed") == 1
   206→        and (not billing_user)
   207→    ):
   208→        frappe.msgprint(
   209→            title=_("Invoice Already Billed"),
   210→            indicator="red",
   211→            msg=_("This order has already been billed. Please reload the page."),
   212→        )
   213→        return {"status": "Failure"}
   214→
   215→    # R41-FIX: Use _get_order_invoice_doc to get the document object, not a dict.
   216→    # The public get_order_invoice returns a dict for frontend safety (BE-R36-003),
   217→    # but sync_order needs the document to call .save() etc.
   218→    invoice = _get_order_invoice_doc(table, invoice, order_type)
   219→
   220→    # R41-FIX: Validate last_invoice belongs to user's branch
   221→    if last_invoice:
   222→        last_inv_branch = frappe.db.get_value("POS Invoice", last_invoice, "branch")
   223→        if last_inv_branch and last_inv_branch != user_branch:
   224→            frappe.throw(_("Invoice does not belong to your branch"), frappe.PermissionError)
   225→
   226→    if last_invoice and last_modified_time:
   227→        lastModifiedTime = invoice.modified
   228→        if isinstance(last_modified_time, str):
   229→            try:
   230→                last_modified_time = datetime.strptime(
   231→                    last_modified_time, "%Y-%m-%d %H:%M:%S.%f"
   232→                )
   233→            except ValueError:
   234→                last_modified_time = datetime.strptime(
   235→                    last_modified_time, "%Y-%m-%d %H:%M:%S"
   236→                )
   237→        if isinstance(lastModifiedTime, str):
   238→            try:
   239→                lastModifiedTime = datetime.strptime(
   240→                    lastModifiedTime, "%Y-%m-%d %H:%M:%S.%f"
   241→                )
   242→            except ValueError:
   243→                lastModifiedTime = datetime.strptime(
   244→                    lastModifiedTime, "%Y-%m-%d %H:%M:%S"
   245→                )
   246→        if lastModifiedTime != last_modified_time:
   247→            frappe.msgprint(
   248→                title=_("Order has been modified"),
   249→                indicator="red",
   250→                msg=_("This order has been modified. Please reload the page to retrieve the latest edits."),
   251→            )
   252→            return {"status": "Failure"}
   253→    else:
   254→        if invoice.name and invoice.invoice_printed == 0 and not billing_user:
   255→            frappe.msgprint(
   256→                title=_("Table Occupied"),
   257→                indicator="red",
   258→                msg=_("{0} is already occupied. Please refresh the page.").format(table),
   259→            )
   260→            return {"status": "Failure"}
   261→
   262→    if not customer:
   263→        frappe.throw(_("Please enter valid customer details"))
   264→    else:
   265→        invoice.customer = customer
   266→
   267→    if order_type:
   268→        invoice.order_type = order_type
   269→
   270→    invoice.mobile_number = frappe.db.get_value("Customer", customer, "mobile_number")
   271→    if comments:
   272→        invoice.custom_comments = comments
   273→    invoice.no_of_pax = no_of_pax
   274→    invoice.pos_profile = pos_profile
   275→    # H-01: Validate cashier and waiter are active users
   276→    for role_user in [("Cashier", cashier), ("Waiter", waiter)]:
   277→        if role_user[1] and not frappe.db.exists("User", {"name": role_user[1], "enabled": 1}):
   278→            frappe.throw(_("Invalid {0}: {1}").format(role_user[0], role_user[1]))
   279→
   280→    invoice.cashier = cashier
   281→    invoice.waiter = waiter
   282→    invoice.custom_aggregator_id = aggregator_id
   283→    invoice.custom_restaurant_room =room
   284→    invoice.restaurant_table = table
   285→    
   286→    if order_type == "Aggregators":
   287→        price_list = frappe.db.get_value("Aggregator Settings",{"customer": customer, "parent": invoice.branch, "parenttype": "Branch"},"price_list",)
   288→        
   289→        if not price_list:
   290→            frappe.throw(_("Price list for customer {0} in branch {1} not found in Aggregator Settings.").format(customer, invoice.branch))
   291→    else:
   292→        price_list = invoice.selling_price_list
   293→
   294→    # dummy payment
   295→    # C-02: Validate mode_of_payment against POS Profile's allowed payment methods
   296→    allowed_mops = frappe.get_all("POS Profile Payment", filters={"parent": pos_profile}, pluck="mode_of_payment")
   297→    if mode_of_payment not in allowed_mops:
   298→        frappe.throw(_("Invalid mode of payment: {0}").format(mode_of_payment))
   299→
   300→    if invoice.invoice_created == 0:
   301→        invoice.append(
   302→            "payments",
   303→            dict(mode_of_payment=mode_of_payment, amount=invoice.grand_total),
   304→        )
   305→        invoice.invoice_created = 1
   306→
   307→    past_item = []
   308→    for item in invoice.items:
   309→        previous_item = {
   310→            "item_code": item.item_code,
   311→            "item_name": item.item_name,
   312→            "qty": item.qty,
   313→            "comments": "",
   314→        }
   315→        past_item.append(previous_item)
   316→        
   317→
   318→    # Conditional checking for 'items' type:
   319→    # - 'ury': JSON passed, hence using isinstance
   320→    # - 'ury_pos': Already formatted list, hence using else
   321→    if isinstance(items, str):
   322→        try:
   323→            items = json.loads(items)
   324→        except json.JSONDecodeError:
   325→            frappe.throw(_("Invalid items data"), frappe.ValidationError)
   326→    invoice.items = []
   327→    
   328→    menu = frappe.db.get_value("URY Menu", {"branch": invoice.branch}, "name")
   329→
   330→    # Batch-fetch courses, prices, and cost_center to avoid N+1 queries
   331→    item_codes = [d.get("item") for d in items]
   332→    item_codes_unique = list(set(item_codes))
   333→
   334→    courses = {}
   335→    if item_codes_unique:
   336→        course_rows = frappe.db.get_all(
   337→            "URY Menu Item",
   338→            filters={"item": ("in", item_codes_unique), "parent": menu},
   339→            fields=["item", "course"],
   340→        )
   341→        courses = {r.item: r.course for r in course_rows}
   342→
   343→    prices = {}
   344→    if item_codes_unique:
   345→        price_rows = frappe.db.get_all(
   346→            "Item Price",
   347→            filters={"item_code": ("in", item_codes_unique), "price_list": price_list},
   348→            fields=["item_code", "price_list_rate"],
   349→        )
   350→        prices = {r.item_code: r.price_list_rate for r in price_rows}
   351→
   352→    cost_center = frappe.db.get_value("POS Profile", pos_profile, "cost_center")
   353→
   354→    for d in items:
   355→        # R47-FIX: Validate item qty is positive (create_order_items in kot_generate
   356→        # validates for KOT creation, but sync_order was missing the same check)
   357→        qty = flt(d.get("qty"))
   358→        if qty <= 0:
   359→            frappe.throw(_("Item '{0}' has invalid quantity ({1}). Quantity must be positive.").format(d.get("item_name", d.get("item")), qty))
   360→        course = courses.get(d.get("item"))
   361→        rate = prices.get(d.get("item"))
   362→        if not rate:
   363→            frappe.throw(_("No item price found for Item: {0} in Price List: {1}. Please check the price list settings.").format(d.get("item"), price_list))
   364→
   365→        invoice.append(
   366→            "items",
   367→            dict(
   368→                item_code=d.get("item"),
   369→                item_name=d.get("item_name"),
   370→                qty=qty,
   371→                **({"custom_course": course} if course else {}),
   372→                comment=d.get("comment"),
   373→                rate=rate,
   374→                price_list_rate=rate,
   375→                base_price_list_rate=rate,
   376→                cost_center=cost_center,
   377→            ),
   378→        )
   379→
   380→    try:
   381→        invoice.save()
   382→    except Exception as e:
   383→        frappe.log_error(frappe.get_traceback(), "Invoice Save Error")
   384→        frappe.throw(_("An error occurred. Please check the error log."))
   385→
   386→
   387→    try:
   388→        kot_execute(invoice.name, customer, table, items, past_item, comments)
   389→
   390→    except Exception as e:
   391→        # Log the KOT error but also notify the user — kitchen won't see this order
   392→        error_msg = f"KOT Creation Failed: {str(e)}"
   393→        frappe.log_error(f"{error_msg}\n{frappe.get_traceback()}", "KOT Error")
   394→        frappe.msgprint(
   395→            title=_("KOT Creation Failed"),
   396→            indicator="orange",
   397→            msg=_("The order was saved but the KOT was not sent to the kitchen. Please check the error log or retry."),
   398→        )
   399→
   400→    # table status
   401→    if invoice.invoice_printed == 0:
   402→        # R47-FIX: Use update_modified=False for auxiliary status updates
   403→        frappe.db.set_value(
   404→            "URY Table", table, {"occupied": 1, "latest_invoice_time": invoice.creation},
   405→            update_modified=False
   406→        )
   407→
   408→    invoice.db_set("owner", frappe.session.user)
   409→    return {
   410→        "name": invoice.name,
   411→        "customer": invoice.customer,
   412→        "grand_total": invoice.grand_total,
   413→        "modified": invoice.modified,
   414→        "invoice_printed": invoice.invoice_printed,
   415→        "restaurant_table": invoice.restaurant_table,
   416→    }

   439→@frappe.whitelist()
   440→def get_restaurant_and_menu_name(table):
   441→    frappe.only_for("Restaurant Manager", "Restaurant User", "Cashier")
   442→    if not table:
   443→        frappe.throw(_("Please select a table"))
   444→
   445→    result = frappe.get_value(
   446→        "URY Table",
   447→        table,
   448→        ["restaurant", "branch", "restaurant_room"],
   449→    )
   450→    if not result:
   451→        frappe.throw(_("URY Table {0} not found").format(table))
   452→    restaurant, branch, room = result
   453→    room_wise_menu = frappe.db.get_value(
   454→        "URY Restaurant",
   455→        restaurant,
   456→        "room_wise_menu",
   457→    )
   458→
   459→    if not room_wise_menu:
   460→        menu = frappe.db.get_value("URY Restaurant", restaurant, "active_menu")
   461→    else:
   462→        menu = frappe.db.get_value(
   463→            "Menu for Room",
   464→            {"parent": restaurant, "room": room},
   465→            "menu",
   466→        )
   467→
   468→    if not menu:
   469→        frappe.throw(
   470→            _("Please set an active menu for Restaurant {0}").format(restaurant)
   471→        )
   472→
   473→    return branch, menu, restaurant

   501→@frappe.whitelist()
   502→def pos_opening_check():
   503→    frappe.only_for("Restaurant Manager", "Restaurant User", "Cashier")
   504→    user = frappe.session.user
   505→    # Handle the administrator case differently
   506→    if user == "Administrator":
   507→        return {
   508→            "opening_exists": False,  # Assuming no POS opening entry is needed for Administrator
   509→            "cashier": None,
   510→            "pos_profile": None,
   511→        }
   512→    
   513→    details = getBranchRoom()
   514→    if not details:
   515→        frappe.throw(_("No room/branch configuration found for the current user."))
   516→    room = details[0].get('name')
   517→    branch = details[0].get('branch')
   518→    
   519→    pos_opening_list = frappe.db.sql("""
   520→        SELECT DISTINCT `tabPOS Opening Entry`.name 
   521→        FROM `tabPOS Opening Entry`
   522→        INNER JOIN `tabMultiple Rooms` 
   523→        ON `tabMultiple Rooms`.parent = `tabPOS Opening Entry`.name
   524→        WHERE `tabPOS Opening Entry`.branch = %s
   525→        AND `tabPOS Opening Entry`.status = 'Open'
   526→        AND `tabPOS Opening Entry`.docstatus = 1
   527→        AND `tabMultiple Rooms`.room = %s
   528→    """, (branch, room), as_dict=True)
   529→    
   530→    
   531→    result = {

   738→def make_invoice(customer, payments, cashier, pos_profile, additionalDiscount=None, table=None, invoice=None):
   739→    frappe.only_for("Restaurant Manager", "Restaurant User", "Cashier")
   740→
   741→    # R39-FIX: Validate POS Profile belongs to user's branch
   742→    pos_profile_branch = frappe.db.get_value("POS Profile", pos_profile, "branch")
   743→    user_branch = _get_user_branch()
   744→    if pos_profile_branch != user_branch:
   745→        frappe.throw(_("POS Profile does not belong to your branch"), frappe.PermissionError)
   746→
   747→    # Validate additional discount
   748→    if additionalDiscount is not None:
   749→        additionalDiscount = flt(additionalDiscount)
   750→        if additionalDiscount < 0 or additionalDiscount > 100:
   751→            frappe.throw(_("Additional discount must be between 0 and 100"))
   752→
   753→    # Check if discounts are enabled for this POS Profile (BE-R36-006)
   754→    if additionalDiscount:
   755→        enable_discount = frappe.db.get_value("POS Profile", pos_profile, "custom_enable_discount")
   756→        if not enable_discount:
   757→            frappe.throw(_("Discounts are not enabled for this POS Profile"), frappe.PermissionError)
   758→
   759→    # Validate payments
   760→    if isinstance(payments, str):
   761→        try:
   762→            payments = json.loads(payments)
   763→        except json.JSONDecodeError:
   764→            frappe.throw(_("Invalid payments data"), frappe.ValidationError)
   765→    if not payments:
   766→        frappe.throw(_("At least one payment is required"))
   767→    # H-03: Validate each payment's mode_of_payment against POS Profile's configured payment methods
   768→    allowed_mops = frappe.get_all("POS Profile Payment", filters={"parent": pos_profile}, pluck="mode_of_payment")
   769→    for p in payments:
   770→        amount = flt(p.get("amount", 0))
   771→        if amount < 0:
   772→            frappe.throw(_("Payment amount cannot be negative"))
   773→        mop = p.get("mode_of_payment")
   774→        if mop not in allowed_mops:
   775→            frappe.throw(_("Invalid mode of payment: {0}").format(mop))
   776→
   777→    # R44-FIX: At least one of invoice or table must be provided — otherwise
   778→    # we would create an orphaned invoice with no way to reference it.
   779→    if not invoice and not table:
   780→        frappe.throw(_("Invoice or table is required to make a payment"), frappe.ValidationError)
   781→
   782→    # R43-FIX: Guard against None invoice — get_value with None returns None/error
   783→    order_type = None
   784→    if invoice:
   785→        order_type = frappe.get_value("POS Invoice", invoice, "order_type")
   786→    # R41-FIX: Use _get_order_invoice_doc to get the document object, not a dict.
   787→    # make_invoice needs to call .save() and .submit() on the document.
   788→    invoice = _get_order_invoice_doc(table, invoice, order_type, "Payments")
   789→
   790→    if table:
   791→        _, _, restaurant = get_restaurant_and_menu_name(table)
   792→        invoice.restaurant = restaurant
   793→    elif not invoice.restaurant:
   794→        restaurant = frappe.db.get_value("URY Restaurant", {"branch": invoice.branch}, "name")
   795→        if restaurant:
   796→            invoice.restaurant = restaurant
   797→
   798→    invoice.customer = customer
   799→    invoice.pos_profile = pos_profile
   800→    invoice.additional_discount_percentage=additionalDiscount
   801→    invoice.calculate_taxes_and_totals()
   802→
   803→    # H-07: Validate that payment amounts match the grand total (prevent negative invoice from discount)
   804→    total_payments = flt(sum(flt(p.get("amount", 0)) for p in payments))
   805→    if flt(total_payments - invoice.grand_total, 2) < 0:
   806→        frappe.throw(_("Total payment amount ({0}) is less than grand total ({1}). Discount may have created a negative balance.").format(total_payments, invoice.grand_total))
   807→
   808→    invoice.payments = []
   809→
   810→    for d in payments:
   811→        invoice.append(
   812→            "payments", dict(mode_of_payment=d["mode_of_payment"], amount=d["amount"])
   813→        )
   814→
   815→    # invoice.owner = owner
   816→    invoice.save()
   817→    try:
   818→        invoice.submit()
   819→    except Exception as e:
   820→        frappe.log_error(frappe.get_traceback(), "Invoice Submit Error")
   821→        frappe.throw(_("An error occurred. Please check the error log."))
