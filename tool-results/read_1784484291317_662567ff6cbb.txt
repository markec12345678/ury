   231→   231→function debounce(fn, delay) {
   232→   232→    let timer = null;
   233→   233→    const debounced = function(...args) {
   234→   234→        clearTimeout(timer);
   235→   235→        timer = setTimeout(() => fn.apply(this, args), delay);
   236→   236→    };
   237→   237→    debounced.cancel = () => { clearTimeout(timer); timer = null; };
   238→   238→    return debounced;

   660→   660→      const hoursRemaining = Math.floor(timeDifference / 3600000);
   661→   661→      const minutesRemaining = Math.floor((timeDifference % 3600000) / 60000);

   668→   668→    fetchKOTWithRetry(maxRetries = 3, initialDelay = 2000) {
   669→   669→      // R41-FIX: Clear any pending retry timer before starting a new retry chain.
   670→   670→      // Without this, calling fetchKOTWithRetry() while a previous retry timer is

   676→   676→          if (retriesLeft <= 0 || !this._isMounted) throw err;

   682→   682→            }, delay);
   683→   683→          });
   684→   684→        });
   685→   685→      };
