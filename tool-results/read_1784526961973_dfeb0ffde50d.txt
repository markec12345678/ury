   317→   317→// ---- Period Helpers ----
   318→   318→
   319→   319→function getPreviousPeriodDates(
   320→   320→  period: ReportPeriod,
   321→   321→  customFromDate?: string | null,
   322→   322→  customToDate?: string | null
   323→   323→): { prevFrom: string; prevTo: string } {
   324→   324→  const today = new Date();
   325→   325→
   326→   326→  // If custom date range, calculate the previous equivalent period
   327→   327→  if (customFromDate && customToDate) {
   328→   328→    const from = new Date(customFromDate);
   329→   329→    const to = new Date(customToDate);
   330→   330→    const diffMs = to.getTime() - from.getTime();
   331→   331→    const prevTo = new Date(from.getTime() - 1); // Day before from
   332→   332→    const prevFrom = new Date(prevTo.getTime() - diffMs);
   333→   333→    return {
   334→   334→      prevFrom: prevFrom.toISOString().split('T')[0],
   335→   335→      prevTo: prevTo.toISOString().split('T')[0],
   336→   336→    };
   337→   337→  }
   338→   338→
   339→   339→  switch (period) {
   340→   340→    case 'daily':
   341→   341→    case 'yesterday': {
   342→   342→      // Previous day
   343→   343→      const prev = new Date(today);
   344→   344→      prev.setDate(prev.getDate() - 1);
   345→   345→      const dateStr = prev.toISOString().split('T')[0];
   346→   346→      return { prevFrom: dateStr, prevTo: dateStr };
   347→   347→    }
   348→   348→    case 'weekly': {
   349→   349→      // Previous week (same day range)
   350→   350→      const prevWeekStart = new Date(today);
   351→   351→      prevWeekStart.setDate(prevWeekStart.getDate() - 7 - prevWeekStart.getDay());
   352→   352→      const prevWeekEnd = new Date(prevWeekStart);
   353→   353→      prevWeekEnd.setDate(prevWeekEnd.getDate() + 6);
   354→   354→      return {
   355→   355→        prevFrom: prevWeekStart.toISOString().split('T')[0],
   356→   356→        prevTo: prevWeekEnd.toISOString().split('T')[0],
   357→   357→      };
   358→   358→    }
   359→   359→    case 'monthly': {
   360→   360→      // Previous month
   361→   361→      const prevMonth = new Date(today.getFullYear(), today.getMonth() - 1, 1);
   362→   362→      const prevMonthEnd = new Date(today.getFullYear(), today.getMonth(), 0);
   363→   363→      return {
   364→   364→        prevFrom: prevMonth.toISOString().split('T')[0],
   365→   365→        prevTo: prevMonthEnd.toISOString().split('T')[0],
   366→   366→      };
   367→   367→    }
   368→   368→    case 'last_month': {
   369→   369→      // Two months ago
   370→   370→      const twoMonthsAgo = new Date(today.getFullYear(), today.getMonth() - 2, 1);
   371→   371→      const twoMonthsAgoEnd = new Date(today.getFullYear(), today.getMonth() - 1, 0);
   372→   372→      return {
   373→   373→        prevFrom: twoMonthsAgo.toISOString().split('T')[0],
   374→   374→        prevTo: twoMonthsAgoEnd.toISOString().split('T')[0],
   375→   375→      };
   376→   376→    }
   377→   377→    case 'last_7_days': {
   378→   378→      const prevEnd = new Date(today);
   379→   379→      prevEnd.setDate(prevEnd.getDate() - 8);
   380→   380→      const prevStart = new Date(prevEnd);
   381→   381→      prevStart.setDate(prevStart.getDate() - 6);
   382→   382→      return {
   383→   383→        prevFrom: prevStart.toISOString().split('T')[0],
   384→   384→        prevTo: prevEnd.toISOString().split('T')[0],
   385→   385→      };
   386→   386→    }
   387→   387→    case 'last_30_days': {
   388→   388→      const prevEnd = new Date(today);
   389→   389→      prevEnd.setDate(prevEnd.getDate() - 31);
   390→   390→      const prevStart = new Date(prevEnd);
   391→   391→      prevStart.setDate(prevStart.getDate() - 29);
   392→   392→      return {
   393→   393→        prevFrom: prevStart.toISOString().split('T')[0],
   394→   394→        prevTo: prevEnd.toISOString().split('T')[0],
   395→   395→      };
   396→   396→    }
   397→   397→    default: {
   398→   398→      const prev = new Date(today);
   399→   399→      prev.setDate(prev.getDate() - 1);
   400→   400→      const dateStr = prev.toISOString().split('T')[0];
   401→   401→      return { prevFrom: dateStr, prevTo: dateStr };
   402→   402→    }
   403→   403→  }
   404→   404→}

   408→   408→function escapeCsvField(field: string | number): string {
   409→   409→  const str = String(field);
   410→   410→  if (str.includes(',') || str.includes('"') || str.includes('\n')) {
   411→   411→    return `"${str.replace(/"/g, '""')}"`;
   412→   412→  }
   413→   413→  return str;
   414→   414→}
   415→   415→
   416→   416→function csvRow(...fields: (string | number)[]): string {
   417→   417→  return fields.map(escapeCsvField).join(',');
   418→   418→}

   420→   420→function downloadCsv(content: string, filename: string): void {
   421→   421→  const BOM = '\uFEFF'; // UTF-8 BOM for Excel compatibility
   422→   422→  const blob = new Blob([BOM + content], { type: 'text/csv;charset=utf-8;' });
   423→   423→  const url = URL.createObjectURL(blob);
   424→   424→  const link = document.createElement('a');
   425→   425→  link.setAttribute('href', url);
   426→   426→  link.setAttribute('download', filename);
   427→   427→  document.body.appendChild(link);
   428→   428→  link.click();
   429→   429→  document.body.removeChild(link);
   430→   430→  URL.revokeObjectURL(url);
   431→   431→}

   433→   433→function generateSalesCsv(report: SalesReport): string {
   434→   434→  const lines: string[] = [];
   435→   435→
   436→   436→  // Header
   437→   437→  lines.push(csvRow('URY Restaurant - Sales Report'));
   438→   438→  lines.push(csvRow('Period', `${report.from_date} to ${report.to_date}`));
   439→   439→  lines.push(csvRow('Generated', new Date().toLocaleString()));
   440→   440→  lines.push('');
   441→   441→
   442→   442→  // Summary
   443→   443→  lines.push(csvRow('--- Summary ---'));
   444→   444→  lines.push(csvRow('Metric', 'Value'));
   445→   445→  lines.push(csvRow('Total Revenue', report.summary.total_revenue.toFixed(2)));
   446→   446→  lines.push(csvRow('Total Orders', report.summary.total_orders));
   447→   447→  lines.push(csvRow('Avg Order Value', report.summary.avg_order_value.toFixed(2)));
   448→   448→  lines.push(csvRow('Net Revenue', report.summary.net_revenue.toFixed(2)));
   449→   449→  lines.push(csvRow('Total Tax', report.summary.total_tax.toFixed(2)));
   450→   450→  lines.push(csvRow('Unique Customers', report.summary.unique_customers));
   451→   451→  lines.push('');
   452→   452→
   453→   453→  // Item Sales
   454→   454→  if (report.item_sales.length > 0) {
   455→   455→    lines.push(csvRow('--- Item-wise Sales ---'));
   456→   456→    lines.push(csvRow('Item', 'Qty', 'Avg Rate', 'Total Amount'));
   457→   457→    report.item_sales.forEach((item) => {
   458→   458→      lines.push(csvRow(item.item_name, item.total_qty, item.avg_rate.toFixed(2), item.total_amount.toFixed(2)));
   459→   459→    });
   460→   460→    lines.push('');
   461→   461→  }
   462→   462→
   463→   463→  // Order Type Sales
   464→   464→  if (report.order_type_sales.length > 0) {
   465→   465→    lines.push(csvRow('--- Order Type Breakdown ---'));
   466→   466→    lines.push(csvRow('Order Type', 'Orders', 'Revenue'));
   467→   467→    report.order_type_sales.forEach((ot) => {
   468→   468→      lines.push(csvRow(ot.order_type, ot.order_count, ot.revenue.toFixed(2)));
   469→   469→    });
   470→   470→    lines.push('');
   471→   471→  }
   472→   472→
   473→   473→  // Cancelled Orders
   474→   474→  if (report.cancelled_orders.count > 0) {
   475→   475→    lines.push(csvRow('--- Cancelled Orders ---'));
   476→   476→    lines.push(csvRow('Count', 'Amount'));
   477→   477→    lines.push(csvRow(report.cancelled_orders.count, report.cancelled_orders.amount.toFixed(2)));
