   567→    calculateTimeRemaining(targetTime) {
   568→      if (!targetTime || !targetTime.includes(":")) return '— : —';
   569→      const currentTime = new Date();

   572→        currentTime.getFullYear(),
   573→        currentTime.getMonth(),
   574→        currentTime.getDate(),
   575→        targetHours,
   576→        targetMinutes,
   577→        targetSeconds
   578→      );
   579→      // Handle cross-midnight KOTs: if target is in the future, it was created yesterday
   580→      if (targetDate > currentTime) {
   581→        targetDate = new Date(targetDate.getTime() - 86400000);
   582→      }
   583→
   584→      const timeDifference = Math.max(0, currentTime - targetDate);
   585→      const hoursRemaining = Math.floor(timeDifference / 3600000);
   586→      const minutesRemaining = Math.floor((timeDifference % 3600000) / 60000);
   587→
   588→      return `${hoursRemaining} : ${String(minutesRemaining).padStart(2, '0')}`;
   589→    },
   590→    fetchkotwithmasonry() {
   591→      return this.fetchKOT().then(() => {
   592→        this.masonryLoading();
   593→      }).catch((e) => { console.error("KOT fetch failed:", e); });
   594→    },
   595→    redirectToLogin() {
   596→      // R38-FIX: Use Vue Router instead of window.location.href to avoid full page reload
   597→      // and preserve app state. The previous approach caused a hard navigation that
   598→      // lost all Vue state and required a full re-initialization.
   599→      this.$router.push({ name: 'Login', query: { route: this.$route.path } });
   600→    },
   601→    masonryLoading(forceRecreate = false) {
   602→      if (this.masonry && !forceRecreate) {
   603→        this.$nextTick(() => {
   604→          if (this.masonry) {
   605→            this.masonry.reloadItems?.();
   606→            this.masonry.layout();
   607→          }
   608→        });
   609→        return;
   610→      }
   611→      if (this.masonry) {
   612→        this.masonry.destroy();
   613→        this.masonry = null;
   614→      }
   615→      this.$nextTick(() => {
   616→        if (!this.$el) return;
   617→        const grid = this.$el.querySelector(".grid");
   618→        if (!grid) return;
   619→        this.masonry = markRaw(new Masonry(grid, {
   620→          itemSelector: ".masonry-item",
   621→          gutter: 28,
   622→        }));
   623→        this.masonry.layout();
   624→      });
   625→    },
   626→    hideAudioAlertMessage() {
   627→      this.showAudioAlertMessage = false;
   628→    },
   629→    handleOnline() {
   630→      // R37-FIX: Guard against post-unmount execution
   631→      if (!this._isMounted) return;
   632→      this.isOnline = true;
   633→      this.setStatusMessage("You are online");
   634→      this.hideStatusMessageAfterDelay();
   635→      this.fetchKOT().then(() => {
   636→        if (!this._isMounted) return;
   637→        this.masonryLoading();
   638→      }).catch((e) => { console.error("KOT fetch failed:", e); });
   639→    },
   640→    handleOffline() {
   641→      // R37-FIX: Guard against post-unmount execution
   642→      if (!this._isMounted) return;
   643→      this.isOnline = false;
   644→      this.setStatusMessage("You are Offline");
   645→    },
   646→    setStatusMessage(message) {
   647→      this.statusMessage = message;
   648→    },
   649→    hideStatusMessageAfterDelay() {
   650→      if (this._statusTimeout) clearTimeout(this._statusTimeout);
   651→      this._statusTimeout = setTimeout(() => {
   652→        // R37-FIX: Guard against mutating unmounted component
   653→        if (!this._isMounted) return;
   654→        this.statusMessage = "";
   655→      }, 3000);
   656→    },
   657→    handleTransitionEnd() {
   658→      if (!this.isOnline) {
   659→        // Reset the status message after transition end
   660→        this.setStatusMessage("");
   661→      }
   662→    },
   663→  },
   664→  created() {
   665→    // R36-FIX: Initialize as non-reactive instance properties (no Proxy overhead)
   666→    this._isMounted = false;
   667→    this._fetchInProgress = null;
   668→    this.notifiedKots = new Set();
   669→    // API client as non-reactive instance property (avoids Proxy overhead)
   670→    this.call = markRaw(frappe.call());
   671→    // R37-FIX: socketHandler as non-reactive to avoid unnecessary Proxy overhead
   672→    this.socketHandler = null;
   673→    // R37-FIX: sortedItems cache map — avoids mutating reactive kot objects in computed
   674→    this._sortedItemsCache = new Map();
   675→  },
   676→  mounted() {
   677→    this._isMounted = true;
   678→    window.addEventListener("online", this.handleOnline);
   679→    window.addEventListener("offline", this.handleOffline);
   680→    document.addEventListener("click", this.hideAudioAlertMessage);
   681→    // R36-FIX: Use route params instead of fragile URL parsing
   682→    const production = this.$route?.params?.production || '';
   683→    const decodedProduction = decodeURIComponent(production);
   684→    this.production = decodedProduction;
   685→
   686→    const debouncedMasonry = debounce(() => { this.masonryLoading(); }, 150);
   687→    this._resizeHandler = debouncedMasonry;
   688→    window.addEventListener("resize", this._resizeHandler);
   689→    this.masonryLoading();
   690→
   691→    // Initialize socket in mounted() so re-mount gets a fresh connection
   692→    this._socketInitPromise = initializeSocket();
   693→
   694→    // Wait for both socket init and auth before attaching listeners
   695→    Promise.all([this._socketInitPromise, this.auth()])
   696→      .then(([sock]) => {
   697→        // R36-FIX: Guard against post-unmount execution
   698→        if (!this._isMounted) return;
   699→        this._socket = sock;
   700→        if (this._socket) this._socket.on('connect_error', (err) => {
   701→          if (!this._isMounted) return;
   702→          console.error("Socket connection error:", err);
   703→          this.setStatusMessage("Connection error. Retrying...");
   704→        });
   705→        if (this._socket) this._socket.on('disconnect', (reason) => {
   706→          if (!this._isMounted) return;
   707→          console.warn("Socket disconnected:", reason);
   708→          this.setStatusMessage("Connection lost. Reconnecting...");
   709→        });
   710→        if (this._socket) this._socket.on('connect', () => {
   711→          if (!this._isMounted) return;
   712→          this.setStatusMessage("Reconnected");
   713→          this.hideStatusMessageAfterDelay();
   714→          // R38-FIX: Re-fetch KOT data after reconnect to sync any missed updates
   715→          // during disconnection period. Without this, the UI could show stale KOTs.
   716→          this.fetchKOT().then(() => { this.masonryLoading(); }).catch(() => {});
   717→        });
   718→
   719→        return this.fetchKOT();
   720→      })
   721→      .then(() => {
   722→        if (!this._isMounted) return;
   723→        if (this.audio_alert === 1) {
   724→          this.showAudioAlertMessage = true;
   725→        }
   726→        this.socketHandler = (doc) => {
   727→          if (!this._isMounted) return;
   728→          try {
   729→            if (this.audio_alert === 1) {
   730→              this.playAlertSound(doc.audio_file);
   731→            }
   732→            // R36-FIX: Namespace localStorage key per production station to avoid cross-tab collision
   733→            let kottime = localStorage.getItem("kot_time_" + this.production);
   734→            if (doc.last_kot_time !== kottime) {
   735→              // Full refresh needed — skip intermediate mutations
   736→              this.fetchKOT().then(() => { this.masonryLoading(); }).catch((e) => { console.error("KOT fetch failed:", e); });
   737→            } else {
   738→              // R36-FIX: Guard against missing doc.kot to prevent TypeError crash
   739→              if (!doc.kot) {
   740→                this.fetchKOT().then(() => { this.masonryLoading(); }).catch((e) => { console.error("KOT fetch failed:", e); });
   741→                return;
   742→              }
   743→              // Incremental update — deduplicate to avoid duplicate cards
   744→              const existingIndex = this.kot.findIndex(k => k.name === doc.kot.name);
   745→              if (existingIndex !== -1) {
   746→                // R36-FIX: Preserve strikethrough state before Object.assign overwrites kot_items
   747→                const strikeMap = new Map(
   748→                  (this.kot[existingIndex].kot_items || []).map(i => [i.name, i.striked])
   749→                );
   750→                Object.assign(this.kot[existingIndex], { timecolor: 'text-black', timeRemaining: '— : —', ...doc.kot });
   751→                // Restore strikethrough state after Object.assign
   752→                if (this.kot[existingIndex].kot_items) {
   753→                  this.kot[existingIndex].kot_items.forEach(i => {
   754→                    if (strikeMap.has(i.name)) i.striked = strikeMap.get(i.name);
   755→                  });
   756→                }
   757→                // R37-FIX: Invalidate sorted cache for this KOT since items may have changed
   758→                this._sortedItemsCache.delete(doc.kot.name);
   759→              } else {
   760→                const newKot = { isRotated: false, showDiv: false, timecolor: 'text-black', timeRemaining: '— : —', ...doc.kot };
   761→                this.kot.unshift(newKot);
   762→              }
   763→              this.updateQtyColorTable();
   764→              this.updateTimeRemaining();
   765→              this.masonryLoading();
   766→            }
   767→            if (this._cancelTimeout) clearTimeout(this._cancelTimeout);
   768→            this._cancelTimeout = setTimeout(() => {
   769→              if (!this._isMounted) return;
   770→              if (doc.kot && doc.kot.type === "Cancelled") {
   771→                this.fetchKOT().then(() => {
   772→                  this.masonryLoading();
   773→                }).catch((e) => { console.error("KOT fetch failed:", e); });
   774→              }
   775→            }, 1500);
   776→            if (doc.kot) {
   777→              try {
   778→                localStorage.setItem("kot_time_" + this.production, doc.kot.time);
   779→              } catch (e) {
   780→                if (import.meta.env?.DEV) console.error('localStorage write failed:', e);
   781→              }
   782→            }
   783→          } catch (err) {
   784→            if (import.meta.env?.DEV) console.error("Socket handler error:", err);
   785→          }
   786→        };
   787→        if (this._socket) this._socket.on(this.kot_channel, this.socketHandler);
   788→      })
   789→      .catch((error) => {
   790→        console.error("Initialization or authentication error:", error);
   791→        if (this._isMounted) this.showModal = true;
   792→      });
   793→    this.timer = setInterval(this.updateTimeRemaining, 60000);
   794→  },
   795→  beforeUnmount() {
   796→    this._isMounted = false;
   797→    window.removeEventListener("online", this.handleOnline);
   798→    window.removeEventListener("offline", this.handleOffline);
   799→    document.removeEventListener("click", this.hideAudioAlertMessage);
   800→    window.removeEventListener("resize", this._resizeHandler);
   801→    if (this.socketHandler && this._socket) {
   802→      this._socket.off(this.kot_channel, this.socketHandler);
   803→    }
   804→    if (this._socket) {
   805→      this._socket.off('connect_error');
   806→      this._socket.off('disconnect');
   807→      this._socket.off('connect');
   808→      this._socket.disconnect();
   809→    }
   810→    this._socketInitPromise = null;
   811→    if (this._cancelTimeout) clearTimeout(this._cancelTimeout);
   812→    if (this._statusTimeout) clearTimeout(this._statusTimeout);
   813→    if (this.timer) clearInterval(this.timer);
   814→    if (this._alertAudio) { this._alertAudio.pause(); this._alertAudio = null; }
   815→  },
   816→  computed: {
   817→    sortedKotItems() {
   818→      // R37-FIX: Pure computed — no side effects on reactive kot objects.
   819→      // Uses instance-level _sortedItemsCache Map (non-reactive) instead of
   820→      // mutating kot._sortedItems / kot._sortKey which caused reactivity issues
   821→      // and stale cache after Object.assign overwrites kot_items in socket handler.
   822→      return (kot) => {
   823→        const items = kot.kot_items || [];
   824→        const key = `${kot.name}:${items.length}:${items.map(i => i.name).join(',')}`;
   825→        const cached = this._sortedItemsCache.get(kot.name);
   826→        if (cached && cached.key === key) {
   827→          return cached.items;
   828→        }
   829→        const sorted = [...items].sort((a, b) => (a.serve_priority || 0) - (b.serve_priority || 0));
   830→        this._sortedItemsCache.set(kot.name, { key, items: sorted });
   831→        return sorted;
   832→      };
   833→    },
   834→    visibleKots() {
   835→      return this.kot.filter(kot => !kot.showDiv && kot.production === this.production);
   836→    },
   837→  },
   838→};
