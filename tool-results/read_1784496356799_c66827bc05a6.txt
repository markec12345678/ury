   230→// R41-FIX: Add cancel() method to debounce return value so pending timers

   232→function debounce(fn, delay) {
   233→    let timer = null;
   234→    const debounced = function(...args) {
   235→        clearTimeout(timer);
   236→        timer = setTimeout(() => fn.apply(this, args), delay);
   237→    };
   238→    debounced.cancel = () => { clearTimeout(timer); timer = null; };
   239→    return debounced;
   240→}

   708→    calculateTimeRemaining(targetTime) {
   709→      if (!targetTime || !targetTime.includes(":")) return '— : —';
   710→      const currentTime = new Date();
   711→      // R39-FIX: Guard against malformed time strings with fewer than 3 parts.
   712→      // Previously, "HH:MM" without seconds would set targetSeconds=undefined,
   713→      // which Date() treats as NaN → Invalid Date → NaN propagation.
   714→      const parts = targetTime.split(":");
   715→      const targetHours = parseInt(parts[0], 10);
   716→      const targetMinutes = parseInt(parts[1], 10);
   717→      const targetSeconds = parseInt(parts[2] || '0', 10);
   718→      if (isNaN(targetHours) || isNaN(targetMinutes) || isNaN(targetSeconds)) return '— : —';
   719→      let targetDate = new Date(
   720→        currentTime.getFullYear(),
   721→        currentTime.getMonth(),
   722→        currentTime.getDate(),
   723→        targetHours,
   724→        targetMinutes,
   725→        targetSeconds
   726→      );
   727→      // Handle cross-midnight KOTs: if target is in the future, it was created yesterday
   728→      if (targetDate > currentTime) {
   729→        targetDate = new Date(targetDate.getTime() - 86400000);
   730→      }
   731→
   732→      const timeDifference = Math.max(0, currentTime - targetDate);
   733→      const hoursRemaining = Math.floor(timeDifference / 3600000);
   734→      const minutesRemaining = Math.floor((timeDifference % 3600000) / 60000);
   735→
   736→      return `${hoursRemaining} : ${String(minutesRemaining).padStart(2, '0')}`;
   737→    },
   738→    // R39-FIX: Retry wrapper for fetchKOT with exponential backoff.
   739→    // Without this, a transient server error leaves the KDS showing stale
   740→    // data indefinitely — the only recovery was a manual page reload.
   741→    fetchKOTWithRetry(maxRetries = 3, initialDelay = 2000) {
   742→      // R41-FIX: Clear any pending retry timer before starting a new retry chain.
   743→      // Without this, calling fetchKOTWithRetry() while a previous retry timer is
   744→      // pending would leave the old timer running, potentially causing duplicate
   745→      // fetchKOT calls when both timers fire.
   746→      if (this._fetchRetryTimer) { clearTimeout(this._fetchRetryTimer); this._fetchRetryTimer = null; }
   747→      // R42-FIX: Bump generation counter so any still-running retry chain from
   748→      // a previous fetchKOTWithRetry call knows it's been superseded. Without
   749→      // this, clearing the timer orphans the old promise — its resolve/reject
   750→      // callbacks are never called, leaking the promise and its closures.
   751→      const generation = ++this._fetchGeneration;
   752→      const attempt = (retriesLeft, delay) => {
   753→        // If a newer fetchKOTWithRetry has started, abort this chain
   754→        if (generation !== this._fetchGeneration) {
   755→          return Promise.reject(new Error('Superseded by newer fetchKOTWithRetry call'));
   756→        }
   757→        return this.fetchKOT().catch(err => {
   758→          if (retriesLeft <= 0 || !this._isMounted || generation !== this._fetchGeneration) throw err;
   759→          if (import.meta.env?.DEV) console.warn(`fetchKOT failed, retrying in ${delay}ms...`, err);
   760→          return new Promise((resolve, reject) => {
   761→            this._fetchRetryTimer = setTimeout(() => {
   762→              if (!this._isMounted || generation !== this._fetchGeneration) { reject(err); return; }
   763→              attempt(retriesLeft - 1, Math.min(delay * 2, 30000)).then(resolve).catch(reject);
   764→            }, delay);
   765→          });
   766→        });
   767→      };
   768→      return attempt(maxRetries, initialDelay);
   769→    },

  1119→              } catch (e) {

  1121→              }
  1122→            }
  1123→          } catch (err) {
  1124→            if (import.meta.env?.DEV) console.error("Socket handler error:", err);
