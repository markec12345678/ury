   233→// R41-FIX: Add cancel() method to debounce return value so pending timers

   235→function debounce(fn, delay) {
   236→    let timer = null;
   237→    const debounced = function(...args) {
   238→        clearTimeout(timer);
   239→        timer = setTimeout(() => fn.apply(this, args), delay);
   240→    };
   241→    debounced.cancel = () => { clearTimeout(timer); timer = null; };
   242→    return debounced;

   777→    calculateTimeRemaining(targetTime) {
   778→      if (!targetTime || !targetTime.includes(":")) return '— : —';
   779→      const currentTime = new Date();
   780→      // R39-FIX: Guard against malformed time strings with fewer than 3 parts.
   781→      // Previously, "HH:MM" without seconds would set targetSeconds=undefined,
   782→      // which Date() treats as NaN → Invalid Date → NaN propagation.
   783→      const parts = targetTime.split(":");
   784→      const targetHours = parseInt(parts[0], 10);
   785→      const targetMinutes = parseInt(parts[1], 10);
   786→      const targetSeconds = parseInt(parts[2] || '0', 10);
   787→      if (isNaN(targetHours) || isNaN(targetMinutes) || isNaN(targetSeconds)) return '— : —';
   788→      let targetDate = new Date(
   789→        currentTime.getFullYear(),
   790→        currentTime.getMonth(),
   791→        currentTime.getDate(),
   792→        targetHours,
   793→        targetMinutes,
   794→        targetSeconds
   795→      );
   796→      // Handle cross-midnight KOTs: if target is in the future, it was created yesterday
   797→      if (targetDate > currentTime) {
   798→        targetDate = new Date(targetDate.getTime() - 86400000);
   799→      }
   800→
   801→      const timeDifference = Math.max(0, currentTime - targetDate);
   802→      const hoursRemaining = Math.floor(timeDifference / 3600000);
   803→      const minutesRemaining = Math.floor((timeDifference % 3600000) / 60000);
   804→
   805→      return `${hoursRemaining} : ${String(minutesRemaining).padStart(2, '0')}`;
   806→    },

   808→    // Without this, a transient server error leaves the KDS showing stale
   809→    // data indefinitely — the only recovery was a manual page reload.
   810→    fetchKOTWithRetry(maxRetries = 3, initialDelay = 2000) {
   811→      // R41-FIX: Clear any pending retry timer before starting a new retry chain.
   812→      // Without this, calling fetchKOTWithRetry() while a previous retry timer is
   813→      // pending would leave the old timer running, potentially causing duplicate
   814→      // fetchKOT calls when both timers fire.
   815→      if (this._fetchRetryTimer) { clearTimeout(this._fetchRetryTimer); this._fetchRetryTimer = null; }
   816→      // R42-FIX: Bump generation counter so any still-running retry chain from
   817→      // a previous fetchKOTWithRetry call knows it's been superseded. Without
   818→      // this, clearing the timer orphans the old promise — its resolve/reject
   819→      // callbacks are never called, leaking the promise and its closures.
   820→      const generation = ++this._fetchGeneration;
   821→      const attempt = (retriesLeft, delay) => {
   822→        // If a newer fetchKOTWithRetry has started, abort this chain
   823→        if (generation !== this._fetchGeneration) {
   824→          return Promise.reject(new Error('Superseded by newer fetchKOTWithRetry call'));
   825→        }
   826→        return this.fetchKOT().catch(err => {
   827→          if (retriesLeft <= 0 || !this._isMounted || generation !== this._fetchGeneration) throw err;
   828→          if (import.meta.env?.DEV) console.warn(`fetchKOT failed, retrying in ${delay}ms...`, err);
   829→          return new Promise((resolve, reject) => {
   830→            this._fetchRetryTimer = setTimeout(() => {
   831→              if (!this._isMounted || generation !== this._fetchGeneration) { reject(err); return; }
   832→              attempt(retriesLeft - 1, Math.min(delay * 2, 30000)).then(resolve).catch(reject);
   833→            }, delay);
   834→          });
   835→        });
   836→      };
   837→      return attempt(maxRetries, initialDelay);
   838→    },
