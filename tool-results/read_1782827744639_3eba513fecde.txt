    77→
    78→type SortField = 'name' | 'price' | 'availability';
    79→type SortDirection = 'asc' | 'desc';
    80→
    81→export function MenuManagementTab() {
    82→  const {
    83→    menuCourses,
    84→    menuItems,
    85→    currency,
    86→    addMenuItem,
    87→    updateMenuItem,
    88→    deleteMenuItem,
    89→    toggleMenuItemAvailability,
    90→    addMenuCourse,
    91→    updateMenuCourse,
    92→    deleteMenuCourse,
    93→    loadMenuFromDB,
    94→  } = useURYStore();
    95→
    96→  const [searchQuery, setSearchQuery] = useState('');
    97→  const [activeCourse, setActiveCourse] = useState<string>('all');
    98→  const [viewMode, setViewMode] = useState<'grid' | 'table'>('table');
    99→  const [isLoading, setIsLoading] = useState(false);
   100→
   101→  // Sorting state
   102→  const [sortField, setSortField] = useState<SortField>('name');
   103→  const [sortDirection, setSortDirection] = useState<SortDirection>('asc');
   104→
   105→  // Menu Item Dialog state
   106→  const [menuItemDialogOpen, setMenuItemDialogOpen] = useState(false);
   107→  const [editingItem, setEditingItem] = useState<MenuItem | null>(null);
   108→  const [menuItemForm, setMenuItemForm] = useState<MenuFormData>({
   109→    name: '',
   110→    nameLocal: '',
   111→    course: '',
   112→    price: 0,
   113→    description: '',
   114→    isVeg: true,
   115→    isAvailable: true,
   116→    tags: [],
   117→  });
   118→  const [tagInput, setTagInput] = useState('');
   119→
   120→  // Form validation tracking
   121→  const [attemptedSubmit, setAttemptedSubmit] = useState(false);
   122→
   123→  // Course Dialog state
   124→  const [courseDialogOpen, setCourseDialogOpen] = useState(false);
   125→  const [editingCourse, setEditingCourse] = useState<MenuCourse | null>(null);
   126→  const [courseForm, setCourseForm] = useState<CourseFormData>({
   127→    name: '',
   128→    priority: 0,
   129→  });
   130→  const [courseFormAttempted, setCourseFormAttempted] = useState(false);
   131→
   132→  // Delete confirmation
   133→  const [deleteTarget, setDeleteTarget] = useState<{ type: 'item' | 'course'; id: string; name: string } | null>(null);
   134→
   135→  // ── Load data on mount ─────────────────────────────────
   136→  useEffect(() => {
   137→    const loadData = async () => {
   138→      setIsLoading(true);
   139→      try {
   140→        await loadMenuFromDB();
   141→      } catch {
   142→        // Silently handle - store falls back to mock data
   143→      } finally {
   144→        setIsLoading(false);
   145→      }
   146→    };
   147→    loadData();
   148→  }, [loadMenuFromDB]);
   149→
   150→  // ── Sync from DB handler ───────────────────────────────
   151→  const handleSyncFromDB = useCallback(async () => {
   152→    setIsLoading(true);
   153→    try {
   154→      await loadMenuFromDB();
   155→    } catch {
   156→      // Silently handle
   157→    } finally {
   158→      setIsLoading(false);
   159→    }
   160→  }, [loadMenuFromDB]);
   161→
   162→  // ── Sorting handler ────────────────────────────────────
   163→  const handleSort = useCallback((field: SortField) => {
   164→    if (sortField === field) {
   165→      setSortDirection((prev) => (prev === 'asc' ? 'desc' : 'asc'));
   166→    } else {
   167→      setSortField(field);
   168→      setSortDirection('asc');
   169→    }
   170→  }, [sortField]);
   171→
   172→  // ── Bulk availability toggle ───────────────────────────
   173→  const handleBulkToggle = useCallback((courseId: string) => {
   174→    const courseItems = menuItems.filter((i) => i.course === courseId);
   175→    if (courseItems.length === 0) return;
   176→
   177→    const allAvailable = courseItems.every((i) => i.isAvailable);
   178→    // If all available → make all unavailable; otherwise → make all available
   179→    courseItems.forEach((item) => {
   180→      if (allAvailable && item.isAvailable) {
   181→        toggleMenuItemAvailability(item.id);
   182→      } else if (!allAvailable && !item.isAvailable) {
   183→        toggleMenuItemAvailability(item.id);
   184→      }
   185→    });
   186→  }, [menuItems, toggleMenuItemAvailability]);
   187→
   188→  // Filter and sort items
   189→  const filteredItems = useMemo(() => {
   190→    let items = [...menuItems];
   191→
   192→    // Filter by course
   193→    if (activeCourse !== 'all') {
   194→      items = items.filter((item) => item.course === activeCourse);
   195→    }
   196→
   197→    // Filter by search
   198→    if (searchQuery.trim()) {
   199→      const q = searchQuery.toLowerCase();
   200→      items = items.filter(
   201→        (item) =>
   202→          item.name.toLowerCase().includes(q) ||
   203→          item.courseName.toLowerCase().includes(q) ||
   204→          item.description?.toLowerCase().includes(q) ||
   205→          item.tags?.some((t) => t.toLowerCase().includes(q))
   206→      );
   207→    }
   208→
   209→    // Sort
   210→    items.sort((a, b) => {
   211→      let cmp = 0;
   212→      switch (sortField) {
   213→        case 'name':
   214→          cmp = a.name.localeCompare(b.name);
   215→          break;
   216→        case 'price':
   217→          cmp = a.price - b.price;
   218→          break;
   219→        case 'availability':
   220→          cmp = Number(b.isAvailable) - Number(a.isAvailable);
   221→          break;
   222→      }
   223→      return sortDirection === 'asc' ? cmp : -cmp;
   224→    });
   225→
   226→    return items;
   227→  }, [menuItems, activeCourse, searchQuery, sortField, sortDirection]);
   228→
   229→  // ── Statistics ─────────────────────────────────────────
   230→  const totalItems = menuItems.length;
   231→  const availableItems = menuItems.filter((i) => i.isAvailable).length;
   232→  const unavailableItems = totalItems - availableItems;
   233→  const totalCategories = menuCourses.length;
   234→
   235→  // ── Menu Item CRUD handlers ─────────────────────────────
   236→
   237→  const openCreateItemDialog = () => {
   238→    setEditingItem(null);
   239→    setMenuItemForm({
   240→      name: '',
   241→      nameLocal: '',
   242→      course: menuCourses[0]?.id || '',
   243→      price: 0,
   244→      description: '',
   245→      isVeg: true,
   246→      isAvailable: true,
   247→      tags: [],
   248→    });
   249→    setTagInput('');
   250→    setAttemptedSubmit(false);
   251→    setMenuItemDialogOpen(true);
   252→  };
   253→
   254→  const openEditItemDialog = (item: MenuItem) => {
   255→    setEditingItem(item);
   256→    setMenuItemForm({
   257→      name: item.name,
   258→      nameLocal: item.nameHi || '',
   259→      course: item.course,
   260→      price: item.price,
   261→      description: item.description || '',
   262→      isVeg: item.isVeg,
   263→      isAvailable: item.isAvailable,
   264→      tags: item.tags || [],
   265→    });
   266→    setTagInput('');
   267→    setAttemptedSubmit(false);
   268→    setMenuItemDialogOpen(true);
   269→  };
   270→
   271→  const handleDuplicateItem = (item: MenuItem) => {
   272→    const duplicatedForm: MenuFormData = {
   273→      name: `${item.name} (kopija)`,
   274→      nameLocal: item.nameHi || '',
   275→      course: item.course,
   276→      price: item.price,
   277→      description: item.description || '',
   278→      isVeg: item.isVeg,
   279→      isAvailable: item.isAvailable,
   280→      tags: item.tags ? [...item.tags] : [],
   281→    };
   282→    addMenuItem(duplicatedForm);
   283→  };
   284→
   285→  const handleSaveMenuItem = () => {
   286→    setAttemptedSubmit(true);
   287→    if (!menuItemForm.name.trim() || menuItemForm.price <= 0 || !menuItemForm.course) return;
   288→    if (editingItem) {
   289→      updateMenuItem(editingItem.id, menuItemForm);
   290→    } else {
   291→      addMenuItem(menuItemForm);
   292→    }
   293→    setMenuItemDialogOpen(false);
   294→    setAttemptedSubmit(false);
   295→  };
   296→
   297→  const handleAddTag = () => {
   298→    const tag = tagInput.trim().toLowerCase();
   299→    if (tag && !menuItemForm.tags?.includes(tag)) {
   300→      setMenuItemForm((prev) => ({
   301→        ...prev,
   302→        tags: [...(prev.tags || []), tag],
   303→      }));
   304→    }
   305→    setTagInput('');
   306→  };
   307→
   308→  const handleRemoveTag = (tag: string) => {
   309→    setMenuItemForm((prev) => ({
   310→      ...prev,
   311→      tags: prev.tags?.filter((t) => t !== tag) || [],
   312→    }));
   313→  };
   314→
   315→  // ── Course CRUD handlers ────────────────────────────────
   316→
   317→  const openCreateCourseDialog = () => {
   318→    setEditingCourse(null);
   319→    setCourseForm({
   320→      name: '',
   321→      priority: menuCourses.length + 1,
   322→    });
   323→    setCourseFormAttempted(false);
   324→    setCourseDialogOpen(true);
   325→  };
   326→
   327→  const openEditCourseDialog = (course: MenuCourse) => {
   328→    setEditingCourse(course);
   329→    setCourseForm({
   330→      name: course.name,
   331→      priority: course.priority,
   332→    });
   333→    setCourseFormAttempted(false);
   334→    setCourseDialogOpen(true);
   335→  };
   336→
   337→  const handleSaveCourse = () => {
   338→    setCourseFormAttempted(true);
   339→    if (!courseForm.name.trim()) return;
   340→    if (editingCourse) {
   341→      updateMenuCourse(editingCourse.id, courseForm);
   342→    } else {
   343→      addMenuCourse(courseForm);
   344→    }
   345→    setCourseDialogOpen(false);
   346→  };
   347→
   348→  // ── Validation helpers ─────────────────────────────────
   349→  const isNameInvalid = attemptedSubmit && !menuItemForm.name.trim();
   350→  const isPriceInvalid = attemptedSubmit && menuItemForm.price <= 0;
   351→  const isCourseInvalid = attemptedSubmit && !menuItemForm.course;
   352→  const isCourseNameInvalid = courseFormAttempted && !courseForm.name.trim();
   353→
   354→  // ── Sort icon helper ───────────────────────────────────
   355→  const SortIcon = ({ field }: { field: SortField }) => {
   356→    if (sortField !== field) return <ArrowUpDown className="h-3 w-3 ml-1 opacity-40" />;
   357→    return sortDirection === 'asc' ? <ArrowUp className="h-3 w-3 ml-1" /> : <ArrowDown className="h-3 w-3 ml-1" />;
   358→  };
   359→
   360→  return (
   361→    <TooltipProvider>
   362→      <div className="space-y-5">
   363→        {/* ── Gradient Header Card ───────────────────────── */}
   364→        <Card className="overflow-hidden border-0 shadow-lg">
   365→          <div className="bg-gradient-to-r from-emerald-600 via-emerald-500 to-teal-500 p-6 text-white">
   366→            <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   367→              <div>
   368→                <h2 className="text-xl font-bold tracking-tight">Upravljanje jedilnika</h2>
   369→                <p className="text-emerald-100 text-sm mt-1">Upravljajte artikle, kategorije in razpoložljivost</p>
   370→              </div>
   371→              <Button
   372→                variant="secondary"
   373→                size="sm"
   374→                className="bg-white/20 hover:bg-white/30 text-white border-white/30 backdrop-blur-sm self-start sm:self-auto"
   375→                onClick={handleSyncFromDB}
   376→                disabled={isLoading}
   377→              >
   378→                <RefreshCw className={`h-4 w-4 mr-2 ${isLoading ? 'animate-spin' : ''}`} />
   379→                Sinhroniziraj iz PB
   380→              </Button>
   381→            </div>
   382→          </div>
   383→          {/* Statistics Summary */}
   384→          <CardContent className="p-0">
   385→            <div className="grid grid-cols-2 sm:grid-cols-4 divide-x divide-y sm:divide-y-0 border-t border-border">
   386→              <div className="p-4 flex items-center gap-3">
   387→                <div className="rounded-lg bg-emerald-100 p-2">
   388→                  <Package className="h-5 w-5 text-emerald-600" />
   389→                </div>
   390→                <div>
   391→                  <p className="text-xs text-muted-foreground font-medium">Skupaj artiklov</p>
   392→                  <p className="text-2xl font-bold">{totalItems}</p>
   393→                </div>
   394→              </div>
   395→              <div className="p-4 flex items-center gap-3">
   396→                <div className="rounded-lg bg-green-100 p-2">
   397→                  <CheckCircle2 className="h-5 w-5 text-green-600" />
   398→                </div>
   399→                <div>
   400→                  <p className="text-xs text-muted-foreground font-medium">Na voljo</p>
   401→                  <p className="text-2xl font-bold text-green-700">{availableItems}</p>
   402→                </div>
   403→              </div>
   404→              <div className="p-4 flex items-center gap-3">
   405→                <div className="rounded-lg bg-red-100 p-2">
   406→                  <XCircle className="h-5 w-5 text-red-500" />
   407→                </div>
   408→                <div>
   409→                  <p className="text-xs text-muted-foreground font-medium">Ni na voljo</p>
   410→                  <p className="text-2xl font-bold text-red-600">{unavailableItems}</p>
   411→                </div>
   412→              </div>
   413→              <div className="p-4 flex items-center gap-3">
   414→                <div className="rounded-lg bg-amber-100 p-2">
   415→                  <LayoutGrid className="h-5 w-5 text-amber-600" />
   416→                </div>
   417→                <div>
   418→                  <p className="text-xs text-muted-foreground font-medium">Kategorije</p>
   419→                  <p className="text-2xl font-bold text-amber-700">{totalCategories}</p>
   420→                </div>
   421→              </div>
   422→            </div>
   423→          </CardContent>
   424→        </Card>
   425→
   426→        {/* ── Action Bar ─────────────────────────────────── */}
   427→        <Card className="shadow-sm">
   428→          <CardContent className="p-4">
   429→            <div className="flex flex-wrap items-center gap-3">
   430→              <div className="relative flex-1 min-w-[200px]">
   431→                <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   432→                <Input
   433→                  placeholder="Išči artikle, kategorije, oznake..."
   434→                  value={searchQuery}
   435→                  onChange={(e) => setSearchQuery(e.target.value)}
   436→                  className="pl-9"
   437→                />
   438→                {searchQuery && (
   439→                  <Button variant="ghost" size="icon" className="absolute right-1 top-1/2 -translate-y-1/2 h-7 w-7" onClick={() => setSearchQuery('')}>
   440→                    <X className="h-3.5 w-3.5" />
   441→                  </Button>
   442→                )}
   443→              </div>
   444→              <Button size="sm" className="bg-emerald-600 hover:bg-emerald-700" onClick={openCreateItemDialog}>
   445→                <Plus className="h-4 w-4 mr-1" />
   446→                Nov artikel
   447→              </Button>
   448→              <Button variant="outline" size="sm" onClick={openCreateCourseDialog}>
   449→                <Plus className="h-4 w-4 mr-1" />
   450→                Nova kategorija
   451→              </Button>
   452→              <div className="flex items-center gap-1 border rounded-md p-0.5">
   453→                <Button variant={viewMode === 'table' ? 'default' : 'ghost'} size="sm" className="h-7 px-2" onClick={() => setViewMode('table')}>
   454→                  <ArrowUpDown className="h-3.5 w-3.5" />
   455→                </Button>
   456→                <Button variant={viewMode === 'grid' ? 'default' : 'ghost'} size="sm" className="h-7 px-2" onClick={() => setViewMode('grid')}>
   457→                  <GripVertical className="h-3.5 w-3.5" />
   458→                </Button>
   459→              </div>
   460→            </div>
   461→
   462→            {/* Sorting Options */}
   463→            <div className="flex flex-wrap items-center gap-2 mt-3 pt-3 border-t">
   464→              <span className="text-xs text-muted-foreground font-medium">Razvrsti:</span>
   465→              <Button
   466→                variant={sortField === 'name' ? 'default' : 'outline'}
   467→                size="sm"
   468→                className={`h-7 text-xs ${sortField === 'name' ? 'bg-emerald-600 hover:bg-emerald-700' : ''}`}
   469→                onClick={() => handleSort('name')}
   470→              >
   471→                Ime <SortIcon field="name" />
   472→              </Button>
   473→              <Button
   474→                variant={sortField === 'price' ? 'default' : 'outline'}
   475→                size="sm"
   476→                className={`h-7 text-xs ${sortField === 'price' ? 'bg-emerald-600 hover:bg-emerald-700' : ''}`}
   477→                onClick={() => handleSort('price')}
   478→              >
   479→                Cena <SortIcon field="price" />
   480→              </Button>
   481→              <Button
   482→                variant={sortField === 'availability' ? 'default' : 'outline'}
   483→                size="sm"
   484→                className={`h-7 text-xs ${sortField === 'availability' ? 'bg-emerald-600 hover:bg-emerald-700' : ''}`}
   485→                onClick={() => handleSort('availability')}
   486→              >
   487→                Razpoložljivost <SortIcon field="availability" />
   488→              </Button>
   489→            </div>
   490→          </CardContent>
   491→        </Card>
   492→
   493→        {/* ── Course Filters with Bulk Toggle ────────────── */}
   494→        <div className="flex flex-wrap gap-2">
   495→          <Button
   496→            variant={activeCourse === 'all' ? 'default' : 'outline'}
   497→            size="sm"
   498→            className={activeCourse === 'all' ? 'bg-emerald-600 hover:bg-emerald-700' : ''}
   499→            onClick={() => setActiveCourse('all')}
   500→          >
   501→            Vse ({menuItems.length})
   502→          </Button>
   503→          {menuCourses.map((course) => {
   504→            const count = menuItems.filter((i) => i.course === course.id).length;
   505→            const courseItems = menuItems.filter((i) => i.course === course.id);
   506→            const allAvailable = courseItems.length > 0 && courseItems.every((i) => i.isAvailable);
   507→            const noneAvailable = courseItems.length > 0 && courseItems.every((i) => !i.isAvailable);
   508→            return (
   509→              <div key={course.id} className="flex items-center gap-1">
   510→                <Button
   511→                  variant={activeCourse === course.id ? 'default' : 'outline'}
   512→                  size="sm"
   513→                  className={activeCourse === course.id ? 'bg-emerald-600 hover:bg-emerald-700' : ''}
   514→                  onClick={() => setActiveCourse(course.id)}
   515→                >
   516→                  {course.name} ({count})
   517→                </Button>
   518→                <Tooltip>
   519→                  <TooltipTrigger asChild>
   520→                    <Button
   521→                      variant="ghost"
   522→                      size="icon"
   523→                      className="h-7 w-7"
   524→                      onClick={() => handleBulkToggle(course.id)}
   525→                      disabled={count === 0}
   526→                    >
   527→                      {allAvailable ? (
   528→                        <EyeOff className="h-3.5 w-3.5 text-amber-500" />
   529→                      ) : noneAvailable ? (
   530→                        <Eye className="h-3.5 w-3.5 text-green-500" />
   531→                      ) : (
   532→                        <SlidersHorizontal className="h-3.5 w-3.5 text-muted-foreground" />
   533→                      )}
   534→                    </Button>
   535→                  </TooltipTrigger>
   536→                  <TooltipContent>
   537→                    <p>
   538→                      {allAvailable
   539→                        ? 'Onemogoči vse v kategoriji'
   540→                        : 'Omogoči vse v kategoriji'}
   541→                    </p>
   542→                  </TooltipContent>
   543→                </Tooltip>
   544→              </div>
   545→            );
   546→          })}
   547→        </div>
   548→
   549→        {/* ── Course Management Section ───────────────────── */}
   550→        <Card className="shadow-sm">
   551→          <CardHeader className="pb-2 bg-gradient-to-r from-muted/80 to-muted/40 rounded-t-lg">
   552→            <div className="flex items-center justify-between">
   553→              <CardTitle className="text-base font-semibold">Kategorije jedilnika</CardTitle>
   554→              <Badge variant="secondary" className="text-xs">{menuCourses.length} kategorij</Badge>
   555→            </div>
   556→          </CardHeader>
   557→          <CardContent className="pt-4">
   558→            {menuCourses.length === 0 ? (
   559→              <p className="text-sm text-muted-foreground text-center py-4">Ni še dodanih kategorij</p>
   560→            ) : (
   561→              <div className="flex flex-wrap gap-2">
   562→                {menuCourses.map((course) => (
   563→                  <div
   564→                    key={course.id}
   565→                    className="flex items-center gap-2 px-3 py-1.5 rounded-lg border bg-muted/50 text-sm hover:bg-muted transition-colors"
