   238→// R41-FIX: Add cancel() method to debounce return value so pending timers

   240→function debounce(fn, delay) {
   241→    let timer = null;
   242→    const debounced = function(...args) {
   243→        clearTimeout(timer);
   244→        timer = setTimeout(() => fn.apply(this, args), delay);
   245→    };
   246→    debounced.cancel = () => { clearTimeout(timer); timer = null; };
   247→    return debounced;

   851→    calculateTimeRemaining(targetTime) {
   852→      if (!targetTime || !targetTime.includes(":")) return '— : —';
   853→      const currentTime = new Date();
   854→      // R39-FIX: Guard against malformed time strings with fewer than 3 parts.
   855→      // Previously, "HH:MM" without seconds would set targetSeconds=undefined,
   856→      // which Date() treats as NaN → Invalid Date → NaN propagation.
   857→      const parts = targetTime.split(":");
   858→      const targetHours = parseInt(parts[0], 10);
   859→      const targetMinutes = parseInt(parts[1], 10);
   860→      const targetSeconds = parseInt(parts[2] || '0', 10);
   861→      if (isNaN(targetHours) || isNaN(targetMinutes) || isNaN(targetSeconds)) return '— : —';
   862→      let targetDate = new Date(
   863→        currentTime.getFullYear(),
   864→        currentTime.getMonth(),
   865→        currentTime.getDate(),
   866→        targetHours,
   867→        targetMinutes,
   868→        targetSeconds
   869→      );
   870→      // Handle cross-midnight KOTs: if target is in the future, it was created yesterday
   871→      if (targetDate > currentTime) {
   872→        targetDate = new Date(targetDate.getTime() - 86400000);
   873→      }
   874→
   875→      const timeDifference = Math.max(0, currentTime - targetDate);
   876→      const hoursRemaining = Math.floor(timeDifference / 3600000);
   877→      const minutesRemaining = Math.floor((timeDifference % 3600000) / 60000);
   878→
   879→      return `${hoursRemaining} : ${String(minutesRemaining).padStart(2, '0')}`;
   880→    },
   881→    // R39-FIX: Retry wrapper for fetchKOT with exponential backoff.
   882→    // Without this, a transient server error leaves the KDS showing stale
   883→    // data indefinitely — the only recovery was a manual page reload.
   884→    fetchKOTWithRetry(maxRetries = 3, initialDelay = 2000) {
   885→      // R41-FIX: Clear any pending retry timer before starting a new retry chain.
   886→      // Without this, calling fetchKOTWithRetry() while a previous retry timer is
   887→      // pending would leave the old timer running, potentially causing duplicate
   888→      // fetchKOT calls when both timers fire.
   889→      if (this._fetchRetryTimer) { clearTimeout(this._fetchRetryTimer); this._fetchRetryTimer = null; }
   890→      // R42-FIX: Bump generation counter so any still-running retry chain from
   891→      // a previous fetchKOTWithRetry call knows it's been superseded. Without
   892→      // this, clearing the timer orphans the old promise — its resolve/reject
   893→      // callbacks are never called, leaking the promise and its closures.
   894→      const generation = ++this._fetchGeneration;
   895→      const attempt = (retriesLeft, delay) => {
   896→        // If a newer fetchKOTWithRetry has started, abort this chain
   897→        if (generation !== this._fetchGeneration) {
   898→          return Promise.reject(new Error('Superseded by newer fetchKOTWithRetry call'));
   899→        }
   900→        return this.fetchKOT().catch(err => {
   901→          if (retriesLeft <= 0 || !this._isMounted || generation !== this._fetchGeneration) throw err;
   902→          if (import.meta.env?.DEV) console.warn(`fetchKOT failed, retrying in ${delay}ms...`, err);
   903→          return new Promise((resolve, reject) => {
   904→            this._fetchRetryTimer = setTimeout(() => {
   905→              if (!this._isMounted || generation !== this._fetchGeneration) { reject(err); return; }
   906→              attempt(retriesLeft - 1, Math.min(delay * 2, 30000)).then(resolve).catch(reject);
   907→            }, delay);
   908→          });
   909→        });
   910→      };
   911→      return attempt(maxRetries, initialDelay);
   912→    },

  1438→    // R39-FIX: sortedKotItems moved from computed (which returned a function) to
  1439→    // a method. Previously, the computed returned a new function object every time
  1440→    // its reactive dependencies (this.kot) changed, which happened every minute
  1441→    // via updateTimeRemaining. Each new function object was unnecessary churn.
  1442→    // As a method, the function identity is stable and the manual _sortedItemsCache
  1443→    // still provides memoization. The template call `sortedKotItems(kot)` works
  1444→    // identically for both computed-returning-function and method.
  1445→    visibleKots() {
  1446→      // R39-FIX: Removed !kot.showDiv filter — showDiv was always false (dead code).
  1447→      // When production is empty (root route "/"), show ALL KOTs instead of filtering
  1448→      // by empty string which would exclude KOTs that have a station assigned.
  1449→      if (!this.production) {
  1450→        return this.kot;
  1451→      }
  1452→      return this.kot.filter(kot => kot.production === this.production);
  1453→    },
  1454→  },
  1455→  watch: {
  1456→    // R47-FIX (M4): Prevent background scrolling when auth modal is open
  1457→    // R48-FIX (M2): Auto-focus first focusable element when modal opens.
  1458→    // R48-FIX (M3): Return focus to previously active element on close.
  1459→    showModal(val) {
  1460→      document.body.style.overflow = val ? 'hidden' : '';
  1461→      if (val) {
  1462→        // R48-FIX (M3): Capture the element that had focus before modal opened
  1463→        this._preModalFocus = document.activeElement;
  1464→        // R48-FIX (M2): Auto-focus the modal or first focusable child
  1465→        this.$nextTick(() => {
  1466→          const modal = this.$refs.authModal;
  1467→          if (modal) {
  1468→            const first = modal.querySelector('button, [href], input, select, textarea, [tabindex]:not([tabindex="-1"])');
  1469→            (first || modal).focus();
  1470→          }
  1471→        });
  1472→      } else {
  1473→        // R48-FIX (M3): Restore focus to the element that was active before modal opened
  1474→        if (this._preModalFocus && typeof this._preModalFocus.focus === 'function' && document.contains(this._preModalFocus)) {
  1475→          this.$nextTick(() => {
  1476→            if (this._preModalFocus) this._preModalFocus.focus();
  1477→            this._preModalFocus = null;
  1478→          });
  1479→        }
  1480→      }
  1481→    },
  1482→    // R41-FIX: Vue reuses component instances when navigating between routes
  1483→    // that use the same component (Home). Without this watcher, navigating
  1484→    // from /station/kitchen to /station/bar would NOT re-mount the KOT
  1485→    // component, so this.production would still be "kitchen" and the KDS
  1486→    // would filter for the wrong station.
  1487→    '$route.params.production'(newVal) {
  1488→      // R42-FIX: decodeURIComponent can throw URIError on malformed input
  1489→      let decoded;
  1490→      try {
  1491→        decoded = decodeURIComponent(newVal || '');
  1492→      } catch (e) {
  1493→        if (import.meta.env?.DEV) console.warn('Failed to decode production param:', newVal, e);
  1494→        decoded = newVal || '';
  1495→      }
  1496→      if (decoded === this.production) return;
  1497→      this.production = decoded;
  1498→      // Re-register socket handler on the new channel
  1499→      if (this.kot_channel && this._socket && this.socketHandler) {
  1500→        this._socket.off(this.kot_channel, this.socketHandler);
  1501→      }
  1502→      // R48-FIX (M1): Abort any in-flight fetch for the old station before switching.
  1503→      if (this._fetchAbortController) {
  1504→        this._fetchAbortController.abort();
  1505→        this._fetchAbortController = null;
  1506→      }
  1507→      // R46-FIX (H1): Invalidate stale fetch promise so a fresh one is created
  1508→      // for the new station. Without this, fetchKOT() returns the in-flight
  1509→      // promise for the OLD station, resulting in wrong data + wrong socket channel.
  1510→      // R47-FIX (H2): Increment fetch generation so the old fetch's .then() discards results
  1511→      this._fetchId++;
  1512→      this._fetchInProgress = null;
  1513→      // fetchKOT will set the new kot_channel and re-register the handler
  1514→      // R45-FIX: Handle fetch failure on station change — previously .catch(() => {})
  1515→      // silently swallowed errors, leaving the user with an empty KDS and no feedback.
  1516→      this.fetchKOTWithRetry().catch((error) => {
  1517→        this._handleFetchError(error, "Failed to load station data. Click Refresh.");
  1518→      });
  1519→    },
  1520→  },
  1521→};
