    97→// ── UI Preference Persistence ────────────────────────────
    98→const UI_PREFS_KEY = 'ury_ui_prefs';
    99→
   100→interface UIPreferences {
   101→  darkMode: boolean;
   102→  activeTab: string;
   103→}
   104→
   105→function loadUIPrefs(): Partial<UIPreferences> {
   106→  if (typeof window === 'undefined') return {};
   107→  try {
   108→    const raw = localStorage.getItem(UI_PREFS_KEY);
   109→    if (!raw) return {};
   110→    return JSON.parse(raw);
   111→  } catch {
   112→    return {};
   113→  }
   114→}
   115→
   116→function saveUIPrefs(prefs: Partial<UIPreferences>): void {
   117→  if (typeof window === 'undefined') return;
   118→  try {
   119→    localStorage.setItem(UI_PREFS_KEY, JSON.stringify(prefs));
   120→  } catch {
   121→    // ignore storage errors
   122→  }
   123→}

   419→    // ── Main Data Refresh ──────────────────────────────
   420→
   421→    refreshData: async () => {
   422→      const { isConnected } = get();
   423→      if (!isConnected) return;
   424→
   425→      set({ isRefreshing: true });
   426→
   427→      try {
   428→        const client = getFrappeClient();
   429→
   430→        // Run all fetches in parallel for better performance
   431→        const [
   432→          kotResult,
   433→          tablesResult,
   434→          invoicesResult,
   435→          roomsResult,
   436→          plResult,
   437→          shiftResult,
   438→        ] = await Promise.allSettled([
   439→          client.getKOTList(),
   440→          client.getRestaurantTables(),
   441→          client.getInvoiceForCashier('Paid', '', 20, 0),
   442→          client.getRestaurantRooms(),
   443→          client.getDailyPL(),
   444→          client.getPOSOpeningEntry('Open'),
   445→        ]);
   446→
   447→        // ── Process KOTs ────────────────────────────────

   449→          const transformedKOTs: KOTCard[] = Array.isArray(kotResult.value.data)

   452→                orderNo: String(kot.order_no || kot.orderNo || ''),
   453→                table: String(kot.restaurant_table || kot.table || ''),

   461→                  : [],
   462→                timePlaced: String(kot.time_placed || kot.timePlaced || ''),
   463→                elapsed: Number(kot.elapsed || 0),

   470→        }
   471→
   472→        // ── Process Tables ──────────────────────────────

   482→                orderTotal: t.order_total ? Number(t.order_total) : undefined,
   483→                customer: t.customer ? String(t.customer) : undefined,
   484→              }))
   485→            : [];
   486→          set({ tables: transformedTables });
   487→        }
   488→
   489→        // ── Process Invoices → Recent Orders + KPIs ─────
   490→        if (invoicesResult.status === 'fulfilled' && invoicesResult.value?.data) {
   491→          const invoices = Array.isArray(invoicesResult.value.data)
   492→            ? invoicesResult.value.data
   493→            : [];
   494→
   495→          // Recent orders
   496→          const transformedOrders: RecentOrder[] = invoices.map((inv: Record<string, unknown>) => ({
   497→            invoice: String(inv.name || ''),
   498→            customer: String(inv.customer || ''),
   499→            type: (inv.type === 'Dine-in' ? 'Dine-in' : inv.type === 'Takeaway' ? 'Takeaway' : 'Delivery') as RecentOrder['type'],
   500→            amount: Number(inv.grand_total || 0),
   501→            status: (inv.status as RecentOrder['status']) || 'Paid',
   502→            time: inv.posting_time ? String(inv.posting_time).slice(0, 5) : '',
   503→          }));
   504→          set({ recentOrders: transformedOrders });
   505→
   506→          // Compute KPIs from invoices
   507→          const totalSales = invoices.reduce((sum: number, inv: Record<string, unknown>) => sum + Number(inv.grand_total || 0), 0);
   508→          const totalOrders = invoices.length;
   509→          const occupiedTables = invoices.filter((inv: Record<string, unknown>) => inv.type === 'Dine-in').length;
   510→          const allTables = get().tables;
   511→          set({
   512→            kpis: {
   513→              dailySales: totalSales,
   514→              totalOrders,
   515→              avgBill: totalOrders > 0 ? Math.round(totalSales / totalOrders) : 0,

   517→              totalTables: allTables.length || 32,
   518→            },
   519→          });
   520→
   521→          // Compute hourly sales from invoices
   522→          const hourMap = new Map<string, { dineIn: number; takeaway: number }>();
   523→          // Initialize all hours from 11:00 to 22:00
   524→          for (let h = 11; h <= 22; h++) {
   525→            const key = `${String(h).padStart(2, '0')}:00`;
   526→            hourMap.set(key, { dineIn: 0, takeaway: 0 });
   527→          }
   528→          // Aggregate invoice totals by hour
   529→          invoices.forEach((inv: Record<string, unknown>) => {
   530→            const timeStr = String(inv.posting_time || '').slice(0, 5);
   531→            if (!timeStr) return;
   532→            const hour = timeStr.split(':')[0];
   533→            const key = `${hour}:00`;
   534→            const entry = hourMap.get(key);
   535→            if (entry) {
   536→              const amount = Number(inv.grand_total || 0);
   537→              if (inv.type === 'Takeaway' || inv.type === 'Delivery') {
   538→                entry.takeaway += amount;
   539→              } else {
   540→                entry.dineIn += amount;
   541→              }
   542→            }
   543→          });
   544→          const hourlyData: HourlySales[] = Array.from(hourMap.entries()).map(([hour, data]) => ({
   545→            hour,
   546→            dineIn: data.dineIn,
   547→            takeaway: data.takeaway,
   548→          }));
   549→          if (hourlyData.length > 0) set({ hourlySales: hourlyData });
   550→        }
   551→
   552→        // ── Process Rooms ───────────────────────────────
   553→        if (roomsResult.status === 'fulfilled' && roomsResult.value?.data) {
   554→          const transformedRooms: Room[] = Array.isArray(roomsResult.value.data)
   555→            ? roomsResult.value.data.map((r: Record<string, unknown>) => ({
   556→                id: String(r.name || r.id || '').toLowerCase().replace(/\s+/g, '-'),
   557→                name: String(r.room_name || r.name || ''),
   558→                tables: Number(r.no_of_tables || r.tables || 0),
   559→              }))
   560→            : [];
   561→          if (transformedRooms.length > 0) set({ rooms: transformedRooms });
   562→        }
   563→
   564→        // ── Process P&L ─────────────────────────────────
   565→        if (plResult.status === 'fulfilled' && plResult.value?.data) {
   566→          const plDocs = Array.isArray(plResult.value.data) ? plResult.value.data : [];
   567→          if (plDocs.length > 0) {
   568→            // Use the latest P&L document
   569→            const latest = plDocs[0] as Record<string, unknown>;
   570→
   571→            // P&L Summary
   572→            const grossSales = Number(latest.gross_sales || latest.total_revenue || 0);
   573→            const cogs = Number(latest.cogs || latest.cost_of_goods_sold || 0);
   574→            const grossProfit = grossSales - cogs;
   575→            const netProfit = Number(latest.net_profit || grossProfit - Number(latest.total_expenses || 0));
   576→            set({
   577→              plSummary: { grossSales, cogs, grossProfit, netProfit },
   578→            });
   579→
   580→            // Daily P&L from multiple docs
   581→            const dailyData: DailyPL[] = plDocs.slice(0, 7).reverse().map((doc: Record<string, unknown>) => {
   582→              const date = String(doc.date || doc.posting_date || '');
   583→              const dayName = date
   584→                ? new Date(date).toLocaleDateString('sl-SI', { weekday: 'short' })
   585→                : '';
   586→              return {
   587→                day: dayName,
   588→                revenue: Number(doc.gross_sales || doc.total_revenue || 0),
   589→                costs: Number(doc.cogs || doc.total_expenses || 0),
   590→              };
   591→            });
   592→            if (dailyData.length > 0) set({ dailyPL: dailyData });
   593→
   594→            // Expense breakdown — try to extract from P&L doc
   595→            const breakdownData: ExpenseBreakdown[] = [];
   596→            if (latest.breakup || latest.expense_breakdown) {
   597→              const items = Array.isArray(latest.breakup || latest.expense_breakdown)
   598→                ? (latest.breakup || latest.expense_breakdown) as Record<string, unknown>[]
   599→                : [];
   600→              const colors = ['#ef4444', '#f97316', '#eab308', '#8b5cf6', '#10b981'];
   601→              items.forEach((item, i) => {
   602→                breakdownData.push({
   603→                  name: String(item.expense_type || item.category || item.name || ''),
   604→                  value: Number(item.percentage || item.value || 0),
   605→                  color: colors[i % colors.length],
   606→                });
   607→              });
   608→            }
   609→            if (breakdownData.length > 0) set({ expenseBreakdown: breakdownData });
   610→
   611→            // Line items
   612→            const lines: PLLineItem[] = [];
   613→            if (latest.breakup || latest.line_items) {
   614→              const items = Array.isArray(latest.breakup || latest.line_items)
   615→                ? (latest.breakup || latest.line_items) as Record<string, unknown>[]
   616→                : [];
   617→              items.forEach((item) => {
   618→                lines.push({
   619→                  label: String(item.label || item.expense_type || item.name || ''),
   620→                  value: Number(item.amount || item.value || 0),
   621→                  bold: Boolean(item.is_total || item.bold || false),
   622→                });
   623→              });
   624→            }
   625→            if (lines.length > 0) set({ plLineItems: lines });
   626→          }
   627→        }
   628→
   629→        // ── Process Shift / Cashier ─────────────────────
   630→        if (shiftResult.status === 'fulfilled' && shiftResult.value?.data) {
   631→          const openings = Array.isArray(shiftResult.value.data) ? shiftResult.value.data : [];
   632→          if (openings.length > 0) {
   633→            const latestOpening = openings[0] as Record<string, unknown>;
   634→            const openedAt = String(latestOpening.period_start || latestOpening.opening_time || '').slice(0, 5);
   635→            const openedBy = String(latestOpening.user || latestOpening.opened_by || '');
   636→            const openingBalance = Number(latestOpening.opening_amount || latestOpening.balance_details?.[0]?.amount || 0);
   637→
   638→            set({
   639→              shiftInfo: {
   640→                status: 'open',
   641→                openedAt: openedAt || '09:00',
   642→                closesAt: '23:00',
   643→                openedBy: openedBy || 'Administrator',
   644→                openingBalance,
   645→              },
   646→            });
   647→
   648→            // Build cashier data from POS opening entry
   649→            const cashierList: CashierData[] = [];
   650→            const balanceRows = Array.isArray(latestOpening.balance_details)
   651→              ? latestOpening.balance_details as Record<string, unknown>[]
   652→              : [];
   653→            if (balanceRows.length > 0) {
   654→              balanceRows.forEach((row) => {
   655→                cashierList.push({
   656→                  name: String(row.user || openedBy || 'Cashier'),
   657→                  role: 'URY Cashier',
   658→                  openedAt: openedAt || '09:00',
   659→                  status: 'active',
   660→                  openingBalance: Number(row.opening_amount || 0),
   661→                  currentTotal: Number(row.closing_amount || row.opening_amount || 0),
   662→                  cashPayments: Number(row.cash_amount || 0),
   663→                  cardPayments: Number(row.card_amount || 0),
   664→                  upiPayments: Number(row.upi_amount || 0),
   665→                  ordersProcessed: Number(row.orders_processed || 0),
   666→                  room: String(row.room || 'Glavna dvorana'),
   667→                });
   668→              });
   669→            }
   670→            if (cashierList.length > 0) set({ cashiers: cashierList });
   671→          } else {
   672→            // No open shift found
   673→            set({
   674→              shiftInfo: {
   675→                status: 'closed',
   676→                openedAt: '',
   677→                closesAt: '',
   678→                openedBy: '',
   679→                openingBalance: 0,
   680→              },
   681→            });
   682→          }
   683→        }
   684→
   685→        // Mark refresh time
   686→        set({ lastRefreshed: new Date(), isRefreshing: false });
   687→      } catch (err) {
   688→        console.error('Failed to refresh data:', err);
   689→        set({ isRefreshing: false });
   690→      }
   691→    },
   692→
   693→    // ── Targeted Refresh: KPIs only ────────────────────
   694→    refreshKPIs: async () => {
   695→      const { isConnected } = get();
   696→      if (!isConnected) return;
   697→
   698→      try {
   699→        const client = getFrappeClient();
   700→        const invoicesResult = await client.getInvoiceForCashier('Paid', '', 50, 0).catch(() => null);
   701→        if (invoicesResult?.data && Array.isArray(invoicesResult.data)) {
   702→          const invoices = invoicesResult.data as Record<string, unknown>[];
   703→          const totalSales = invoices.reduce((sum, inv) => sum + Number(inv.grand_total || 0), 0);
   704→          const totalOrders = invoices.length;
   705→          const occupiedTables = invoices.filter((inv) => inv.type === 'Dine-in').length;
   706→          const allTables = get().tables;
   707→          set({
   708→            kpis: {
   709→              dailySales: totalSales,
   710→              totalOrders,
   711→              avgBill: totalOrders > 0 ? Math.round(totalSales / totalOrders) : 0,
   712→              occupiedTables,
   713→              totalTables: allTables.length || 32,
   714→            },
   715→          });
   716→        }
   717→      } catch (err) {
   718→        console.error('Failed to refresh KPIs:', err);
   719→      }
   720→    },
   721→
   722→    // ── Targeted Refresh: P&L only ─────────────────────
   723→    refreshPL: async () => {
   724→      const { isConnected } = get();
   725→      if (!isConnected) return;
   726→
   727→      try {
   728→        const client = getFrappeClient();
   729→        const plResult = await client.getDailyPL().catch(() => null);
   730→        if (plResult?.data && Array.isArray(plResult.data)) {
   731→          const plDocs = plResult.data as Record<string, unknown>[];
   732→          if (plDocs.length > 0) {
   733→            const latest = plDocs[0];
   734→            const grossSales = Number(latest.gross_sales || latest.total_revenue || 0);
   735→            const cogs = Number(latest.cogs || latest.cost_of_goods_sold || 0);
   736→            const grossProfit = grossSales - cogs;
   737→            const netProfit = Number(latest.net_profit || grossProfit - Number(latest.total_expenses || 0));
   738→            set({ plSummary: { grossSales, cogs, grossProfit, netProfit } });
   739→
   740→            const dailyData: DailyPL[] = plDocs.slice(0, 7).reverse().map((doc) => {
   741→              const date = String(doc.date || doc.posting_date || '');
   742→              return {
   743→                day: date ? new Date(date).toLocaleDateString('sl-SI', { weekday: 'short' }) : '',
   744→                revenue: Number(doc.gross_sales || doc.total_revenue || 0),
   745→                costs: Number(doc.cogs || doc.total_expenses || 0),
   746→              };
   747→            });
   748→            if (dailyData.length > 0) set({ dailyPL: dailyData });
   749→          }
   750→        }
   751→      } catch (err) {
   752→        console.error('Failed to refresh P&L:', err);
   753→      }
   754→    },

   797→    updateKOTStatus: (id, status) => {

   855→    startAutoRefresh: () => {

   878→      const duration = toast.duration ?? 5000;

   884→        toastTimers.delete(id);

   886→      toastTimers.set(id, timer);

   891→      const timer = toastTimers.get(id);
   892→      if (timer) {

   894→        toastTimers.delete(id);
   895→      }
   896→    },
