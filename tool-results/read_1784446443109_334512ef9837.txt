   480→   480→          const savedState = localStorage.getItem(
   481→   481→            `${kot.name}_${kotitem.name}_strike`
   482→   482→          );
   483→   483→          if (savedState) {
   484→   484→            try {
   485→   485→              kotitem.striked = JSON.parse(savedState);
   486→   486→            } catch (e) {
   487→   487→              kotitem.striked = false;
   488→   488→            }
   489→   489→          }
   490→   490→          this.calculateQty(
   491→   491→            kotitem,
   492→   492→            kotitem.quantity,
   493→   493→            kot.type,
   494→   494→            kotitem.cancelled_qty
   495→   495→          );
   496→   496→        });
   497→   497→      });
   498→   498→    },
   499→   499→    calculateQty(kotitem, qty, type, cancelled_qty) {
   500→   500→      // R37-FIX: Guard against negative quantities from cancelled_qty > qty
   501→   501→      if (type === "Partially cancelled" || type === "Cancelled") {
   502→   502→        kotitem.qty = Math.max(0, qty - cancelled_qty);
   503→   503→      } else {
   504→   504→        kotitem.qty = qty;
   505→   505→      }
   506→   506→    },
   507→   507→    removeAllItemsFromLocalStorage(kot) {
   508→   508→      // Get all keys in local storage
   509→   509→      const keys = Object.keys(localStorage);
   510→   510→      // Remove keys that start with `${kot.name}_`
   511→   511→      keys.forEach((key) => {
   512→   512→        if (key.startsWith(`${kot.name}_`)) {
   513→   513→          localStorage.removeItem(key);
   514→   514→        }
   515→   515→      });
   516→   516→    },
   517→   517→
   518→   518→    updateTimeRemaining() {
   519→   519→      this.kot.forEach((kot) => {
   520→   520→        kot.timeRemaining = this.calculateTimeRemaining(kot.time);
   521→   521→
   522→   522→        const timeRemaining = kot.timeRemaining.split(":");
   523→   523→        const minutes =
   524→   524→          parseInt(timeRemaining[0], 10) * 60 + parseInt(timeRemaining[1], 10);
   525→   525→
   526→   526→        // R36-FIX: Handle NaN from invalid time format — treat as elapsed time exceeded
   527→   527→        const validMinutes = isNaN(minutes) ? Infinity : minutes;
   528→   528→
   529→   529→        if (
   530→   530→          validMinutes === Number(this.kot_alert_time) &&
   531→   531→          kot.type !== "Cancelled" &&
   532→   532→          kot.type !== "Partially cancelled" &&
   533→   533→          !this.notifiedKots.has(kot.name)
   534→   534→        ) {
   535→   535→          this.notifiedKots.add(kot.name);
   536→   536→          this.orderDelayNotify(kot);
   537→   537→        }
   538→   538→        if (validMinutes >= this.kot_alert_time) {
   539→   539→          kot.timecolor = "text-[#DC0000]";
   540→   540→        } else {
   541→   541→          kot.timecolor = "text-black";
   542→   542→        }
   543→   543→      });
   544→   544→    },
   545→   545→    calculateTimeRemaining(targetTime) {
   546→   546→      if (!targetTime || !targetTime.includes(":")) return '— : —';
   547→   547→      const currentTime = new Date();
   548→   548→      const [targetHours, targetMinutes, targetSeconds] = targetTime.split(":");
   549→   549→      let targetDate = new Date(
   550→   550→        currentTime.getFullYear(),
   551→   551→        currentTime.getMonth(),
   552→   552→        currentTime.getDate(),
   553→   553→        targetHours,
   554→   554→        targetMinutes,
   555→   555→        targetSeconds
   556→   556→      );
   557→   557→      // Handle cross-midnight KOTs: if target is in the future, it was created yesterday
   558→   558→      if (targetDate > currentTime) {
   559→   559→        targetDate = new Date(targetDate.getTime() - 86400000);
   560→   560→      }
   561→   561→
   562→   562→      const timeDifference = Math.max(0, currentTime - targetDate);
   563→   563→      const hoursRemaining = Math.floor(timeDifference / 3600000);
   564→   564→      const minutesRemaining = Math.floor((timeDifference % 3600000) / 60000);
   565→   565→
   566→   566→      return `${hoursRemaining} : ${String(minutesRemaining).padStart(2, '0')}`;
   567→   567→    },
   568→   568→    fetchkotwithmasonry() {
   569→   569→      return this.fetchKOT().then(() => {
   570→   570→        this.masonryLoading();
   571→   571→      }).catch((e) => { console.error("KOT fetch failed:", e); });
   572→   572→    },

   584→   584→          }

   590→   590→        this.masonry = null;
   591→   591→      }
   592→   592→      this.$nextTick(() => {
   593→   593→        if (!this.$el) return;
   594→   594→        const grid = this.$el.querySelector(".grid");
   595→   595→        if (!grid) return;
   596→   596→        this.masonry = markRaw(new Masonry(grid, {
   597→   597→          itemSelector: ".masonry-item",
   598→   598→          gutter: 28,
   599→   599→        }));
   600→   600→        this.masonry.layout();
   601→   601→      });
   602→   602→    },

   605→   605→    },

   609→   609→      this.isOnline = true;

   644→   644→    this._fetchInProgress = null;

   653→   653→  mounted() {
   654→   654→    this._isMounted = true;
   655→   655→    window.addEventListener("online", this.handleOnline);
   656→   656→    window.addEventListener("offline", this.handleOffline);

   658→   658→    // R36-FIX: Use route params instead of fragile URL parsing
   659→   659→    const production = this.$route?.params?.production || '';
   660→   660→    const decodedProduction = decodeURIComponent(production);
   661→   661→    this.production = decodedProduction;
   662→   662→

   673→   673→      .then(([sock]) => {

   681→   681→        });

   683→   683→          if (!this._isMounted) return;
   684→   684→          console.warn("Socket disconnected:", reason);
   685→   685→          this.setStatusMessage("Connection lost. Reconnecting...");

   688→   688→          if (!this._isMounted) return;
   689→   689→          this.setStatusMessage("Reconnected");
   690→   690→          this.hideStatusMessageAfterDelay();
   691→   691→        });

   693→   693→        return this.fetchKOT();
   694→   694→      })
   695→   695→      .then(() => {

   701→   701→          if (!this._isMounted) return;

   703→   703→            if (this.audio_alert === 1) {
   704→   704→              this.playAlertSound(doc.audio_file);
   705→   705→            }

   707→   707→            let kottime = localStorage.getItem("kot_time_" + this.production);
   708→   708→            if (doc.last_kot_time !== kottime) {
   709→   709→              // Full refresh needed — skip intermediate mutations
   710→   710→              this.fetchKOT().then(() => { this.masonryLoading(); }).catch((e) => { console.error("KOT fetch failed:", e); });
   711→   711→            } else {
   712→   712→              // R36-FIX: Guard against missing doc.kot to prevent TypeError crash
   713→   713→              if (!doc.kot) {
   714→   714→                this.fetchKOT().then(() => { this.masonryLoading(); }).catch((e) => { console.error("KOT fetch failed:", e); });
   715→   715→                return;
   716→   716→              }
   717→   717→              // Incremental update — deduplicate to avoid duplicate cards
   718→   718→              const existingIndex = this.kot.findIndex(k => k.name === doc.kot.name);
   719→   719→              if (existingIndex !== -1) {
   720→   720→                // R36-FIX: Preserve strikethrough state before Object.assign overwrites kot_items

   732→   732→                this._sortedItemsCache.delete(doc.kot.name);

   734→   734→                const newKot = { isRotated: false, showDiv: false, timecolor: 'text-black', timeRemaining: '— : —', ...doc.kot };
   735→   735→                this.kot.unshift(newKot);
   736→   736→              }

   738→   738→              this.updateTimeRemaining();
   739→   739→              this.masonryLoading();

   753→   753→              } catch (e) {

   756→   756→            }
   757→   757→          } catch (err) {

   759→   759→          }
   760→   760→        };

   762→   762→      })

   764→   764→        console.error("Initialization or authentication error:", error);

   768→   768→  },

   790→   790→  computed: {
   791→   791→    sortedKotItems() {
   792→   792→      // R37-FIX: Pure computed — no side effects on reactive kot objects.
   793→   793→      // Uses instance-level _sortedItemsCache Map (non-reactive) instead of
   794→   794→      // mutating kot._sortedItems / kot._sortKey which caused reactivity issues
   795→   795→      // and stale cache after Object.assign overwrites kot_items in socket handler.

   806→   806→      };
   807→   807→    },
