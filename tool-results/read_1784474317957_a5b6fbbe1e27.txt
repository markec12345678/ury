   229→function debounce(fn, delay) {
   230→    let timer = null;
   231→    return function(...args) {
   232→        clearTimeout(timer);
   233→        timer = setTimeout(() => fn.apply(this, args), delay);
   234→    };

   565→    removeAllItemsFromLocalStorage(kot) {
   566→      // R38-FIX: Remove strike-state keys for this KOT's items by known key pattern
   567→      // instead of iterating ALL localStorage keys (which may include unrelated Frappe keys).
   568→      if (!kot.kot_items) return;
   569→      kot.kot_items.forEach((kotitem) => {
   570→        try {
   571→          localStorage.removeItem(`${kot.name}_${kotitem.name}_strike`);
   572→        } catch (e) {
   573→          if (import.meta.env?.DEV) console.error('localStorage removeItem failed:', e);
   574→        }
   575→      });
   576→    },
   577→
   578→    updateTimeRemaining() {
   579→      this.kot.forEach((kot) => {
   580→        kot.timeRemaining = this.calculateTimeRemaining(kot.time);
   581→
   582→        const timeRemaining = kot.timeRemaining.split(":");
   583→        const minutes =
   584→          parseInt(timeRemaining[0], 10) * 60 + parseInt(timeRemaining[1], 10);
   585→
   586→        // R36-FIX: Handle NaN from invalid time format — treat as elapsed time exceeded
   587→        const validMinutes = isNaN(minutes) ? Infinity : minutes;
   588→
   589→        if (
   590→          validMinutes === Number(this.kot_alert_time) &&
   591→          kot.type !== "Cancelled" &&
   592→          kot.type !== "Partially cancelled" &&
   593→          !this.notifiedKots.has(kot.name)
   594→        ) {
   595→          this.notifiedKots.add(kot.name);
   596→          this.orderDelayNotify(kot);
   597→        }
   598→        // R38-FIX: Guard against empty/falsy kot_alert_time. An empty string
   599→        // coerces to 0 in >= comparison, making ALL KOTs show red time.
   600→        const alertThreshold = Number(this.kot_alert_time);
   601→        if (alertThreshold > 0 && validMinutes >= alertThreshold) {
   602→          kot.timecolor = "text-[#DC0000]";
   603→        } else {
   604→          kot.timecolor = "text-black";
   605→        }
   606→      });
   607→    },
   608→    calculateTimeRemaining(targetTime) {
   609→      if (!targetTime || !targetTime.includes(":")) return '— : —';
   610→      const currentTime = new Date();
   611→      // R39-FIX: Guard against malformed time strings with fewer than 3 parts.
   612→      // Previously, "HH:MM" without seconds would set targetSeconds=undefined,
   613→      // which Date() treats as NaN → Invalid Date → NaN propagation.
   614→      const parts = targetTime.split(":");
   615→      const targetHours = parseInt(parts[0], 10);
   616→      const targetMinutes = parseInt(parts[1], 10);
   617→      const targetSeconds = parseInt(parts[2] || '0', 10);
   618→      if (isNaN(targetHours) || isNaN(targetMinutes) || isNaN(targetSeconds)) return '— : —';
   619→      let targetDate = new Date(
   620→        currentTime.getFullYear(),
   621→        currentTime.getMonth(),
   622→        currentTime.getDate(),
   623→        targetHours,
   624→        targetMinutes,
   625→        targetSeconds
   626→      );
   627→      // Handle cross-midnight KOTs: if target is in the future, it was created yesterday
   628→      if (targetDate > currentTime) {
   629→        targetDate = new Date(targetDate.getTime() - 86400000);
   630→      }
   631→
   632→      const timeDifference = Math.max(0, currentTime - targetDate);
   633→      const hoursRemaining = Math.floor(timeDifference / 3600000);
   634→      const minutesRemaining = Math.floor((timeDifference % 3600000) / 60000);
   635→
   636→      return `${hoursRemaining} : ${String(minutesRemaining).padStart(2, '0')}`;
   637→    },
   638→    // R39-FIX: Retry wrapper for fetchKOT with exponential backoff.
   639→    // Without this, a transient server error leaves the KDS showing stale
   640→    // data indefinitely — the only recovery was a manual page reload.
   641→    fetchKOTWithRetry(maxRetries = 3, initialDelay = 2000) {
   642→      const attempt = (retriesLeft, delay) => {
   643→        return this.fetchKOT().catch(err => {
   644→          if (retriesLeft <= 0 || !this._isMounted) throw err;
   645→          if (import.meta.env?.DEV) console.warn(`fetchKOT failed, retrying in ${delay}ms...`, err);
   646→          return new Promise((resolve, reject) => {
   647→            this._fetchRetryTimer = setTimeout(() => {
   648→              if (!this._isMounted) { reject(err); return; }
   649→              attempt(retriesLeft - 1, Math.min(delay * 2, 30000)).then(resolve).catch(reject);
   650→            }, delay);
   651→          });
   652→        });
   653→      };
   654→      return attempt(maxRetries, initialDelay);
   655→    },
   656→    fetchkotwithmasonry() {
   657→      return this.fetchKOTWithRetry().then(() => {
   658→        this.masonryLoading();
   659→      }).catch((e) => { console.error("KOT fetch failed:", e); });
   660→    },
   661→    redirectToLogin() {
   662→      // R38-FIX: Use Vue Router instead of window.location.href to avoid full page reload
   663→      // and preserve app state. The previous approach caused a hard navigation that
   664→      // lost all Vue state and required a full re-initialization.
   665→      this.$router.push({ name: 'Login', query: { route: this.$route.path } });
   666→    },
   667→    masonryLoading(forceRecreate = false) {
   668→      // R39-FIX: Use non-reactive _masonry instead of reactive masonry from data()
   669→      if (this._masonry && !forceRecreate) {
   670→        this.$nextTick(() => {
   671→          if (this._masonry) {
   672→            this._masonry.reloadItems?.();
   673→            this._masonry.layout();
   674→          }
   675→        });
   676→        return;
   677→      }
   678→      if (this._masonry) {
   679→        this._masonry.destroy();
   680→        this._masonry = null;
   681→      }
   682→      this.$nextTick(() => {
   683→        if (!this.$el) return;
   684→        const grid = this.$el.querySelector(".grid");
   685→        if (!grid) return;
   686→        this._masonry = markRaw(new Masonry(grid, {
   687→          itemSelector: ".masonry-item",
   688→          gutter: 28,
   689→        }));
   690→        this._masonry.layout();
   691→      });
   692→    },
   693→    hideAudioAlertMessage() {
   694→      this.showAudioAlertMessage = false;
   695→    },
   696→    handleOnline() {
   697→      // R37-FIX: Guard against post-unmount execution
   698→      if (!this._isMounted) return;
   699→      this.isOnline = true;
   700→      this.setStatusMessage("You are online");
   701→      this.hideStatusMessageAfterDelay();
   702→      // R39-FIX: Use retry wrapper for transient network errors on reconnect
   703→      this.fetchKOTWithRetry().then(() => {
   704→        if (!this._isMounted) return;
   705→        this.masonryLoading();
   706→      }).catch((e) => { console.error("KOT fetch failed:", e); });
   707→    },
   708→    handleOffline() {
   709→      // R37-FIX: Guard against post-unmount execution
   710→      if (!this._isMounted) return;
   711→      this.isOnline = false;
   712→      this.setStatusMessage("You are Offline");
   713→    },
   714→    setStatusMessage(message) {
   715→      this.statusMessage = message;
   716→    },
   717→    hideStatusMessageAfterDelay() {
   718→      if (this._statusTimeout) clearTimeout(this._statusTimeout);
   719→      this._statusTimeout = setTimeout(() => {
   720→        // R37-FIX: Guard against mutating unmounted component
   721→        if (!this._isMounted) return;
   722→        this.statusMessage = "";
   723→      }, 3000);
   724→    },
   725→    handleTransitionEnd() {
   726→      if (!this.isOnline) {
   727→        // Reset the status message after transition end
   728→        this.setStatusMessage("");
   729→      }
   730→    },
   731→  },
   732→  created() {
   733→    // R36-FIX: Initialize as non-reactive instance properties (no Proxy overhead)
   734→    this._isMounted = false;
   735→    this._fetchInProgress = null;
   736→    this.notifiedKots = new Set();
   737→    // API client as non-reactive instance property (avoids Proxy overhead)
   738→    this.call = markRaw(frappe.call());
   739→    // R37-FIX: socketHandler as non-reactive to avoid unnecessary Proxy overhead
   740→    this.socketHandler = null;
   741→    // R37-FIX: sortedItems cache map — avoids mutating reactive kot objects in computed
   742→    this._sortedItemsCache = new Map();
   743→    // R39-FIX: masonry as non-reactive instance property — Masonry objects are large
   744→    // and should not be tracked by Vue's reactivity system. Previously in data(),
   745→    // it triggered unnecessary Proxy overhead and could cause cascading re-renders
   746→    // when the masonry reference changed during reloadItems/layout calls.
   747→    this._masonry = null;
   748→    // R39-FIX: Track socket init retry timer for cleanup
   749→    this._socketRetryTimer = null;
   750→  },
   751→  mounted() {
   752→    this._isMounted = true;
   753→    window.addEventListener("online", this.handleOnline);
   754→    window.addEventListener("offline", this.handleOffline);
   755→    document.addEventListener("click", this.hideAudioAlertMessage);
   756→    // R36-FIX: Use route params instead of fragile URL parsing
   757→    const production = this.$route?.params?.production || '';
   758→    const decodedProduction = decodeURIComponent(production);
   759→    this.production = decodedProduction;
   760→
   761→    const debouncedMasonry = debounce(() => { this.masonryLoading(); }, 150);
   762→    this._resizeHandler = debouncedMasonry;
   763→    window.addEventListener("resize", this._resizeHandler);
   764→    this.masonryLoading();
   765→
   766→    // R39-FIX: Handle bfcache restore — when the browser restores this page from
   767→    // back-forward cache, the socket may be stale and timers may not fire.
   768→    // Re-initialize socket and re-fetch KOT data on pageshow with persisted=true.
   769→    this._pageshowHandler = (event) => {
   770→      if (event.persisted) {
   771→        // Page was restored from bfcache — re-fetch data and re-init socket
   772→        // R39-FIX: Use retry wrapper for bfcache restore re-fetch
   773→        this.fetchKOTWithRetry().then(() => { this.masonryLoading(); }).catch(() => {});
   774→        if (this._socket && !this._socket.connected) {
   775→          this._socket.connect();
   776→        }
   777→      }
   778→    };
   779→    window.addEventListener('pageshow', this._pageshowHandler);
   780→
   781→    // R39-FIX: Socket init with retry — if fetchSiteName fails (network issue,
   782→    // server down), retry with exponential backoff instead of giving up forever.
   783→    // Without this, a transient site-name fetch failure would leave the KDS
   784→    // without real-time updates until a full page reload.
   785→    const initSocketWithRetry = (retries = 0) => {
   786→      return initializeSocket().then(sock => {
   787→        if (sock) return sock;
   788→        // initializeSocket returned null (siteName fetch failed) — retry
   789→        const delay = Math.min(2000 * Math.pow(2, retries), 30000);
   790→        if (!this._isMounted) return null;
   791→        return new Promise(resolve => {
   792→          this._socketRetryTimer = setTimeout(() => {
   793→            if (!this._isMounted) { resolve(null); return; }
   794→            resolve(initSocketWithRetry(retries + 1));
   795→          }, delay);
   796→        });
   797→      });
   798→    };
   799→    this._socketInitPromise = initSocketWithRetry();
   800→
   801→    // Wait for both socket init and auth before attaching listeners
   802→    Promise.all([this._socketInitPromise, this.auth()])
   803→      .then(([sock]) => {
   804→        // R39-FIX: If component unmounted during init, disconnect the orphaned
   805→        // socket to prevent a connection leak. Previously, the socket created by
   806→        // initializeSocket() was never stored in this._socket and thus never
   807→        // disconnected in beforeUnmount.
   808→        if (!this._isMounted) {
   809→          if (sock) sock.disconnect();
   810→          return;
   811→        }
   812→        this._socket = sock;
   813→        if (this._socket) this._socket.on('connect_error', (err) => {
   814→          if (!this._isMounted) return;
   815→          console.error("Socket connection error:", err);
   816→          this.setStatusMessage("Connection error. Retrying...");
   817→        });
   818→        if (this._socket) this._socket.on('disconnect', (reason) => {
   819→          if (!this._isMounted) return;
   820→          console.warn("Socket disconnected:", reason);
   821→          this.setStatusMessage("Connection lost. Reconnecting...");
   822→        });
   823→        if (this._socket) this._socket.on('connect', () => {
   824→          if (!this._isMounted) return;
   825→          this.setStatusMessage("Reconnected");
   826→          this.hideStatusMessageAfterDelay();
   827→          // R38-FIX: Re-fetch KOT data after reconnect to sync any missed updates
   828→          // during disconnection period. Without this, the UI could show stale KOTs.
   829→          // R39-FIX: Use retry wrapper for transient errors on reconnect
   830→          this.fetchKOTWithRetry().then(() => { this.masonryLoading(); }).catch(() => {});
   831→        });
   832→
   833→        // R39-FIX: Use retry wrapper for initial fetch — transient server errors
   834→        // should not leave the KDS permanently blank
   835→        return this.fetchKOTWithRetry();
   836→      })
   837→      .then(() => {
   838→        if (!this._isMounted) return;
   839→        if (this.audio_alert === 1) {
   840→          this.showAudioAlertMessage = true;
   841→        }
   842→        this.socketHandler = (doc) => {
   843→          if (!this._isMounted) return;
   844→          // R39-FIX: Guard against null/undefined doc from malformed socket messages
   845→          if (!doc) return;
   846→          try {
   847→            if (this.audio_alert === 1) {
   848→              this.playAlertSound(doc.audio_file);
   849→            }
   850→            // R36-FIX: Namespace localStorage key per production station to avoid cross-tab collision
   851→            // R39-FIX: Normalize null kottime from localStorage — if getItem returns null
   852→            // (no previous value stored) and doc.last_kot_time is also null, they'd be
   853→            // equal and fall through to the incremental path incorrectly. Treat null
   854→            // localStorage value as a signal to do a full refresh.
   855→            let kottime = localStorage.getItem("kot_time_" + this.production);
   856→            if (doc.last_kot_time !== kottime || kottime === null) {
   857→              // Full refresh needed — skip intermediate mutations
   858→              this.fetchKOT().then(() => { this.masonryLoading(); }).catch((e) => { console.error("KOT fetch failed:", e); });
   859→            } else {
   860→              // R36-FIX: Guard against missing doc.kot to prevent TypeError crash
   861→              if (!doc.kot) {
   862→                this.fetchKOT().then(() => { this.masonryLoading(); }).catch((e) => { console.error("KOT fetch failed:", e); });
   863→                return;
   864→              }
   865→              // Incremental update — deduplicate to avoid duplicate cards
   866→              const existingIndex = this.kot.findIndex(k => k.name === doc.kot.name);
   867→              if (existingIndex !== -1) {
   868→                // R36-FIX: Preserve strikethrough state before Object.assign overwrites kot_items
   869→                const strikeMap = new Map(
   870→                  (this.kot[existingIndex].kot_items || []).map(i => [i.name, i.striked])
   871→                );
   872→                Object.assign(this.kot[existingIndex], { timecolor: 'text-black', timeRemaining: '— : —', ...doc.kot });
   873→                // Restore strikethrough state after Object.assign
   874→                if (this.kot[existingIndex].kot_items) {
   875→                  this.kot[existingIndex].kot_items.forEach(i => {
   876→                    if (strikeMap.has(i.name)) i.striked = strikeMap.get(i.name);
   877→                  });
   878→                }
   879→                // R37-FIX: Invalidate sorted cache for this KOT since items may have changed
   880→                this._sortedItemsCache.delete(doc.kot.name);
   881→              } else {
   882→                // R39-FIX: Removed showDiv: false — dead code
   883→                const newKot = { isRotated: false, timecolor: 'text-black', timeRemaining: '— : —', ...doc.kot };
   884→                this.kot.unshift(newKot);
   885→              }
   886→              this.updateQtyColorTable();
   887→              this.updateTimeRemaining();
   888→              this.masonryLoading();
   889→            }
   890→            if (this._cancelTimeout) clearTimeout(this._cancelTimeout);
   891→            // R39-FIX: Only schedule cancel re-fetch if the KOT is actually a cancellation.
   892→            // Previously, _cancelTimeout was set for ALL socket events, causing an
   893→            // unnecessary re-fetch 1.5s after every non-cancel update. Now we only
   894→            // set the timeout when doc.kot.type is "Cancelled", matching the check inside.
   895→            if (doc.kot && doc.kot.type === "Cancelled") {
   896→              this._cancelTimeout = setTimeout(() => {
   897→                if (!this._isMounted) return;
   898→                this.fetchKOT().then(() => {
   899→                  this.masonryLoading();
   900→                }).catch((e) => { console.error("KOT fetch failed:", e); });
   901→              }, 1500);
   902→            }
   903→            if (doc.kot) {
   904→              try {
   905→                localStorage.setItem("kot_time_" + this.production, doc.kot.time);
   906→              } catch (e) {
   907→                if (import.meta.env?.DEV) console.error('localStorage write failed:', e);
   908→              }
   909→            }
   910→          } catch (err) {
   911→            if (import.meta.env?.DEV) console.error("Socket handler error:", err);
   912→          }
   913→        };
   914→        if (this._socket) this._socket.on(this.kot_channel, this.socketHandler);
   915→      })
   916→      .catch((error) => {
   917→        console.error("Initialization or authentication error:", error);
   918→        if (this._isMounted) this.showModal = true;
   919→      });
   920→    this.timer = setInterval(this.updateTimeRemaining, 60000);
   921→  },
   922→  beforeUnmount() {
   923→    this._isMounted = false;
   924→    window.removeEventListener("online", this.handleOnline);
   925→    window.removeEventListener("offline", this.handleOffline);
   926→    document.removeEventListener("click", this.hideAudioAlertMessage);
   927→    window.removeEventListener("resize", this._resizeHandler);
   928→    if (this._pageshowHandler) window.removeEventListener('pageshow', this._pageshowHandler);
   929→    if (this.socketHandler && this._socket) {
   930→      this._socket.off(this.kot_channel, this.socketHandler);
   931→    }
   932→    if (this._socket) {
   933→      this._socket.off('connect_error');
   934→      this._socket.off('disconnect');
   935→      this._socket.off('connect');
   936→      this._socket.disconnect();
   937→    }
   938→    this._socketInitPromise = null;
   939→    if (this._cancelTimeout) clearTimeout(this._cancelTimeout);
   940→    if (this._statusTimeout) clearTimeout(this._statusTimeout);
   941→    if (this._socketRetryTimer) clearTimeout(this._socketRetryTimer);
   942→    if (this._fetchRetryTimer) clearTimeout(this._fetchRetryTimer);
   943→    if (this.timer) clearInterval(this.timer);
   944→    if (this._alertAudio) { this._alertAudio.pause(); this._alertAudio = null; }
   945→    // R39-FIX: Destroy non-reactive masonry instance on unmount
   946→    if (this._masonry) {
   947→      this._masonry.destroy();
   948→      this._masonry = null;
   949→    }
   950→  },
   951→  computed: {
   952→    // R39-FIX: sortedKotItems moved from computed (which returned a function) to
   953→    // a method. Previously, the computed returned a new function object every time
   954→    // its reactive dependencies (this.kot) changed, which happened every minute
   955→    // via updateTimeRemaining. Each new function object was unnecessary churn.
   956→    // As a method, the function identity is stable and the manual _sortedItemsCache
   957→    // still provides memoization. The template call `sortedKotItems(kot)` works
   958→    // identically for both computed-returning-function and method.
   959→    visibleKots() {
   960→      // R39-FIX: Removed !kot.showDiv filter — showDiv was always false (dead code).
   961→      // When production is empty (root route "/"), show ALL KOTs instead of filtering
   962→      // by empty string which would exclude KOTs that have a station assigned.
   963→      if (!this.production) {
   964→        return this.kot;
   965→      }
   966→      return this.kot.filter(kot => kot.production === this.production);
   967→    },
   968→  },
