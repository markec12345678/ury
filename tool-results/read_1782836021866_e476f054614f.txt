   310→// ---- Period Helpers ----
   311→
   312→function getPreviousPeriodDates(
   313→  period: ReportPeriod,
   314→  customFromDate?: string | null,
   315→  customToDate?: string | null
   316→): { prevFrom: string; prevTo: string } {
   317→  const today = new Date();
   318→
   319→  // If custom date range, calculate the previous equivalent period
   320→  if (customFromDate && customToDate) {
   321→    const from = new Date(customFromDate);
   322→    const to = new Date(customToDate);
   323→    const diffMs = to.getTime() - from.getTime();
   324→    const prevTo = new Date(from.getTime() - 1); // Day before from
   325→    const prevFrom = new Date(prevTo.getTime() - diffMs);
   326→    return {
   327→      prevFrom: prevFrom.toISOString().split('T')[0],
   328→      prevTo: prevTo.toISOString().split('T')[0],
   329→    };
   330→  }
   331→
   332→  switch (period) {
   333→    case 'daily':
   334→    case 'yesterday': {
   335→      // Previous day
   336→      const prev = new Date(today);
   337→      prev.setDate(prev.getDate() - 1);
   338→      const dateStr = prev.toISOString().split('T')[0];
   339→      return { prevFrom: dateStr, prevTo: dateStr };
   340→    }
   341→    case 'weekly': {
   342→      // Previous week (same day range)
   343→      const prevWeekStart = new Date(today);
   344→      prevWeekStart.setDate(prevWeekStart.getDate() - 7 - prevWeekStart.getDay());
   345→      const prevWeekEnd = new Date(prevWeekStart);
   346→      prevWeekEnd.setDate(prevWeekEnd.getDate() + 6);
   347→      return {
   348→        prevFrom: prevWeekStart.toISOString().split('T')[0],
   349→        prevTo: prevWeekEnd.toISOString().split('T')[0],
   350→      };
   351→    }
   352→    case 'monthly': {
   353→      // Previous month
   354→      const prevMonth = new Date(today.getFullYear(), today.getMonth() - 1, 1);
   355→      const prevMonthEnd = new Date(today.getFullYear(), today.getMonth(), 0);
   356→      return {
   357→        prevFrom: prevMonth.toISOString().split('T')[0],
   358→        prevTo: prevMonthEnd.toISOString().split('T')[0],
   359→      };
   360→    }
   361→    case 'last_month': {
   362→      // Two months ago
   363→      const twoMonthsAgo = new Date(today.getFullYear(), today.getMonth() - 2, 1);
   364→      const twoMonthsAgoEnd = new Date(today.getFullYear(), today.getMonth() - 1, 0);
   365→      return {
   366→        prevFrom: twoMonthsAgo.toISOString().split('T')[0],
   367→        prevTo: twoMonthsAgoEnd.toISOString().split('T')[0],
   368→      };
   369→    }
   370→    case 'last_7_days': {
   371→      const prevEnd = new Date(today);
   372→      prevEnd.setDate(prevEnd.getDate() - 8);
   373→      const prevStart = new Date(prevEnd);
   374→      prevStart.setDate(prevStart.getDate() - 6);
   375→      return {
   376→        prevFrom: prevStart.toISOString().split('T')[0],
   377→        prevTo: prevEnd.toISOString().split('T')[0],
   378→      };
   379→    }
   380→    case 'last_30_days': {
   381→      const prevEnd = new Date(today);
   382→      prevEnd.setDate(prevEnd.getDate() - 31);
   383→      const prevStart = new Date(prevEnd);
   384→      prevStart.setDate(prevStart.getDate() - 29);
   385→      return {
   386→        prevFrom: prevStart.toISOString().split('T')[0],
   387→        prevTo: prevEnd.toISOString().split('T')[0],
   388→      };
   389→    }
   390→    default: {
   391→      const prev = new Date(today);
   392→      prev.setDate(prev.getDate() - 1);
   393→      const dateStr = prev.toISOString().split('T')[0];
   394→      return { prevFrom: dateStr, prevTo: dateStr };
   395→    }
   396→  }
   397→}

   401→function escapeCsvField(field: string | number): string {
   402→  const str = String(field);
   403→  if (str.includes(',') || str.includes('"') || str.includes('\n')) {
   404→    return `"${str.replace(/"/g, '""')}"`;
   405→  }
   406→  return str;
   407→}
   408→
   409→function csvRow(...fields: (string | number)[]): string {
   410→  return fields.map(escapeCsvField).join(',');
   411→}
   412→
   413→function downloadCsv(content: string, filename: string): void {
   414→  const BOM = '\uFEFF'; // UTF-8 BOM for Excel compatibility
   415→  const blob = new Blob([BOM + content], { type: 'text/csv;charset=utf-8;' });
   416→  const url = URL.createObjectURL(blob);
   417→  const link = document.createElement('a');
   418→  link.setAttribute('href', url);
   419→  link.setAttribute('download', filename);
   420→  document.body.appendChild(link);
   421→  link.click();
   422→  document.body.removeChild(link);
   423→  URL.revokeObjectURL(url);
   424→}

   426→function generateSalesCsv(report: SalesReport): string {
   427→  const lines: string[] = [];
   428→
   429→  // Header
   430→  lines.push(csvRow('URY Restaurant - Sales Report'));
   431→  lines.push(csvRow('Period', `${report.from_date} to ${report.to_date}`));
   432→  lines.push(csvRow('Generated', new Date().toLocaleString()));
   433→  lines.push('');
   434→
   435→  // Summary
   436→  lines.push(csvRow('--- Summary ---'));
   437→  lines.push(csvRow('Metric', 'Value'));
   438→  lines.push(csvRow('Total Revenue', report.summary.total_revenue.toFixed(2)));
   439→  lines.push(csvRow('Total Orders', report.summary.total_orders));
   440→  lines.push(csvRow('Avg Order Value', report.summary.avg_order_value.toFixed(2)));
   441→  lines.push(csvRow('Net Revenue', report.summary.net_revenue.toFixed(2)));
   442→  lines.push(csvRow('Total Tax', report.summary.total_tax.toFixed(2)));
   443→  lines.push(csvRow('Unique Customers', report.summary.unique_customers));
   444→  lines.push('');
   445→
   446→  // Item Sales
   447→  if (report.item_sales.length > 0) {
   448→    lines.push(csvRow('--- Item-wise Sales ---'));
   449→    lines.push(csvRow('Item', 'Qty', 'Avg Rate', 'Total Amount'));
   450→    report.item_sales.forEach((item) => {
   451→      lines.push(csvRow(item.item_name, item.total_qty, item.avg_rate.toFixed(2), item.total_amount.toFixed(2)));
   452→    });
   453→    lines.push('');
   454→  }
   455→
   456→  // Order Type Sales
   457→  if (report.order_type_sales.length > 0) {
   458→    lines.push(csvRow('--- Order Type Breakdown ---'));
   459→    lines.push(csvRow('Order Type', 'Orders', 'Revenue'));
   460→    report.order_type_sales.forEach((ot) => {
   461→      lines.push(csvRow(ot.order_type, ot.order_count, ot.revenue.toFixed(2)));
   462→    });
   463→    lines.push('');
   464→  }
   465→
   466→  // Cancelled Orders
   467→  if (report.cancelled_orders.count > 0) {
   468→    lines.push(csvRow('--- Cancelled Orders ---'));
   469→    lines.push(csvRow('Count', 'Amount'));
   470→    lines.push(csvRow(report.cancelled_orders.count, report.cancelled_orders.amount.toFixed(2)));
   471→    lines.push('');
   472→  }
   473→
   474→  return lines.join('\n');
   475→}

   509→    });
   510→    lines.push('');
   511→  }
   512→
   513→  return lines.join('\n');
