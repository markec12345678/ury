   234→// R41-FIX: Add cancel() method to debounce return value so pending timers

   236→function debounce(fn, delay) {
   237→    let timer = null;
   238→    const debounced = function(...args) {
   239→        clearTimeout(timer);
   240→        timer = setTimeout(() => fn.apply(this, args), delay);
   241→    };
   242→    debounced.cancel = () => { clearTimeout(timer); timer = null; };
   243→    return debounced;
   244→}

   783→    fetchKOTWithRetry(maxRetries = 3, initialDelay = 2000) {
   784→      // R41-FIX: Clear any pending retry timer before starting a new retry chain.
   785→      // Without this, calling fetchKOTWithRetry() while a previous retry timer is
   786→      // pending would leave the old timer running, potentially causing duplicate
   787→      // fetchKOT calls when both timers fire.
   788→      if (this._fetchRetryTimer) { clearTimeout(this._fetchRetryTimer); this._fetchRetryTimer = null; }
   789→      // R42-FIX: Bump generation counter so any still-running retry chain from
   790→      // a previous fetchKOTWithRetry call knows it's been superseded. Without
   791→      // this, clearing the timer orphans the old promise — its resolve/reject
   792→      // callbacks are never called, leaking the promise and its closures.
   793→      const generation = ++this._fetchGeneration;
   794→      const attempt = (retriesLeft, delay) => {
   795→        // If a newer fetchKOTWithRetry has started, abort this chain
   796→        if (generation !== this._fetchGeneration) {
   797→          return Promise.reject(new Error('Superseded by newer fetchKOTWithRetry call'));
   798→        }
   799→        return this.fetchKOT().catch(err => {
   800→          if (retriesLeft <= 0 || !this._isMounted || generation !== this._fetchGeneration) throw err;
   801→          if (import.meta.env?.DEV) console.warn(`fetchKOT failed, retrying in ${delay}ms...`, err);
   802→          return new Promise((resolve, reject) => {
   803→            this._fetchRetryTimer = setTimeout(() => {
   804→              if (!this._isMounted || generation !== this._fetchGeneration) { reject(err); return; }
   805→              attempt(retriesLeft - 1, Math.min(delay * 2, 30000)).then(resolve).catch(reject);
   806→            }, delay);
   807→          });
   808→        });
   809→      };
   810→      return attempt(maxRetries, initialDelay);
   811→    },
