    20→def _get_order_invoice_doc(table=None, invoiceNo=None, order_type=None, is_payment=None):
    21→    """Internal: returns POS Invoice document object for server-side manipulation.
    22→    Used by sync_order and make_invoice which need the doc, not a dict."""
    23→    # Verify user has access to the table's room (BE-R36-017)
    24→    if table:
    25→        from ury.ury_pos.api import _get_user_branch_rooms
    26→        table_room = frappe.db.get_value("URY Table", table, "restaurant_room")
    27→        if table_room:
    28→            user_rooms = [r.room for r in _get_user_branch_rooms() if r.room]
    29→            if user_rooms and table_room not in user_rooms:
    30→                frappe.throw(_("You do not have access to this table's room"), frappe.PermissionError)
    31→
    32→    if table:
    33→        # Lock the table row to prevent concurrent invoice creation (BE-R36-001)
    34→        frappe.db.sql(
    35→            "SELECT name FROM `tabURY Table` WHERE name = %s FOR UPDATE",
    36→            (table,), as_dict=True
    37→        )
    38→        
    39→        if is_payment == "Payments":
    40→            invoice_name = frappe.get_value(
    41→                "POS Invoice", dict(restaurant_table=table, docstatus=0, name=invoiceNo)
    42→            )
    43→            
    44→        else:
    45→            if invoiceNo:
    46→                invoice_name = frappe.get_value(
    47→                    "POS Invoice",
    48→                    dict(restaurant_table=table, docstatus=0, name=invoiceNo),
    49→                )
    50→               
    51→            else:
    52→                invoice_name = frappe.get_value(
    53→                    "POS Invoice",
    54→                    dict(restaurant_table=table, docstatus=0, invoice_printed=0),
    55→                )
    56→                
    57→        branch, menu_name, restaurant = get_restaurant_and_menu_name(table)
    58→
    59→        if invoice_name:
    60→            invoice = frappe.get_doc("POS Invoice", invoice_name)
    61→
    62→        else:
    63→            invoice = frappe.new_doc("POS Invoice")
    64→
    65→            invoice.naming_series = frappe.db.get_value(
    66→                "URY Restaurant", restaurant, "invoice_series_prefix"
    67→            )
    68→
    69→            invoice.is_pos = 1
    70→            invoice.update_stock = 1
    71→            invoice.restaurant = restaurant
    72→            invoice.branch = branch
    73→
    74→            is_take_away = frappe.db.get_value("URY Table", table, "is_take_away")
    75→            if is_take_away == 1:
    76→                invoice.order_type = "Take Away"
    77→            else:
    78→                invoice.order_type= "Dine In"
    79→
    80→        invoice.taxes_and_charges = frappe.db.get_value(
    81→            "URY Restaurant", restaurant, "default_tax_template"
    82→        )
    83→
    84→        invoice.selling_price_list = frappe.db.get_value(
    85→            "Price List", dict(restaurant_menu=menu_name, enabled=1)
    86→        )
    87→
    88→    else:
    89→
    90→        invoice_name = frappe.get_value(
    91→                "POS Invoice", dict(docstatus=0, name=invoiceNo)
    92→            )
    93→            
    94→        if invoice_name:
    95→            invoice = frappe.get_doc("POS Invoice", invoice_name)
    96→            
    97→
    98→        else:
    99→            invoice = frappe.new_doc("POS Invoice")
   100→            invoice.is_pos = 1
   101→            invoice.update_stock = 1
   102→        
   103→        branch = _get_user_branch()
   104→        restaurant = frappe.db.get_value("URY Restaurant", {"branch": branch}, "name")
   105→
   106→        menu=get_menu_name(order_type)
   107→
   108→        if (order_type == "Aggregators" and frappe.db.get_value("Branch", branch, "custom_no_taxes") == 0) or order_type != "Aggregators":
   109→            invoice.taxes_and_charges = frappe.db.get_value("URY Restaurant", restaurant, "default_tax_template")
   110→        
   111→        invoice.selling_price_list = frappe.db.get_value(
   112→            "Price List", dict(restaurant_menu=menu, enabled=1)
   113→        )
   114→        # R43-FIX: Set branch on non-table invoices to prevent branchless invoices
   115→        invoice.branch = branch
   116→
   117→    return invoice

   163→@frappe.whitelist()
   164→def sync_order(
   165→    items,
   166→    cashier,
   167→    owner,
   168→    mode_of_payment,
   169→    customer,
   170→    no_of_pax,
   171→    last_invoice,
   172→    waiter,
   173→    pos_profile,
   174→    last_modified_time=None,
   175→    table=None,
   176→    invoice=None,
   177→    comments=None,
   178→    order_type=None,
   179→    aggregator_id=None,
   180→    room=None
   181→):
   182→    frappe.only_for("Restaurant Manager", "Restaurant User", "Cashier")
   183→    # Validate pos_profile belongs to user's branch (BE-R36-002)
   184→    pos_profile_branch = frappe.db.get_value("POS Profile", pos_profile, "branch")
   185→    user_branch = _get_user_branch()
   186→    if pos_profile_branch != user_branch:
   187→        frappe.throw(_("POS Profile does not belong to your branch"), frappe.PermissionError)
   188→    user_role = frappe.get_roles()
   189→    billing_roles = frappe.get_all(
   190→        "POS Profile Role",
   191→        filters={"parent": pos_profile},
   192→        fields=["role"],
   193→        pluck="role",
   194→    )
   195→    billing_user = bool(set(user_role).intersection(billing_roles))
   196→
   197→    # Check if the last invoice was already billed
   198→    if (
   199→        last_invoice
   200→        and frappe.db.get_value("POS Invoice", last_invoice, "invoice_printed") == 1
   201→        and (not billing_user)
   202→    ):
   203→        frappe.msgprint(
   204→            title=_("Invoice Already Billed"),
   205→            indicator="red",
   206→            msg=_("This order has already been billed. Please reload the page."),
   207→        )
   208→        return {"status": "Failure"}
   209→
   210→    # R41-FIX: Use _get_order_invoice_doc to get the document object, not a dict.
   211→    # The public get_order_invoice returns a dict for frontend safety (BE-R36-003),
   212→    # but sync_order needs the document to call .save() etc.
   213→    invoice = _get_order_invoice_doc(table, invoice, order_type)
   214→
   215→    # R41-FIX: Validate last_invoice belongs to user's branch
   216→    if last_invoice:
   217→        last_inv_branch = frappe.db.get_value("POS Invoice", last_invoice, "branch")
   218→        if last_inv_branch and last_inv_branch != user_branch:
   219→            frappe.throw(_("Invoice does not belong to your branch"), frappe.PermissionError)
   220→
   221→    if last_invoice and last_modified_time:
   222→        lastModifiedTime = invoice.modified
   223→        from datetime import datetime
   224→
   225→        if isinstance(last_modified_time, str):
   226→            try:
   227→                last_modified_time = datetime.strptime(
   228→                    last_modified_time, "%Y-%m-%d %H:%M:%S.%f"
   229→                )
   230→            except ValueError:
   231→                last_modified_time = datetime.strptime(
   232→                    last_modified_time, "%Y-%m-%d %H:%M:%S"
   233→                )
   234→        if isinstance(lastModifiedTime, str):
   235→            try:
   236→                lastModifiedTime = datetime.strptime(
   237→                    lastModifiedTime, "%Y-%m-%d %H:%M:%S.%f"
   238→                )
   239→            except ValueError:
   240→                lastModifiedTime = datetime.strptime(
   241→                    lastModifiedTime, "%Y-%m-%d %H:%M:%S"
   242→                )
   243→        if lastModifiedTime != last_modified_time:
   244→            frappe.msgprint(
   245→                title=_("Order has been modified"),
   246→                indicator="red",
   247→                msg=_("This order has been modified. Please reload the page to retrieve the latest edits."),
   248→            )
   249→            return {"status": "Failure"}
   250→    else:
   251→        if invoice.name and invoice.invoice_printed == 0 and not billing_user:
   252→            frappe.msgprint(
   253→                title=_("Table Occupied"),
   254→                indicator="red",
   255→                msg=_("{0} is already occupied. Please refresh the page.").format(table),
   256→            )
   257→            return {"status": "Failure"}
   258→
   259→    if not customer:
   260→        frappe.throw(_("Please enter valid customer details"))
   261→    else:
   262→        invoice.customer = customer
   263→
   264→    if order_type:
   265→        invoice.order_type = order_type
   266→
   267→    invoice.mobile_number = frappe.db.get_value("Customer", customer, "mobile_number")
   268→    if comments:
   269→        invoice.custom_comments = comments
   270→    invoice.no_of_pax = no_of_pax
   271→    invoice.pos_profile = pos_profile
   272→    invoice.cashier = cashier
   273→    invoice.waiter = waiter
   274→    invoice.custom_aggregator_id = aggregator_id
   275→    invoice.custom_restaurant_room =room
   276→    invoice.restaurant_table = table
   277→    
   278→    if order_type == "Aggregators":
   279→        price_list = frappe.db.get_value("Aggregator Settings",{"customer": customer, "parent": invoice.branch, "parenttype": "Branch"},"price_list",)
   280→        
   281→        if not price_list:
   282→            frappe.throw(_("Price list for customer {0} in branch {1} not found in Aggregator Settings.").format(customer, invoice.branch))
   283→    else:
   284→        price_list = invoice.selling_price_list
   285→
   286→    # dummy payment
   287→    if invoice.invoice_created == 0:
   288→        invoice.append(
   289→            "payments",
   290→            dict(mode_of_payment=mode_of_payment, amount=invoice.grand_total),
   291→        )
   292→        invoice.invoice_created = 1
   293→
   294→    past_item = []
   295→    for item in invoice.items:
   296→        previous_item = {
   297→            "item_code": item.item_code,
   298→            "item_name": item.item_name,
   299→            "qty": item.qty,
   300→            "comments": "",
   301→        }
   302→        past_item.append(previous_item)
   303→        
   304→
   305→    # Conditional checking for 'items' type:
   306→    # - 'ury': JSON passed, hence using isinstance
   307→    # - 'ury_pos': Already formatted list, hence using else
   308→    if isinstance(items, str):
   309→        items = json.loads(items)
   310→    invoice.items = []
   311→    
   312→    menu = frappe.db.get_value("URY Menu", {"branch": invoice.branch}, "name")
   313→
   314→    # Batch-fetch courses, prices, and cost_center to avoid N+1 queries
   315→    item_codes = [d.get("item") for d in items]
   316→    item_codes_unique = list(set(item_codes))
   317→
   318→    courses = {}
   319→    if item_codes_unique:
   320→        course_rows = frappe.db.get_all(
   321→            "URY Menu Item",
   322→            filters={"item": ("in", item_codes_unique), "parent": menu},
   323→            fields=["item", "course"],
   324→        )
   325→        courses = {r.item: r.course for r in course_rows}
   326→
   327→    prices = {}
   328→    if item_codes_unique:
   329→        price_rows = frappe.db.get_all(
   330→            "Item Price",
   331→            filters={"item_code": ("in", item_codes_unique), "price_list": price_list},
   332→            fields=["item_code", "price_list_rate"],
   333→        )
   334→        prices = {r.item_code: r.price_list_rate for r in price_rows}
   335→
   336→    cost_center = frappe.db.get_value("POS Profile", pos_profile, "cost_center")
   337→
   338→    for d in items:
   339→        course = courses.get(d.get("item"))
   340→        rate = prices.get(d.get("item"))
   341→        if not rate:
   342→            frappe.throw(_("No item price found for Item: {0} in Price List: {1}. Please check the price list settings.").format(d.get("item"), price_list))
   343→
   344→        invoice.append(
   345→            "items",
   346→            dict(
   347→                item_code=d.get("item"),
   348→                item_name=d.get("item_name"),
   349→                qty=d.get("qty"),
   350→                **({"custom_course": course} if course else {}),
   351→                comment=d.get("comment"),
   352→                rate=rate,
   353→                price_list_rate=rate,
   354→                base_price_list_rate=rate,
   355→                cost_center=cost_center,
   356→            ),
   357→        )
   358→
   359→    try:
   360→        invoice.save()
   361→    except Exception as e:
   362→        frappe.log_error(frappe.get_traceback(), "Invoice Save Error")
   363→        frappe.throw(_("An error occurred. Please check the error log."))
   364→
   365→
   366→    try:
   367→        kot_execute(invoice.name, customer, table, items, past_item, comments)
   368→
   369→    except Exception as e:
   370→        # Log the KOT error but also notify the user — kitchen won't see this order
   371→        error_msg = f"KOT Creation Failed: {str(e)}"
   372→        frappe.log_error(f"{error_msg}\n{frappe.get_traceback()}", "KOT Error")
   373→        frappe.msgprint(
   374→            title=_("KOT Creation Failed"),
   375→            indicator="orange",
   376→            msg=_("The order was saved but the KOT was not sent to the kitchen. Please check the error log or retry."),
   377→        )
   378→
   379→    # table status
   380→    if invoice.invoice_printed == 0:
   381→        frappe.db.set_value(
   382→            "URY Table", table, {"occupied": 1, "latest_invoice_time": invoice.creation}
   383→        )
   384→
   385→    invoice.db_set("owner", frappe.session.user)
   386→    return {
   387→        "name": invoice.name,
   388→        "customer": invoice.customer,
   389→        "grand_total": invoice.grand_total,
   390→        "modified": invoice.modified,
   391→        "invoice_printed": invoice.invoice_printed,
   392→        "restaurant_table": invoice.restaurant_table,
   393→    }

   416→@frappe.whitelist()
   417→def get_restaurant_and_menu_name(table):
   418→    frappe.only_for("Restaurant Manager", "Restaurant User", "Cashier")
   419→    if not table:
   420→        frappe.throw(_("Please select a table"))
   421→
   422→    result = frappe.get_value(
   423→        "URY Table",
   424→        table,
   425→        ["restaurant", "branch", "restaurant_room"],
   426→    )
   427→    if not result:
   428→        frappe.throw(_("URY Table {0} not found").format(table))
   429→    restaurant, branch, room = result
   430→    room_wise_menu = frappe.db.get_value(
   431→        "URY Restaurant",
   432→        restaurant,
   433→        "room_wise_menu",
   434→    )
   435→
   436→    if not room_wise_menu:
   437→        menu = frappe.db.get_value("URY Restaurant", restaurant, "active_menu")
   438→    else:
   439→        menu = frappe.db.get_value(
   440→            "Menu for Room",
   441→            {"parent": restaurant, "room": room},
   442→            "menu",
   443→        )
   444→
   445→    if not menu:
   446→        frappe.throw(
   447→            _("Please set an active menu for Restaurant {0}").format(restaurant)
   448→        )
   449→
   450→    return branch, menu, restaurant

   478→@frappe.whitelist()
   479→def pos_opening_check():
   480→    frappe.only_for("Restaurant Manager", "Restaurant User", "Cashier")
   481→    user = frappe.session.user
   482→    # Handle the administrator case differently
   483→    if user == "Administrator":
   484→        return {
   485→            "opening_exists": False,  # Assuming no POS opening entry is needed for Administrator
   486→            "cashier": None,
   487→            "pos_profile": None,
   488→        }
   489→    
   490→    details = getBranchRoom()
   491→    if not details:
   492→        frappe.throw(_("No room/branch configuration found for the current user."))
   493→    room = details[0].get('name')
   494→    branch = details[0].get('branch')
   495→    
   496→    pos_opening_list = frappe.db.sql("""
   497→        SELECT DISTINCT `tabPOS Opening Entry`.name 
   498→        FROM `tabPOS Opening Entry`
   499→        INNER JOIN `tabMultiple Rooms` 
   500→        ON `tabMultiple Rooms`.parent = `tabPOS Opening Entry`.name
   501→        WHERE `tabPOS Opening Entry`.branch = %s
   502→        AND `tabPOS Opening Entry`.status = 'Open'
   503→        AND `tabPOS Opening Entry`.docstatus = 1
   504→        AND `tabMultiple Rooms`.room = %s
   505→    """, (branch, room), as_dict=True)
   506→    
   507→    
   508→    result = {
   509→        "opening_exists": len(pos_opening_list) > 0,
   510→        "cashier": None,
   511→        "pos_profile": None,
   512→    }
   513→
   514→    if result["opening_exists"]:
   515→        opening_vals = frappe.db.get_value(
   516→            "POS Opening Entry", pos_opening_list[0].name,
   517→            ["user", "pos_profile"],
   518→            as_dict=True,
   519→        )
   520→        result["cashier"] = opening_vals.user
   521→        result["pos_profile"] = opening_vals.pos_profile
   522→        
   523→    return result

   526→@frappe.whitelist()
   527→def table_transfer(table, newTable, invoice):
   528→    frappe.only_for("Restaurant Manager", "Restaurant User", "Cashier")
   529→    if not frappe.has_permission("POS Invoice", "write", invoice):
   530→        frappe.throw(_("Not permitted to transfer tables"), frappe.PermissionError)
   531→    # R39-FIX: Validate invoice belongs to user's branch
   532→    inv_branch = frappe.db.get_value("POS Invoice", invoice, "branch")
   533→    if not inv_branch:
   534→        frappe.throw(_("POS Invoice {0} not found").format(invoice))
   535→    user_branch = _get_user_branch()
   536→    if inv_branch != user_branch:
   537→        frappe.throw(_("You do not have access to invoices from another branch"), frappe.PermissionError)
   538→    # Lock both source and destination tables to prevent race conditions (BE-R36-007)
   539→    table_data = frappe.db.sql(
   540→        """SELECT name, restaurant_room, occupied FROM `tabURY Table`
   541→           WHERE name IN %s FOR UPDATE""",
   542→        ((table, newTable),), as_dict=True
   543→    )

   709→def make_invoice(customer, payments, cashier, pos_profile, additionalDiscount=None, table=None, invoice=None):
   710→    frappe.only_for("Restaurant Manager", "Restaurant User", "Cashier")
   711→
   712→    # R39-FIX: Validate POS Profile belongs to user's branch
   713→    pos_profile_branch = frappe.db.get_value("POS Profile", pos_profile, "branch")
   714→    user_branch = _get_user_branch()
   715→    if pos_profile_branch != user_branch:
   716→        frappe.throw(_("POS Profile does not belong to your branch"), frappe.PermissionError)
   717→
   718→    # Validate additional discount
   719→    if additionalDiscount is not None:
   720→        additionalDiscount = flt(additionalDiscount)
   721→        if additionalDiscount < 0 or additionalDiscount > 100:
   722→            frappe.throw(_("Additional discount must be between 0 and 100"))
   723→
   724→    # Check if discounts are enabled for this POS Profile (BE-R36-006)
   725→    if additionalDiscount:
   726→        enable_discount = frappe.db.get_value("POS Profile", pos_profile, "custom_enable_discount")
   727→        if not enable_discount:
   728→            frappe.throw(_("Discounts are not enabled for this POS Profile"), frappe.PermissionError)
   729→

   731→    if isinstance(payments, str):
   732→        payments = json.loads(payments)
   733→    if not payments:
   734→        frappe.throw(_("At least one payment is required"))
   735→    for p in payments:
   736→        amount = flt(p.get("amount", 0))
   737→        if amount < 0:
   738→            frappe.throw(_("Payment amount cannot be negative"))
   739→
   740→    # R43-FIX: Guard against None invoice — get_value with None returns None/error
   741→    order_type = None
   742→    if invoice:
   743→        order_type = frappe.get_value("POS Invoice", invoice, "order_type")

   751→    elif not invoice.restaurant:
   752→        restaurant = frappe.db.get_value("URY Restaurant", {"branch": invoice.branch}, "name")
   753→        if restaurant:
   754→            invoice.restaurant = restaurant
   755→

   757→    invoice.pos_profile = pos_profile
   758→    invoice.additional_discount_percentage=additionalDiscount
   759→    invoice.calculate_taxes_and_totals()
   760→
   761→    invoice.payments = []
   762→
   763→    for d in payments:
   764→        invoice.append(
   765→            "payments", dict(mode_of_payment=d["mode_of_payment"], amount=d["amount"])
   766→        )
