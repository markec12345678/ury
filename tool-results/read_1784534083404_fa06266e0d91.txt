   319→// ---- Period Helpers ----
   320→
   321→function getPreviousPeriodDates(
   322→  period: ReportPeriod,
   323→  customFromDate?: string | null,
   324→  customToDate?: string | null
   325→): { prevFrom: string; prevTo: string } {
   326→  const today = new Date();
   327→
   328→  // If custom date range, calculate the previous equivalent period
   329→  if (customFromDate && customToDate) {
   330→    const from = new Date(customFromDate);
   331→    const to = new Date(customToDate);
   332→    const diffMs = to.getTime() - from.getTime();
   333→    const prevTo = new Date(from.getTime() - 1); // Day before from
   334→    const prevFrom = new Date(prevTo.getTime() - diffMs);
   335→    return {
   336→      prevFrom: prevFrom.toISOString().split('T')[0],
   337→      prevTo: prevTo.toISOString().split('T')[0],
   338→    };
   339→  }
   340→
   341→  switch (period) {
   342→    case 'daily':
   343→    case 'yesterday': {
   344→      // Previous day
   345→      const prev = new Date(today);
   346→      prev.setDate(prev.getDate() - 1);
   347→      const dateStr = prev.toISOString().split('T')[0];
   348→      return { prevFrom: dateStr, prevTo: dateStr };
   349→    }
   350→    case 'weekly': {
   351→      // Previous week (same day range)
   352→      const prevWeekStart = new Date(today);
   353→      prevWeekStart.setDate(prevWeekStart.getDate() - 7 - prevWeekStart.getDay());
   354→      const prevWeekEnd = new Date(prevWeekStart);
   355→      prevWeekEnd.setDate(prevWeekEnd.getDate() + 6);
   356→      return {
   357→        prevFrom: prevWeekStart.toISOString().split('T')[0],
   358→        prevTo: prevWeekEnd.toISOString().split('T')[0],
   359→      };
   360→    }
   361→    case 'monthly': {
   362→      // Previous month
   363→      const prevMonth = new Date(today.getFullYear(), today.getMonth() - 1, 1);
   364→      const prevMonthEnd = new Date(today.getFullYear(), today.getMonth(), 0);
   365→      return {
   366→        prevFrom: prevMonth.toISOString().split('T')[0],
   367→        prevTo: prevMonthEnd.toISOString().split('T')[0],
   368→      };
   369→    }
   370→    case 'last_month': {
   371→      // Two months ago
   372→      const twoMonthsAgo = new Date(today.getFullYear(), today.getMonth() - 2, 1);
   373→      const twoMonthsAgoEnd = new Date(today.getFullYear(), today.getMonth() - 1, 0);
   374→      return {
   375→        prevFrom: twoMonthsAgo.toISOString().split('T')[0],
   376→        prevTo: twoMonthsAgoEnd.toISOString().split('T')[0],
   377→      };
   378→    }
   379→    case 'last_7_days': {
   380→      const prevEnd = new Date(today);
   381→      prevEnd.setDate(prevEnd.getDate() - 8);
   382→      const prevStart = new Date(prevEnd);
   383→      prevStart.setDate(prevStart.getDate() - 6);
   384→      return {
   385→        prevFrom: prevStart.toISOString().split('T')[0],
   386→        prevTo: prevEnd.toISOString().split('T')[0],
   387→      };
   388→    }
   389→    case 'last_30_days': {
   390→      const prevEnd = new Date(today);
   391→      prevEnd.setDate(prevEnd.getDate() - 31);
   392→      const prevStart = new Date(prevEnd);
   393→      prevStart.setDate(prevStart.getDate() - 29);
   394→      return {
   395→        prevFrom: prevStart.toISOString().split('T')[0],
   396→        prevTo: prevEnd.toISOString().split('T')[0],
   397→      };
   398→    }
   399→    default: {
   400→      const prev = new Date(today);
   401→      prev.setDate(prev.getDate() - 1);
   402→      const dateStr = prev.toISOString().split('T')[0];
   403→      return { prevFrom: dateStr, prevTo: dateStr };
   404→    }
   405→  }
   406→}

   410→function escapeCsvField(field: string | number): string {
   411→  const str = String(field);
   412→  if (str.includes(',') || str.includes('"') || str.includes('\n')) {
   413→    return `"${str.replace(/"/g, '""')}"`;
   414→  }
   415→  return str;
   416→}
   417→
   418→function csvRow(...fields: (string | number)[]): string {
   419→  return fields.map(escapeCsvField).join(',');
   420→}
   421→
   422→function downloadCsv(content: string, filename: string): void {
   423→  const BOM = '\uFEFF'; // UTF-8 BOM for Excel compatibility
   424→  const blob = new Blob([BOM + content], { type: 'text/csv;charset=utf-8;' });
   425→  const url = URL.createObjectURL(blob);
   426→  const link = document.createElement('a');
   427→  link.setAttribute('href', url);
   428→  link.setAttribute('download', filename);
   429→  document.body.appendChild(link);
   430→  link.click();
   431→  document.body.removeChild(link);
   432→  URL.revokeObjectURL(url);
   433→}

   435→function generateSalesCsv(report: SalesReport): string {
   436→  const lines: string[] = [];
   437→
   438→  // Header
   439→  lines.push(csvRow('URY Restaurant - Sales Report'));
   440→  lines.push(csvRow('Period', `${report.from_date} to ${report.to_date}`));
   441→  lines.push(csvRow('Generated', new Date().toLocaleString()));
   442→  lines.push('');
   443→
   444→  // Summary
   445→  lines.push(csvRow('--- Summary ---'));
   446→  lines.push(csvRow('Metric', 'Value'));
   447→  lines.push(csvRow('Total Revenue', report.summary.total_revenue.toFixed(2)));
   448→  lines.push(csvRow('Total Orders', report.summary.total_orders));
   449→  lines.push(csvRow('Avg Order Value', report.summary.avg_order_value.toFixed(2)));
   450→  lines.push(csvRow('Net Revenue', report.summary.net_revenue.toFixed(2)));
   451→  lines.push(csvRow('Total Tax', report.summary.total_tax.toFixed(2)));
   452→  lines.push(csvRow('Unique Customers', report.summary.unique_customers));
   453→  lines.push('');
   454→
   455→  // Item Sales
   456→  if (report.item_sales.length > 0) {
   457→    lines.push(csvRow('--- Item-wise Sales ---'));
   458→    lines.push(csvRow('Item', 'Qty', 'Avg Rate', 'Total Amount'));
   459→    report.item_sales.forEach((item) => {
   460→      lines.push(csvRow(item.item_name, item.total_qty, item.avg_rate.toFixed(2), item.total_amount.toFixed(2)));
   461→    });
   462→    lines.push('');
   463→  }
   464→
   465→  // Order Type Sales
   466→  if (report.order_type_sales.length > 0) {
   467→    lines.push(csvRow('--- Order Type Breakdown ---'));
   468→    lines.push(csvRow('Order Type', 'Orders', 'Revenue'));
   469→    report.order_type_sales.forEach((ot) => {
   470→      lines.push(csvRow(ot.order_type, ot.order_count, ot.revenue.toFixed(2)));
   471→    });
   472→    lines.push('');
   473→  }
   474→
   475→  // Cancelled Orders
   476→  if (report.cancelled_orders.count > 0) {
   477→    lines.push(csvRow('--- Cancelled Orders ---'));
   478→    lines.push(csvRow('Count', 'Amount'));
   479→    lines.push(csvRow(report.cancelled_orders.count, report.cancelled_orders.amount.toFixed(2)));
   480→    lines.push('');
   481→  }
   482→
   483→  return lines.join('\n');
   484→}

   564→  lines.push(csvRow('--- Stock Levels ---'));
   565→  lines.push(csvRow('Item', 'Code', 'Current Stock', 'Reorder Level', 'Unit', 'Valuation Rate', 'Stock Value', 'Status'));
   566→  report.items.forEach((item) => {
   567→    lines.push(csvRow(
   568→      item.item_name,
   569→      item.item_code,
   570→      item.current_stock,
   571→      item.reorder_level,
   572→      item.stock_uom,
   573→      item.valuation_rate.toFixed(2),
   574→      item.stock_value.toFixed(2),
   575→      item.status
   576→    ));
   577→  });
   578→
   579→  return lines.join('\n');
   580→}
