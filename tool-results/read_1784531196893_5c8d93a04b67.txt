   230→// R41-FIX: Add cancel() method to debounce return value so pending timers

   232→function debounce(fn, delay) {
   233→    let timer = null;
   234→    const debounced = function(...args) {
   235→        clearTimeout(timer);
   236→        timer = setTimeout(() => fn.apply(this, args), delay);
   237→    };
   238→    debounced.cancel = () => { clearTimeout(timer); timer = null; };
   239→    return debounced;
   240→}

   784→    calculateTimeRemaining(targetTime) {
   785→      if (!targetTime || !targetTime.includes(":")) return '— : —';
   786→      const currentTime = new Date();
   787→      // R39-FIX: Guard against malformed time strings with fewer than 3 parts.
   788→      // Previously, "HH:MM" without seconds would set targetSeconds=undefined,
   789→      // which Date() treats as NaN → Invalid Date → NaN propagation.
   790→      const parts = targetTime.split(":");
   791→      const targetHours = parseInt(parts[0], 10);
   792→      const targetMinutes = parseInt(parts[1], 10);
   793→      const targetSeconds = parseInt(parts[2] || '0', 10);
   794→      if (isNaN(targetHours) || isNaN(targetMinutes) || isNaN(targetSeconds)) return '— : —';
   795→      let targetDate = new Date(
   796→        currentTime.getFullYear(),
   797→        currentTime.getMonth(),
   798→        currentTime.getDate(),
   799→        targetHours,
   800→        targetMinutes,
   801→        targetSeconds
   802→      );

   804→      if (targetDate > currentTime) {
   805→        targetDate = new Date(targetDate.getTime() - 86400000);
   806→      }
   807→

   809→      const hoursRemaining = Math.floor(timeDifference / 3600000);
   810→      const minutesRemaining = Math.floor((timeDifference % 3600000) / 60000);
   811→
   812→      return `${hoursRemaining} : ${String(minutesRemaining).padStart(2, '0')}`;

   815→    // Without this, a transient server error leaves the KDS showing stale
   816→    // data indefinitely — the only recovery was a manual page reload.
   817→    fetchKOTWithRetry(maxRetries = 3, initialDelay = 2000) {
   818→      // R41-FIX: Clear any pending retry timer before starting a new retry chain.
   819→      // Without this, calling fetchKOTWithRetry() while a previous retry timer is
   820→      // pending would leave the old timer running, potentially causing duplicate
   821→      // fetchKOT calls when both timers fire.
   822→      if (this._fetchRetryTimer) { clearTimeout(this._fetchRetryTimer); this._fetchRetryTimer = null; }
   823→      // R42-FIX: Bump generation counter so any still-running retry chain from
   824→      // a previous fetchKOTWithRetry call knows it's been superseded. Without
   825→      // this, clearing the timer orphans the old promise — its resolve/reject
   826→      // callbacks are never called, leaking the promise and its closures.
   827→      const generation = ++this._fetchGeneration;
   828→      const attempt = (retriesLeft, delay) => {
   829→        // If a newer fetchKOTWithRetry has started, abort this chain
   830→        if (generation !== this._fetchGeneration) {
   831→          return Promise.reject(new Error('Superseded by newer fetchKOTWithRetry call'));
   832→        }
   833→        return this.fetchKOT().catch(err => {
   834→          if (retriesLeft <= 0 || !this._isMounted || generation !== this._fetchGeneration) throw err;
   835→          if (import.meta.env?.DEV) console.warn(`fetchKOT failed, retrying in ${delay}ms...`, err);
   836→          return new Promise((resolve, reject) => {
   837→            this._fetchRetryTimer = setTimeout(() => {
   838→              if (!this._isMounted || generation !== this._fetchGeneration) { reject(err); return; }
   839→              attempt(retriesLeft - 1, Math.min(delay * 2, 30000)).then(resolve).catch(reject);
   840→            }, delay);
   841→          });
   842→        });
   843→      };
   844→      return attempt(maxRetries, initialDelay);
   845→    },
