   495→   495→        const timeRemaining = kot.timeRemaining.split(":");
   496→   496→        const minutes =
   497→   497→          parseInt(timeRemaining[0], 10) * 60 + parseInt(timeRemaining[1], 10);
   498→   498→
   499→   499→        // R36-FIX: Handle NaN from invalid time format — treat as elapsed time exceeded
   500→   500→        const validMinutes = isNaN(minutes) ? Infinity : minutes;
   501→   501→
   502→   502→        if (
   503→   503→          validMinutes === Number(this.kot_alert_time) &&
   504→   504→          kot.type !== "Cancelled" &&
   505→   505→          kot.type !== "Partially cancelled" &&
   506→   506→          !this.notifiedKots.has(kot.name)
   507→   507→        ) {
   508→   508→          this.notifiedKots.add(kot.name);
   509→   509→          this.orderDelayNotify(kot);
   510→   510→        }
   511→   511→        if (validMinutes >= this.kot_alert_time) {
   512→   512→          kot.timecolor = "text-[#DC0000]";
   513→   513→        } else {
   514→   514→          kot.timecolor = "text-black";
   515→   515→        }
   516→   516→      });
   517→   517→    },
   518→   518→    calculateTimeRemaining(targetTime) {
   519→   519→      if (!targetTime || !targetTime.includes(":")) return '— : —';
   520→   520→      const currentTime = new Date();
   521→   521→      const [targetHours, targetMinutes, targetSeconds] = targetTime.split(":");
   522→   522→      let targetDate = new Date(
   523→   523→        currentTime.getFullYear(),
   524→   524→        currentTime.getMonth(),
   525→   525→        currentTime.getDate(),
   526→   526→        targetHours,
   527→   527→        targetMinutes,
   528→   528→        targetSeconds
   529→   529→      );
   530→   530→      // Handle cross-midnight KOTs: if target is in the future, it was created yesterday
   531→   531→      if (targetDate > currentTime) {
   532→   532→        targetDate = new Date(targetDate.getTime() - 86400000);
   533→   533→      }
   534→   534→
   535→   535→      const timeDifference = Math.max(0, currentTime - targetDate);
   536→   536→      const hoursRemaining = Math.floor(timeDifference / 3600000);
   537→   537→      const minutesRemaining = Math.floor((timeDifference % 3600000) / 60000);
   538→   538→
   539→   539→      return `${hoursRemaining} : ${String(minutesRemaining).padStart(2, '0')}`;
   540→   540→    },
   541→   541→    fetchkotwithmasonry() {
   542→   542→      return this.fetchKOT().then(() => {
   543→   543→        this.masonryLoading();
   544→   544→      }).catch((e) => { console.error("KOT fetch failed:", e); });
   545→   545→    },

   547→   547→      const currentDomain = window.location.origin;

   549→   549→        currentDomain + "/login?redirect-to=" + encodeURIComponent("URYMosaic/" + this.production);

   559→   559→        return;
   560→   560→      }
   561→   561→      if (this.masonry) {
   562→   562→        this.masonry.destroy();
   563→   563→        this.masonry = null;
   564→   564→      }
   565→   565→      this.$nextTick(() => {

   567→   567→        const grid = this.$el.querySelector(".grid");
   568→   568→        if (!grid) return;
   569→   569→        this.masonry = markRaw(new Masonry(grid, {
   570→   570→          itemSelector: ".masonry-item",
   571→   571→          gutter: 28,
   572→   572→        }));

   574→   574→      });
   575→   575→    },

   615→   615→  mounted() {
   616→   616→    this._isMounted = true;

   618→   618→    window.addEventListener("offline", this.handleOffline);

   620→   620→    // R36-FIX: Use route params instead of fragile URL parsing
   621→   621→    const production = this.$route?.params?.production || '';
   622→   622→    const decodedProduction = decodeURIComponent(production);

   626→   626→    this._resizeHandler = debouncedMasonry;
   627→   627→    window.addEventListener("resize", this._resizeHandler);
   628→   628→    this.masonryLoading();
   629→   629→
   630→   630→    // Initialize socket in mounted() so re-mount gets a fresh connection
   631→   631→    this._socketInitPromise = initializeSocket();
   632→   632→

   635→   635→      .then(([sock]) => {
   636→   636→        // R36-FIX: Guard against post-unmount execution
   637→   637→        if (!this._isMounted) return;
   638→   638→        this._socket = sock;
   639→   639→        if (this._socket) this._socket.on('connect_error', (err) => {
   640→   640→          if (!this._isMounted) return;

   645→   645→          if (!this._isMounted) return;
   646→   646→          console.warn("Socket disconnected:", reason);
   647→   647→          this.setStatusMessage("Connection lost. Reconnecting...");
   648→   648→        });
   649→   649→        if (this._socket) this._socket.on('connect', () => {
   650→   650→          if (!this._isMounted) return;
   651→   651→          this.setStatusMessage("Reconnected");
   652→   652→          this.hideStatusMessageAfterDelay();
   653→   653→        });
   654→   654→
   655→   655→        return this.fetchKOT();
   656→   656→      })
   657→   657→      .then(() => {
   658→   658→        if (!this._isMounted) return;
   659→   659→        if (this.audio_alert === 1) {
   660→   660→          this.showAudioAlertMessage = true;
   661→   661→        }

   663→   663→          if (!this._isMounted) return;

   665→   665→            if (this.audio_alert === 1) {
   666→   666→              this.playAlertSound(doc.audio_file);
   667→   667→            }
   668→   668→            // R36-FIX: Namespace localStorage key per production station to avoid cross-tab collision
   669→   669→            let kottime = localStorage.getItem("kot_time_" + this.production);
   670→   670→            if (doc.last_kot_time !== kottime) {
   671→   671→              // Full refresh needed — skip intermediate mutations
   672→   672→              this.fetchKOT().then(() => { this.masonryLoading(); }).catch((e) => { console.error("KOT fetch failed:", e); });
   673→   673→            } else {
   674→   674→              // R36-FIX: Guard against missing doc.kot to prevent TypeError crash
   675→   675→              if (!doc.kot) {
   676→   676→                this.fetchKOT().then(() => { this.masonryLoading(); }).catch((e) => { console.error("KOT fetch failed:", e); });
   677→   677→                return;
   678→   678→              }
   679→   679→              // Incremental update — deduplicate to avoid duplicate cards
   680→   680→              const existingIndex = this.kot.findIndex(k => k.name === doc.kot.name);
   681→   681→              if (existingIndex !== -1) {
   682→   682→                // R36-FIX: Preserve strikethrough state before Object.assign overwrites kot_items
   683→   683→                const strikeMap = new Map(
   684→   684→                  (this.kot[existingIndex].kot_items || []).map(i => [i.name, i.striked])
   685→   685→                );
   686→   686→                Object.assign(this.kot[existingIndex], { timecolor: 'text-black', timeRemaining: '— : —', ...doc.kot });
   687→   687→                // Restore strikethrough state after Object.assign
   688→   688→                if (this.kot[existingIndex].kot_items) {
   689→   689→                  this.kot[existingIndex].kot_items.forEach(i => {
   690→   690→                    if (strikeMap.has(i.name)) i.striked = strikeMap.get(i.name);
   691→   691→                  });
   692→   692→                }
   693→   693→              } else {
   694→   694→                const newKot = { isRotated: false, showDiv: false, timecolor: 'text-black', timeRemaining: '— : —', ...doc.kot };
   695→   695→                this.kot.unshift(newKot);
   696→   696→              }
   697→   697→              this.updateQtyColorTable();
   698→   698→              this.updateTimeRemaining();
   699→   699→              this.masonryLoading();
   700→   700→            }

   702→   702→            this._cancelTimeout = setTimeout(() => {
   703→   703→              if (!this._isMounted) return;
   704→   704→              if (doc.kot && doc.kot.type === "Cancelled") {
   705→   705→                this.fetchKOT().then(() => {
   706→   706→                  this.masonryLoading();
   707→   707→                }).catch((e) => { console.error("KOT fetch failed:", e); });
   708→   708→              }

   711→   711→              try {
   712→   712→                localStorage.setItem("kot_time_" + this.production, doc.kot.time);
   713→   713→              } catch (e) {
   714→   714→                if (import.meta.env?.DEV) console.error('localStorage write failed:', e);
   715→   715→              }
   716→   716→            }
   717→   717→          } catch (err) {
   718→   718→            if (import.meta.env?.DEV) console.error("Socket handler error:", err);
   719→   719→          }
   720→   720→        };
   721→   721→        if (this._socket) this._socket.on(this.kot_channel, this.socketHandler);
   722→   722→      })
   723→   723→      .catch((error) => {
   724→   724→        console.error("Initialization or authentication error:", error);
   725→   725→        if (this._isMounted) this.showModal = true;
   726→   726→      });

   728→   728→  },

   749→   749→  },
