   313→function getPreviousPeriodDates(
   314→  period: ReportPeriod,
   315→  customFromDate?: string | null,
   316→  customToDate?: string | null
   317→): { prevFrom: string; prevTo: string } {
   318→  const today = new Date();
   319→
   320→  // If custom date range, calculate the previous equivalent period
   321→  if (customFromDate && customToDate) {
   322→    const from = new Date(customFromDate);
   323→    const to = new Date(customToDate);
   324→    const diffMs = to.getTime() - from.getTime();
   325→    const prevTo = new Date(from.getTime() - 1); // Day before from
   326→    const prevFrom = new Date(prevTo.getTime() - diffMs);
   327→    return {
   328→      prevFrom: prevFrom.toISOString().split('T')[0],
   329→      prevTo: prevTo.toISOString().split('T')[0],
   330→    };
   331→  }
   332→
   333→  switch (period) {
   334→    case 'daily':
   335→    case 'yesterday': {
   336→      // Previous day
   337→      const prev = new Date(today);
   338→      prev.setDate(prev.getDate() - 1);
   339→      const dateStr = prev.toISOString().split('T')[0];
   340→      return { prevFrom: dateStr, prevTo: dateStr };
   341→    }
   342→    case 'weekly': {
   343→      // Previous week (same day range)
   344→      const prevWeekStart = new Date(today);
   345→      prevWeekStart.setDate(prevWeekStart.getDate() - 7 - prevWeekStart.getDay());
   346→      const prevWeekEnd = new Date(prevWeekStart);
   347→      prevWeekEnd.setDate(prevWeekEnd.getDate() + 6);
   348→      return {
   349→        prevFrom: prevWeekStart.toISOString().split('T')[0],
   350→        prevTo: prevWeekEnd.toISOString().split('T')[0],
   351→      };
   352→    }
   353→    case 'monthly': {
   354→      // Previous month
   355→      const prevMonth = new Date(today.getFullYear(), today.getMonth() - 1, 1);
   356→      const prevMonthEnd = new Date(today.getFullYear(), today.getMonth(), 0);
   357→      return {
   358→        prevFrom: prevMonth.toISOString().split('T')[0],
   359→        prevTo: prevMonthEnd.toISOString().split('T')[0],
   360→      };
   361→    }
   362→    case 'last_month': {
   363→      // Two months ago
   364→      const twoMonthsAgo = new Date(today.getFullYear(), today.getMonth() - 2, 1);
   365→      const twoMonthsAgoEnd = new Date(today.getFullYear(), today.getMonth() - 1, 0);
   366→      return {
   367→        prevFrom: twoMonthsAgo.toISOString().split('T')[0],
   368→        prevTo: twoMonthsAgoEnd.toISOString().split('T')[0],
   369→      };
   370→    }
   371→    case 'last_7_days': {
   372→      const prevEnd = new Date(today);
   373→      prevEnd.setDate(prevEnd.getDate() - 8);
   374→      const prevStart = new Date(prevEnd);
   375→      prevStart.setDate(prevStart.getDate() - 6);
   376→      return {
   377→        prevFrom: prevStart.toISOString().split('T')[0],
   378→        prevTo: prevEnd.toISOString().split('T')[0],
   379→      };
   380→    }
   381→    case 'last_30_days': {
   382→      const prevEnd = new Date(today);
   383→      prevEnd.setDate(prevEnd.getDate() - 31);
   384→      const prevStart = new Date(prevEnd);
   385→      prevStart.setDate(prevStart.getDate() - 29);
   386→      return {
   387→        prevFrom: prevStart.toISOString().split('T')[0],
   388→        prevTo: prevEnd.toISOString().split('T')[0],
   389→      };
   390→    }
   391→    default: {
   392→      const prev = new Date(today);
   393→      prev.setDate(prev.getDate() - 1);
   394→      const dateStr = prev.toISOString().split('T')[0];
   395→      return { prevFrom: dateStr, prevTo: dateStr };
   396→    }
   397→  }
   398→}

   402→function escapeCsvField(field: string | number): string {
   403→  const str = String(field);
   404→  if (str.includes(',') || str.includes('"') || str.includes('\n')) {
   405→    return `"${str.replace(/"/g, '""')}"`;
   406→  }
   407→  return str;
   408→}
   409→
   410→function csvRow(...fields: (string | number)[]): string {
   411→  return fields.map(escapeCsvField).join(',');
   412→}
   413→
   414→function downloadCsv(content: string, filename: string): void {
   415→  const BOM = '\uFEFF'; // UTF-8 BOM for Excel compatibility
   416→  const blob = new Blob([BOM + content], { type: 'text/csv;charset=utf-8;' });
   417→  const url = URL.createObjectURL(blob);
   418→  const link = document.createElement('a');
   419→  link.setAttribute('href', url);
   420→  link.setAttribute('download', filename);
   421→  document.body.appendChild(link);
   422→  link.click();
   423→  document.body.removeChild(link);
   424→  URL.revokeObjectURL(url);
   425→}

   427→function generateSalesCsv(report: SalesReport): string {
   428→  const lines: string[] = [];
   429→
   430→  // Header
   431→  lines.push(csvRow('URY Restaurant - Sales Report'));
   432→  lines.push(csvRow('Period', `${report.from_date} to ${report.to_date}`));
   433→  lines.push(csvRow('Generated', new Date().toLocaleString()));
   434→  lines.push('');
   435→
   436→  // Summary
   437→  lines.push(csvRow('--- Summary ---'));
   438→  lines.push(csvRow('Metric', 'Value'));
   439→  lines.push(csvRow('Total Revenue', report.summary.total_revenue.toFixed(2)));
   440→  lines.push(csvRow('Total Orders', report.summary.total_orders));
   441→  lines.push(csvRow('Avg Order Value', report.summary.avg_order_value.toFixed(2)));
   442→  lines.push(csvRow('Net Revenue', report.summary.net_revenue.toFixed(2)));
   443→  lines.push(csvRow('Total Tax', report.summary.total_tax.toFixed(2)));
   444→  lines.push(csvRow('Unique Customers', report.summary.unique_customers));
   445→  lines.push('');
   446→
   447→  // Item Sales
   448→  if (report.item_sales.length > 0) {
   449→    lines.push(csvRow('--- Item-wise Sales ---'));
   450→    lines.push(csvRow('Item', 'Qty', 'Avg Rate', 'Total Amount'));
   451→    report.item_sales.forEach((item) => {
   452→      lines.push(csvRow(item.item_name, item.total_qty, item.avg_rate.toFixed(2), item.total_amount.toFixed(2)));
   453→    });
   454→    lines.push('');
   455→  }
   456→
   457→  // Order Type Sales
   458→  if (report.order_type_sales.length > 0) {
   459→    lines.push(csvRow('--- Order Type Breakdown ---'));
   460→    lines.push(csvRow('Order Type', 'Orders', 'Revenue'));
   461→    report.order_type_sales.forEach((ot) => {
   462→      lines.push(csvRow(ot.order_type, ot.order_count, ot.revenue.toFixed(2)));
   463→    });
   464→    lines.push('');
   465→  }
   466→
   467→  // Cancelled Orders
   468→  if (report.cancelled_orders.count > 0) {
   469→    lines.push(csvRow('--- Cancelled Orders ---'));
   470→    lines.push(csvRow('Count', 'Amount'));
   471→    lines.push(csvRow(report.cancelled_orders.count, report.cancelled_orders.amount.toFixed(2)));
   472→    lines.push('');
   473→  }
   474→
   475→  return lines.join('\n');
   476→}

   625→function addReportFooter(doc: jsPDF): void {
   626→  const pageCount = doc.getNumberOfPages();
   627→
   628→  for (let i = 1; i <= pageCount; i++) {
   629→    doc.setPage(i);
   630→
