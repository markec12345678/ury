   829→   829→    fetchKOTWithRetry(maxRetries = 3, initialDelay = 2000) {
   830→   830→      // R41-FIX: Clear any pending retry timer before starting a new retry chain.
   831→   831→      // Without this, calling fetchKOTWithRetry() while a previous retry timer is
   832→   832→      // pending would leave the old timer running, potentially causing duplicate
   833→   833→      // fetchKOT calls when both timers fire.
   834→   834→      if (this._fetchRetryTimer) { clearTimeout(this._fetchRetryTimer); this._fetchRetryTimer = null; }
   835→   835→      // R42-FIX: Bump generation counter so any still-running retry chain from
   836→   836→      // a previous fetchKOTWithRetry call knows it's been superseded. Without
   837→   837→      // this, clearing the timer orphans the old promise — its resolve/reject
   838→   838→      // callbacks are never called, leaking the promise and its closures.
   839→   839→      const generation = ++this._fetchGeneration;
   840→   840→      const attempt = (retriesLeft, delay) => {
   841→   841→        // If a newer fetchKOTWithRetry has started, abort this chain
   842→   842→        if (generation !== this._fetchGeneration) {
   843→   843→          return Promise.reject(new Error('Superseded by newer fetchKOTWithRetry call'));
   844→   844→        }
   845→   845→        return this.fetchKOT().catch(err => {
   846→   846→          if (retriesLeft <= 0 || !this._isMounted || generation !== this._fetchGeneration) throw err;
   847→   847→          if (import.meta.env?.DEV) console.warn(`fetchKOT failed, retrying in ${delay}ms...`, err);
   848→   848→          return new Promise((resolve, reject) => {
   849→   849→            this._fetchRetryTimer = setTimeout(() => {
   850→   850→              if (!this._isMounted || generation !== this._fetchGeneration) { reject(err); return; }
   851→   851→              attempt(retriesLeft - 1, Math.min(delay * 2, 30000)).then(resolve).catch(reject);
   852→   852→            }, delay);
