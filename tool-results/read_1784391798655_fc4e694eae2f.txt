    12→    12→def inner_bom_process(buying_price_list, bom):
    13→    13→    unset_bom_items = []
    14→    14→    bom_buying_price = 0
    15→    15→
    16→    16→    for bom_item in bom.items:
    17→    17→        bom_item_qty = bom_item.qty
    18→    18→        bom_item_name = bom_item.item_name
    19→    19→        boms = frappe.db.get_all("BOM", fields=["name", "item"], filters={'item': bom_item.item_code, 'is_active': 1, 'is_default': 1, 'docstatus': 1})
    20→    20→        
    21→    21→        if len(boms) > 0:
    22→    22→            inner_bom = frappe.get_doc("BOM", boms[0].name)
    23→    23→            inner_bom_data = inner_inner_bom_process(buying_price_list, inner_bom)
    24→    24→            inner_bom_buying_price = inner_bom_data['bom_buying_price']
    25→    25→            inner_unset_bom_items = inner_bom_data['unset_bom_items']
    26→    26→            bom_buying_price += float(inner_bom_buying_price) * bom_item_qty
    27→    27→
    28→    28→            for item in inner_unset_bom_items:
    29→    29→                if item not in unset_bom_items:
    30→    30→                    unset_bom_items.append(item)
    31→    31→        else:
    32→    32→            bom_items_price = frappe.db.get_all("Item Price", fields=['name', 'price_list_rate'], filters={'price_list': buying_price_list, 'item_code': bom_item.item_code})
    33→    33→
    34→    34→            if len(bom_items_price) == 0:
    35→    35→                if bom_item_name not in unset_bom_items:
    36→    36→                    unset_bom_items.append(bom_item_name)
    37→    37→            else:
    38→    38→                bom_buying_price += float(bom_items_price[0].price_list_rate) * bom_item_qty
    39→    39→
    40→    40→    if not bom.quantity:
    41→    41→        return {"bom_buying_price": 0, "unset_bom_items": unset_bom_items}
    42→    42→    bom_buying_price = bom_buying_price / bom.quantity
    43→    43→    return {"bom_buying_price": bom_buying_price, "unset_bom_items": unset_bom_items}
