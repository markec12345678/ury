   183→   183→    aggregator_id=None,
   184→   184→    room=None
   185→   185→):
   186→   186→    frappe.only_for("Restaurant Manager", "Restaurant User", "Cashier")
   187→   187→    # Validate pos_profile belongs to user's branch (BE-R36-002)
   188→   188→    pos_profile_branch = frappe.db.get_value("POS Profile", pos_profile, "branch")
   189→   189→    user_branch = _get_user_branch()
   190→   190→    if pos_profile_branch != user_branch:
   191→   191→        frappe.throw(_("POS Profile does not belong to your branch"), frappe.PermissionError)
   192→   192→    user_role = frappe.get_roles()
   193→   193→    billing_roles = frappe.get_all(
   194→   194→        "POS Profile Role",
   195→   195→        filters={"parent": pos_profile},
   196→   196→        fields=["role"],
   197→   197→        pluck="role",
   198→   198→    )
   199→   199→    billing_user = bool(set(user_role).intersection(billing_roles))
   200→   200→
   201→   201→    # Check if the last invoice was already billed
   202→   202→    if (
   203→   203→        last_invoice
   204→   204→        and frappe.db.get_value("POS Invoice", last_invoice, "invoice_printed") == 1
   205→   205→        and (not billing_user)
   206→   206→    ):
   207→   207→        frappe.msgprint(
   208→   208→            title=_("Invoice Already Billed"),
   209→   209→            indicator="red",
   210→   210→            msg=_("This order has already been billed. Please reload the page."),
   211→   211→        )
   212→   212→        return {"status": "Failure"}
   213→   213→
   214→   214→    # R41-FIX: Use _get_order_invoice_doc to get the document object, not a dict.
   215→   215→    # The public get_order_invoice returns a dict for frontend safety (BE-R36-003),
   216→   216→    # but sync_order needs the document to call .save() etc.
   217→   217→    invoice = _get_order_invoice_doc(table, invoice, order_type)
   218→   218→
   219→   219→    # R41-FIX: Validate last_invoice belongs to user's branch
   220→   220→    if last_invoice:
   221→   221→        last_inv_branch = frappe.db.get_value("POS Invoice", last_invoice, "branch")
   222→   222→        if last_inv_branch and last_inv_branch != user_branch:
   223→   223→            frappe.throw(_("Invoice does not belong to your branch"), frappe.PermissionError)
   224→   224→

   227→   227→        from datetime import datetime
   228→   228→
   229→   229→        if isinstance(last_modified_time, str):
   230→   230→            try:
   231→   231→                last_modified_time = datetime.strptime(
   232→   232→                    last_modified_time, "%Y-%m-%d %H:%M:%S.%f"
   233→   233→                )
   234→   234→            except ValueError:
   235→   235→                last_modified_time = datetime.strptime(
   236→   236→                    last_modified_time, "%Y-%m-%d %H:%M:%S"
   237→   237→                )
   238→   238→        if isinstance(lastModifiedTime, str):
   239→   239→            try:
   240→   240→                lastModifiedTime = datetime.strptime(
   241→   241→                    lastModifiedTime, "%Y-%m-%d %H:%M:%S.%f"
   242→   242→                )
   243→   243→            except ValueError:
   244→   244→                lastModifiedTime = datetime.strptime(
   245→   245→                    lastModifiedTime, "%Y-%m-%d %H:%M:%S"
   246→   246→                )
   247→   247→        if lastModifiedTime != last_modified_time:
   248→   248→            frappe.msgprint(
   249→   249→                title=_("Order has been modified"),
   250→   250→                indicator="red",
   251→   251→                msg=_("This order has been modified. Please reload the page to retrieve the latest edits."),
   252→   252→            )
   253→   253→            return {"status": "Failure"}
   254→   254→    else:
   255→   255→        if invoice.name and invoice.invoice_printed == 0 and not billing_user:
   256→   256→            frappe.msgprint(
   257→   257→                title=_("Table Occupied"),
   258→   258→                indicator="red",
   259→   259→                msg=_("{0} is already occupied. Please refresh the page.").format(table),
   260→   260→            )
   261→   261→            return {"status": "Failure"}
   262→   262→
   263→   263→    if not customer:
   264→   264→        frappe.throw(_("Please enter valid customer details"))
   265→   265→    else:
   266→   266→        invoice.customer = customer
   267→   267→
   268→   268→    if order_type:
   269→   269→        invoice.order_type = order_type
   270→   270→
   271→   271→    invoice.mobile_number = frappe.db.get_value("Customer", customer, "mobile_number")
   272→   272→    if comments:
   273→   273→        invoice.custom_comments = comments
   274→   274→    invoice.no_of_pax = no_of_pax
   275→   275→    invoice.pos_profile = pos_profile
   276→   276→    invoice.cashier = cashier
   277→   277→    invoice.waiter = waiter
   278→   278→    invoice.custom_aggregator_id = aggregator_id
   279→   279→    invoice.custom_restaurant_room =room
   280→   280→    invoice.restaurant_table = table
   281→   281→    
   282→   282→    if order_type == "Aggregators":
   283→   283→        price_list = frappe.db.get_value("Aggregator Settings",{"customer": customer, "parent": invoice.branch, "parenttype": "Branch"},"price_list",)
   284→   284→        
   285→   285→        if not price_list:
   286→   286→            frappe.throw(_("Price list for customer {0} in branch {1} not found in Aggregator Settings.").format(customer, invoice.branch))
   287→   287→    else:
   288→   288→        price_list = invoice.selling_price_list
   289→   289→
   290→   290→    # dummy payment
   291→   291→    if invoice.invoice_created == 0:
   292→   292→        invoice.append(
   293→   293→            "payments",
   294→   294→            dict(mode_of_payment=mode_of_payment, amount=invoice.grand_total),

   296→   296→        invoice.invoice_created = 1
   297→   297→
   298→   298→    past_item = []
   299→   299→    for item in invoice.items:
   300→   300→        previous_item = {
   301→   301→            "item_code": item.item_code,

   304→   304→            "comments": "",
   305→   305→        }
   306→   306→        past_item.append(previous_item)
   307→   307→        
   308→   308→
   309→   309→    # Conditional checking for 'items' type:
   310→   310→    # - 'ury': JSON passed, hence using isinstance
   311→   311→    # - 'ury_pos': Already formatted list, hence using else
   312→   312→    if isinstance(items, str):
   313→   313→        items = json.loads(items)
   314→   314→    invoice.items = []
   315→   315→    
   316→   316→    menu = frappe.db.get_value("URY Menu", {"branch": invoice.branch}, "name")
   317→   317→
   318→   318→    # Batch-fetch courses, prices, and cost_center to avoid N+1 queries
   319→   319→    item_codes = [d.get("item") for d in items]
   320→   320→    item_codes_unique = list(set(item_codes))
   321→   321→
   322→   322→    courses = {}
   323→   323→    if item_codes_unique:
   324→   324→        course_rows = frappe.db.get_all(
   325→   325→            "URY Menu Item",

   381→   381→        )
   382→   382→
   383→   383→    # table status

   386→   386→            "URY Table", table, {"occupied": 1, "latest_invoice_time": invoice.creation}
   387→   387→        )
   388→   388→
   389→   389→    invoice.db_set("owner", frappe.session.user)
   390→   390→    return {
   391→   391→        "name": invoice.name,
   392→   392→        "customer": invoice.customer,
   393→   393→        "grand_total": invoice.grand_total,
   394→   394→        "modified": invoice.modified,
   395→   395→        "invoice_printed": invoice.invoice_printed,
   396→   396→        "restaurant_table": invoice.restaurant_table,
   397→   397→    }

   430→   430→    )
   431→   431→    if not result:
   432→   432→        frappe.throw(_("URY Table {0} not found").format(table))
   433→   433→    restaurant, branch, room = result
   434→   434→    room_wise_menu = frappe.db.get_value(
   435→   435→        "URY Restaurant",
   436→   436→        restaurant,
   437→   437→        "room_wise_menu",
   438→   438→    )
   439→   439→

   441→   441→        menu = frappe.db.get_value("URY Restaurant", restaurant, "active_menu")
   442→   442→    else:
   443→   443→        menu = frappe.db.get_value(
   444→   444→            "Menu for Room",
   445→   445→            {"parent": restaurant, "room": room},
   446→   446→            "menu",
   447→   447→        )
   448→   448→
