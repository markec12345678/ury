    11→def _inner_bom_process(buying_price_list, bom, depth=0, max_depth=10, visited=None):
    12→    if visited is None:
    13→        visited = set()
    14→    unset_bom_items = []
    15→    bom_buying_price = 0
    16→
    17→    if depth > max_depth:
    18→        frappe.log_error(
    19→            f"BOM recursion depth exceeded max_depth={max_depth} for BOM {bom.name}",
    20→            "BOM Depth Limit Warning"
    21→        )
    22→        return {"bom_buying_price": bom_buying_price, "unset_bom_items": unset_bom_items}
    23→
    24→    if bom.name in visited:
    25→        frappe.log_error(
    26→            f"Circular BOM reference detected: {bom.name} already visited",
    27→            "BOM Circular Reference Warning"
    28→        )
    29→        return {"bom_buying_price": bom_buying_price, "unset_bom_items": unset_bom_items}
    30→
    31→    visited = visited | {bom.name}
    32→
    33→    for bom_item in bom.items:
    34→        bom_item_qty = bom_item.qty
    35→        bom_item_name = bom_item.item_name
    36→        boms = frappe.db.get_all("BOM", fields=["name", "item"], filters={'item': bom_item.item_code, 'is_active': 1, 'is_default': 1, 'docstatus': 1})
    37→        
    38→        if len(boms) > 0:
    39→            inner_bom = frappe.get_doc("BOM", boms[0].name)
    40→            inner_bom_data = _inner_bom_process(buying_price_list, inner_bom, depth=depth+1, max_depth=max_depth, visited=visited)
    41→            inner_bom_buying_price = inner_bom_data['bom_buying_price']
    42→            inner_unset_bom_items = inner_bom_data['unset_bom_items']
    43→            bom_buying_price += float(inner_bom_buying_price) * bom_item_qty
    44→
    45→            for item in inner_unset_bom_items:
    46→                if item not in unset_bom_items:
    47→                    unset_bom_items.append(item)
    48→        else:
    49→            bom_items_price = frappe.db.get_all("Item Price", fields=['name', 'price_list_rate'], filters={'price_list': buying_price_list, 'item_code': bom_item.item_code})
    50→
    51→            if len(bom_items_price) == 0:
    52→                if bom_item_name not in unset_bom_items:
    53→                    unset_bom_items.append(bom_item_name)
    54→            else:
    55→                bom_buying_price += float(bom_items_price[0].price_list_rate) * bom_item_qty
    56→
    57→    if not bom.quantity:
    58→        return {"bom_buying_price": 0, "unset_bom_items": unset_bom_items}
    59→    bom_buying_price = bom_buying_price / bom.quantity
    60→    return {"bom_buying_price": bom_buying_price, "unset_bom_items": unset_bom_items}
