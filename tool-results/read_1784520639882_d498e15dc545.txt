    20→def _get_order_invoice_doc(table=None, invoiceNo=None, order_type=None, is_payment=None):
    21→    """Internal: returns POS Invoice document object for server-side manipulation.
    22→    Used by sync_order and make_invoice which need the doc, not a dict."""
    23→    # Verify user has access to the table's room (BE-R36-017)
    24→    if table:
    25→        from ury.ury_pos.api import _get_user_branch_rooms
    26→        table_room = frappe.db.get_value("URY Table", table, "restaurant_room")
    27→        if table_room:
    28→            user_rooms = [r.room for r in _get_user_branch_rooms() if r.room]
    29→            if user_rooms and table_room not in user_rooms:
    30→                frappe.throw(_("You do not have access to this table's room"), frappe.PermissionError)
    31→
    32→    if table:
    33→        # Lock the table row to prevent concurrent invoice creation (BE-R36-001)
    34→        frappe.db.sql(
    35→            "SELECT name FROM `tabURY Table` WHERE name = %s FOR UPDATE",
    36→            (table,), as_dict=True
    37→        )
    38→        
    39→        if is_payment == "Payments":
    40→            invoice_name = frappe.get_value(
    41→                "POS Invoice", dict(restaurant_table=table, docstatus=0, name=invoiceNo)
    42→            )
    43→            
    44→        else:
    45→            if invoiceNo:
    46→                invoice_name = frappe.get_value(
    47→                    "POS Invoice",
    48→                    dict(restaurant_table=table, docstatus=0, name=invoiceNo),
    49→                )
    50→               
    51→            else:
    52→                invoice_name = frappe.get_value(
    53→                    "POS Invoice",
    54→                    dict(restaurant_table=table, docstatus=0, invoice_printed=0),
    55→                )
    56→                
    57→        branch, menu_name, restaurant = get_restaurant_and_menu_name(table)
    58→
    59→        if invoice_name:
    60→            invoice = frappe.get_doc("POS Invoice", invoice_name)
    61→
    62→        else:
    63→            invoice = frappe.new_doc("POS Invoice")
    64→
    65→            invoice.naming_series = frappe.db.get_value(
    66→                "URY Restaurant", restaurant, "invoice_series_prefix"
    67→            )
    68→
    69→            invoice.is_pos = 1
    70→            invoice.update_stock = 1
    71→            invoice.restaurant = restaurant
    72→            invoice.branch = branch
    73→
    74→            is_take_away = frappe.db.get_value("URY Table", table, "is_take_away")
    75→            if is_take_away == 1:
    76→                invoice.order_type = "Take Away"
    77→            else:
    78→                invoice.order_type= "Dine In"
    79→
    80→        invoice.taxes_and_charges = frappe.db.get_value(
    81→            "URY Restaurant", restaurant, "default_tax_template"
    82→        )
    83→
    84→        invoice.selling_price_list = frappe.db.get_value(
    85→            "Price List", dict(restaurant_menu=menu_name, enabled=1)
    86→        )
    87→
    88→    else:
    89→
    90→        invoice_name = frappe.get_value(
    91→                "POS Invoice", dict(docstatus=0, name=invoiceNo)
    92→            )
    93→            
    94→        if invoice_name:
    95→            invoice = frappe.get_doc("POS Invoice", invoice_name)
    96→            # R44-FIX: Validate that the loaded invoice belongs to the user's branch
    97→            user_branch = _get_user_branch()
    98→            if invoice.branch and invoice.branch != user_branch:
    99→                frappe.throw(_("You do not have access to invoices from another branch"), frappe.PermissionError)
   100→            
   101→
   102→        else:
   103→            invoice = frappe.new_doc("POS Invoice")
   104→            invoice.is_pos = 1
   105→            invoice.update_stock = 1
   106→        
   107→        branch = _get_user_branch()
   108→        restaurant = frappe.db.get_value("URY Restaurant", {"branch": branch}, "name")
   109→
   110→        menu=get_menu_name(order_type)
   111→
   112→        if (order_type == "Aggregators" and frappe.db.get_value("Branch", branch, "custom_no_taxes") == 0) or order_type != "Aggregators":
   113→            invoice.taxes_and_charges = frappe.db.get_value("URY Restaurant", restaurant, "default_tax_template")
   114→        
   115→        invoice.selling_price_list = frappe.db.get_value(
   116→            "Price List", dict(restaurant_menu=menu, enabled=1)
   117→        )
   118→        # R43-FIX: Set branch on non-table invoices to prevent branchless invoices
   119→        invoice.branch = branch
   120→
   121→    return invoice

   167→@frappe.whitelist()
   168→def sync_order(
   169→    items,
   170→    cashier,
   171→    owner,
   172→    mode_of_payment,
   173→    customer,
   174→    no_of_pax,
   175→    last_invoice,
   176→    waiter,
   177→    pos_profile,
   178→    last_modified_time=None,
   179→    table=None,
   180→    invoice=None,
   181→    comments=None,
   182→    order_type=None,
   183→    aggregator_id=None,
   184→    room=None
   185→):
   186→    frappe.only_for("Restaurant Manager", "Restaurant User", "Cashier")
   187→    # Validate pos_profile belongs to user's branch (BE-R36-002)
   188→    pos_profile_branch = frappe.db.get_value("POS Profile", pos_profile, "branch")
   189→    user_branch = _get_user_branch()
   190→    if pos_profile_branch != user_branch:
   191→        frappe.throw(_("POS Profile does not belong to your branch"), frappe.PermissionError)
   192→    user_role = frappe.get_roles()
   193→    billing_roles = frappe.get_all(
   194→        "POS Profile Role",
   195→        filters={"parent": pos_profile},
   196→        fields=["role"],
   197→        pluck="role",
   198→    )
   199→    billing_user = bool(set(user_role).intersection(billing_roles))
   200→
   201→    # Check if the last invoice was already billed
   202→    if (
   203→        last_invoice
   204→        and frappe.db.get_value("POS Invoice", last_invoice, "invoice_printed") == 1
   205→        and (not billing_user)
   206→    ):
   207→        frappe.msgprint(
   208→            title=_("Invoice Already Billed"),
   209→            indicator="red",
   210→            msg=_("This order has already been billed. Please reload the page."),
   211→        )
   212→        return {"status": "Failure"}
   213→
   214→    # R41-FIX: Use _get_order_invoice_doc to get the document object, not a dict.
   215→    # The public get_order_invoice returns a dict for frontend safety (BE-R36-003),
   216→    # but sync_order needs the document to call .save() etc.
   217→    invoice = _get_order_invoice_doc(table, invoice, order_type)
   218→
   219→    # R41-FIX: Validate last_invoice belongs to user's branch
   220→    if last_invoice:
   221→        last_inv_branch = frappe.db.get_value("POS Invoice", last_invoice, "branch")
   222→        if last_inv_branch and last_inv_branch != user_branch:
   223→            frappe.throw(_("Invoice does not belong to your branch"), frappe.PermissionError)
   224→
   225→    if last_invoice and last_modified_time:
   226→        lastModifiedTime = invoice.modified
   227→        from datetime import datetime
   228→
   229→        if isinstance(last_modified_time, str):
   230→            try:
   231→                last_modified_time = datetime.strptime(
   232→                    last_modified_time, "%Y-%m-%d %H:%M:%S.%f"
   233→                )
   234→            except ValueError:
   235→                last_modified_time = datetime.strptime(
   236→                    last_modified_time, "%Y-%m-%d %H:%M:%S"
   237→                )
   238→        if isinstance(lastModifiedTime, str):
   239→            try:
   240→                lastModifiedTime = datetime.strptime(
   241→                    lastModifiedTime, "%Y-%m-%d %H:%M:%S.%f"
   242→                )
   243→            except ValueError:
   244→                lastModifiedTime = datetime.strptime(
   245→                    lastModifiedTime, "%Y-%m-%d %H:%M:%S"
   246→                )
   247→        if lastModifiedTime != last_modified_time:
   248→            frappe.msgprint(
   249→                title=_("Order has been modified"),
   250→                indicator="red",
   251→                msg=_("This order has been modified. Please reload the page to retrieve the latest edits."),
   252→            )
   253→            return {"status": "Failure"}
   254→    else:
   255→        if invoice.name and invoice.invoice_printed == 0 and not billing_user:
   256→            frappe.msgprint(
   257→                title=_("Table Occupied"),
   258→                indicator="red",
   259→                msg=_("{0} is already occupied. Please refresh the page.").format(table),
   260→            )
   261→            return {"status": "Failure"}
   262→
   263→    if not customer:
   264→        frappe.throw(_("Please enter valid customer details"))
   265→    else:
   266→        invoice.customer = customer
   267→
   268→    if order_type:
   269→        invoice.order_type = order_type
   270→
   271→    invoice.mobile_number = frappe.db.get_value("Customer", customer, "mobile_number")
   272→    if comments:
   273→        invoice.custom_comments = comments
   274→    invoice.no_of_pax = no_of_pax
   275→    invoice.pos_profile = pos_profile
   276→    invoice.cashier = cashier
   277→    invoice.waiter = waiter
   278→    invoice.custom_aggregator_id = aggregator_id
   279→    invoice.custom_restaurant_room =room
   280→    invoice.restaurant_table = table
   281→    
   282→    if order_type == "Aggregators":
   283→        price_list = frappe.db.get_value("Aggregator Settings",{"customer": customer, "parent": invoice.branch, "parenttype": "Branch"},"price_list",)
   284→        
   285→        if not price_list:
   286→            frappe.throw(_("Price list for customer {0} in branch {1} not found in Aggregator Settings.").format(customer, invoice.branch))
   287→    else:
   288→        price_list = invoice.selling_price_list
   289→
   290→    # dummy payment
   291→    if invoice.invoice_created == 0:
   292→        invoice.append(
   293→            "payments",
   294→            dict(mode_of_payment=mode_of_payment, amount=invoice.grand_total),
   295→        )
   296→        invoice.invoice_created = 1
   297→
   298→    past_item = []
   299→    for item in invoice.items:
   300→        previous_item = {
   301→            "item_code": item.item_code,
   302→            "item_name": item.item_name,
   303→            "qty": item.qty,
   304→            "comments": "",
   305→        }
   306→        past_item.append(previous_item)
   307→        
   308→
   309→    # Conditional checking for 'items' type:
   310→    # - 'ury': JSON passed, hence using isinstance
   311→    # - 'ury_pos': Already formatted list, hence using else
   312→    if isinstance(items, str):
   313→        items = json.loads(items)
   314→    invoice.items = []
   315→    
   316→    menu = frappe.db.get_value("URY Menu", {"branch": invoice.branch}, "name")
   317→
   318→    # Batch-fetch courses, prices, and cost_center to avoid N+1 queries
   319→    item_codes = [d.get("item") for d in items]
   320→    item_codes_unique = list(set(item_codes))
   321→
   322→    courses = {}
   323→    if item_codes_unique:
   324→        course_rows = frappe.db.get_all(
   325→            "URY Menu Item",
   326→            filters={"item": ("in", item_codes_unique), "parent": menu},
   327→            fields=["item", "course"],
   328→        )
   329→        courses = {r.item: r.course for r in course_rows}
   330→
   331→    prices = {}
   332→    if item_codes_unique:
   333→        price_rows = frappe.db.get_all(
   334→            "Item Price",
   335→            filters={"item_code": ("in", item_codes_unique), "price_list": price_list},
   336→            fields=["item_code", "price_list_rate"],
   337→        )
   338→        prices = {r.item_code: r.price_list_rate for r in price_rows}
   339→
   340→    cost_center = frappe.db.get_value("POS Profile", pos_profile, "cost_center")
   341→
   342→    for d in items:
   343→        course = courses.get(d.get("item"))
   344→        rate = prices.get(d.get("item"))
   345→        if not rate:
   346→            frappe.throw(_("No item price found for Item: {0} in Price List: {1}. Please check the price list settings.").format(d.get("item"), price_list))
   347→
   348→        invoice.append(
   349→            "items",
   350→            dict(
   351→                item_code=d.get("item"),
   352→                item_name=d.get("item_name"),
   353→                qty=d.get("qty"),
   354→                **({"custom_course": course} if course else {}),
   355→                comment=d.get("comment"),
   356→                rate=rate,
   357→                price_list_rate=rate,
   358→                base_price_list_rate=rate,
   359→                cost_center=cost_center,
   360→            ),
   361→        )
   362→
   363→    try:
   364→        invoice.save()
   365→    except Exception as e:
   366→        frappe.log_error(frappe.get_traceback(), "Invoice Save Error")
   367→        frappe.throw(_("An error occurred. Please check the error log."))
   368→
   369→
   370→    try:
   371→        kot_execute(invoice.name, customer, table, items, past_item, comments)
   372→
   373→    except Exception as e:
   374→        # Log the KOT error but also notify the user — kitchen won't see this order
   375→        error_msg = f"KOT Creation Failed: {str(e)}"
   376→        frappe.log_error(f"{error_msg}\n{frappe.get_traceback()}", "KOT Error")
   377→        frappe.msgprint(
   378→            title=_("KOT Creation Failed"),
   379→            indicator="orange",
   380→            msg=_("The order was saved but the KOT was not sent to the kitchen. Please check the error log or retry."),
   381→        )
   382→
   383→    # table status
   384→    if invoice.invoice_printed == 0:
   385→        frappe.db.set_value(
   386→            "URY Table", table, {"occupied": 1, "latest_invoice_time": invoice.creation}
   387→        )
   388→
   389→    invoice.db_set("owner", frappe.session.user)
   390→    return {
   391→        "name": invoice.name,
   392→        "customer": invoice.customer,
   393→        "grand_total": invoice.grand_total,
   394→        "modified": invoice.modified,
   395→        "invoice_printed": invoice.invoice_printed,
   396→        "restaurant_table": invoice.restaurant_table,
   397→    }

   420→@frappe.whitelist()
   421→def get_restaurant_and_menu_name(table):
   422→    frappe.only_for("Restaurant Manager", "Restaurant User", "Cashier")
   423→    if not table:
   424→        frappe.throw(_("Please select a table"))
   425→
   426→    result = frappe.get_value(
   427→        "URY Table",
   428→        table,
   429→        ["restaurant", "branch", "restaurant_room"],
   430→    )
   431→    if not result:
   432→        frappe.throw(_("URY Table {0} not found").format(table))
   433→    restaurant, branch, room = result
   434→    room_wise_menu = frappe.db.get_value(
   435→        "URY Restaurant",
   436→        restaurant,
   437→        "room_wise_menu",
   438→    )
   439→
   440→    if not room_wise_menu:
   441→        menu = frappe.db.get_value("URY Restaurant", restaurant, "active_menu")
   442→    else:
   443→        menu = frappe.db.get_value(
   444→            "Menu for Room",
   445→            {"parent": restaurant, "room": room},
   446→            "menu",
   447→        )
   448→
   449→    if not menu:
   450→        frappe.throw(
   451→            _("Please set an active menu for Restaurant {0}").format(restaurant)
   452→        )
   453→
   454→    return branch, menu, restaurant

   482→@frappe.whitelist()
   483→def pos_opening_check():
   484→    frappe.only_for("Restaurant Manager", "Restaurant User", "Cashier")
   485→    user = frappe.session.user
   486→    # Handle the administrator case differently
   487→    if user == "Administrator":
   488→        return {
   489→            "opening_exists": False,  # Assuming no POS opening entry is needed for Administrator
   490→            "cashier": None,
   491→            "pos_profile": None,
   492→        }
   493→    
   494→    details = getBranchRoom()
   495→    if not details:
   496→        frappe.throw(_("No room/branch configuration found for the current user."))
   497→    room = details[0].get('name')
   498→    branch = details[0].get('branch')
   499→    
   500→    pos_opening_list = frappe.db.sql("""
   501→        SELECT DISTINCT `tabPOS Opening Entry`.name 
   502→        FROM `tabPOS Opening Entry`
   503→        INNER JOIN `tabMultiple Rooms` 
   504→        ON `tabMultiple Rooms`.parent = `tabPOS Opening Entry`.name
   505→        WHERE `tabPOS Opening Entry`.branch = %s
   506→        AND `tabPOS Opening Entry`.status = 'Open'
   507→        AND `tabPOS Opening Entry`.docstatus = 1
   508→        AND `tabMultiple Rooms`.room = %s
   509→    """, (branch, room), as_dict=True)
   510→    
   511→    
   512→    result = {
   513→        "opening_exists": len(pos_opening_list) > 0,
   514→        "cashier": None,
   515→        "pos_profile": None,
   516→    }
   517→
   518→    if result["opening_exists"]:
   519→        opening_vals = frappe.db.get_value(
   520→            "POS Opening Entry", pos_opening_list[0].name,
   521→            ["user", "pos_profile"],
   522→            as_dict=True,
   523→        )
   524→        result["cashier"] = opening_vals.user
   525→        result["pos_profile"] = opening_vals.pos_profile
   526→        
   527→    return result

   533→    if not frappe.has_permission("POS Invoice", "write", invoice):

   537→    if not inv_branch:

   712→@frappe.whitelist()
   713→def make_invoice(customer, payments, cashier, pos_profile, additionalDiscount=None, table=None, invoice=None):
   714→    frappe.only_for("Restaurant Manager", "Restaurant User", "Cashier")
   715→
   716→    # R39-FIX: Validate POS Profile belongs to user's branch
   717→    pos_profile_branch = frappe.db.get_value("POS Profile", pos_profile, "branch")
   718→    user_branch = _get_user_branch()
   719→    if pos_profile_branch != user_branch:
   720→        frappe.throw(_("POS Profile does not belong to your branch"), frappe.PermissionError)
   721→
   722→    # Validate additional discount
   723→    if additionalDiscount is not None:
   724→        additionalDiscount = flt(additionalDiscount)
   725→        if additionalDiscount < 0 or additionalDiscount > 100:
   726→            frappe.throw(_("Additional discount must be between 0 and 100"))
   727→
   728→    # Check if discounts are enabled for this POS Profile (BE-R36-006)
   729→    if additionalDiscount:
   730→        enable_discount = frappe.db.get_value("POS Profile", pos_profile, "custom_enable_discount")
   731→        if not enable_discount:
   732→            frappe.throw(_("Discounts are not enabled for this POS Profile"), frappe.PermissionError)
   733→
   734→    # Validate payments
   735→    if isinstance(payments, str):
   736→        payments = json.loads(payments)
   737→    if not payments:
   738→        frappe.throw(_("At least one payment is required"))
   739→    for p in payments:
   740→        amount = flt(p.get("amount", 0))
   741→        if amount < 0:
   742→            frappe.throw(_("Payment amount cannot be negative"))
   743→
   744→    # R44-FIX: At least one of invoice or table must be provided — otherwise
   745→    # we would create an orphaned invoice with no way to reference it.
   746→    if not invoice and not table:
   747→        frappe.throw(_("Invoice or table is required to make a payment"), frappe.ValidationError)
   748→
   749→    # R43-FIX: Guard against None invoice — get_value with None returns None/error
   750→    order_type = None
   751→    if invoice:
   752→        order_type = frappe.get_value("POS Invoice", invoice, "order_type")
   753→    # R41-FIX: Use _get_order_invoice_doc to get the document object, not a dict.
   754→    # make_invoice needs to call .save() and .submit() on the document.
   755→    invoice = _get_order_invoice_doc(table, invoice, order_type, "Payments")
   756→
   757→    if table:
   758→        _, _, restaurant = get_restaurant_and_menu_name(table)
   759→        invoice.restaurant = restaurant
   760→    elif not invoice.restaurant:
   761→        restaurant = frappe.db.get_value("URY Restaurant", {"branch": invoice.branch}, "name")
   762→        if restaurant:
   763→            invoice.restaurant = restaurant
   764→
   765→    invoice.customer = customer
   766→    invoice.pos_profile = pos_profile
   767→    invoice.additional_discount_percentage=additionalDiscount
   768→    invoice.calculate_taxes_and_totals()
   769→
   770→    invoice.payments = []
   771→
   772→    for d in payments:
   773→        invoice.append(
   774→            "payments", dict(mode_of_payment=d["mode_of_payment"], amount=d["amount"])
   775→        )
   776→
   777→    # invoice.owner = owner
   778→    invoice.save()
   779→    try:
   780→        invoice.submit()
   781→    except Exception as e:
   782→        frappe.log_error(frappe.get_traceback(), "Invoice Submit Error")
   783→        frappe.throw(_("An error occurred. Please check the error log."))
