   230→// R41-FIX: Add cancel() method to debounce return value so pending timers
   231→// can be cleaned up on component unmount, preventing post-unmount masonry calls.
   232→function debounce(fn, delay) {
   233→    let timer = null;
   234→    const debounced = function(...args) {
   235→        clearTimeout(timer);
   236→        timer = setTimeout(() => fn.apply(this, args), delay);
   237→    };
   238→    debounced.cancel = () => { clearTimeout(timer); timer = null; };
   239→    return debounced;
   240→}

   711→    calculateTimeRemaining(targetTime) {
   712→      if (!targetTime || !targetTime.includes(":")) return '— : —';
   713→      const currentTime = new Date();
   714→      // R39-FIX: Guard against malformed time strings with fewer than 3 parts.
   715→      // Previously, "HH:MM" without seconds would set targetSeconds=undefined,
   716→      // which Date() treats as NaN → Invalid Date → NaN propagation.
   717→      const parts = targetTime.split(":");
   718→      const targetHours = parseInt(parts[0], 10);
   719→      const targetMinutes = parseInt(parts[1], 10);
   720→      const targetSeconds = parseInt(parts[2] || '0', 10);
   721→      if (isNaN(targetHours) || isNaN(targetMinutes) || isNaN(targetSeconds)) return '— : —';
   722→      let targetDate = new Date(
   723→        currentTime.getFullYear(),
   724→        currentTime.getMonth(),
   725→        currentTime.getDate(),
   726→        targetHours,
   727→        targetMinutes,
   728→        targetSeconds
   729→      );
   730→      // Handle cross-midnight KOTs: if target is in the future, it was created yesterday
   731→      if (targetDate > currentTime) {
   732→        targetDate = new Date(targetDate.getTime() - 86400000);
   733→      }
   734→
   735→      const timeDifference = Math.max(0, currentTime - targetDate);
   736→      const hoursRemaining = Math.floor(timeDifference / 3600000);
   737→      const minutesRemaining = Math.floor((timeDifference % 3600000) / 60000);
   738→
   739→      return `${hoursRemaining} : ${String(minutesRemaining).padStart(2, '0')}`;

   742→    // Without this, a transient server error leaves the KDS showing stale
   743→    // data indefinitely — the only recovery was a manual page reload.
   744→    fetchKOTWithRetry(maxRetries = 3, initialDelay = 2000) {
   745→      // R41-FIX: Clear any pending retry timer before starting a new retry chain.
   746→      // Without this, calling fetchKOTWithRetry() while a previous retry timer is
   747→      // pending would leave the old timer running, potentially causing duplicate
   748→      // fetchKOT calls when both timers fire.
   749→      if (this._fetchRetryTimer) { clearTimeout(this._fetchRetryTimer); this._fetchRetryTimer = null; }
   750→      // R42-FIX: Bump generation counter so any still-running retry chain from
   751→      // a previous fetchKOTWithRetry call knows it's been superseded. Without
   752→      // this, clearing the timer orphans the old promise — its resolve/reject
   753→      // callbacks are never called, leaking the promise and its closures.
   754→      const generation = ++this._fetchGeneration;
   755→      const attempt = (retriesLeft, delay) => {
   756→        // If a newer fetchKOTWithRetry has started, abort this chain
   757→        if (generation !== this._fetchGeneration) {
   758→          return Promise.reject(new Error('Superseded by newer fetchKOTWithRetry call'));
   759→        }
   760→        return this.fetchKOT().catch(err => {
   761→          if (retriesLeft <= 0 || !this._isMounted || generation !== this._fetchGeneration) throw err;
   762→          if (import.meta.env?.DEV) console.warn(`fetchKOT failed, retrying in ${delay}ms...`, err);
   763→          return new Promise((resolve, reject) => {
   764→            this._fetchRetryTimer = setTimeout(() => {
   765→              if (!this._isMounted || generation !== this._fetchGeneration) { reject(err); return; }
   766→              attempt(retriesLeft - 1, Math.min(delay * 2, 30000)).then(resolve).catch(reject);
   767→            }, delay);
   768→          });
   769→        });
   770→      };
   771→      return attempt(maxRetries, initialDelay);
   772→    },

  1018→    // init. Previously, data only loaded after BOTH socket AND auth completed,
  1019→    // causing a blank KDS when socket init was slow (siteName fetch + WebSocket
  1020→    // handshake can take 2-5s). Now the early fetch shows data immediately, and
  1021→    // the socket chain's fetchKOTWithRetry is skipped if data is already loaded.
  1022→    authPromise.then(() => {
  1023→      if (!this._isMounted) return;
  1024→      this.fetchKOTWithRetry().catch(() => {});
  1025→    }).catch(() => {});
  1026→
  1027→    this._socketInitPromise
  1028→      .then((sock) => {
  1029→        // R40-FIX: If component unmounted during socket init, disconnect immediately
  1030→        if (!this._isMounted) {
  1031→          if (sock) sock.disconnect();
  1032→          return null;
  1033→        }
  1034→        // R40-FIX: Store socket early so beforeUnmount can always clean it up
  1035→        this._socket = sock;
  1036→        // Now wait for auth to complete
  1037→        return authPromise;
  1038→      })
  1039→      .then(() => {
  1040→        if (!this._isMounted || !this._socket) return;
  1041→        this._socket.on('connect_error', (err) => {
  1042→          if (!this._isMounted) return;
  1043→          // R42-FIX: Only update status message if it's different from current.
  1044→          // During extended outages, socket.io retries every 1-5 seconds, and
  1045→          // each failed attempt triggered setStatusMessage, causing a Vue
  1046→          // re-render each time. Since the message text is identical, skip
  1047→          // the reactive assignment to avoid unnecessary re-renders.
  1048→          if (this.statusMessage !== "Connection error. Retrying...") {
  1049→            console.error("Socket connection error:", err);
  1050→            this.setStatusMessage("Connection error. Retrying...");
  1051→          }
  1052→        });
  1053→        this._socket.on('disconnect', (reason) => {
  1054→          if (!this._isMounted) return;
  1055→          console.warn("Socket disconnected:", reason);
  1056→          // R42-FIX: Skip reactive assignment if message is already showing
  1057→          // the same text — avoids unnecessary Vue re-renders during flapping.
  1058→          if (this.statusMessage !== "Connection lost. Reconnecting...") {
  1059→            this.setStatusMessage("Connection lost. Reconnecting...");
  1060→          }
  1061→        });
  1062→        this._socket.on('connect', () => {
  1063→          if (!this._isMounted) return;
  1064→          this.setStatusMessage("Reconnected");
  1065→          this.hideStatusMessageAfterDelay();
  1066→          // R38-FIX: Re-fetch KOT data after reconnect to sync any missed updates
  1067→          // during disconnection period. Without this, the UI could show stale KOTs.
  1068→          // R39-FIX: Use retry wrapper for transient errors on reconnect
  1069→          // R43-FIX: Remove redundant .then(() => masonryLoading()) — fetchKOT()
  1070→          // already calls masonryLoading(true) on success.
  1071→          // R45-FIX: Handle fetch failure — previously .catch(() => {}) silently
  1072→          // swallowed errors. During extended outages, the user's session may have
  1073→          // expired, so auth errors need to show the login modal. Other errors
  1074→          // show a status message so the user knows data refresh failed.
  1075→          this.fetchKOTWithRetry().catch((error) => {
  1076→            this._handleFetchError(error, "Reconnected but data refresh failed. Click Refresh.");
  1077→          });
  1078→        });
  1079→
  1080→        // R39-FIX: Use retry wrapper for initial fetch — transient server errors
  1081→        // should not leave the KDS permanently blank.
  1082→        // R44-FIX: Skip if early fetch (started after auth) already loaded data.
  1083→        // This avoids a redundant full refresh when the early fetch succeeded.
  1084→        if (this.kot.length === 0) {
  1085→          return this.fetchKOTWithRetry();
  1086→        }
  1087→      })
  1088→      .then(() => {
  1089→        if (!this._isMounted || !this._socket) return;
  1090→        if (this.audio_alert === 1) {
  1091→          this.showAudioAlertMessage = true;
  1092→        }
  1093→        this.socketHandler = (doc) => {
  1094→          if (!this._isMounted) return;
  1095→          // R39-FIX: Guard against null/undefined doc from malformed socket messages
  1096→          if (!doc) return;
  1097→          try {
  1098→            if (this.audio_alert === 1) {
  1099→              this.playAlertSound(doc.audio_file);
  1100→            }
  1101→            // R36-FIX: Namespace localStorage key per production station to avoid cross-tab collision
  1102→            // R39-FIX: Normalize null kottime from localStorage — if getItem returns null
  1103→            // (no previous value stored) and doc.last_kot_time is also null, they'd be
  1104→            // equal and fall through to the incremental path incorrectly. Treat null
  1105→            // localStorage value as a signal to do a full refresh.
  1106→            let kottime = localStorage.getItem("kot_time_" + this.production);
  1107→            if (doc.last_kot_time !== kottime || kottime === null) {
  1108→              // R43-FIX: Move localStorage sentinel update AFTER fetchKOT succeeds.
  1109→              // Previously, localStorage.setItem ran before fetchKOT, so if the fetch
  1110→              // failed, the sentinel was already updated. Subsequent socket events
  1111→              // would take the incremental path instead of the full-refresh path,
  1112→              // potentially missing KOTs that were added/removed server-side.
  1113→              // R43-FIX: Remove redundant masonryLoading() — fetchKOT() already calls
  1114→              // masonryLoading(true) at the end of its .then() handler.
  1115→              this.fetchKOT().then(() => {
  1116→                if (doc.kot && doc.kot.time != null) {
  1117→                  try {
  1118→                    localStorage.setItem("kot_time_" + this.production, doc.kot.time);
  1119→                  } catch (e) {
  1120→                    if (import.meta.env?.DEV) console.error('localStorage write failed:', e);
  1121→                  }
  1122→                }
  1123→              }).catch((e) => { console.error("KOT fetch failed:", e); });
  1124→              return;
  1125→            }
  1126→            // R36-FIX: Guard against missing doc.kot to prevent TypeError crash
  1127→            // R44-FIX: Also guard against non-object doc.kot (e.g., array, string).
  1128→            // If the server sends a malformed doc.kot, accessing .name would
  1129→            // return undefined, causing findIndex to fail and the bad data to
  1130→            // be unshifted into the KOT array, corrupting the display.
  1131→            if (!doc.kot || typeof doc.kot !== 'object' || Array.isArray(doc.kot)) {
  1132→              this.fetchKOT().catch((e) => { console.error("KOT fetch failed:", e); });
  1133→              return;
  1134→            }
  1135→            // Incremental update — deduplicate to avoid duplicate cards
  1136→            const existingIndex = this.kot.findIndex(k => k.name === doc.kot.name);
  1137→            let targetKot;
  1138→            if (existingIndex !== -1) {
  1139→              targetKot = this.kot[existingIndex];
  1140→              // R36-FIX: Preserve strikethrough state before Object.assign overwrites kot_items
  1141→              const strikeMap = new Map(
  1142→                (targetKot.kot_items || []).map(i => [i.name, i.striked])
  1143→              );
  1144→              // R43-FIX: Exclude 'name' from the spread to prevent overwriting the KOT's
  1145→              // primary key. If the server sends a different name in doc.kot, the KOT
  1146→              // would become orphaned — findIndex in serveOrder/confirmOrder uses the
  1147→              // original name as the lookup key.
  1148→              const { name: _incomingName, ...kotData } = doc.kot;
  1149→              Object.assign(targetKot, { timecolor: 'text-black', timeRemaining: '— : —', ...kotData });
  1150→              // Restore strikethrough state after Object.assign
  1151→              if (targetKot.kot_items) {
  1152→                targetKot.kot_items.forEach(i => {
  1153→                  if (strikeMap.has(i.name)) i.striked = strikeMap.get(i.name);
  1154→                });
  1155→              }
  1156→              // R37-FIX: Invalidate sorted cache for this KOT since items may have changed
  1157→              this._sortedItemsCache.delete(doc.kot.name);
  1158→            } else {
  1159→              // R39-FIX: Removed showDiv: false — dead code
  1160→              targetKot = { isRotated: false, timecolor: 'text-black', timeRemaining: '— : —', ...doc.kot };
  1161→              this.kot.unshift(targetKot);
  1162→            }
  1163→            // R44-FIX: Use targeted single-KOT methods instead of processing ALL KOTs.
  1164→            // The incremental path only modifies one KOT, so O(1) is sufficient.
  1165→            this._updateSingleKotQtyColor(targetKot);
  1166→            this._updateSingleKotTimeRemaining(targetKot);
  1167→            // R40-FIX: Use debounced masonry layout for rapid socket events
  1168→            this._debouncedMasonryLayout();
  1169→            // R41-FIX: Cancel timeout and localStorage write are now ONLY in the incremental path.
  1170→            // Previously these ran after both full-refresh and incremental branches.
  1171→            if (this._cancelTimeout) clearTimeout(this._cancelTimeout);
  1172→            // R39-FIX: Only schedule cancel re-fetch if the KOT is actually a cancellation.
  1173→            if (doc.kot.type === "Cancelled") {
  1174→              this._cancelTimeout = setTimeout(() => {
  1175→                if (!this._isMounted) return;
  1176→                // R43-FIX: Remove redundant masonryLoading() — fetchKOT() calls it.
  1177→                this.fetchKOT().catch((e) => { console.error("KOT fetch failed:", e); });
  1178→              }, 1500);
  1179→            }
  1180→            // R41-FIX: Guard against storing null/undefined as string "null"/"undefined"
  1181→            // in localStorage. These corrupted sentinel values would cause the
  1182→            // next socket event to always take the full-refresh path.
  1183→            if (doc.kot.time != null) {
  1184→              try {
  1185→                localStorage.setItem("kot_time_" + this.production, doc.kot.time);
  1186→              } catch (e) {
  1187→                if (import.meta.env?.DEV) console.error('localStorage write failed:', e);
  1188→              }
  1189→            }
  1190→          } catch (err) {
  1191→            if (import.meta.env?.DEV) console.error("Socket handler error:", err);
  1192→          }
  1193→        };
  1194→        if (this._socket) this._socket.on(this.kot_channel, this.socketHandler);
  1195→      })
  1196→      .catch((error) => {
  1197→        console.error("Initialization or authentication error:", error);
  1198→        // R40-FIX: this._socket is now stored early, so beforeUnmount will
  1199→        // disconnect it. No need for manual cleanup here.
  1200→        if (this._isMounted) {
  1201→          // R41-FIX: Differentiate auth errors from network/server errors.
  1202→          // Auth failure → "Not Permitted" with login prompt.
  1203→          // Network/server failure → "Connection Error" with retry guidance.
  1204→          // R45-FIX: Use centralized _isAuthError instead of duplicated inline check
  1205→          if (this._isAuthError(error)) {
  1206→            this.modalTitle = "Not Permitted";
  1207→            this.modalMessage = "Log in to access this page.";
  1208→          } else {
  1209→            this.modalTitle = "Connection Error";
  1210→            this.modalMessage = "Unable to reach the server. Please check your connection and try again.";
  1211→          }
  1212→          this.showModal = true;
  1213→        }
  1214→      });
  1215→    this.timer = setInterval(this.updateTimeRemaining, 60000);
  1216→  },

  1219→    window.removeEventListener("online", this.handleOnline);
  1220→    window.removeEventListener("offline", this.handleOffline);

  1222→    window.removeEventListener("resize", this._resizeHandler);

  1225→      this._socket.off(this.kot_channel, this.socketHandler);
  1226→    }
  1227→    if (this._socket) {
  1228→      this._socket.off('connect_error');
  1229→      this._socket.off('disconnect');
  1230→      this._socket.off('connect');

  1233→    this._socketInitPromise = null;
  1234→    if (this._cancelTimeout) clearTimeout(this._cancelTimeout);

  1240→    // attempting DOM operations on a detached element tree.

  1242→    if (this._resizeHandler) this._resizeHandler.cancel?.();

  1244→    // R42-FIX: Clear src and call load() after pause() to release the audio

  1246→    // even after pause(), leaking the audio file's network resources.

  1248→      this._alertAudio.pause();

  1250→      this._alertAudio.load();
  1251→      this._alertAudio = null;
  1252→    }
  1253→    // R39-FIX: Destroy non-reactive masonry instance on unmount
  1254→    if (this._masonry) {
  1255→      this._masonry.destroy();
  1256→      this._masonry = null;
  1257→    }
  1258→  },
  1259→  computed: {
  1260→    // R39-FIX: sortedKotItems moved from computed (which returned a function) to
  1261→    // a method. Previously, the computed returned a new function object every time
  1262→    // its reactive dependencies (this.kot) changed, which happened every minute

  1264→    // As a method, the function identity is stable and the manual _sortedItemsCache
  1265→    // still provides memoization. The template call `sortedKotItems(kot)` works
  1266→    // identically for both computed-returning-function and method.

  1268→      // R39-FIX: Removed !kot.showDiv filter — showDiv was always false (dead code).

  1271→      if (!this.production) {
  1272→        return this.kot;
  1273→      }
  1274→      return this.kot.filter(kot => kot.production === this.production);

  1276→  },
  1277→  watch: {
  1278→    // R41-FIX: Vue reuses component instances when navigating between routes
  1279→    // that use the same component (Home). Without this watcher, navigating
  1280→    // from /station/kitchen to /station/bar would NOT re-mount the KOT

  1283→    '$route.params.production'(newVal) {

  1286→      try {
  1287→        decoded = decodeURIComponent(newVal || '');
  1288→      } catch (e) {

  1290→        decoded = newVal || '';
  1291→      }
  1292→      if (decoded === this.production) return;
  1293→      this.production = decoded;
  1294→      // Re-register socket handler on the new channel
  1295→      if (this.kot_channel && this._socket && this.socketHandler) {
  1296→        this._socket.off(this.kot_channel, this.socketHandler);
  1297→      }
  1298→      // fetchKOT will set the new kot_channel and re-register the handler
  1299→      // R45-FIX: Handle fetch failure on station change — previously .catch(() => {})
  1300→      // silently swallowed errors, leaving the user with an empty KDS and no feedback.
  1301→      this.fetchKOTWithRetry().catch((error) => {
  1302→        this._handleFetchError(error, "Failed to load station data. Click Refresh.");
  1303→      });
  1304→    },
  1305→  },
