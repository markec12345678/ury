   316→   316→// ---- Period Helpers ----
   317→   317→
   318→   318→function getPreviousPeriodDates(
   319→   319→  period: ReportPeriod,
   320→   320→  customFromDate?: string | null,
   321→   321→  customToDate?: string | null
   322→   322→): { prevFrom: string; prevTo: string } {
   323→   323→  const today = new Date();
   324→   324→
   325→   325→  // If custom date range, calculate the previous equivalent period
   326→   326→  if (customFromDate && customToDate) {
   327→   327→    const from = new Date(customFromDate);
   328→   328→    const to = new Date(customToDate);
   329→   329→    const diffMs = to.getTime() - from.getTime();
   330→   330→    const prevTo = new Date(from.getTime() - 1); // Day before from
   331→   331→    const prevFrom = new Date(prevTo.getTime() - diffMs);
   332→   332→    return {
   333→   333→      prevFrom: prevFrom.toISOString().split('T')[0],
   334→   334→      prevTo: prevTo.toISOString().split('T')[0],
   335→   335→    };
   336→   336→  }
   337→   337→
   338→   338→  switch (period) {
   339→   339→    case 'daily':
   340→   340→    case 'yesterday': {
   341→   341→      // Previous day
   342→   342→      const prev = new Date(today);
   343→   343→      prev.setDate(prev.getDate() - 1);
   344→   344→      const dateStr = prev.toISOString().split('T')[0];
   345→   345→      return { prevFrom: dateStr, prevTo: dateStr };
   346→   346→    }
   347→   347→    case 'weekly': {
   348→   348→      // Previous week (same day range)
   349→   349→      const prevWeekStart = new Date(today);
   350→   350→      prevWeekStart.setDate(prevWeekStart.getDate() - 7 - prevWeekStart.getDay());
   351→   351→      const prevWeekEnd = new Date(prevWeekStart);
   352→   352→      prevWeekEnd.setDate(prevWeekEnd.getDate() + 6);
   353→   353→      return {
   354→   354→        prevFrom: prevWeekStart.toISOString().split('T')[0],
   355→   355→        prevTo: prevWeekEnd.toISOString().split('T')[0],
   356→   356→      };
   357→   357→    }
   358→   358→    case 'monthly': {
   359→   359→      // Previous month
   360→   360→      const prevMonth = new Date(today.getFullYear(), today.getMonth() - 1, 1);
   361→   361→      const prevMonthEnd = new Date(today.getFullYear(), today.getMonth(), 0);
   362→   362→      return {
   363→   363→        prevFrom: prevMonth.toISOString().split('T')[0],
   364→   364→        prevTo: prevMonthEnd.toISOString().split('T')[0],
   365→   365→      };
   366→   366→    }
   367→   367→    case 'last_month': {
   368→   368→      // Two months ago
   369→   369→      const twoMonthsAgo = new Date(today.getFullYear(), today.getMonth() - 2, 1);
   370→   370→      const twoMonthsAgoEnd = new Date(today.getFullYear(), today.getMonth() - 1, 0);
   371→   371→      return {
   372→   372→        prevFrom: twoMonthsAgo.toISOString().split('T')[0],
   373→   373→        prevTo: twoMonthsAgoEnd.toISOString().split('T')[0],
   374→   374→      };
   375→   375→    }
   376→   376→    case 'last_7_days': {
   377→   377→      const prevEnd = new Date(today);
   378→   378→      prevEnd.setDate(prevEnd.getDate() - 8);
   379→   379→      const prevStart = new Date(prevEnd);
   380→   380→      prevStart.setDate(prevStart.getDate() - 6);
   381→   381→      return {
   382→   382→        prevFrom: prevStart.toISOString().split('T')[0],
   383→   383→        prevTo: prevEnd.toISOString().split('T')[0],
   384→   384→      };
   385→   385→    }
   386→   386→    case 'last_30_days': {
   387→   387→      const prevEnd = new Date(today);
   388→   388→      prevEnd.setDate(prevEnd.getDate() - 31);
   389→   389→      const prevStart = new Date(prevEnd);
   390→   390→      prevStart.setDate(prevStart.getDate() - 29);
   391→   391→      return {
   392→   392→        prevFrom: prevStart.toISOString().split('T')[0],
   393→   393→        prevTo: prevEnd.toISOString().split('T')[0],
   394→   394→      };
   395→   395→    }
   396→   396→    default: {
   397→   397→      const prev = new Date(today);
   398→   398→      prev.setDate(prev.getDate() - 1);
   399→   399→      const dateStr = prev.toISOString().split('T')[0];
   400→   400→      return { prevFrom: dateStr, prevTo: dateStr };
   401→   401→    }
   402→   402→  }
   403→   403→}

   407→   407→function escapeCsvField(field: string | number): string {
   408→   408→  const str = String(field);
   409→   409→  if (str.includes(',') || str.includes('"') || str.includes('\n')) {
   410→   410→    return `"${str.replace(/"/g, '""')}"`;
   411→   411→  }
   412→   412→  return str;
   413→   413→}
   414→   414→
   415→   415→function csvRow(...fields: (string | number)[]): string {
   416→   416→  return fields.map(escapeCsvField).join(',');
   417→   417→}

   419→   419→function downloadCsv(content: string, filename: string): void {
   420→   420→  const BOM = '\uFEFF'; // UTF-8 BOM for Excel compatibility
   421→   421→  const blob = new Blob([BOM + content], { type: 'text/csv;charset=utf-8;' });
   422→   422→  const url = URL.createObjectURL(blob);
   423→   423→  const link = document.createElement('a');
   424→   424→  link.setAttribute('href', url);
   425→   425→  link.setAttribute('download', filename);
   426→   426→  document.body.appendChild(link);
   427→   427→  link.click();
   428→   428→  document.body.removeChild(link);
   429→   429→  URL.revokeObjectURL(url);
   430→   430→}

   432→   432→function generateSalesCsv(report: SalesReport): string {
   433→   433→  const lines: string[] = [];
   434→   434→
   435→   435→  // Header
   436→   436→  lines.push(csvRow('URY Restaurant - Sales Report'));
   437→   437→  lines.push(csvRow('Period', `${report.from_date} to ${report.to_date}`));
   438→   438→  lines.push(csvRow('Generated', new Date().toLocaleString()));
   439→   439→  lines.push('');
   440→   440→
   441→   441→  // Summary
   442→   442→  lines.push(csvRow('--- Summary ---'));
   443→   443→  lines.push(csvRow('Metric', 'Value'));
   444→   444→  lines.push(csvRow('Total Revenue', report.summary.total_revenue.toFixed(2)));
   445→   445→  lines.push(csvRow('Total Orders', report.summary.total_orders));
   446→   446→  lines.push(csvRow('Avg Order Value', report.summary.avg_order_value.toFixed(2)));
   447→   447→  lines.push(csvRow('Net Revenue', report.summary.net_revenue.toFixed(2)));
   448→   448→  lines.push(csvRow('Total Tax', report.summary.total_tax.toFixed(2)));
   449→   449→  lines.push(csvRow('Unique Customers', report.summary.unique_customers));
   450→   450→  lines.push('');
   451→   451→
   452→   452→  // Item Sales
   453→   453→  if (report.item_sales.length > 0) {
   454→   454→    lines.push(csvRow('--- Item-wise Sales ---'));
   455→   455→    lines.push(csvRow('Item', 'Qty', 'Avg Rate', 'Total Amount'));
   456→   456→    report.item_sales.forEach((item) => {
   457→   457→      lines.push(csvRow(item.item_name, item.total_qty, item.avg_rate.toFixed(2), item.total_amount.toFixed(2)));
   458→   458→    });
   459→   459→    lines.push('');
   460→   460→  }
   461→   461→
   462→   462→  // Order Type Sales
   463→   463→  if (report.order_type_sales.length > 0) {
   464→   464→    lines.push(csvRow('--- Order Type Breakdown ---'));
   465→   465→    lines.push(csvRow('Order Type', 'Orders', 'Revenue'));
   466→   466→    report.order_type_sales.forEach((ot) => {
   467→   467→      lines.push(csvRow(ot.order_type, ot.order_count, ot.revenue.toFixed(2)));
   468→   468→    });
   469→   469→    lines.push('');
   470→   470→  }
   471→   471→
   472→   472→  // Cancelled Orders
   473→   473→  if (report.cancelled_orders.count > 0) {
   474→   474→    lines.push(csvRow('--- Cancelled Orders ---'));
   475→   475→    lines.push(csvRow('Count', 'Amount'));
   476→   476→    lines.push(csvRow(report.cancelled_orders.count, report.cancelled_orders.amount.toFixed(2)));
   477→   477→    lines.push('');
   478→   478→  }
   479→   479→
   480→   480→  return lines.join('\n');
   481→   481→}

   483→   483→function generateExpenseCsv(report: ExpenseReport): string {
