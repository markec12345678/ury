   233→// R41-FIX: Add cancel() method to debounce return value so pending timers

   235→function debounce(fn, delay) {
   236→    let timer = null;
   237→    const debounced = function(...args) {
   238→        clearTimeout(timer);
   239→        timer = setTimeout(() => fn.apply(this, args), delay);
   240→    };
   241→    debounced.cancel = () => { clearTimeout(timer); timer = null; };
   242→    return debounced;
   243→}

   780→    // Without this, a transient server error leaves the KDS showing stale
   781→    // data indefinitely — the only recovery was a manual page reload.
   782→    fetchKOTWithRetry(maxRetries = 3, initialDelay = 2000) {
   783→      // R41-FIX: Clear any pending retry timer before starting a new retry chain.
   784→      // Without this, calling fetchKOTWithRetry() while a previous retry timer is
   785→      // pending would leave the old timer running, potentially causing duplicate
   786→      // fetchKOT calls when both timers fire.
   787→      if (this._fetchRetryTimer) { clearTimeout(this._fetchRetryTimer); this._fetchRetryTimer = null; }
   788→      // R42-FIX: Bump generation counter so any still-running retry chain from
   789→      // a previous fetchKOTWithRetry call knows it's been superseded. Without
   790→      // this, clearing the timer orphans the old promise — its resolve/reject
   791→      // callbacks are never called, leaking the promise and its closures.
   792→      const generation = ++this._fetchGeneration;
   793→      const attempt = (retriesLeft, delay) => {
   794→        // If a newer fetchKOTWithRetry has started, abort this chain
   795→        if (generation !== this._fetchGeneration) {
   796→          return Promise.reject(new Error('Superseded by newer fetchKOTWithRetry call'));
   797→        }
   798→        return this.fetchKOT().catch(err => {
   799→          if (retriesLeft <= 0 || !this._isMounted || generation !== this._fetchGeneration) throw err;
   800→          if (import.meta.env?.DEV) console.warn(`fetchKOT failed, retrying in ${delay}ms...`, err);
   801→          return new Promise((resolve, reject) => {
   802→            this._fetchRetryTimer = setTimeout(() => {
   803→              if (!this._isMounted || generation !== this._fetchGeneration) { reject(err); return; }
   804→              attempt(retriesLeft - 1, Math.min(delay * 2, 30000)).then(resolve).catch(reject);
   805→            }, delay);
   806→          });
   807→        });
   808→      };
   809→      return attempt(maxRetries, initialDelay);
   810→    },
