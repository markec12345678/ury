   229→function debounce(fn, delay) {
   230→    let timer = null;
   231→    return function(...args) {
   232→        clearTimeout(timer);
   233→        timer = setTimeout(() => fn.apply(this, args), delay);
   234→    };

   560→      const key = `${kot.name}:${items.length}:${items.map(i => i.name).join(',')}`;
   561→      const cached = this._sortedItemsCache.get(kot.name);
   562→      if (cached && cached.key === key) {
   563→        return cached.items;
   564→      }
   565→      const sorted = [...items].sort((a, b) => (a.serve_priority || 0) - (b.serve_priority || 0));
   566→      this._sortedItemsCache.set(kot.name, { key, items: sorted });
   567→      return sorted;
   568→    },
   569→    calculateQty(kotitem, qty, type, cancelled_qty) {
   570→      // R37-FIX: Guard against negative quantities from cancelled_qty > qty
   571→      if (type === "Partially cancelled" || type === "Cancelled") {
   572→        kotitem.qty = Math.max(0, qty - cancelled_qty);
   573→      } else {
   574→        kotitem.qty = qty;
   575→      }
   576→    },
   577→    removeAllItemsFromLocalStorage(kot) {
   578→      // R38-FIX: Remove strike-state keys for this KOT's items by known key pattern
   579→      // instead of iterating ALL localStorage keys (which may include unrelated Frappe keys).
   580→      if (!kot.kot_items) return;
   581→      kot.kot_items.forEach((kotitem) => {
   582→        try {
   583→          localStorage.removeItem(`${kot.name}_${kotitem.name}_strike`);
   584→        } catch (e) {
   585→          if (import.meta.env?.DEV) console.error('localStorage removeItem failed:', e);
   586→        }
   587→      });
   588→    },
   589→
   590→    updateTimeRemaining() {
   591→      this.kot.forEach((kot) => {
   592→        kot.timeRemaining = this.calculateTimeRemaining(kot.time);
   593→
   594→        const timeRemaining = kot.timeRemaining.split(":");
   595→        const minutes =
   596→          parseInt(timeRemaining[0], 10) * 60 + parseInt(timeRemaining[1], 10);
   597→
   598→        // R36-FIX: Handle NaN from invalid time format — treat as elapsed time exceeded
   599→        const validMinutes = isNaN(minutes) ? Infinity : minutes;
   600→
   601→        if (
   602→          validMinutes === Number(this.kot_alert_time) &&
   603→          kot.type !== "Cancelled" &&
   604→          kot.type !== "Partially cancelled" &&
   605→          !this.notifiedKots.has(kot.name)
   606→        ) {
   607→          this.notifiedKots.add(kot.name);
   608→          this.orderDelayNotify(kot);
   609→        }
   610→        // R38-FIX: Guard against empty/falsy kot_alert_time. An empty string
   611→        // coerces to 0 in >= comparison, making ALL KOTs show red time.
   612→        const alertThreshold = Number(this.kot_alert_time);
   613→        if (alertThreshold > 0 && validMinutes >= alertThreshold) {
   614→          kot.timecolor = "text-[#DC0000]";
   615→        } else {
   616→          kot.timecolor = "text-black";
   617→        }
   618→      });
   619→    },
   620→    calculateTimeRemaining(targetTime) {
   621→      if (!targetTime || !targetTime.includes(":")) return '— : —';
   622→      const currentTime = new Date();
   623→      // R39-FIX: Guard against malformed time strings with fewer than 3 parts.
   624→      // Previously, "HH:MM" without seconds would set targetSeconds=undefined,
   625→      // which Date() treats as NaN → Invalid Date → NaN propagation.
   626→      const parts = targetTime.split(":");
   627→      const targetHours = parseInt(parts[0], 10);
   628→      const targetMinutes = parseInt(parts[1], 10);
   629→      const targetSeconds = parseInt(parts[2] || '0', 10);
   630→      if (isNaN(targetHours) || isNaN(targetMinutes) || isNaN(targetSeconds)) return '— : —';
   631→      let targetDate = new Date(
   632→        currentTime.getFullYear(),
   633→        currentTime.getMonth(),
   634→        currentTime.getDate(),
   635→        targetHours,
   636→        targetMinutes,
   637→        targetSeconds
   638→      );
   639→      // Handle cross-midnight KOTs: if target is in the future, it was created yesterday
   640→      if (targetDate > currentTime) {
   641→        targetDate = new Date(targetDate.getTime() - 86400000);
   642→      }
   643→
   644→      const timeDifference = Math.max(0, currentTime - targetDate);
   645→      const hoursRemaining = Math.floor(timeDifference / 3600000);
   646→      const minutesRemaining = Math.floor((timeDifference % 3600000) / 60000);
   647→
   648→      return `${hoursRemaining} : ${String(minutesRemaining).padStart(2, '0')}`;
   649→    },
   650→    // R39-FIX: Retry wrapper for fetchKOT with exponential backoff.
   651→    // Without this, a transient server error leaves the KDS showing stale
   652→    // data indefinitely — the only recovery was a manual page reload.
   653→    fetchKOTWithRetry(maxRetries = 3, initialDelay = 2000) {
   654→      const attempt = (retriesLeft, delay) => {
   655→        return this.fetchKOT().catch(err => {
   656→          if (retriesLeft <= 0 || !this._isMounted) throw err;
   657→          if (import.meta.env?.DEV) console.warn(`fetchKOT failed, retrying in ${delay}ms...`, err);
   658→          return new Promise((resolve, reject) => {
   659→            this._fetchRetryTimer = setTimeout(() => {
   660→              if (!this._isMounted) { reject(err); return; }
   661→              attempt(retriesLeft - 1, Math.min(delay * 2, 30000)).then(resolve).catch(reject);
   662→            }, delay);
   663→          });
   664→        });
   665→      };
   666→      return attempt(maxRetries, initialDelay);
   667→    },
   668→    // R40-FIX: Removed redundant masonryLoading() call — fetchKOT() already
   669→    // calls masonryLoading(true) at the end of its .then() handler, so the
   670→    // extra call here caused a double layout calculation.
   671→    fetchkotwithmasonry() {
   672→      return this.fetchKOTWithRetry()
   673→        .catch((e) => { console.error("KOT fetch failed:", e); });
   674→    },
   675→    redirectToLogin() {
   676→      // R38-FIX: Use Vue Router instead of window.location.href to avoid full page reload
   677→      // and preserve app state. The previous approach caused a hard navigation that
   678→      // lost all Vue state and required a full re-initialization.
   679→      this.$router.push({ name: 'Login', query: { route: this.$route.path } });

   681→    masonryLoading(forceRecreate = false) {
   682→      // R40-FIX: Early return if unmounted — prevents post-unmount masonry
   683→      // operations from debounced callbacks, $nextTick, and delayed socket events.
   684→      if (!this._isMounted) return;
   685→      // R39-FIX: Use non-reactive _masonry instead of reactive masonry from data()
   686→      if (this._masonry && !forceRecreate) {
   687→        this.$nextTick(() => {
   688→          if (this._masonry) {
   689→            this._masonry.reloadItems?.();
   690→            this._masonry.layout();
   691→          }
   692→        });
   693→        return;
   694→      }
   695→      if (this._masonry) {
   696→        this._masonry.destroy();
   697→        this._masonry = null;
   698→      }
   699→      this.$nextTick(() => {
   700→        if (!this.$el) return;
   701→        const grid = this.$el.querySelector(".grid");
   702→        if (!grid) return;
   703→        this._masonry = markRaw(new Masonry(grid, {
   704→          itemSelector: ".masonry-item",
   705→          gutter: 28,
   706→        }));
   707→        this._masonry.layout();
   708→      });
   709→    },
   710→    hideAudioAlertMessage() {
   711→      this.showAudioAlertMessage = false;
   712→    },
   713→    handleOnline() {
   714→      // R37-FIX: Guard against post-unmount execution
   715→      if (!this._isMounted) return;
   716→      this.isOnline = true;
   717→      this.setStatusMessage("You are online");
   718→      this.hideStatusMessageAfterDelay();
   719→      // R39-FIX: Use retry wrapper for transient network errors on reconnect
   720→      this.fetchKOTWithRetry().then(() => {
   721→        if (!this._isMounted) return;
   722→        this.masonryLoading();
   723→      }).catch((e) => { console.error("KOT fetch failed:", e); });
   724→    },
   725→    handleOffline() {
   726→      // R37-FIX: Guard against post-unmount execution
   727→      if (!this._isMounted) return;
   728→      this.isOnline = false;
   729→      this.setStatusMessage("You are Offline");
   730→    },
   731→    setStatusMessage(message) {
   732→      this.statusMessage = message;
   733→    },
   734→    hideStatusMessageAfterDelay() {
   735→      if (this._statusTimeout) clearTimeout(this._statusTimeout);
   736→      this._statusTimeout = setTimeout(() => {
   737→        // R37-FIX: Guard against mutating unmounted component
   738→        if (!this._isMounted) return;
   739→        this.statusMessage = "";
   740→      }, 3000);
   741→    },
   742→    handleTransitionEnd() {
   743→      if (!this.isOnline) {
   744→        // Reset the status message after transition end
   745→        this.setStatusMessage("");
   746→      }
   747→    },

   749→  created() {
   750→    // R36-FIX: Initialize as non-reactive instance properties (no Proxy overhead)
   751→    this._isMounted = false;
   752→    this._fetchInProgress = null;
   753→    this.notifiedKots = new Set();
   754→    // API client as non-reactive instance property (avoids Proxy overhead)
   755→    this.call = markRaw(frappe.call());
   756→    // R37-FIX: socketHandler as non-reactive to avoid unnecessary Proxy overhead
   757→    this.socketHandler = null;
   758→    // R37-FIX: sortedItems cache map — avoids mutating reactive kot objects in computed
   759→    this._sortedItemsCache = new Map();
   760→    // R39-FIX: masonry as non-reactive instance property — Masonry objects are large
   761→    // and should not be tracked by Vue's reactivity system. Previously in data(),
   762→    // it triggered unnecessary Proxy overhead and could cause cascading re-renders
   763→    // when the masonry reference changed during reloadItems/layout calls.
   764→    this._masonry = null;
   765→    // R39-FIX: Track socket init retry timer for cleanup
   766→    this._socketRetryTimer = null;
   767→    // R40-FIX: Debounced masonry layout for socket events — prevents layout
   768→    // thrashing when multiple socket events arrive in rapid succession.
   769→    this._debouncedMasonryLayout = debounce(() => { this.masonryLoading(); }, 100);
   770→    // R40-FIX: Recursion guard for errorCaptured
   771→    this._errorCapturedRecovering = false;

   773→  mounted() {
   774→    this._isMounted = true;
   775→    window.addEventListener("online", this.handleOnline);
   776→    window.addEventListener("offline", this.handleOffline);
   777→    document.addEventListener("click", this.hideAudioAlertMessage);
   778→    // R36-FIX: Use route params instead of fragile URL parsing
   779→    const production = this.$route?.params?.production || '';
   780→    const decodedProduction = decodeURIComponent(production);
   781→    this.production = decodedProduction;
   782→
   783→    const debouncedMasonry = debounce(() => { this.masonryLoading(); }, 150);
   784→    this._resizeHandler = debouncedMasonry;
   785→    window.addEventListener("resize", this._resizeHandler);
   786→    this.masonryLoading();
   787→
   788→    // R39-FIX: Handle bfcache restore — when the browser restores this page from
   789→    // back-forward cache, the socket may be stale and timers may not fire.
   790→    // Re-initialize socket and re-fetch KOT data on pageshow with persisted=true.
   791→    this._pageshowHandler = (event) => {
   792→      if (event.persisted) {
   793→        // Page was restored from bfcache — re-fetch data and re-init socket
   794→        // R39-FIX: Use retry wrapper for bfcache restore re-fetch
   795→        this.fetchKOTWithRetry().then(() => { this.masonryLoading(); }).catch(() => {});
   796→        if (this._socket && !this._socket.connected) {
   797→          this._socket.connect();
   798→        }
   799→      }
   800→    };
   801→    window.addEventListener('pageshow', this._pageshowHandler);
   802→
   803→    // R39-FIX: Socket init with retry — if fetchSiteName fails (network issue,
   804→    // server down), retry with exponential backoff instead of giving up forever.
   805→    // Without this, a transient site-name fetch failure would leave the KDS
   806→    // without real-time updates until a full page reload.
   807→    const initSocketWithRetry = (retries = 0) => {
   808→      return initializeSocket().then(sock => {
   809→        if (sock) return sock;
   810→        // initializeSocket returned null (siteName fetch failed) — retry
   811→        const delay = Math.min(2000 * Math.pow(2, retries), 30000);
   812→        if (!this._isMounted) return null;
   813→        return new Promise(resolve => {
   814→          this._socketRetryTimer = setTimeout(() => {
   815→            if (!this._isMounted) { resolve(null); return; }
   816→            resolve(initSocketWithRetry(retries + 1));
   817→          }, delay);
   818→        });
   819→      });
   820→    };
   821→    // R40-FIX: Run socket init and auth in parallel, but store the socket
   822→    // in this._socket as soon as it resolves — even before auth completes.
   823→    // Previously, Promise.all meant the socket was only stored after BOTH
   824→    // socket init AND auth succeeded. If auth() rejected, the socket was
   825→    // orphaned (never stored in this._socket) and thus never disconnected
   826→    // in beforeUnmount, causing a connection leak.
   827→    const authPromise = this.auth();
   828→    this._socketInitPromise = initSocketWithRetry();
   829→
   830→    this._socketInitPromise
   831→      .then((sock) => {
   832→        // R40-FIX: If component unmounted during socket init, disconnect immediately
   833→        if (!this._isMounted) {
   834→          if (sock) sock.disconnect();
   835→          return null;
   836→        }
   837→        // R40-FIX: Store socket early so beforeUnmount can always clean it up
   838→        this._socket = sock;
   839→        // Now wait for auth to complete
   840→        return authPromise;
   841→      })
   842→      .then(() => {
   843→        if (!this._isMounted || !this._socket) return;
   844→        this._socket.on('connect_error', (err) => {
   845→          if (!this._isMounted) return;
   846→          console.error("Socket connection error:", err);
   847→          this.setStatusMessage("Connection error. Retrying...");
   848→        });
   849→        this._socket.on('disconnect', (reason) => {
   850→          if (!this._isMounted) return;
   851→          console.warn("Socket disconnected:", reason);
   852→          this.setStatusMessage("Connection lost. Reconnecting...");
   853→        });
   854→        this._socket.on('connect', () => {
   855→          if (!this._isMounted) return;
   856→          this.setStatusMessage("Reconnected");
   857→          this.hideStatusMessageAfterDelay();
   858→          // R38-FIX: Re-fetch KOT data after reconnect to sync any missed updates
   859→          // during disconnection period. Without this, the UI could show stale KOTs.
   860→          // R39-FIX: Use retry wrapper for transient errors on reconnect
   861→          this.fetchKOTWithRetry().then(() => { this.masonryLoading(); }).catch(() => {});
   862→        });
   863→
   864→        // R39-FIX: Use retry wrapper for initial fetch — transient server errors
   865→        // should not leave the KDS permanently blank
   866→        return this.fetchKOTWithRetry();
   867→      })
   868→      .then(() => {
   869→        if (!this._isMounted || !this._socket) return;
   870→        if (this.audio_alert === 1) {
   871→          this.showAudioAlertMessage = true;
   872→        }
   873→        this.socketHandler = (doc) => {
   874→          if (!this._isMounted) return;
   875→          // R39-FIX: Guard against null/undefined doc from malformed socket messages
   876→          if (!doc) return;
   877→          try {
   878→            if (this.audio_alert === 1) {
   879→              this.playAlertSound(doc.audio_file);
   880→            }
   881→            // R36-FIX: Namespace localStorage key per production station to avoid cross-tab collision
   882→            // R39-FIX: Normalize null kottime from localStorage — if getItem returns null
   883→            // (no previous value stored) and doc.last_kot_time is also null, they'd be
   884→            // equal and fall through to the incremental path incorrectly. Treat null
   885→            // localStorage value as a signal to do a full refresh.
   886→            let kottime = localStorage.getItem("kot_time_" + this.production);
   887→            if (doc.last_kot_time !== kottime || kottime === null) {
   888→              // Full refresh needed — skip intermediate mutations
   889→              this.fetchKOT().then(() => { this.masonryLoading(); }).catch((e) => { console.error("KOT fetch failed:", e); });
   890→            } else {
   891→              // R36-FIX: Guard against missing doc.kot to prevent TypeError crash
   892→              if (!doc.kot) {
   893→                this.fetchKOT().then(() => { this.masonryLoading(); }).catch((e) => { console.error("KOT fetch failed:", e); });
   894→                return;
   895→              }
   896→              // Incremental update — deduplicate to avoid duplicate cards
   897→              const existingIndex = this.kot.findIndex(k => k.name === doc.kot.name);
   898→              if (existingIndex !== -1) {
   899→                // R36-FIX: Preserve strikethrough state before Object.assign overwrites kot_items
   900→                const strikeMap = new Map(
   901→                  (this.kot[existingIndex].kot_items || []).map(i => [i.name, i.striked])
   902→                );
   903→                Object.assign(this.kot[existingIndex], { timecolor: 'text-black', timeRemaining: '— : —', ...doc.kot });
   904→                // Restore strikethrough state after Object.assign
   905→                if (this.kot[existingIndex].kot_items) {
   906→                  this.kot[existingIndex].kot_items.forEach(i => {
   907→                    if (strikeMap.has(i.name)) i.striked = strikeMap.get(i.name);
   908→                  });
   909→                }
   910→                // R37-FIX: Invalidate sorted cache for this KOT since items may have changed
   911→                this._sortedItemsCache.delete(doc.kot.name);
   912→              } else {
   913→                // R39-FIX: Removed showDiv: false — dead code
   914→                const newKot = { isRotated: false, timecolor: 'text-black', timeRemaining: '— : —', ...doc.kot };
   915→                this.kot.unshift(newKot);
   916→              }
   917→              this.updateQtyColorTable();
   918→              this.updateTimeRemaining();
   919→              // R40-FIX: Use debounced masonry layout for rapid socket events
   920→              this._debouncedMasonryLayout();
   921→            }
   922→            if (this._cancelTimeout) clearTimeout(this._cancelTimeout);
   923→            // R39-FIX: Only schedule cancel re-fetch if the KOT is actually a cancellation.
   924→            // Previously, _cancelTimeout was set for ALL socket events, causing an
   925→            // unnecessary re-fetch 1.5s after every non-cancel update. Now we only
   926→            // set the timeout when doc.kot.type is "Cancelled", matching the check inside.
   927→            if (doc.kot && doc.kot.type === "Cancelled") {
   928→              this._cancelTimeout = setTimeout(() => {
   929→                if (!this._isMounted) return;
   930→                this.fetchKOT().then(() => {
   931→                  this.masonryLoading();
   932→                }).catch((e) => { console.error("KOT fetch failed:", e); });
   933→              }, 1500);
   934→            }
   935→            if (doc.kot) {
   936→              try {
   937→                localStorage.setItem("kot_time_" + this.production, doc.kot.time);
   938→              } catch (e) {
   939→                if (import.meta.env?.DEV) console.error('localStorage write failed:', e);
   940→              }
   941→            }
   942→          } catch (err) {
   943→            if (import.meta.env?.DEV) console.error("Socket handler error:", err);
   944→          }
   945→        };
   946→        if (this._socket) this._socket.on(this.kot_channel, this.socketHandler);
   947→      })
   948→      .catch((error) => {
   949→        console.error("Initialization or authentication error:", error);
   950→        // R40-FIX: this._socket is now stored early, so beforeUnmount will
   951→        // disconnect it. No need for manual cleanup here.
   952→        if (this._isMounted) this.showModal = true;
   953→      });
   954→    this.timer = setInterval(this.updateTimeRemaining, 60000);
   955→  },
   956→  beforeUnmount() {
   957→    this._isMounted = false;
   958→    window.removeEventListener("online", this.handleOnline);
   959→    window.removeEventListener("offline", this.handleOffline);
   960→    document.removeEventListener("click", this.hideAudioAlertMessage);
   961→    window.removeEventListener("resize", this._resizeHandler);
   962→    if (this._pageshowHandler) window.removeEventListener('pageshow', this._pageshowHandler);
   963→    if (this.socketHandler && this._socket) {
   964→      this._socket.off(this.kot_channel, this.socketHandler);
   965→    }
   966→    if (this._socket) {
   967→      this._socket.off('connect_error');
   968→      this._socket.off('disconnect');
   969→      this._socket.off('connect');
   970→      this._socket.disconnect();
   971→    }
   972→    this._socketInitPromise = null;
   973→    if (this._cancelTimeout) clearTimeout(this._cancelTimeout);
   974→    if (this._statusTimeout) clearTimeout(this._statusTimeout);
   975→    if (this._socketRetryTimer) clearTimeout(this._socketRetryTimer);
   976→    if (this._fetchRetryTimer) clearTimeout(this._fetchRetryTimer);
   977→    if (this.timer) clearInterval(this.timer);
   978→    if (this._alertAudio) { this._alertAudio.pause(); this._alertAudio = null; }
   979→    // R39-FIX: Destroy non-reactive masonry instance on unmount
   980→    if (this._masonry) {
   981→      this._masonry.destroy();
   982→      this._masonry = null;
   983→    }
   984→  },
   985→  computed: {
   986→    // R39-FIX: sortedKotItems moved from computed (which returned a function) to
   987→    // a method. Previously, the computed returned a new function object every time
   988→    // its reactive dependencies (this.kot) changed, which happened every minute
   989→    // via updateTimeRemaining. Each new function object was unnecessary churn.
   990→    // As a method, the function identity is stable and the manual _sortedItemsCache
   991→    // still provides memoization. The template call `sortedKotItems(kot)` works
   992→    // identically for both computed-returning-function and method.
   993→    visibleKots() {
   994→      // R39-FIX: Removed !kot.showDiv filter — showDiv was always false (dead code).
   995→      // When production is empty (root route "/"), show ALL KOTs instead of filtering
   996→      // by empty string which would exclude KOTs that have a station assigned.
   997→      if (!this.production) {
   998→        return this.kot;
   999→      }
  1000→      return this.kot.filter(kot => kot.production === this.production);
  1001→    },
  1002→  },
