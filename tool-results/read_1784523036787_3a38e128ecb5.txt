   230→// R41-FIX: Add cancel() method to debounce return value so pending timers

   232→function debounce(fn, delay) {
   233→    let timer = null;
   234→    const debounced = function(...args) {
   235→        clearTimeout(timer);
   236→        timer = setTimeout(() => fn.apply(this, args), delay);
   237→    };
   238→    debounced.cancel = () => { clearTimeout(timer); timer = null; };
   239→    return debounced;
   240→}

   751→    fetchKOTWithRetry(maxRetries = 3, initialDelay = 2000) {
   752→      // R41-FIX: Clear any pending retry timer before starting a new retry chain.
   753→      // Without this, calling fetchKOTWithRetry() while a previous retry timer is
   754→      // pending would leave the old timer running, potentially causing duplicate
   755→      // fetchKOT calls when both timers fire.
   756→      if (this._fetchRetryTimer) { clearTimeout(this._fetchRetryTimer); this._fetchRetryTimer = null; }
   757→      // R42-FIX: Bump generation counter so any still-running retry chain from
   758→      // a previous fetchKOTWithRetry call knows it's been superseded. Without
   759→      // this, clearing the timer orphans the old promise — its resolve/reject
   760→      // callbacks are never called, leaking the promise and its closures.
   761→      const generation = ++this._fetchGeneration;
   762→      const attempt = (retriesLeft, delay) => {
   763→        // If a newer fetchKOTWithRetry has started, abort this chain
   764→        if (generation !== this._fetchGeneration) {
   765→          return Promise.reject(new Error('Superseded by newer fetchKOTWithRetry call'));
   766→        }
   767→        return this.fetchKOT().catch(err => {
   768→          if (retriesLeft <= 0 || !this._isMounted || generation !== this._fetchGeneration) throw err;
   769→          if (import.meta.env?.DEV) console.warn(`fetchKOT failed, retrying in ${delay}ms...`, err);
   770→          return new Promise((resolve, reject) => {
   771→            this._fetchRetryTimer = setTimeout(() => {
   772→              if (!this._isMounted || generation !== this._fetchGeneration) { reject(err); return; }
   773→              attempt(retriesLeft - 1, Math.min(delay * 2, 30000)).then(resolve).catch(reject);
   774→            }, delay);
   775→          });
   776→        });
   777→      };
   778→      return attempt(maxRetries, initialDelay);
   779→    },

  1014→        if (!this._isMounted) return null;
  1015→        return new Promise(resolve => {
  1016→          this._socketRetryTimer = setTimeout(() => {
  1017→            if (!this._isMounted) { resolve(null); return; }
  1018→            resolve(initSocketWithRetry(retries + 1));
  1019→          }, delay);
  1020→        });
  1021→      });
  1022→    };
  1023→    // R40-FIX: Run socket init and auth in parallel, but store the socket
  1024→    // in this._socket as soon as it resolves — even before auth completes.
  1025→    // Previously, Promise.all meant the socket was only stored after BOTH
  1026→    // socket init AND auth succeeded. If auth() rejected, the socket was
  1027→    // orphaned (never stored in this._socket) and thus never disconnected
  1028→    // in beforeUnmount, causing a connection leak.
  1029→    const authPromise = this.auth();
  1030→    this._socketInitPromise = initSocketWithRetry();
  1031→
  1032→    // R44-FIX: Start data fetch as soon as auth succeeds, independently of socket
  1033→    // init. Previously, data only loaded after BOTH socket AND auth completed,
  1034→    // causing a blank KDS when socket init was slow (siteName fetch + WebSocket
  1035→    // handshake can take 2-5s). Now the early fetch shows data immediately, and
  1036→    // the socket chain's fetchKOTWithRetry is skipped if data is already loaded.
  1037→    authPromise.then(() => {
  1038→      if (!this._isMounted) return;
  1039→      this.fetchKOTWithRetry().catch(() => {});
  1040→    }).catch(() => {});
  1041→
  1042→    this._socketInitPromise
  1043→      .then((sock) => {

  1045→        if (!this._isMounted) {
  1046→          if (sock) sock.disconnect();
  1047→          return null;
  1048→        }
  1049→        // R40-FIX: Store socket early so beforeUnmount can always clean it up
  1050→        this._socket = sock;
  1051→        // Now wait for auth to complete
  1052→        return authPromise;
  1053→      })
  1054→      .then(() => {
  1055→        if (!this._isMounted || !this._socket) return;
  1056→        this._socket.on('connect_error', (err) => {
  1057→          if (!this._isMounted) return;

  1059→          // During extended outages, socket.io retries every 1-5 seconds, and
  1060→          // each failed attempt triggered setStatusMessage, causing a Vue
  1061→          // re-render each time. Since the message text is identical, skip
  1062→          // the reactive assignment to avoid unnecessary re-renders.

  1064→            console.error("Socket connection error:", err);
  1065→            this.setStatusMessage("Connection error. Retrying...");
  1066→          }
  1067→        });
  1068→        this._socket.on('disconnect', (reason) => {
  1069→          if (!this._isMounted) return;
  1070→          console.warn("Socket disconnected:", reason);
  1071→          // R42-FIX: Skip reactive assignment if message is already showing
  1072→          // the same text — avoids unnecessary Vue re-renders during flapping.
  1073→          if (this.statusMessage !== "Connection lost. Reconnecting...") {
  1074→            this.setStatusMessage("Connection lost. Reconnecting...");
  1075→          }
  1076→        });

  1078→          if (!this._isMounted) return;
  1079→          this.setStatusMessage("Reconnected");
  1080→          this.hideStatusMessageAfterDelay();
  1081→          // R38-FIX: Re-fetch KOT data after reconnect to sync any missed updates
  1082→          // during disconnection period. Without this, the UI could show stale KOTs.
  1083→          // R39-FIX: Use retry wrapper for transient errors on reconnect
  1084→          // R43-FIX: Remove redundant .then(() => masonryLoading()) — fetchKOT()
  1085→          // already calls masonryLoading(true) on success.
  1086→          // R45-FIX: Handle fetch failure — previously .catch(() => {}) silently
  1087→          // swallowed errors. During extended outages, the user's session may have
  1088→          // expired, so auth errors need to show the login modal. Other errors
  1089→          // show a status message so the user knows data refresh failed.
  1090→          this.fetchKOTWithRetry().catch((error) => {
  1091→            this._handleFetchError(error, "Reconnected but data refresh failed. Click Refresh.");
  1092→          });
  1093→        });
  1094→
  1095→        // R39-FIX: Use retry wrapper for initial fetch — transient server errors
  1096→        // should not leave the KDS permanently blank.
  1097→        // R44-FIX: Skip if early fetch (started after auth) already loaded data.
  1098→        // This avoids a redundant full refresh when the early fetch succeeded.
  1099→        if (this.kot.length === 0) {
  1100→          return this.fetchKOTWithRetry();
  1101→        }
  1102→      })
  1103→      .then(() => {
  1104→        if (!this._isMounted || !this._socket) return;
  1105→        if (this.audio_alert === 1) {
  1106→          this.showAudioAlertMessage = true;
  1107→        }
  1108→        this.socketHandler = (doc) => {
  1109→          if (!this._isMounted) return;
  1110→          // R39-FIX: Guard against null/undefined doc from malformed socket messages
  1111→          if (!doc) return;
  1112→          try {
  1113→            if (this.audio_alert === 1) {
  1114→              this.playAlertSound(doc.audio_file);
  1115→            }
  1116→            // R36-FIX: Namespace localStorage key per production station to avoid cross-tab collision
  1117→            // R39-FIX: Normalize null kottime from localStorage — if getItem returns null
  1118→            // (no previous value stored) and doc.last_kot_time is also null, they'd be
  1119→            // equal and fall through to the incremental path incorrectly. Treat null
  1120→            // localStorage value as a signal to do a full refresh.
  1121→            let kottime = localStorage.getItem("kot_time_" + this.production);
  1122→            if (doc.last_kot_time !== kottime || kottime === null) {
  1123→              // R43-FIX: Move localStorage sentinel update AFTER fetchKOT succeeds.
  1124→              // Previously, localStorage.setItem ran before fetchKOT, so if the fetch
  1125→              // failed, the sentinel was already updated. Subsequent socket events
  1126→              // would take the incremental path instead of the full-refresh path,
  1127→              // potentially missing KOTs that were added/removed server-side.
  1128→              // R43-FIX: Remove redundant masonryLoading() — fetchKOT() already calls
  1129→              // masonryLoading(true) at the end of its .then() handler.
  1130→              this.fetchKOT().then(() => {
  1131→                if (doc.kot && doc.kot.time != null) {
  1132→                  try {
  1133→                    localStorage.setItem("kot_time_" + this.production, doc.kot.time);
  1134→                  } catch (e) {
  1135→                    if (import.meta.env?.DEV) console.error('localStorage write failed:', e);
  1136→                  }
  1137→                }
  1138→              }).catch((e) => { console.error("KOT fetch failed:", e); });
  1139→              return;
  1140→            }
  1141→            // R36-FIX: Guard against missing doc.kot to prevent TypeError crash
  1142→            // R44-FIX: Also guard against non-object doc.kot (e.g., array, string).
  1143→            // If the server sends a malformed doc.kot, accessing .name would

  1145→            // be unshifted into the KOT array, corrupting the display.

  1147→              this.fetchKOT().catch((e) => { console.error("KOT fetch failed:", e); });
  1148→              return;
  1149→            }
  1150→            // Incremental update — deduplicate to avoid duplicate cards
  1151→            const existingIndex = this.kot.findIndex(k => k.name === doc.kot.name);
  1152→            let targetKot;
  1153→            if (existingIndex !== -1) {
  1154→              targetKot = this.kot[existingIndex];
  1155→              // R36-FIX: Preserve strikethrough state before Object.assign overwrites kot_items
  1156→              const strikeMap = new Map(
  1157→                (targetKot.kot_items || []).map(i => [i.name, i.striked])
  1158→              );
  1159→              // R43-FIX: Exclude 'name' from the spread to prevent overwriting the KOT's
  1160→              // primary key. If the server sends a different name in doc.kot, the KOT
  1161→              // would become orphaned — findIndex in serveOrder/confirmOrder uses the
  1162→              // original name as the lookup key.
  1163→              const { name: _incomingName, ...kotData } = doc.kot;
  1164→              Object.assign(targetKot, { timecolor: 'text-black', timeRemaining: '— : —', ...kotData });

  1166→              if (targetKot.kot_items) {
  1167→                targetKot.kot_items.forEach(i => {
  1168→                  if (strikeMap.has(i.name)) i.striked = strikeMap.get(i.name);
  1169→                });
  1170→              }
  1171→              // R37-FIX: Invalidate sorted cache for this KOT since items may have changed
  1172→              this._sortedItemsCache.delete(doc.kot.name);
  1173→            } else {
  1174→              // R39-FIX: Removed showDiv: false — dead code
  1175→              targetKot = { isRotated: false, timecolor: 'text-black', timeRemaining: '— : —', ...doc.kot };
  1176→              this.kot.unshift(targetKot);
  1177→            }
  1178→            // R44-FIX: Use targeted single-KOT methods instead of processing ALL KOTs.
  1179→            // The incremental path only modifies one KOT, so O(1) is sufficient.
  1180→            this._updateSingleKotQtyColor(targetKot);
  1181→            this._updateSingleKotTimeRemaining(targetKot);
  1182→            // R40-FIX: Use debounced masonry layout for rapid socket events
  1183→            this._debouncedMasonryLayout();
  1184→            // R41-FIX: Cancel timeout and localStorage write are now ONLY in the incremental path.
  1185→            // Previously these ran after both full-refresh and incremental branches.
  1186→            if (this._cancelTimeout) clearTimeout(this._cancelTimeout);
  1187→            // R39-FIX: Only schedule cancel re-fetch if the KOT is actually a cancellation.
  1188→            if (doc.kot.type === "Cancelled") {
  1189→              this._cancelTimeout = setTimeout(() => {
  1190→                if (!this._isMounted) return;
  1191→                // R43-FIX: Remove redundant masonryLoading() — fetchKOT() calls it.
  1192→                this.fetchKOT().catch((e) => { console.error("KOT fetch failed:", e); });
  1193→              }, 1500);

  1195→            // R41-FIX: Guard against storing null/undefined as string "null"/"undefined"
  1196→            // in localStorage. These corrupted sentinel values would cause the
  1197→            // next socket event to always take the full-refresh path.
  1198→            if (doc.kot.time != null) {
  1199→              try {
  1200→                localStorage.setItem("kot_time_" + this.production, doc.kot.time);
  1201→              } catch (e) {
  1202→                if (import.meta.env?.DEV) console.error('localStorage write failed:', e);

  1204→            }
  1205→          } catch (err) {
  1206→            if (import.meta.env?.DEV) console.error("Socket handler error:", err);

  1208→        };
  1209→        if (this._socket) this._socket.on(this.kot_channel, this.socketHandler);
  1210→      })
