    18→def _get_order_invoice_doc(table=None, invoiceNo=None, order_type=None, is_payment=None):
    19→    """Internal: returns POS Invoice document object for server-side manipulation.
    20→    Used by sync_order and make_invoice which need the doc, not a dict."""
    21→    # Verify user has access to the table's room (BE-R36-017)
    22→    if table:
    23→        from ury.ury_pos.api import _get_user_branch_rooms
    24→        table_room = frappe.db.get_value("URY Table", table, "restaurant_room")
    25→        if table_room:
    26→            user_rooms = [r.room for r in _get_user_branch_rooms() if r.room]
    27→            if user_rooms and table_room not in user_rooms:
    28→                frappe.throw(_("You do not have access to this table's room"), frappe.PermissionError)
    29→
    30→    if table:
    31→        # Lock the table row to prevent concurrent invoice creation (BE-R36-001)
    32→        frappe.db.sql(
    33→            "SELECT name FROM `tabURY Table` WHERE name = %s FOR UPDATE",
    34→            (table,), as_dict=True
    35→        )
    36→        
    37→        if is_payment == "Payments":
    38→            invoice_name = frappe.get_value(
    39→                "POS Invoice", dict(restaurant_table=table, docstatus=0, name=invoiceNo)
    40→            )
    41→            
    42→        else:
    43→            if invoiceNo:
    44→                invoice_name = frappe.get_value(
    45→                    "POS Invoice",
    46→                    dict(restaurant_table=table, docstatus=0, name=invoiceNo),
    47→                )
    48→               
    49→            else:
    50→                invoice_name = frappe.get_value(
    51→                    "POS Invoice",
    52→                    dict(restaurant_table=table, docstatus=0, invoice_printed=0),
    53→                )
    54→                
    55→        branch, menu_name, restaurant = get_restaurant_and_menu_name(table)
    56→
    57→        if invoice_name:
    58→            invoice = frappe.get_doc("POS Invoice", invoice_name)
    59→
    60→        else:
    61→            invoice = frappe.new_doc("POS Invoice")
    62→
    63→            invoice.naming_series = frappe.db.get_value(
    64→                "URY Restaurant", restaurant, "invoice_series_prefix"
    65→            )
    66→
    67→            invoice.is_pos = 1
    68→            invoice.update_stock = 1
    69→            invoice.restaurant = restaurant
    70→            invoice.branch = branch
    71→
    72→            is_take_away = frappe.db.get_value("URY Table", table, "is_take_away")
    73→            if is_take_away == 1:
    74→                invoice.order_type = "Take Away"
    75→            else:
    76→                invoice.order_type= "Dine In"
    77→
    78→        invoice.taxes_and_charges = frappe.db.get_value(
    79→            "URY Restaurant", restaurant, "default_tax_template"
    80→        )
    81→
    82→        invoice.selling_price_list = frappe.db.get_value(
    83→            "Price List", dict(restaurant_menu=menu_name, enabled=1)
    84→        )
    85→
    86→    else:
    87→
    88→        invoice_name = frappe.get_value(
    89→                "POS Invoice", dict(docstatus=0, name=invoiceNo)
    90→            )
    91→            
    92→        if invoice_name:
    93→            invoice = frappe.get_doc("POS Invoice", invoice_name)
    94→            
    95→
    96→        else:
    97→            invoice = frappe.new_doc("POS Invoice")
    98→            invoice.is_pos = 1
    99→            invoice.update_stock = 1
   100→        
   101→        branch = getBranch()
   102→        restaurant = frappe.db.get_value("URY Restaurant", {"branch": branch}, "name")
   103→   
   104→        menu=get_menu_name(order_type)
   105→ 
   106→        if (order_type == "Aggregators" and frappe.db.get_value("Branch", branch, "custom_no_taxes") == 0) or order_type != "Aggregators":
   107→            invoice.taxes_and_charges = frappe.db.get_value("URY Restaurant", restaurant, "default_tax_template")
   108→        
   109→        invoice.selling_price_list = frappe.db.get_value(
   110→            "Price List", dict(restaurant_menu=menu, enabled=1)
   111→        )
   112→
   113→    return invoice

   159→@frappe.whitelist()
   160→def sync_order(
   161→    items,
   162→    cashier,
   163→    owner,
   164→    mode_of_payment,
   165→    customer,
   166→    no_of_pax,
   167→    last_invoice,
   168→    waiter,
   169→    pos_profile,
   170→    last_modified_time=None,
   171→    table=None,
   172→    invoice=None,
   173→    comments=None,
   174→    order_type=None,
   175→    aggregator_id=None,
   176→    room=None
   177→):
   178→    frappe.only_for("Restaurant Manager", "Restaurant User", "Cashier")
   179→    # Validate pos_profile belongs to user's branch (BE-R36-002)
   180→    pos_profile_branch = frappe.db.get_value("POS Profile", pos_profile, "branch")
   181→    user_branch = getBranch()
   182→    if pos_profile_branch != user_branch:
   183→        frappe.throw(_("POS Profile does not belong to your branch"), frappe.PermissionError)
   184→    user_role = frappe.get_roles()
   185→    billing_roles = frappe.get_all(
   186→        "POS Profile Role",
   187→        filters={"parent": pos_profile},
   188→        fields=["role"],
   189→        pluck="role",
   190→    )
   191→    billing_user = bool(set(user_role).intersection(billing_roles))
   192→
   193→    # Check if the last invoice was already billed
   194→    if (
   195→        last_invoice
   196→        and frappe.db.get_value("POS Invoice", last_invoice, "invoice_printed") == 1
   197→        and (not billing_user)
   198→    ):
   199→        frappe.msgprint(
   200→            title=_("Invoice Already Billed"),
   201→            indicator="red",
   202→            msg=_("This order has already been billed. Please reload the page."),
   203→        )
   204→        return {"status": "Failure"}
   205→
   206→    # R41-FIX: Use _get_order_invoice_doc to get the document object, not a dict.
   207→    # The public get_order_invoice returns a dict for frontend safety (BE-R36-003),
   208→    # but sync_order needs the document to call .save() etc.
   209→    invoice = _get_order_invoice_doc(table, invoice, order_type)
   210→
   211→    # R41-FIX: Validate last_invoice belongs to user's branch
   212→    if last_invoice:
   213→        last_inv_branch = frappe.db.get_value("POS Invoice", last_invoice, "branch")
   214→        if last_inv_branch and last_inv_branch != user_branch:
   215→            frappe.throw(_("Invoice does not belong to your branch"), frappe.PermissionError)
   216→
   217→    if last_invoice and last_modified_time:
   218→        lastModifiedTime = invoice.modified
   219→        from datetime import datetime
   220→
   221→        if isinstance(last_modified_time, str):
   222→            try:
   223→                last_modified_time = datetime.strptime(
   224→                    last_modified_time, "%Y-%m-%d %H:%M:%S.%f"
   225→                )
   226→            except ValueError:
   227→                last_modified_time = datetime.strptime(
   228→                    last_modified_time, "%Y-%m-%d %H:%M:%S"
   229→                )
   230→        if isinstance(lastModifiedTime, str):
   231→            try:
   232→                lastModifiedTime = datetime.strptime(
   233→                    lastModifiedTime, "%Y-%m-%d %H:%M:%S.%f"
   234→                )
   235→            except ValueError:
   236→                lastModifiedTime = datetime.strptime(
   237→                    lastModifiedTime, "%Y-%m-%d %H:%M:%S"
   238→                )
   239→        if lastModifiedTime != last_modified_time:
   240→            frappe.msgprint(
   241→                title=_("Order has been modified"),
   242→                indicator="red",
   243→                msg=_("This order has been modified. Please reload the page to retrieve the latest edits."),
   244→            )
   245→            return {"status": "Failure"}
   246→    else:
   247→        if invoice.name and invoice.invoice_printed == 0 and not billing_user:
   248→            frappe.msgprint(
   249→                title=_("Table Occupied"),
   250→                indicator="red",
   251→                msg=_("{0} is already occupied. Please refresh the page.").format(table),
   252→            )
   253→            return {"status": "Failure"}
   254→
   255→    if not customer:
   256→        frappe.throw(_("Please enter valid customer details"))
   257→    else:
   258→        invoice.customer = customer
   259→
   260→    if order_type:
   261→        invoice.order_type = order_type
   262→
   263→    invoice.mobile_number = frappe.db.get_value("Customer", customer, "mobile_number")
   264→    if comments:
   265→        invoice.custom_comments = comments
   266→    invoice.no_of_pax = no_of_pax
   267→    invoice.pos_profile = pos_profile
   268→    invoice.cashier = cashier
   269→    invoice.waiter = waiter
   270→    invoice.custom_aggregator_id = aggregator_id
   271→    invoice.custom_restaurant_room =room
   272→    invoice.restaurant_table = table
   273→    
   274→    if order_type == "Aggregators":
   275→        price_list = frappe.db.get_value("Aggregator Settings",{"customer": customer, "parent": invoice.branch, "parenttype": "Branch"},"price_list",)
   276→        
   277→        if not price_list:
   278→            frappe.throw(_("Price list for customer {0} in branch {1} not found in Aggregator Settings.").format(customer, invoice.branch))
   279→    else:
   280→        price_list = invoice.selling_price_list
   281→
   282→    # dummy payment
   283→    if invoice.invoice_created == 0:
   284→        invoice.append(
   285→            "payments",
   286→            dict(mode_of_payment=mode_of_payment, amount=invoice.grand_total),
   287→        )
   288→        invoice.invoice_created = 1
   289→
   290→    past_item = []
   291→    for item in invoice.items:
   292→        previous_item = {
   293→            "item_code": item.item_code,
   294→            "item_name": item.item_name,
   295→            "qty": item.qty,
   296→            "comments": "",
   297→        }
   298→        past_item.append(previous_item)
   299→        
   300→
   301→    # Conditional checking for 'items' type:
   302→    # - 'ury': JSON passed, hence using isinstance
   303→    # - 'ury_pos': Already formatted list, hence using else
   304→    if isinstance(items, str):
   305→        items = json.loads(items)
   306→    invoice.items = []
   307→    
   308→    menu = frappe.db.get_value("URY Menu", {"branch": invoice.branch}, "name")
   309→
   310→    # Batch-fetch courses, prices, and cost_center to avoid N+1 queries
   311→    item_codes = [d.get("item") for d in items]
   312→    item_codes_unique = list(set(item_codes))
   313→
   314→    courses = {}
   315→    if item_codes_unique:
   316→        course_rows = frappe.db.get_all(
   317→            "URY Menu Item",
   318→            filters={"item": ("in", item_codes_unique), "parent": menu},
   319→            fields=["item", "course"],
   320→        )
   321→        courses = {r.item: r.course for r in course_rows}
   322→
   323→    prices = {}
   324→    if item_codes_unique:
   325→        price_rows = frappe.db.get_all(
   326→            "Item Price",
   327→            filters={"item_code": ("in", item_codes_unique), "price_list": price_list},
   328→            fields=["item_code", "price_list_rate"],
   329→        )
   330→        prices = {r.item_code: r.price_list_rate for r in price_rows}
   331→
   332→    cost_center = frappe.db.get_value("POS Profile", pos_profile, "cost_center")
   333→
   334→    for d in items:
   335→        course = courses.get(d.get("item"))
   336→        rate = prices.get(d.get("item"))
   337→        if not rate:
   338→            frappe.throw(_("No item price found for Item: {0} in Price List: {1}. Please check the price list settings.").format(d.get("item"), price_list))
   339→
   340→        invoice.append(
   341→            "items",
   342→            dict(
   343→                item_code=d.get("item"),
   344→                item_name=d.get("item_name"),
   345→                qty=d.get("qty"),
   346→                **({"custom_course": course} if course else {}),
   347→                comment=d.get("comment"),
   348→                rate=rate,
   349→                price_list_rate=rate,
   350→                base_price_list_rate=rate,
   351→                cost_center=cost_center,
   352→            ),
   353→        )
   354→
   355→    try:
   356→        invoice.save()
   357→    except Exception as e:
   358→        frappe.log_error(str(e))
   359→        frappe.throw(_("An error occurred. Please check the error log."))   
   360→
   361→
   362→    try:
   363→        kot_execute(invoice.name, customer, table, items, past_item, comments)
   364→
   365→    except Exception as e:
   366→        # Log the KOT error but also notify the user — kitchen won't see this order
   367→        error_msg = f"KOT Creation Failed: {str(e)}"            
   368→        frappe.log_error(error_msg, "KOT Error")
   369→        frappe.msgprint(
   370→            title=_("KOT Creation Failed"),
   371→            indicator="orange",
   372→            msg=_("The order was saved but the KOT was not sent to the kitchen. Please check the error log or retry."),
   373→        )
   374→
   375→    # table status
   376→    if invoice.invoice_printed == 0:
   377→        frappe.db.set_value(
   378→            "URY Table", table, {"occupied": 1, "latest_invoice_time": invoice.creation}
   379→        )
   380→
   381→    invoice.db_set("owner", frappe.session.user)
   382→    return {
   383→        "name": invoice.name,
   384→        "customer": invoice.customer,
   385→        "grand_total": invoice.grand_total,
   386→        "modified": invoice.modified,
   387→        "invoice_printed": invoice.invoice_printed,
   388→        "restaurant_table": invoice.restaurant_table,
   389→    }

   412→@frappe.whitelist()
   413→def get_restaurant_and_menu_name(table):
   414→    frappe.only_for("Restaurant Manager", "Restaurant User", "Cashier")
   415→    if not table:
   416→        frappe.throw(_("Please select a table"))
   417→
   418→    result = frappe.get_value(
   419→        "URY Table",
   420→        table,
   421→        ["restaurant", "branch", "restaurant_room"],
   422→    )
   423→    if not result:
   424→        frappe.throw(_("URY Table {0} not found").format(table))
   425→    restaurant, branch, room = result
   426→    room_wise_menu = frappe.db.get_value(
   427→        "URY Restaurant",
   428→        restaurant,
   429→        "room_wise_menu",
   430→    )
   431→
   432→    if not room_wise_menu:
   433→        menu = frappe.db.get_value("URY Restaurant", restaurant, "active_menu")
   434→    else:
   435→        menu = frappe.db.get_value(
   436→            "Menu for Room",
   437→            {"parent": restaurant, "room": room},
   438→            "menu",
   439→        )
   440→
   441→    if not menu:
   442→        frappe.throw(
   443→            _("Please set an active menu for Restaurant {0}").format(restaurant)
   444→        )
   445→
   446→    return branch, menu, restaurant

   474→@frappe.whitelist()
   475→def pos_opening_check():
   476→    frappe.only_for("Restaurant Manager", "Restaurant User", "Cashier")
   477→    user = frappe.session.user
   478→    # Handle the administrator case differently
   479→    if user == "Administrator":
   480→        return {
   481→            "opening_exists": False,  # Assuming no POS opening entry is needed for Administrator
   482→            "cashier": None,
   483→            "pos_profile": None,
   484→        }
   485→    
   486→    details = getBranchRoom()
   487→    if not details:
   488→        frappe.throw(_("No room/branch configuration found for the current user."))
   489→    room = details[0].get('name')
   490→    branch = details[0].get('branch')
   491→    
   492→    pos_opening_list = frappe.db.sql("""
   493→        SELECT DISTINCT `tabPOS Opening Entry`.name 
   494→        FROM `tabPOS Opening Entry`
   495→        INNER JOIN `tabMultiple Rooms` 
   496→        ON `tabMultiple Rooms`.parent = `tabPOS Opening Entry`.name
   497→        WHERE `tabPOS Opening Entry`.branch = %s
   498→        AND `tabPOS Opening Entry`.status = 'Open'
   499→        AND `tabPOS Opening Entry`.docstatus = 1
   500→        AND `tabMultiple Rooms`.room = %s
   501→    """, (branch, room), as_dict=True)
   502→    
   503→    
   504→    result = {
   505→        "opening_exists": len(pos_opening_list) > 0,
   506→        "cashier": None,
   507→        "pos_profile": None,
   508→    }
   509→
   510→    if result["opening_exists"]:
   511→        opening_vals = frappe.db.get_value(
   512→            "POS Opening Entry", pos_opening_list[0].name,
   513→            ["user", "pos_profile"],
   514→            as_dict=True,
   515→        )
   516→        result["cashier"] = opening_vals.user
   517→        result["pos_profile"] = opening_vals.pos_profile
   518→        
   519→    return result

   522→@frappe.whitelist()
   523→def table_transfer(table, newTable, invoice):
   524→    frappe.only_for("Restaurant Manager", "Restaurant User", "Cashier")
   525→    if not frappe.has_permission("POS Invoice", "write", invoice):
   526→        frappe.throw(_("Not permitted to transfer tables"), frappe.PermissionError)
   527→    # R39-FIX: Validate invoice belongs to user's branch
   528→    inv_branch = frappe.db.get_value("POS Invoice", invoice, "branch")
   529→    if not inv_branch:
   530→        frappe.throw(_("POS Invoice {0} not found").format(invoice))
   531→    user_branch = getBranch()
   532→    if inv_branch != user_branch:
   533→        frappe.throw(_("You do not have access to invoices from another branch"), frappe.PermissionError)
   534→    # Lock both source and destination tables to prevent race conditions (BE-R36-007)
   535→    table_data = frappe.db.sql(
   536→        """SELECT name, restaurant_room, occupied FROM `tabURY Table`
   537→           WHERE name IN %s FOR UPDATE""",
   538→        ((table, newTable),), as_dict=True
   539→    )
   540→    table_map = {t.name: t for t in table_data}
   541→    current_room = table_map[table].restaurant_room if table in table_map else frappe.db.get_value("URY Table", table, "restaurant_room")
   542→    new_table_data = [table_map[newTable]] if newTable in table_map else []

   707→
   708→    # R39-FIX: Validate POS Profile belongs to user's branch

   717→        if additionalDiscount < 0 or additionalDiscount > 100:
