   229→// R41-FIX: Add cancel() method to debounce return value so pending timers

   231→function debounce(fn, delay) {
   232→    let timer = null;
   233→    const debounced = function(...args) {
   234→        clearTimeout(timer);
   235→        timer = setTimeout(() => fn.apply(this, args), delay);
   236→    };
   237→    debounced.cancel = () => { clearTimeout(timer); timer = null; };
   238→    return debounced;
   239→}

   635→    calculateTimeRemaining(targetTime) {
   636→      if (!targetTime || !targetTime.includes(":")) return '— : —';
   637→      const currentTime = new Date();
   638→      // R39-FIX: Guard against malformed time strings with fewer than 3 parts.
   639→      // Previously, "HH:MM" without seconds would set targetSeconds=undefined,
   640→      // which Date() treats as NaN → Invalid Date → NaN propagation.
   641→      const parts = targetTime.split(":");
   642→      const targetHours = parseInt(parts[0], 10);
   643→      const targetMinutes = parseInt(parts[1], 10);
   644→      const targetSeconds = parseInt(parts[2] || '0', 10);
   645→      if (isNaN(targetHours) || isNaN(targetMinutes) || isNaN(targetSeconds)) return '— : —';
   646→      let targetDate = new Date(
   647→        currentTime.getFullYear(),
   648→        currentTime.getMonth(),
   649→        currentTime.getDate(),
   650→        targetHours,
   651→        targetMinutes,
   652→        targetSeconds
   653→      );
   654→      // Handle cross-midnight KOTs: if target is in the future, it was created yesterday
   655→      if (targetDate > currentTime) {
   656→        targetDate = new Date(targetDate.getTime() - 86400000);
   657→      }
   658→
   659→      const timeDifference = Math.max(0, currentTime - targetDate);
   660→      const hoursRemaining = Math.floor(timeDifference / 3600000);
   661→      const minutesRemaining = Math.floor((timeDifference % 3600000) / 60000);
   662→
   663→      return `${hoursRemaining} : ${String(minutesRemaining).padStart(2, '0')}`;
   664→    },

   669→      // R41-FIX: Clear any pending retry timer before starting a new retry chain.
   670→      // Without this, calling fetchKOTWithRetry() while a previous retry timer is
   671→      // pending would leave the old timer running, potentially causing duplicate
   672→      // fetchKOT calls when both timers fire.

   675→        return this.fetchKOT().catch(err => {
   676→          if (retriesLeft <= 0 || !this._isMounted) throw err;
