   230→// R41-FIX: Add cancel() method to debounce return value so pending timers

   232→function debounce(fn, delay) {
   233→    let timer = null;
   234→    const debounced = function(...args) {
   235→        clearTimeout(timer);
   236→        timer = setTimeout(() => fn.apply(this, args), delay);
   237→    };
   238→    debounced.cancel = () => { clearTimeout(timer); timer = null; };
   239→    return debounced;
   240→}

   709→    calculateTimeRemaining(targetTime) {
   710→      if (!targetTime || !targetTime.includes(":")) return '— : —';
   711→      const currentTime = new Date();
   712→      // R39-FIX: Guard against malformed time strings with fewer than 3 parts.
   713→      // Previously, "HH:MM" without seconds would set targetSeconds=undefined,
   714→      // which Date() treats as NaN → Invalid Date → NaN propagation.
   715→      const parts = targetTime.split(":");
   716→      const targetHours = parseInt(parts[0], 10);
   717→      const targetMinutes = parseInt(parts[1], 10);
   718→      const targetSeconds = parseInt(parts[2] || '0', 10);
   719→      if (isNaN(targetHours) || isNaN(targetMinutes) || isNaN(targetSeconds)) return '— : —';
   720→      let targetDate = new Date(
   721→        currentTime.getFullYear(),
   722→        currentTime.getMonth(),
   723→        currentTime.getDate(),
   724→        targetHours,
   725→        targetMinutes,
   726→        targetSeconds
   727→      );
   728→      // Handle cross-midnight KOTs: if target is in the future, it was created yesterday
   729→      if (targetDate > currentTime) {
   730→        targetDate = new Date(targetDate.getTime() - 86400000);
   731→      }
   732→
   733→      const timeDifference = Math.max(0, currentTime - targetDate);
   734→      const hoursRemaining = Math.floor(timeDifference / 3600000);
   735→      const minutesRemaining = Math.floor((timeDifference % 3600000) / 60000);
   736→
   737→      return `${hoursRemaining} : ${String(minutesRemaining).padStart(2, '0')}`;
   738→    },

   740→    // Without this, a transient server error leaves the KDS showing stale
   741→    // data indefinitely — the only recovery was a manual page reload.
   742→    fetchKOTWithRetry(maxRetries = 3, initialDelay = 2000) {
   743→      // R41-FIX: Clear any pending retry timer before starting a new retry chain.
   744→      // Without this, calling fetchKOTWithRetry() while a previous retry timer is
   745→      // pending would leave the old timer running, potentially causing duplicate
   746→      // fetchKOT calls when both timers fire.
   747→      if (this._fetchRetryTimer) { clearTimeout(this._fetchRetryTimer); this._fetchRetryTimer = null; }
   748→      // R42-FIX: Bump generation counter so any still-running retry chain from
   749→      // a previous fetchKOTWithRetry call knows it's been superseded. Without
   750→      // this, clearing the timer orphans the old promise — its resolve/reject
   751→      // callbacks are never called, leaking the promise and its closures.
   752→      const generation = ++this._fetchGeneration;
   753→      const attempt = (retriesLeft, delay) => {
   754→        // If a newer fetchKOTWithRetry has started, abort this chain
   755→        if (generation !== this._fetchGeneration) {
   756→          return Promise.reject(new Error('Superseded by newer fetchKOTWithRetry call'));
   757→        }
   758→        return this.fetchKOT().catch(err => {
   759→          if (retriesLeft <= 0 || !this._isMounted || generation !== this._fetchGeneration) throw err;
   760→          if (import.meta.env?.DEV) console.warn(`fetchKOT failed, retrying in ${delay}ms...`, err);
   761→          return new Promise((resolve, reject) => {
   762→            this._fetchRetryTimer = setTimeout(() => {
   763→              if (!this._isMounted || generation !== this._fetchGeneration) { reject(err); return; }
   764→              attempt(retriesLeft - 1, Math.min(delay * 2, 30000)).then(resolve).catch(reject);
   765→            }, delay);
   766→          });
   767→        });
   768→      };
   769→      return attempt(maxRetries, initialDelay);
   770→    },

  1019→            this.setStatusMessage("Connection error. Retrying...");
  1020→          }
  1021→        });

  1023→          if (!this._isMounted) return;
  1024→          console.warn("Socket disconnected:", reason);
  1025→          // R42-FIX: Skip reactive assignment if message is already showing
  1026→          // the same text — avoids unnecessary Vue re-renders during flapping.

  1028→            this.setStatusMessage("Connection lost. Reconnecting...");
  1029→          }
  1030→        });

  1032→          if (!this._isMounted) return;
  1033→          this.setStatusMessage("Reconnected");
  1034→          this.hideStatusMessageAfterDelay();
  1035→          // R38-FIX: Re-fetch KOT data after reconnect to sync any missed updates
  1036→          // during disconnection period. Without this, the UI could show stale KOTs.
  1037→          // R39-FIX: Use retry wrapper for transient errors on reconnect
  1038→          // R43-FIX: Remove redundant .then(() => masonryLoading()) — fetchKOT()
  1039→          // already calls masonryLoading(true) on success.
  1040→          this.fetchKOTWithRetry().catch(() => {});
  1041→        });
  1042→
  1043→        // R39-FIX: Use retry wrapper for initial fetch — transient server errors
  1044→        // should not leave the KDS permanently blank.
  1045→        // R44-FIX: Skip if early fetch (started after auth) already loaded data.
  1046→        // This avoids a redundant full refresh when the early fetch succeeded.
  1047→        if (this.kot.length === 0) {
  1048→          return this.fetchKOTWithRetry();
  1049→        }
  1050→      })
  1051→      .then(() => {
  1052→        if (!this._isMounted || !this._socket) return;
  1053→        if (this.audio_alert === 1) {
  1054→          this.showAudioAlertMessage = true;
  1055→        }
  1056→        this.socketHandler = (doc) => {
  1057→          if (!this._isMounted) return;
  1058→          // R39-FIX: Guard against null/undefined doc from malformed socket messages
  1059→          if (!doc) return;
  1060→          try {
  1061→            if (this.audio_alert === 1) {
  1062→              this.playAlertSound(doc.audio_file);
  1063→            }
  1064→            // R36-FIX: Namespace localStorage key per production station to avoid cross-tab collision
  1065→            // R39-FIX: Normalize null kottime from localStorage — if getItem returns null
  1066→            // (no previous value stored) and doc.last_kot_time is also null, they'd be
  1067→            // equal and fall through to the incremental path incorrectly. Treat null
  1068→            // localStorage value as a signal to do a full refresh.
  1069→            let kottime = localStorage.getItem("kot_time_" + this.production);
  1070→            if (doc.last_kot_time !== kottime || kottime === null) {
  1071→              // R43-FIX: Move localStorage sentinel update AFTER fetchKOT succeeds.
  1072→              // Previously, localStorage.setItem ran before fetchKOT, so if the fetch
  1073→              // failed, the sentinel was already updated. Subsequent socket events
  1074→              // would take the incremental path instead of the full-refresh path,
  1075→              // potentially missing KOTs that were added/removed server-side.
  1076→              // R43-FIX: Remove redundant masonryLoading() — fetchKOT() already calls
  1077→              // masonryLoading(true) at the end of its .then() handler.
  1078→              this.fetchKOT().then(() => {
  1079→                if (doc.kot && doc.kot.time != null) {
  1080→                  try {
  1081→                    localStorage.setItem("kot_time_" + this.production, doc.kot.time);
  1082→                  } catch (e) {
  1083→                    if (import.meta.env?.DEV) console.error('localStorage write failed:', e);
  1084→                  }
  1085→                }
  1086→              }).catch((e) => { console.error("KOT fetch failed:", e); });
  1087→              return;
  1088→            }
  1089→            // R36-FIX: Guard against missing doc.kot to prevent TypeError crash
  1090→            // R44-FIX: Also guard against non-object doc.kot (e.g., array, string).
  1091→            // If the server sends a malformed doc.kot, accessing .name would
  1092→            // return undefined, causing findIndex to fail and the bad data to
  1093→            // be unshifted into the KOT array, corrupting the display.
  1094→            if (!doc.kot || typeof doc.kot !== 'object' || Array.isArray(doc.kot)) {
  1095→              this.fetchKOT().catch((e) => { console.error("KOT fetch failed:", e); });
  1096→              return;
  1097→            }
  1098→            // Incremental update — deduplicate to avoid duplicate cards
  1099→            const existingIndex = this.kot.findIndex(k => k.name === doc.kot.name);
  1100→            let targetKot;
  1101→            if (existingIndex !== -1) {
  1102→              targetKot = this.kot[existingIndex];
  1103→              // R36-FIX: Preserve strikethrough state before Object.assign overwrites kot_items
  1104→              const strikeMap = new Map(
  1105→                (targetKot.kot_items || []).map(i => [i.name, i.striked])
  1106→              );
  1107→              // R43-FIX: Exclude 'name' from the spread to prevent overwriting the KOT's
  1108→              // primary key. If the server sends a different name in doc.kot, the KOT
  1109→              // would become orphaned — findIndex in serveOrder/confirmOrder uses the
  1110→              // original name as the lookup key.
  1111→              const { name: _incomingName, ...kotData } = doc.kot;
  1112→              Object.assign(targetKot, { timecolor: 'text-black', timeRemaining: '— : —', ...kotData });
  1113→              // Restore strikethrough state after Object.assign
  1114→              if (targetKot.kot_items) {
  1115→                targetKot.kot_items.forEach(i => {
  1116→                  if (strikeMap.has(i.name)) i.striked = strikeMap.get(i.name);
  1117→                });
  1118→              }
  1119→              // R37-FIX: Invalidate sorted cache for this KOT since items may have changed
  1120→              this._sortedItemsCache.delete(doc.kot.name);
  1121→            } else {
  1122→              // R39-FIX: Removed showDiv: false — dead code
  1123→              targetKot = { isRotated: false, timecolor: 'text-black', timeRemaining: '— : —', ...doc.kot };
  1124→              this.kot.unshift(targetKot);
  1125→            }
  1126→            // R44-FIX: Use targeted single-KOT methods instead of processing ALL KOTs.
  1127→            // The incremental path only modifies one KOT, so O(1) is sufficient.
  1128→            this._updateSingleKotQtyColor(targetKot);
  1129→            this._updateSingleKotTimeRemaining(targetKot);
  1130→            // R40-FIX: Use debounced masonry layout for rapid socket events
  1131→            this._debouncedMasonryLayout();
  1132→            // R41-FIX: Cancel timeout and localStorage write are now ONLY in the incremental path.
  1133→            // Previously these ran after both full-refresh and incremental branches.
  1134→            if (this._cancelTimeout) clearTimeout(this._cancelTimeout);
  1135→            // R39-FIX: Only schedule cancel re-fetch if the KOT is actually a cancellation.
  1136→            if (doc.kot.type === "Cancelled") {
  1137→              this._cancelTimeout = setTimeout(() => {
  1138→                if (!this._isMounted) return;
  1139→                // R43-FIX: Remove redundant masonryLoading() — fetchKOT() calls it.
  1140→                this.fetchKOT().catch((e) => { console.error("KOT fetch failed:", e); });
  1141→              }, 1500);
  1142→            }
  1143→            // R41-FIX: Guard against storing null/undefined as string "null"/"undefined"
  1144→            // in localStorage. These corrupted sentinel values would cause the
  1145→            // next socket event to always take the full-refresh path.
  1146→            if (doc.kot.time != null) {
  1147→              try {
  1148→                localStorage.setItem("kot_time_" + this.production, doc.kot.time);
  1149→              } catch (e) {
  1150→                if (import.meta.env?.DEV) console.error('localStorage write failed:', e);
  1151→              }
  1152→            }
  1153→          } catch (err) {
  1154→            if (import.meta.env?.DEV) console.error("Socket handler error:", err);
  1155→          }
  1156→        };
  1157→        if (this._socket) this._socket.on(this.kot_channel, this.socketHandler);
  1158→      })
  1159→      .catch((error) => {
  1160→        console.error("Initialization or authentication error:", error);
  1161→        // R40-FIX: this._socket is now stored early, so beforeUnmount will
  1162→        // disconnect it. No need for manual cleanup here.
  1163→        if (this._isMounted) {
  1164→          // R41-FIX: Differentiate auth errors from network/server errors.
  1165→          // Auth failure → "Not Permitted" with login prompt.
  1166→          // Network/server failure → "Connection Error" with retry guidance.
  1167→          const isAuthError = error && (

  1169→            error.httpStatus === 403 ||
  1170→            /auth/i.test(String(error.message || ''))

  1174→            this.modalMessage = "Log in to access this page.";
  1175→          } else {
  1176→            this.modalTitle = "Connection Error";
  1177→            this.modalMessage = "Unable to reach the server. Please check your connection and try again.";
  1178→          }
  1179→          this.showModal = true;
  1180→        }
  1181→      });
  1182→    this.timer = setInterval(this.updateTimeRemaining, 60000);
  1183→  },
