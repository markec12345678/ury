    20→def _get_order_invoice_doc(table=None, invoiceNo=None, order_type=None, is_payment=None):
    21→    """Internal: returns POS Invoice document object for server-side manipulation.
    22→    Used by sync_order and make_invoice which need the doc, not a dict."""
    23→    # Verify user has access to the table's room (BE-R36-017)
    24→    if table:
    25→        from ury.ury_pos.api import _get_user_branch_rooms
    26→        table_room = frappe.db.get_value("URY Table", table, "restaurant_room")
    27→        if table_room:
    28→            user_rooms = [r.room for r in _get_user_branch_rooms() if r.room]
    29→            if user_rooms and table_room not in user_rooms:
    30→                frappe.throw(_("You do not have access to this table's room"), frappe.PermissionError)
    31→
    32→    if table:
    33→        # Lock the table row to prevent concurrent invoice creation (BE-R36-001)
    34→        frappe.db.sql(
    35→            "SELECT name FROM `tabURY Table` WHERE name = %s FOR UPDATE",
    36→            (table,), as_dict=True
    37→        )
    38→        
    39→        if is_payment == "Payments":
    40→            invoice_name = frappe.get_value(
    41→                "POS Invoice", dict(restaurant_table=table, docstatus=0, name=invoiceNo)
    42→            )
    43→            
    44→        else:
    45→            if invoiceNo:
    46→                invoice_name = frappe.get_value(
    47→                    "POS Invoice",
    48→                    dict(restaurant_table=table, docstatus=0, name=invoiceNo),
    49→                )
    50→               
    51→            else:
    52→                invoice_name = frappe.get_value(
    53→                    "POS Invoice",
    54→                    dict(restaurant_table=table, docstatus=0, invoice_printed=0),
    55→                )
    56→                
    57→        branch, menu_name, restaurant = get_restaurant_and_menu_name(table)
    58→
    59→        if invoice_name:
    60→            invoice = frappe.get_doc("POS Invoice", invoice_name)
    61→
    62→        else:
    63→            invoice = frappe.new_doc("POS Invoice")
    64→
    65→            invoice.naming_series = frappe.db.get_value(
    66→                "URY Restaurant", restaurant, "invoice_series_prefix"
    67→            )
    68→
    69→            invoice.is_pos = 1
    70→            invoice.update_stock = 1
    71→            invoice.restaurant = restaurant
    72→            invoice.branch = branch
    73→
    74→            is_take_away = frappe.db.get_value("URY Table", table, "is_take_away")
    75→            if is_take_away == 1:
    76→                invoice.order_type = "Take Away"
    77→            else:
    78→                invoice.order_type= "Dine In"
    79→
    80→        invoice.taxes_and_charges = frappe.db.get_value(
    81→            "URY Restaurant", restaurant, "default_tax_template"
    82→        )
    83→
    84→        invoice.selling_price_list = frappe.db.get_value(
    85→            "Price List", dict(restaurant_menu=menu_name, enabled=1)
    86→        )
    87→
    88→    else:
    89→
    90→        invoice_name = frappe.get_value(
    91→                "POS Invoice", dict(docstatus=0, name=invoiceNo)
    92→            )
    93→            
    94→        if invoice_name:
    95→            invoice = frappe.get_doc("POS Invoice", invoice_name)
    96→            # R44-FIX: Validate that the loaded invoice belongs to the user's branch
    97→            user_branch = _get_user_branch()
    98→            if invoice.branch and invoice.branch != user_branch:
    99→                frappe.throw(_("You do not have access to invoices from another branch"), frappe.PermissionError)
   100→            
   101→
   102→        else:
   103→            invoice = frappe.new_doc("POS Invoice")
   104→            invoice.is_pos = 1
   105→            invoice.update_stock = 1
   106→        
   107→        branch = _get_user_branch()
   108→        restaurant = frappe.db.get_value("URY Restaurant", {"branch": branch}, "name")
   109→
   110→        menu=get_menu_name(order_type)
   111→
   112→        if (order_type == "Aggregators" and frappe.db.get_value("Branch", branch, "custom_no_taxes") == 0) or order_type != "Aggregators":
   113→            invoice.taxes_and_charges = frappe.db.get_value("URY Restaurant", restaurant, "default_tax_template")
   114→        
   115→        invoice.selling_price_list = frappe.db.get_value(
   116→            "Price List", dict(restaurant_menu=menu, enabled=1)
   117→        )
   118→        # R43-FIX: Set branch on non-table invoices to prevent branchless invoices
   119→        invoice.branch = branch
   120→
   121→    return invoice

   167→@frappe.whitelist()
   168→def sync_order(
   169→    items,
   170→    cashier,
   171→    owner,
   172→    mode_of_payment,
   173→    customer,
   174→    no_of_pax,
   175→    last_invoice,
   176→    waiter,
   177→    pos_profile,
   178→    last_modified_time=None,
   179→    table=None,
   180→    invoice=None,
   181→    comments=None,
   182→    order_type=None,
   183→    aggregator_id=None,
   184→    room=None
   185→):
   186→    frappe.only_for("Restaurant Manager", "Restaurant User", "Cashier")
   187→    # Validate pos_profile belongs to user's branch (BE-R36-002)
   188→    pos_profile_branch = frappe.db.get_value("POS Profile", pos_profile, "branch")
   189→    user_branch = _get_user_branch()
   190→    if pos_profile_branch != user_branch:
   191→        frappe.throw(_("POS Profile does not belong to your branch"), frappe.PermissionError)
   192→    user_role = frappe.get_roles()
   193→    billing_roles = frappe.get_all(
   194→        "POS Profile Role",
   195→        filters={"parent": pos_profile},
   196→        fields=["role"],
   197→        pluck="role",
   198→    )
   199→    billing_user = bool(set(user_role).intersection(billing_roles))
   200→
   201→    # Check if the last invoice was already billed
   202→    if (
   203→        last_invoice
   204→        and frappe.db.get_value("POS Invoice", last_invoice, "invoice_printed") == 1
   205→        and (not billing_user)
   206→    ):
   207→        frappe.msgprint(
   208→            title=_("Invoice Already Billed"),
   209→            indicator="red",
   210→            msg=_("This order has already been billed. Please reload the page."),
   211→        )
   212→        return {"status": "Failure"}
   213→
   214→    # R41-FIX: Use _get_order_invoice_doc to get the document object, not a dict.
   215→    # The public get_order_invoice returns a dict for frontend safety (BE-R36-003),
   216→    # but sync_order needs the document to call .save() etc.
   217→    invoice = _get_order_invoice_doc(table, invoice, order_type)
   218→
   219→    # R41-FIX: Validate last_invoice belongs to user's branch
   220→    if last_invoice:
   221→        last_inv_branch = frappe.db.get_value("POS Invoice", last_invoice, "branch")
   222→        if last_inv_branch and last_inv_branch != user_branch:
   223→            frappe.throw(_("Invoice does not belong to your branch"), frappe.PermissionError)
   224→
   225→    if last_invoice and last_modified_time:
   226→        lastModifiedTime = invoice.modified
   227→        from datetime import datetime
   228→
   229→        if isinstance(last_modified_time, str):
   230→            try:
   231→                last_modified_time = datetime.strptime(
   232→                    last_modified_time, "%Y-%m-%d %H:%M:%S.%f"
   233→                )
   234→            except ValueError:
   235→                last_modified_time = datetime.strptime(
   236→                    last_modified_time, "%Y-%m-%d %H:%M:%S"
   237→                )
   238→        if isinstance(lastModifiedTime, str):
   239→            try:
   240→                lastModifiedTime = datetime.strptime(
   241→                    lastModifiedTime, "%Y-%m-%d %H:%M:%S.%f"
   242→                )
   243→            except ValueError:
   244→                lastModifiedTime = datetime.strptime(
   245→                    lastModifiedTime, "%Y-%m-%d %H:%M:%S"
   246→                )
   247→        if lastModifiedTime != last_modified_time:
   248→            frappe.msgprint(
   249→                title=_("Order has been modified"),
   250→                indicator="red",
   251→                msg=_("This order has been modified. Please reload the page to retrieve the latest edits."),
   252→            )
   253→            return {"status": "Failure"}
   254→    else:
   255→        if invoice.name and invoice.invoice_printed == 0 and not billing_user:
   256→            frappe.msgprint(
   257→                title=_("Table Occupied"),
   258→                indicator="red",
   259→                msg=_("{0} is already occupied. Please refresh the page.").format(table),
   260→            )
   261→            return {"status": "Failure"}
   262→
   263→    if not customer:
   264→        frappe.throw(_("Please enter valid customer details"))
   265→    else:
   266→        invoice.customer = customer
   267→
   268→    if order_type:
   269→        invoice.order_type = order_type
   270→
   271→    invoice.mobile_number = frappe.db.get_value("Customer", customer, "mobile_number")
   272→    if comments:
   273→        invoice.custom_comments = comments
   274→    invoice.no_of_pax = no_of_pax
   275→    invoice.pos_profile = pos_profile
   276→    invoice.cashier = cashier
   277→    invoice.waiter = waiter
   278→    invoice.custom_aggregator_id = aggregator_id
   279→    invoice.custom_restaurant_room =room
   280→    invoice.restaurant_table = table
   281→    
   282→    if order_type == "Aggregators":
   283→        price_list = frappe.db.get_value("Aggregator Settings",{"customer": customer, "parent": invoice.branch, "parenttype": "Branch"},"price_list",)
   284→        
   285→        if not price_list:
   286→            frappe.throw(_("Price list for customer {0} in branch {1} not found in Aggregator Settings.").format(customer, invoice.branch))
   287→    else:
   288→        price_list = invoice.selling_price_list
   289→
   290→    # dummy payment
   291→    if invoice.invoice_created == 0:
   292→        invoice.append(
   293→            "payments",
   294→            dict(mode_of_payment=mode_of_payment, amount=invoice.grand_total),
   295→        )
   296→        invoice.invoice_created = 1
   297→
   298→    past_item = []
   299→    for item in invoice.items:
   300→        previous_item = {
   301→            "item_code": item.item_code,
   302→            "item_name": item.item_name,
   303→            "qty": item.qty,
   304→            "comments": "",
   305→        }
   306→        past_item.append(previous_item)
   307→        
   308→
   309→    # Conditional checking for 'items' type:
   310→    # - 'ury': JSON passed, hence using isinstance
   311→    # - 'ury_pos': Already formatted list, hence using else
   312→    if isinstance(items, str):
   313→        try:
   314→            items = json.loads(items)
   315→        except json.JSONDecodeError:
   316→            frappe.throw(_("Invalid items data"), frappe.ValidationError)
   317→    invoice.items = []
   318→    
   319→    menu = frappe.db.get_value("URY Menu", {"branch": invoice.branch}, "name")
   320→
   321→    # Batch-fetch courses, prices, and cost_center to avoid N+1 queries
   322→    item_codes = [d.get("item") for d in items]
   323→    item_codes_unique = list(set(item_codes))
   324→
   325→    courses = {}
   326→    if item_codes_unique:
   327→        course_rows = frappe.db.get_all(
   328→            "URY Menu Item",
   329→            filters={"item": ("in", item_codes_unique), "parent": menu},
   330→            fields=["item", "course"],
   331→        )
   332→        courses = {r.item: r.course for r in course_rows}
   333→
   334→    prices = {}
   335→    if item_codes_unique:
   336→        price_rows = frappe.db.get_all(
   337→            "Item Price",
   338→            filters={"item_code": ("in", item_codes_unique), "price_list": price_list},
   339→            fields=["item_code", "price_list_rate"],
   340→        )
   341→        prices = {r.item_code: r.price_list_rate for r in price_rows}
   342→
   343→    cost_center = frappe.db.get_value("POS Profile", pos_profile, "cost_center")
   344→
   345→    for d in items:
   346→        course = courses.get(d.get("item"))
   347→        rate = prices.get(d.get("item"))
   348→        if not rate:
   349→            frappe.throw(_("No item price found for Item: {0} in Price List: {1}. Please check the price list settings.").format(d.get("item"), price_list))
   350→
   351→        invoice.append(
   352→            "items",
   353→            dict(
   354→                item_code=d.get("item"),
   355→                item_name=d.get("item_name"),
   356→                qty=d.get("qty"),
   357→                **({"custom_course": course} if course else {}),
   358→                comment=d.get("comment"),
   359→                rate=rate,
   360→                price_list_rate=rate,
   361→                base_price_list_rate=rate,
   362→                cost_center=cost_center,
   363→            ),
   364→        )
   365→
   366→    try:
   367→        invoice.save()
   368→    except Exception as e:
   369→        frappe.log_error(frappe.get_traceback(), "Invoice Save Error")
   370→        frappe.throw(_("An error occurred. Please check the error log."))
   371→
   372→
   373→    try:
   374→        kot_execute(invoice.name, customer, table, items, past_item, comments)
   375→
   376→    except Exception as e:
   377→        # Log the KOT error but also notify the user — kitchen won't see this order
   378→        error_msg = f"KOT Creation Failed: {str(e)}"
   379→        frappe.log_error(f"{error_msg}\n{frappe.get_traceback()}", "KOT Error")
   380→        frappe.msgprint(
   381→            title=_("KOT Creation Failed"),
   382→            indicator="orange",
   383→            msg=_("The order was saved but the KOT was not sent to the kitchen. Please check the error log or retry."),
   384→        )
   385→
   386→    # table status
   387→    if invoice.invoice_printed == 0:
   388→        frappe.db.set_value(
   389→            "URY Table", table, {"occupied": 1, "latest_invoice_time": invoice.creation}
   390→        )
   391→
   392→    invoice.db_set("owner", frappe.session.user)
   393→    return {
   394→        "name": invoice.name,
   395→        "customer": invoice.customer,
   396→        "grand_total": invoice.grand_total,
   397→        "modified": invoice.modified,
   398→        "invoice_printed": invoice.invoice_printed,
   399→        "restaurant_table": invoice.restaurant_table,
   400→    }

   423→@frappe.whitelist()
   424→def get_restaurant_and_menu_name(table):
   425→    frappe.only_for("Restaurant Manager", "Restaurant User", "Cashier")
   426→    if not table:
   427→        frappe.throw(_("Please select a table"))
   428→
   429→    result = frappe.get_value(
   430→        "URY Table",
   431→        table,
   432→        ["restaurant", "branch", "restaurant_room"],
   433→    )
   434→    if not result:
   435→        frappe.throw(_("URY Table {0} not found").format(table))
   436→    restaurant, branch, room = result
   437→    room_wise_menu = frappe.db.get_value(
   438→        "URY Restaurant",
   439→        restaurant,
   440→        "room_wise_menu",
   441→    )
   442→
   443→    if not room_wise_menu:
   444→        menu = frappe.db.get_value("URY Restaurant", restaurant, "active_menu")
   445→    else:
   446→        menu = frappe.db.get_value(
   447→            "Menu for Room",
   448→            {"parent": restaurant, "room": room},
   449→            "menu",
   450→        )
   451→
   452→    if not menu:
   453→        frappe.throw(
   454→            _("Please set an active menu for Restaurant {0}").format(restaurant)
   455→        )
   456→
   457→    return branch, menu, restaurant

   485→@frappe.whitelist()
   486→def pos_opening_check():
   487→    frappe.only_for("Restaurant Manager", "Restaurant User", "Cashier")
   488→    user = frappe.session.user
   489→    # Handle the administrator case differently
   490→    if user == "Administrator":
   491→        return {
   492→            "opening_exists": False,  # Assuming no POS opening entry is needed for Administrator
   493→            "cashier": None,
   494→            "pos_profile": None,
   495→        }
   496→    
   497→    details = getBranchRoom()
   498→    if not details:
   499→        frappe.throw(_("No room/branch configuration found for the current user."))
   500→    room = details[0].get('name')
   501→    branch = details[0].get('branch')
   502→    
   503→    pos_opening_list = frappe.db.sql("""
   504→        SELECT DISTINCT `tabPOS Opening Entry`.name 
   505→        FROM `tabPOS Opening Entry`
   506→        INNER JOIN `tabMultiple Rooms` 
   507→        ON `tabMultiple Rooms`.parent = `tabPOS Opening Entry`.name
   508→        WHERE `tabPOS Opening Entry`.branch = %s
   509→        AND `tabPOS Opening Entry`.status = 'Open'
   510→        AND `tabPOS Opening Entry`.docstatus = 1
   511→        AND `tabMultiple Rooms`.room = %s
   512→    """, (branch, room), as_dict=True)
   513→    
   514→    
   515→    result = {
   516→        "opening_exists": len(pos_opening_list) > 0,
   517→        "cashier": None,
   518→        "pos_profile": None,
   519→    }
   520→
   521→    if result["opening_exists"]:
   522→        opening_vals = frappe.db.get_value(
   523→            "POS Opening Entry", pos_opening_list[0].name,
   524→            ["user", "pos_profile"],
   525→            as_dict=True,
   526→        )
   527→        result["cashier"] = opening_vals.user
   528→        result["pos_profile"] = opening_vals.pos_profile
   529→        
   530→    return result

   533→@frappe.whitelist()
   534→def table_transfer(table, newTable, invoice):
   535→    frappe.only_for("Restaurant Manager", "Restaurant User", "Cashier")
   536→    if not frappe.has_permission("POS Invoice", "write", invoice):
   537→        frappe.throw(_("Not permitted to transfer tables"), frappe.PermissionError)
   538→    # R39-FIX: Validate invoice belongs to user's branch
   539→    inv_branch = frappe.db.get_value("POS Invoice", invoice, "branch")
   540→    if not inv_branch:
   541→        frappe.throw(_("POS Invoice {0} not found").format(invoice))
   542→    user_branch = _get_user_branch()
   543→    if inv_branch != user_branch:

   715→@frappe.whitelist()
   716→def make_invoice(customer, payments, cashier, pos_profile, additionalDiscount=None, table=None, invoice=None):
   717→    frappe.only_for("Restaurant Manager", "Restaurant User", "Cashier")
   718→
   719→    # R39-FIX: Validate POS Profile belongs to user's branch
   720→    pos_profile_branch = frappe.db.get_value("POS Profile", pos_profile, "branch")
   721→    user_branch = _get_user_branch()
   722→    if pos_profile_branch != user_branch:
   723→        frappe.throw(_("POS Profile does not belong to your branch"), frappe.PermissionError)
   724→
   725→    # Validate additional discount
   726→    if additionalDiscount is not None:
   727→        additionalDiscount = flt(additionalDiscount)
   728→        if additionalDiscount < 0 or additionalDiscount > 100:
   729→            frappe.throw(_("Additional discount must be between 0 and 100"))
   730→
   731→    # Check if discounts are enabled for this POS Profile (BE-R36-006)
   732→    if additionalDiscount:
   733→        enable_discount = frappe.db.get_value("POS Profile", pos_profile, "custom_enable_discount")
   734→        if not enable_discount:
   735→            frappe.throw(_("Discounts are not enabled for this POS Profile"), frappe.PermissionError)
   736→
   737→    # Validate payments
   738→    if isinstance(payments, str):
   739→        try:
   740→            payments = json.loads(payments)
   741→        except json.JSONDecodeError:
   742→            frappe.throw(_("Invalid payments data"), frappe.ValidationError)
   743→    if not payments:
   744→        frappe.throw(_("At least one payment is required"))
   745→    for p in payments:
   746→        amount = flt(p.get("amount", 0))
   747→        if amount < 0:
   748→            frappe.throw(_("Payment amount cannot be negative"))
   749→
   750→    # R44-FIX: At least one of invoice or table must be provided — otherwise
   751→    # we would create an orphaned invoice with no way to reference it.
   752→    if not invoice and not table:
   753→        frappe.throw(_("Invoice or table is required to make a payment"), frappe.ValidationError)
   754→
   755→    # R43-FIX: Guard against None invoice — get_value with None returns None/error
   756→    order_type = None
   757→    if invoice:
   758→        order_type = frappe.get_value("POS Invoice", invoice, "order_type")
   759→    # R41-FIX: Use _get_order_invoice_doc to get the document object, not a dict.
   760→    # make_invoice needs to call .save() and .submit() on the document.
   761→    invoice = _get_order_invoice_doc(table, invoice, order_type, "Payments")
   762→
   763→    if table:
   764→        _, _, restaurant = get_restaurant_and_menu_name(table)
   765→        invoice.restaurant = restaurant
   766→    elif not invoice.restaurant:
   767→        restaurant = frappe.db.get_value("URY Restaurant", {"branch": invoice.branch}, "name")
   768→        if restaurant:
   769→            invoice.restaurant = restaurant
   770→
   771→    invoice.customer = customer
   772→    invoice.pos_profile = pos_profile
   773→    invoice.additional_discount_percentage=additionalDiscount
   774→    invoice.calculate_taxes_and_totals()
   775→
   776→    invoice.payments = []
   777→
   778→    for d in payments:
   779→        invoice.append(
   780→            "payments", dict(mode_of_payment=d["mode_of_payment"], amount=d["amount"])
   781→        )
   782→
   783→    # invoice.owner = owner
   784→    invoice.save()
   785→    try:
   786→        invoice.submit()
   787→    except Exception as e:
   788→        frappe.log_error(frappe.get_traceback(), "Invoice Submit Error")
   789→        frappe.throw(_("An error occurred. Please check the error log."))
