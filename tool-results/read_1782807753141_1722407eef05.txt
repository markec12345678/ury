    88→// ── UI Preference Persistence ────────────────────────────
    89→const UI_PREFS_KEY = 'ury_ui_prefs';
    90→
    91→interface UIPreferences {
    92→  darkMode: boolean;
    93→  activeTab: string;
    94→}
    95→
    96→function loadUIPrefs(): Partial<UIPreferences> {
    97→  if (typeof window === 'undefined') return {};
    98→  try {
    99→    const raw = localStorage.getItem(UI_PREFS_KEY);
   100→    if (!raw) return {};
   101→    return JSON.parse(raw);
   102→  } catch {
   103→    return {};
   104→  }
   105→}
   106→
   107→function saveUIPrefs(prefs: Partial<UIPreferences>): void {
   108→  if (typeof window === 'undefined') return;
   109→  try {
   110→    localStorage.setItem(UI_PREFS_KEY, JSON.stringify(prefs));
   111→  } catch {
   112→    // ignore storage errors
   113→  }
   114→}

   397→    // ── Main Data Refresh ──────────────────────────────
   398→
   399→    refreshData: async () => {
   400→      const { isConnected } = get();
   401→      if (!isConnected) return;
   402→
   403→      set({ isRefreshing: true });
   404→
   405→      try {
   406→        const client = getFrappeClient();
   407→
   408→        // Run all fetches in parallel for better performance
   409→        const [
   410→          kotResult,
   411→          tablesResult,
   412→          invoicesResult,
   413→          roomsResult,
   414→          plResult,
   415→          shiftResult,

   417→          client.getKOTList(),
   418→          client.getRestaurantTables(),

   420→          client.getRestaurantRooms(),
   421→          client.getDailyPL(),
   422→          client.getPOSOpeningEntry('Open'),
   423→        ]);
   424→
   425→        // ── Process KOTs ────────────────────────────────
   426→        if (kotResult.status === 'fulfilled' && kotResult.value?.data) {
   427→          const transformedKOTs: KOTCard[] = Array.isArray(kotResult.value.data)
   428→            ? kotResult.value.data.map((kot: Record<string, unknown>) => ({
   429→                id: String(kot.name || kot.id || ''),
   430→                orderNo: String(kot.order_no || kot.orderNo || ''),
   431→                table: String(kot.restaurant_table || kot.table || ''),
   432→                items: Array.isArray(kot.items)
   433→                  ? kot.items.map((item: Record<string, unknown>) => ({
   434→                      name: String(item.item_name || item.name || ''),
   435→                      qty: Number(item.qty || 1),
   436→                      course: item.course ? String(item.course) : undefined,
   437→                      comments: item.comments ? String(item.comments) : undefined,
   438→                    }))
   439→                  : [],
   440→                timePlaced: String(kot.time_placed || kot.timePlaced || ''),
   441→                elapsed: Number(kot.elapsed || 0),
   442→                status: (kot.status as KOTStatus) || 'new',
   443→                production: String(kot.production_unit || kot.production || 'Kuhinja 1') as KOTCard['production'],

   445→              }))
   446→            : [];
   447→          set({ kotCards: transformedKOTs });
   448→        }
   449→
   450→        // ── Process Tables ──────────────────────────────
   451→        if (tablesResult.status === 'fulfilled' && tablesResult.value?.data) {
   452→          const transformedTables: TableData[] = Array.isArray(tablesResult.value.data)

   455→                room: String(t.room || ''),
   456→                status: (t.status as TableData['status']) || 'free',

   460→                orderTotal: t.order_total ? Number(t.order_total) : undefined,
   461→                customer: t.customer ? String(t.customer) : undefined,
   462→              }))
   463→            : [];
   464→          set({ tables: transformedTables });
   465→        }
   466→
   467→        // ── Process Invoices → Recent Orders + KPIs ─────
   468→        if (invoicesResult.status === 'fulfilled' && invoicesResult.value?.data) {
   469→          const invoices = Array.isArray(invoicesResult.value.data)
   470→            ? invoicesResult.value.data
   471→            : [];
   472→
   473→          // Recent orders
   474→          const transformedOrders: RecentOrder[] = invoices.map((inv: Record<string, unknown>) => ({

   476→            customer: String(inv.customer || ''),

   480→            time: inv.posting_time ? String(inv.posting_time).slice(0, 5) : '',

   482→          set({ recentOrders: transformedOrders });
   483→
   484→          // Compute KPIs from invoices

   487→          const occupiedTables = invoices.filter((inv: Record<string, unknown>) => inv.type === 'Dine-in').length;
   488→          const allTables = get().tables;
   489→          set({
   490→            kpis: {
   491→              dailySales: totalSales,
   492→              totalOrders,

   497→          });
   498→
   499→          // Compute hourly sales from invoices
   500→          const hourMap = new Map<string, { dineIn: number; takeaway: number }>();
   501→          // Initialize all hours from 11:00 to 22:00

   503→            const key = `${String(h).padStart(2, '0')}:00`;
   504→            hourMap.set(key, { dineIn: 0, takeaway: 0 });

   509→            if (!timeStr) return;
   510→            const hour = timeStr.split(':')[0];
   511→            const key = `${hour}:00`;
   512→            const entry = hourMap.get(key);
   513→            if (entry) {
   514→              const amount = Number(inv.grand_total || 0);
   515→              if (inv.type === 'Takeaway' || inv.type === 'Delivery') {
   516→                entry.takeaway += amount;
   517→              } else {
   518→                entry.dineIn += amount;
   519→              }
   520→            }
   521→          });
   522→          const hourlyData: HourlySales[] = Array.from(hourMap.entries()).map(([hour, data]) => ({
   523→            hour,
   524→            dineIn: data.dineIn,
   525→            takeaway: data.takeaway,
   526→          }));
   527→          if (hourlyData.length > 0) set({ hourlySales: hourlyData });
   528→        }
   529→
   530→        // ── Process Rooms ───────────────────────────────
   531→        if (roomsResult.status === 'fulfilled' && roomsResult.value?.data) {
   532→          const transformedRooms: Room[] = Array.isArray(roomsResult.value.data)
   533→            ? roomsResult.value.data.map((r: Record<string, unknown>) => ({
   534→                id: String(r.name || r.id || '').toLowerCase().replace(/\s+/g, '-'),
   535→                name: String(r.room_name || r.name || ''),
   536→                tables: Number(r.no_of_tables || r.tables || 0),
   537→              }))
   538→            : [];
   539→          if (transformedRooms.length > 0) set({ rooms: transformedRooms });
   540→        }
   541→
   542→        // ── Process P&L ─────────────────────────────────
   543→        if (plResult.status === 'fulfilled' && plResult.value?.data) {
   544→          const plDocs = Array.isArray(plResult.value.data) ? plResult.value.data : [];
   545→          if (plDocs.length > 0) {
   546→            // Use the latest P&L document
   547→            const latest = plDocs[0] as Record<string, unknown>;
   548→
   549→            // P&L Summary
   550→            const grossSales = Number(latest.gross_sales || latest.total_revenue || 0);
   551→            const cogs = Number(latest.cogs || latest.cost_of_goods_sold || 0);
   552→            const grossProfit = grossSales - cogs;
   553→            const netProfit = Number(latest.net_profit || grossProfit - Number(latest.total_expenses || 0));
   554→            set({
   555→              plSummary: { grossSales, cogs, grossProfit, netProfit },
   556→            });
   557→
   558→            // Daily P&L from multiple docs
   559→            const dailyData: DailyPL[] = plDocs.slice(0, 7).reverse().map((doc: Record<string, unknown>) => {
   560→              const date = String(doc.date || doc.posting_date || '');
   561→              const dayName = date
   562→                ? new Date(date).toLocaleDateString('sl-SI', { weekday: 'short' })
   563→                : '';
   564→              return {
   565→                day: dayName,
   566→                revenue: Number(doc.gross_sales || doc.total_revenue || 0),
   567→                costs: Number(doc.cogs || doc.total_expenses || 0),
   568→              };
   569→            });
   570→            if (dailyData.length > 0) set({ dailyPL: dailyData });
   571→
   572→            // Expense breakdown — try to extract from P&L doc
   573→            const breakdownData: ExpenseBreakdown[] = [];
   574→            if (latest.breakup || latest.expense_breakdown) {
   575→              const items = Array.isArray(latest.breakup || latest.expense_breakdown)
   576→                ? (latest.breakup || latest.expense_breakdown) as Record<string, unknown>[]
   577→                : [];
   578→              const colors = ['#ef4444', '#f97316', '#eab308', '#8b5cf6', '#10b981'];
   579→              items.forEach((item, i) => {
   580→                breakdownData.push({
   581→                  name: String(item.expense_type || item.category || item.name || ''),
   582→                  value: Number(item.percentage || item.value || 0),
   583→                  color: colors[i % colors.length],
   584→                });
   585→              });
   586→            }
   587→            if (breakdownData.length > 0) set({ expenseBreakdown: breakdownData });
   588→
   589→            // Line items
   590→            const lines: PLLineItem[] = [];
   591→            if (latest.breakup || latest.line_items) {
   592→              const items = Array.isArray(latest.breakup || latest.line_items)
   593→                ? (latest.breakup || latest.line_items) as Record<string, unknown>[]
   594→                : [];
   595→              items.forEach((item) => {
   596→                lines.push({
   597→                  label: String(item.label || item.expense_type || item.name || ''),
   598→                  value: Number(item.amount || item.value || 0),
   599→                  bold: Boolean(item.is_total || item.bold || false),
   600→                });
   601→              });
   602→            }
   603→            if (lines.length > 0) set({ plLineItems: lines });
   604→          }
   605→        }
   606→
   607→        // ── Process Shift / Cashier ─────────────────────
   608→        if (shiftResult.status === 'fulfilled' && shiftResult.value?.data) {
   609→          const openings = Array.isArray(shiftResult.value.data) ? shiftResult.value.data : [];
   610→          if (openings.length > 0) {
   611→            const latestOpening = openings[0] as Record<string, unknown>;
   612→            const openedAt = String(latestOpening.period_start || latestOpening.opening_time || '').slice(0, 5);
   613→            const openedBy = String(latestOpening.user || latestOpening.opened_by || '');
   614→            const openingBalance = Number(latestOpening.opening_amount || latestOpening.balance_details?.[0]?.amount || 0);
   615→
   616→            set({
   617→              shiftInfo: {
   618→                status: 'open',
   619→                openedAt: openedAt || '09:00',
   620→                closesAt: '23:00',
   621→                openedBy: openedBy || 'Administrator',
   622→                openingBalance,
   623→              },
   624→            });
   625→
   626→            // Build cashier data from POS opening entry
   627→            const cashierList: CashierData[] = [];
   628→            const balanceRows = Array.isArray(latestOpening.balance_details)
   629→              ? latestOpening.balance_details as Record<string, unknown>[]
   630→              : [];
   631→            if (balanceRows.length > 0) {
   632→              balanceRows.forEach((row) => {
   633→                cashierList.push({
   634→                  name: String(row.user || openedBy || 'Cashier'),
   635→                  role: 'URY Cashier',
   636→                  openedAt: openedAt || '09:00',
   637→                  status: 'active',
   638→                  openingBalance: Number(row.opening_amount || 0),
   639→                  currentTotal: Number(row.closing_amount || row.opening_amount || 0),
   640→                  cashPayments: Number(row.cash_amount || 0),
   641→                  cardPayments: Number(row.card_amount || 0),
   642→                  upiPayments: Number(row.upi_amount || 0),
   643→                  ordersProcessed: Number(row.orders_processed || 0),
   644→                  room: String(row.room || 'Glavna dvorana'),
   645→                });
   646→              });
   647→            }
   648→            if (cashierList.length > 0) set({ cashiers: cashierList });
   649→          } else {
   650→            // No open shift found
   651→            set({
   652→              shiftInfo: {
   653→                status: 'closed',
   654→                openedAt: '',
   655→                closesAt: '',
   656→                openedBy: '',
   657→                openingBalance: 0,
   658→              },
   659→            });
   660→          }
   661→        }
   662→
   663→        // Mark refresh time
   664→        set({ lastRefreshed: new Date(), isRefreshing: false });
   665→      } catch (err) {
   666→        console.error('Failed to refresh data:', err);
   667→        set({ isRefreshing: false });
   668→      }
   669→    },
   670→
   671→    // ── Targeted Refresh: KPIs only ────────────────────
   672→    refreshKPIs: async () => {
   673→      const { isConnected } = get();
   674→      if (!isConnected) return;
   675→
   676→      try {
   677→        const client = getFrappeClient();
   678→        const invoicesResult = await client.getInvoiceForCashier('Paid', '', 50, 0).catch(() => null);
   679→        if (invoicesResult?.data && Array.isArray(invoicesResult.data)) {
   680→          const invoices = invoicesResult.data as Record<string, unknown>[];
   681→          const totalSales = invoices.reduce((sum, inv) => sum + Number(inv.grand_total || 0), 0);
   682→          const totalOrders = invoices.length;
   683→          const occupiedTables = invoices.filter((inv) => inv.type === 'Dine-in').length;
   684→          const allTables = get().tables;
   685→          set({
   686→            kpis: {
   687→              dailySales: totalSales,
   688→              totalOrders,
   689→              avgBill: totalOrders > 0 ? Math.round(totalSales / totalOrders) : 0,
   690→              occupiedTables,
   691→              totalTables: allTables.length || 32,
   692→            },
   693→          });
   694→        }
   695→      } catch (err) {
   696→        console.error('Failed to refresh KPIs:', err);
   697→      }
   698→    },
   699→
   700→    // ── Targeted Refresh: P&L only ─────────────────────
   701→    refreshPL: async () => {
   702→      const { isConnected } = get();
   703→      if (!isConnected) return;
   704→
   705→      try {
   706→        const client = getFrappeClient();
   707→        const plResult = await client.getDailyPL().catch(() => null);
   708→        if (plResult?.data && Array.isArray(plResult.data)) {
   709→          const plDocs = plResult.data as Record<string, unknown>[];
   710→          if (plDocs.length > 0) {
   711→            const latest = plDocs[0];
   712→            const grossSales = Number(latest.gross_sales || latest.total_revenue || 0);
   713→            const cogs = Number(latest.cogs || latest.cost_of_goods_sold || 0);
   714→            const grossProfit = grossSales - cogs;
   715→            const netProfit = Number(latest.net_profit || grossProfit - Number(latest.total_expenses || 0));
   716→            set({ plSummary: { grossSales, cogs, grossProfit, netProfit } });
   717→
   718→            const dailyData: DailyPL[] = plDocs.slice(0, 7).reverse().map((doc) => {
   719→              const date = String(doc.date || doc.posting_date || '');
   720→              return {
   721→                day: date ? new Date(date).toLocaleDateString('sl-SI', { weekday: 'short' }) : '',
   722→                revenue: Number(doc.gross_sales || doc.total_revenue || 0),
   723→                costs: Number(doc.cogs || doc.total_expenses || 0),
   724→              };
   725→            });
   726→            if (dailyData.length > 0) set({ dailyPL: dailyData });
   727→          }
   728→        }
   729→      } catch (err) {
   730→        console.error('Failed to refresh P&L:', err);
   731→      }
   732→    },
   733→
   734→    // ── Targeted Refresh: Shift only ───────────────────
   735→    refreshShift: async () => {

   775→    updateKOTStatus: (id, status) => {

   777→        kotCards: s.kotCards.map((c) =>

   786→        if (status === 'served') {

   788→        } else if (status === 'preparing') {
   789→          // Could call a confirm/update endpoint
   790→        }
   791→      }

   794→    openShift: (openingBalance, openedBy) => {
   795→      const now = new Date();
   796→      const timeStr = now.toLocaleTimeString('sl-SI', { hour: '2-digit', minute: '2-digit' });
   797→      set({

   832→
   833→    startAutoRefresh: () => {
   834→      if (refreshTimer) return; // Already running
   835→      refreshTimer = setInterval(() => {
   836→        const { isConnected } = get();
   837→        if (isConnected) {
   838→          get().refreshData();
   839→        } else {
   840→          get().stopAutoRefresh();

   846→      if (refreshTimer) {
   847→        clearInterval(refreshTimer);
   848→        refreshTimer = null;
   849→      }
   850→    },

   852→    // ── Toast Notifications ──────────────────────────────

   854→    addToast: (toast) => {
   855→      const id = `toast-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`;
   856→      const duration = toast.duration ?? 5000;
   857→      set((s) => ({ toasts: [...s.toasts, { ...toast, id }] }));
   858→
   859→      // Auto-remove after duration
   860→      const timer = setTimeout(() => {
   861→        get().removeToast(id);
   862→        toastTimers.delete(id);
   863→      }, duration);
   864→      toastTimers.set(id, timer);
   865→    },
   866→
   867→    removeToast: (id) => {
   868→      set((s) => ({ toasts: s.toasts.filter((t) => t.id !== id) }));
   869→      const timer = toastTimers.get(id);
   870→      if (timer) {
   871→        clearTimeout(timer);
   872→        toastTimers.delete(id);
   873→      }
   874→    },
   875→  };
