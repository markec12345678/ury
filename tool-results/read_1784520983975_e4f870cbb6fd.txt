   312→// ---- Period Helpers ----
   313→
   314→function getPreviousPeriodDates(
   315→  period: ReportPeriod,
   316→  customFromDate?: string | null,
   317→  customToDate?: string | null
   318→): { prevFrom: string; prevTo: string } {
   319→  const today = new Date();
   320→
   321→  // If custom date range, calculate the previous equivalent period
   322→  if (customFromDate && customToDate) {
   323→    const from = new Date(customFromDate);
   324→    const to = new Date(customToDate);
   325→    const diffMs = to.getTime() - from.getTime();
   326→    const prevTo = new Date(from.getTime() - 1); // Day before from
   327→    const prevFrom = new Date(prevTo.getTime() - diffMs);
   328→    return {
   329→      prevFrom: prevFrom.toISOString().split('T')[0],
   330→      prevTo: prevTo.toISOString().split('T')[0],
   331→    };
   332→  }
   333→
   334→  switch (period) {
   335→    case 'daily':
   336→    case 'yesterday': {
   337→      // Previous day
   338→      const prev = new Date(today);
   339→      prev.setDate(prev.getDate() - 1);
   340→      const dateStr = prev.toISOString().split('T')[0];
   341→      return { prevFrom: dateStr, prevTo: dateStr };
   342→    }
   343→    case 'weekly': {
   344→      // Previous week (same day range)
   345→      const prevWeekStart = new Date(today);
   346→      prevWeekStart.setDate(prevWeekStart.getDate() - 7 - prevWeekStart.getDay());
   347→      const prevWeekEnd = new Date(prevWeekStart);
   348→      prevWeekEnd.setDate(prevWeekEnd.getDate() + 6);
   349→      return {
   350→        prevFrom: prevWeekStart.toISOString().split('T')[0],
   351→        prevTo: prevWeekEnd.toISOString().split('T')[0],
   352→      };
   353→    }
   354→    case 'monthly': {
   355→      // Previous month
   356→      const prevMonth = new Date(today.getFullYear(), today.getMonth() - 1, 1);
   357→      const prevMonthEnd = new Date(today.getFullYear(), today.getMonth(), 0);
   358→      return {
   359→        prevFrom: prevMonth.toISOString().split('T')[0],
   360→        prevTo: prevMonthEnd.toISOString().split('T')[0],
   361→      };
   362→    }
   363→    case 'last_month': {
   364→      // Two months ago
   365→      const twoMonthsAgo = new Date(today.getFullYear(), today.getMonth() - 2, 1);
   366→      const twoMonthsAgoEnd = new Date(today.getFullYear(), today.getMonth() - 1, 0);
   367→      return {
   368→        prevFrom: twoMonthsAgo.toISOString().split('T')[0],
   369→        prevTo: twoMonthsAgoEnd.toISOString().split('T')[0],
   370→      };
   371→    }
   372→    case 'last_7_days': {
   373→      const prevEnd = new Date(today);
   374→      prevEnd.setDate(prevEnd.getDate() - 8);
   375→      const prevStart = new Date(prevEnd);
   376→      prevStart.setDate(prevStart.getDate() - 6);
   377→      return {
   378→        prevFrom: prevStart.toISOString().split('T')[0],
   379→        prevTo: prevEnd.toISOString().split('T')[0],
   380→      };
   381→    }
   382→    case 'last_30_days': {
   383→      const prevEnd = new Date(today);
   384→      prevEnd.setDate(prevEnd.getDate() - 31);
   385→      const prevStart = new Date(prevEnd);
   386→      prevStart.setDate(prevStart.getDate() - 29);
   387→      return {
   388→        prevFrom: prevStart.toISOString().split('T')[0],
   389→        prevTo: prevEnd.toISOString().split('T')[0],
   390→      };
   391→    }
   392→    default: {
   393→      const prev = new Date(today);
   394→      prev.setDate(prev.getDate() - 1);
   395→      const dateStr = prev.toISOString().split('T')[0];
   396→      return { prevFrom: dateStr, prevTo: dateStr };
   397→    }
   398→  }
   399→}

   403→function escapeCsvField(field: string | number): string {
   404→  const str = String(field);
   405→  if (str.includes(',') || str.includes('"') || str.includes('\n')) {
   406→    return `"${str.replace(/"/g, '""')}"`;
   407→  }
   408→  return str;
   409→}
   410→
   411→function csvRow(...fields: (string | number)[]): string {
   412→  return fields.map(escapeCsvField).join(',');
   413→}
   414→
   415→function downloadCsv(content: string, filename: string): void {
   416→  const BOM = '\uFEFF'; // UTF-8 BOM for Excel compatibility
   417→  const blob = new Blob([BOM + content], { type: 'text/csv;charset=utf-8;' });
   418→  const url = URL.createObjectURL(blob);
   419→  const link = document.createElement('a');
   420→  link.setAttribute('href', url);
   421→  link.setAttribute('download', filename);
   422→  document.body.appendChild(link);
   423→  link.click();
   424→  document.body.removeChild(link);
   425→  URL.revokeObjectURL(url);
   426→}

   428→function generateSalesCsv(report: SalesReport): string {
   429→  const lines: string[] = [];
   430→
   431→  // Header
   432→  lines.push(csvRow('URY Restaurant - Sales Report'));
   433→  lines.push(csvRow('Period', `${report.from_date} to ${report.to_date}`));
   434→  lines.push(csvRow('Generated', new Date().toLocaleString()));
   435→  lines.push('');
   436→
   437→  // Summary
   438→  lines.push(csvRow('--- Summary ---'));
   439→  lines.push(csvRow('Metric', 'Value'));
   440→  lines.push(csvRow('Total Revenue', report.summary.total_revenue.toFixed(2)));
   441→  lines.push(csvRow('Total Orders', report.summary.total_orders));
   442→  lines.push(csvRow('Avg Order Value', report.summary.avg_order_value.toFixed(2)));
   443→  lines.push(csvRow('Net Revenue', report.summary.net_revenue.toFixed(2)));
   444→  lines.push(csvRow('Total Tax', report.summary.total_tax.toFixed(2)));
   445→  lines.push(csvRow('Unique Customers', report.summary.unique_customers));
   446→  lines.push('');
   447→
   448→  // Item Sales
   449→  if (report.item_sales.length > 0) {
   450→    lines.push(csvRow('--- Item-wise Sales ---'));
   451→    lines.push(csvRow('Item', 'Qty', 'Avg Rate', 'Total Amount'));
   452→    report.item_sales.forEach((item) => {
   453→      lines.push(csvRow(item.item_name, item.total_qty, item.avg_rate.toFixed(2), item.total_amount.toFixed(2)));
   454→    });
   455→    lines.push('');
   456→  }
   457→
   458→  // Order Type Sales
   459→  if (report.order_type_sales.length > 0) {
   460→    lines.push(csvRow('--- Order Type Breakdown ---'));
   461→    lines.push(csvRow('Order Type', 'Orders', 'Revenue'));
   462→    report.order_type_sales.forEach((ot) => {
   463→      lines.push(csvRow(ot.order_type, ot.order_count, ot.revenue.toFixed(2)));
   464→    });
   465→    lines.push('');
   466→  }
   467→
   468→  // Cancelled Orders
   469→  if (report.cancelled_orders.count > 0) {
   470→    lines.push(csvRow('--- Cancelled Orders ---'));
   471→    lines.push(csvRow('Count', 'Amount'));
   472→    lines.push(csvRow(report.cancelled_orders.count, report.cancelled_orders.amount.toFixed(2)));
   473→    lines.push('');
   474→  }
   475→
   476→  return lines.join('\n');
   477→}

   487→  // Summary

   490→  lines.push(csvRow('Fixed Expenses', report.total_fixed.toFixed(2)));
   491→  lines.push(csvRow('Variable Expenses', report.total_variable.toFixed(2)));
   492→  lines.push(csvRow('Total Expenses', report.total_expenses.toFixed(2)));
   493→  lines.push('');
   494→
   495→  // Fixed Expenses
   496→  if (report.fixed_expenses.length > 0) {
   497→    lines.push(csvRow('--- Fixed Expenses ---'));
   498→    lines.push(csvRow('Type', 'Description', 'Amount'));
   499→    report.fixed_expenses.forEach((exp: ExpenseItem) => {
   500→      lines.push(csvRow(exp.expense_type, exp.description || '', exp.amount.toFixed(2)));
   501→    });
   502→    lines.push('');
   503→  }
   504→
   505→  // Variable Expenses
   506→  if (report.variable_expenses.length > 0) {
   507→    lines.push(csvRow('--- Variable Expenses ---'));
   508→    lines.push(csvRow('Date', 'Type', 'Description', 'Amount'));
   509→    report.variable_expenses.forEach((exp: ExpenseItem) => {
   510→      lines.push(csvRow(exp.date || '', exp.expense_type, exp.description || '', exp.amount.toFixed(2)));
   511→    });
   512→    lines.push('');
   513→  }
   514→
   515→  return lines.join('\n');
   516→}
