   511→        frappe.throw(_("Not permitted to transfer tables"), frappe.PermissionError)
   512→    # Lock both source and destination tables to prevent race conditions (BE-R36-007)
   513→    table_data = frappe.db.sql(
   514→        """SELECT name, restaurant_room, occupied FROM `tabURY Table`
   515→           WHERE name IN %s FOR UPDATE""",
   516→        ((table, newTable),), as_dict=True
   517→    )

   519→    current_room = table_map[table].restaurant_room if table in table_map else frappe.db.get_value("URY Table", table, "restaurant_room")
   520→    new_table_data = [table_map[newTable]] if newTable in table_map else []
   521→    if not new_table_data:
   522→        frappe.throw(_("Table {0} not found").format(newTable))
   523→    new_table_room = new_table_data[0].restaurant_room
   524→    new_table_occupied = new_table_data[0].occupied

   527→    if current_room == new_table_room:
   528→        if new_table_occupied == 1:

   531→        # Update table status
   532→        frappe.db.set_value(
   533→            "URY Table",
   534→            newTable,
   535→            {"occupied": 1, "latest_invoice_time": pos_invoice.creation},
   536→        )
   537→        frappe.db.set_value(
   538→            "URY Table",
   539→            table,
   540→            {"occupied": 0, "latest_invoice_time": None},
   541→        )

   543→        # Update POS Invoice
   544→        pos_invoice.restaurant_table = newTable
   545→        pos_invoice.save()
   546→
   547→        try:
   548→            change_table_in_kot(
   549→                    pos_invoice.name, newTable, pos_invoice.branch

   552→        except Exception as e:
   553→            frappe.log_error(
   554→                f"KOT table transfer failed for {pos_invoice.name}: {frappe.get_traceback()}",
   555→                "KOT Table Transfer Error"

   683→    if isinstance(payments, str):
   684→        payments = json.loads(payments)

   686→        frappe.throw(_("At least one payment is required"))
   687→    for p in payments:
   688→        amount = flt(p.get("amount", 0))
   689→        if amount < 0:
   690→            frappe.throw(_("Payment amount cannot be negative"))
   691→
   692→    order_type = frappe.get_value("POS Invoice", invoice, "order_type")

   697→        invoice.restaurant = restaurant

   700→        if restaurant:

   725→# Cancel KOT Doc Creation
   726→def cancel_kot(invoice_id):
   727→
   728→    pos_invoice = frappe.get_doc("POS Invoice", invoice_id)
   729→    pos_profile_id = pos_invoice.pos_profile
   730→    kot_naming_series = frappe.db.get_value("POS Profile", pos_profile_id, "custom_kot_naming_series")
   731→    cancel_kot_naming_series = "CNCL-" + (kot_naming_series or "")
   732→
   733→    items = []
   734→    # Create a list of items for the canceled KOT
   735→    for item in pos_invoice.items:
   736→        order_item = {
   737→            "item_code": item.item_code,
   738→            "qty": item.qty,
   739→            "item_name": item.item_name,
   740→        }
   741→        items.append(order_item)
   742→
   743→    if pos_invoice.restaurant_table:
   744→        restaurant_table = pos_invoice.restaurant_table
   745→    else:
   746→        restaurant_table = None
   747→
   748→    # Build invoiceItems list for comparison in cancel KOT processing
   749→    invoice_items = [
   750→        {"item_code": item.item_code, "item_name": item.item_name, "qty": item.qty}
   751→        for item in pos_invoice.items
   752→    ]
   753→
   754→    # Process items for a canceled KOT
   755→    process_items_for_cancel_kot(
   756→        invoice_id,
   757→        pos_invoice.customer,
   758→        restaurant_table,
   759→        items,
   760→        "",
   761→        pos_profile_id,
   762→        cancel_kot_naming_series,
   763→        "Cancelled",
   764→        invoice_items,
   765→        pos_invoice,
   766→        pos_invoice.branch,
   767→    )
   768→
   769→    # Set the KOTs associated with the invoice as canceled via batch SQL
   770→    kot_names = frappe.db.get_list(
   771→        "URY KOT",
   772→        filters={
   773→            "invoice": invoice_id,
   774→            "type": ("in", ("New Order", "Order Modified")),
   775→            "docstatus": 1,
   776→        },
   777→        fields=["name"],
   778→        pluck="name",
   779→    )
   780→
   781→    if kot_names:
   782→        for kot_name in kot_names:
   783→            frappe.get_doc("URY KOT", kot_name).cancel()

   789→        "URY KOT",

   792→            "docstatus": 1,
   793→            "order_status": "Ready For Prepare",
   794→            "verified": 0,
   795→        },
   796→        fields=["name", "production"],
   797→    )
   798→
   799→    # Update each KOT's restaurant_table and send a real-time update
   800→    for kot in kot_list:
   801→        frappe.db.set_value("URY KOT", kot.name, "restaurant_table", new_table)
   802→        kot_channel = "{}_{}_{}".format("kot_update", branch, kot.production)
