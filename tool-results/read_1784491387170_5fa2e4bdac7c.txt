   714→   714→    // Without this, a transient server error leaves the KDS showing stale
   715→   715→    // data indefinitely — the only recovery was a manual page reload.
   716→   716→    fetchKOTWithRetry(maxRetries = 3, initialDelay = 2000) {
   717→   717→      // R41-FIX: Clear any pending retry timer before starting a new retry chain.
   718→   718→      // Without this, calling fetchKOTWithRetry() while a previous retry timer is
   719→   719→      // pending would leave the old timer running, potentially causing duplicate
   720→   720→      // fetchKOT calls when both timers fire.
   721→   721→      if (this._fetchRetryTimer) { clearTimeout(this._fetchRetryTimer); this._fetchRetryTimer = null; }
   722→   722→      // R42-FIX: Bump generation counter so any still-running retry chain from
   723→   723→      // a previous fetchKOTWithRetry call knows it's been superseded. Without
   724→   724→      // this, clearing the timer orphans the old promise — its resolve/reject
   725→   725→      // callbacks are never called, leaking the promise and its closures.
   726→   726→      const generation = ++this._fetchGeneration;
   727→   727→      const attempt = (retriesLeft, delay) => {
   728→   728→        // If a newer fetchKOTWithRetry has started, abort this chain
   729→   729→        if (generation !== this._fetchGeneration) {
   730→   730→          return Promise.reject(new Error('Superseded by newer fetchKOTWithRetry call'));
   731→   731→        }
   732→   732→        return this.fetchKOT().catch(err => {
   733→   733→          if (retriesLeft <= 0 || !this._isMounted || generation !== this._fetchGeneration) throw err;
   734→   734→          if (import.meta.env?.DEV) console.warn(`fetchKOT failed, retrying in ${delay}ms...`, err);
   735→   735→          return new Promise((resolve, reject) => {
   736→   736→            this._fetchRetryTimer = setTimeout(() => {
   737→   737→              if (!this._isMounted || generation !== this._fetchGeneration) { reject(err); return; }
   738→   738→              attempt(retriesLeft - 1, Math.min(delay * 2, 30000)).then(resolve).catch(reject);
   739→   739→            }, delay);
   740→   740→          });
   741→   741→        });
   742→   742→      };
   743→   743→      return attempt(maxRetries, initialDelay);
   744→   744→    },
