    12→    12→def inner_bom_process(buying_price_list, bom, depth=0, max_depth=10, visited=None):
    13→    13→    if visited is None:
    14→    14→        visited = set()
    15→    15→    unset_bom_items = []
    16→    16→    bom_buying_price = 0
    17→    17→
    18→    18→    if depth > max_depth:
    19→    19→        frappe.log_error(
    20→    20→            f"BOM recursion depth exceeded max_depth={max_depth} for BOM {bom.name}",
    21→    21→            "BOM Depth Limit Warning"
    22→    22→        )
    23→    23→        return {"bom_buying_price": bom_buying_price, "unset_bom_items": unset_bom_items}
    24→    24→
    25→    25→    if bom.name in visited:
    26→    26→        frappe.log_error(
    27→    27→            f"Circular BOM reference detected: {bom.name} already visited",
    28→    28→            "BOM Circular Reference Warning"
    29→    29→        )
    30→    30→        return {"bom_buying_price": bom_buying_price, "unset_bom_items": unset_bom_items}
    31→    31→
    32→    32→    visited = visited | {bom.name}
    33→    33→
    34→    34→    for bom_item in bom.items:
    35→    35→        bom_item_qty = bom_item.qty
    36→    36→        bom_item_name = bom_item.item_name
    37→    37→        boms = frappe.db.get_all("BOM", fields=["name", "item"], filters={'item': bom_item.item_code, 'is_active': 1, 'is_default': 1, 'docstatus': 1})
    38→    38→        
    39→    39→        if len(boms) > 0:
    40→    40→            inner_bom = frappe.get_doc("BOM", boms[0].name)
    41→    41→            inner_bom_data = inner_bom_process(buying_price_list, inner_bom, depth=depth+1, max_depth=max_depth, visited=visited)
    42→    42→            inner_bom_buying_price = inner_bom_data['bom_buying_price']
    43→    43→            inner_unset_bom_items = inner_bom_data['unset_bom_items']
    44→    44→            bom_buying_price += float(inner_bom_buying_price) * bom_item_qty
    45→    45→
    46→    46→            for item in inner_unset_bom_items:
    47→    47→                if item not in unset_bom_items:
    48→    48→                    unset_bom_items.append(item)
    49→    49→        else:
    50→    50→            bom_items_price = frappe.db.get_all("Item Price", fields=['name', 'price_list_rate'], filters={'price_list': buying_price_list, 'item_code': bom_item.item_code})
    51→    51→
    52→    52→            if len(bom_items_price) == 0:
    53→    53→                if bom_item_name not in unset_bom_items:
    54→    54→                    unset_bom_items.append(bom_item_name)
    55→    55→            else:
    56→    56→                bom_buying_price += float(bom_items_price[0].price_list_rate) * bom_item_qty
    57→    57→
    58→    58→    if not bom.quantity:
    59→    59→        return {"bom_buying_price": 0, "unset_bom_items": unset_bom_items}
    60→    60→    bom_buying_price = bom_buying_price / bom.quantity
    61→    61→    return {"bom_buying_price": bom_buying_price, "unset_bom_items": unset_bom_items}

    64→    64→def inner_inner_bom_process(buying_price_list, bom, depth=0, max_depth=10, visited=None):
    65→    65→    """Leaf-level BOM processor (no further BOM resolution).
    66→    66→    Kept for backward compatibility; inner_bom_process now recurses into itself
    67→    67→    for arbitrary nesting depth with circular-reference and depth-limit guards."""
    68→    68→    if visited is None:
    69→    69→        visited = set()
    70→    70→    unset_bom_items = []
    71→    71→    bom_buying_price = 0
    72→    72→
    73→    73→    if depth > max_depth or bom.name in visited:
    74→    74→        frappe.log_error(
    75→    75→            f"BOM depth/circular check triggered for {bom.name} in inner_inner_bom_process",
    76→    76→            "BOM Safety Check Warning"
    77→    77→        )
    78→    78→        return {"bom_buying_price": bom_buying_price, "unset_bom_items": unset_bom_items}
    79→    79→
    80→    80→    for bom_item in bom.items:
    81→    81→        bom_item_qty = bom_item.qty
    82→    82→        bom_item_name = bom_item.item_name
    83→    83→        bom_items_price = frappe.db.get_all("Item Price", fields=['name', 'price_list_rate'], filters={'price_list': buying_price_list, 'item_code': bom_item.item_code})
    84→    84→        if len(bom_items_price) == 0:
    85→    85→            if bom_item_name not in unset_bom_items:
    86→    86→                unset_bom_items.append(bom_item_name)
    87→    87→        else:
    88→    88→            bom_buying_price += float(bom_items_price[0].price_list_rate) * bom_item_qty
    89→    89→
    90→    90→    if not bom.quantity:
    91→    91→        return {"bom_buying_price": 0, "unset_bom_items": unset_bom_items}
    92→    92→    bom_buying_price = bom_buying_price / bom.quantity
    93→    93→    return {"bom_buying_price": bom_buying_price, "unset_bom_items": unset_bom_items}
