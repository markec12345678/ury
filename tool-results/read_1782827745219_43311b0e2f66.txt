    85→// ── Previous period mock data generator ──────────────────
    86→// Multiplies current data by realistic factors to simulate previous period
    87→function generatePreviousPeriodData(current: ReportSummary): ReportSummary {
    88→  const revenueFactor = 0.85 + Math.random() * 0.2; // 0.85 – 1.05
    89→  const orderFactor = 0.88 + Math.random() * 0.15;
    90→  const discountFactor = 0.9 + Math.random() * 0.2;
    91→  const cancelFactor = 0.8 + Math.random() * 0.4;
    92→
    93→  return {
    94→    ...current,
    95→    totalRevenue: Math.round(current.totalRevenue * revenueFactor),
    96→    totalOrders: Math.round(current.totalOrders * orderFactor),
    97→    avgOrderValue: Math.round((current.totalRevenue * revenueFactor) / (current.totalOrders * orderFactor)),
    98→    totalDiscounts: Math.round(current.totalDiscounts * discountFactor),
    99→    totalCancellations: Math.round(current.totalCancellations * cancelFactor),
   100→    netRevenue: Math.round(
   101→      current.totalRevenue * revenueFactor - current.totalDiscounts * discountFactor
   102→    ),
   103→    topItems: current.topItems.map((item) => ({
   104→      ...item,
   105→      quantity: Math.round(item.quantity * (0.8 + Math.random() * 0.3)),
   106→      revenue: Math.round(item.revenue * (0.8 + Math.random() * 0.3)),
   107→    })),
   108→    dailyBreakdown: current.dailyBreakdown.map((d) => ({
   109→      ...d,
   110→      revenue: Math.round(d.revenue * (0.82 + Math.random() * 0.25)),
   111→      orders: Math.round(d.orders * (0.85 + Math.random() * 0.2)),
   112→    })),
   113→    paymentSplit: current.paymentSplit.map((p) => ({
   114→      ...p,
   115→      count: Math.round(p.count * (0.85 + Math.random() * 0.2)),
   116→      amount: Math.round(p.amount * (0.85 + Math.random() * 0.2)),
   117→    })),
   118→    orderTypeSplit: current.orderTypeSplit.map((o) => ({
   119→      ...o,
   120→      count: Math.round(o.count * (0.85 + Math.random() * 0.2)),
   121→      revenue: Math.round(o.revenue * (0.85 + Math.random() * 0.2)),
   122→    })),
   123→    courseRevenue: current.courseRevenue.map((c) => ({
   124→      ...c,
   125→      revenue: Math.round(c.revenue * (0.82 + Math.random() * 0.25)),
   126→      items: Math.round(c.items * (0.85 + Math.random() * 0.2)),
   127→    })),
   128→  };
   129→}

   131→// ── Generate AI-style insight text ───────────────────────
   132→function generateInsight(
   133→  current: ReportSummary,
   134→  previous: ReportSummary | null,
   135→  currency: string
   136→): string {
   137→  const parts: string[] = [];
   138→
   139→  // Revenue insight
   140→  if (previous) {
   141→    const revDelta = ((current.totalRevenue - previous.totalRevenue) / previous.totalRevenue) * 100;
   142→    if (revDelta > 0) {
   143→      parts.push(`Prihodki so se povečali za ${revDelta.toFixed(1)}% v primerjavi s prejšnjim obdobjem`);
   144→    } else if (revDelta < 0) {
   145→      parts.push(`Prihodki so se zmanjšali za ${Math.abs(revDelta).toFixed(1)}% v primerjavi s prejšnjim obdobjem`);
   146→    } else {
   147→      parts.push('Prihodki so ostali enaki kot v prejšnjem obdobju');
   148→    }
   149→
   150→    // Orders insight
   151→    const ordDelta = ((current.totalOrders - previous.totalOrders) / previous.totalOrders) * 100;
   152→    if (Math.abs(ordDelta) > 3) {
   153→      parts.push(`število naročil se je ${ordDelta > 0 ? 'povečalo' : 'zmanjšalo'} za ${Math.abs(ordDelta).toFixed(1)}%`);
   154→    }
   155→  } else {
   156→    parts.push(`Skupni prihodki znašajo ${currency}${current.totalRevenue.toLocaleString('en-IN')} s ${current.totalOrders} naročili`);
   157→  }
   158→
   159→  // Top item insight
   160→  if (current.topItems.length > 0) {
   161→    const top = current.topItems[0];
   162→    parts.push(`${top.name} ostaja najbolj prodajan artikel (${top.quantity}x, ${currency}${top.revenue.toLocaleString('en-IN')})`);
   163→  }
   164→
   165→  // Net margin insight
   166→  const netMargin = current.totalRevenue > 0 ? (current.netRevenue / current.totalRevenue) * 100 : 0;
   167→  if (netMargin > 90) {
   168→    parts.push(`Neto marža je odlična (${netMargin.toFixed(1)}%)`);
   169→  } else if (netMargin > 80) {
   170→    parts.push(`Neto marža je dobra (${netMargin.toFixed(1)}%)`);
   171→  } else {
   172→    parts.push(`Neto marža znaša ${netMargin.toFixed(1)}% — priporočljivo je zmanjšati popuste in preklice`);
   173→  }
   174→
   175→  // Cancellations insight
   176→  if (current.totalCancellations > 5) {
   177→    parts.push(`Pozor: ${current.totalCancellations} preklicanih naročil — preverite vzroke`);
   178→  }
   179→
   180→  return parts.join('. ') + '.';
   181→}

   183→// ── Delta arrow component ────────────────────────────────
   184→function DeltaArrow({ current, previous, invert = false, suffix = '%' }: {
   185→  current: number;
   186→  previous: number;
   187→  invert?: boolean;
   188→  suffix?: string;
   189→}) {
   190→  if (previous === 0) return <span className="text-xs text-muted-foreground">—</span>;
   191→  const delta = ((current - previous) / previous) * 100;
   192→  const isPositive = invert ? delta < 0 : delta > 0;
   193→  const isNeutral = Math.abs(delta) < 0.5;
   194→
   195→  if (isNeutral) {
   196→    return (
   197→      <span className="inline-flex items-center gap-0.5 text-xs text-muted-foreground">
   198→        <Minus className="h-3 w-3" />
   199→        {delta.toFixed(1)}{suffix}
   200→      </span>
   201→    );
   202→  }
   203→
   204→  return (
   205→    <span className={`inline-flex items-center gap-0.5 text-xs font-medium ${isPositive ? 'text-emerald-600' : 'text-red-500'}`}>
   206→      {isPositive ? <ArrowUpRight className="h-3 w-3" /> : <ArrowDownRight className="h-3 w-3" />}
   207→      {Math.abs(delta).toFixed(1)}{suffix}
   208→    </span>
   209→  );
   210→}
