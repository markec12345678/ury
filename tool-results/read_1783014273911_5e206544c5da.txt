   266→  updateQuantity: (uniqueId: string, quantity: number) => Promise<void>;
   267→  clearOrder: () => Promise<void>;
   268→  initializeCart: () => Promise<void>;
   269→  getCartTotals: () => CartTotals;
   270→  itemExistsInCart: (uniqueId: string) => boolean;
   271→  validateQuantity: (quantity: number) => boolean;
   272→  getItemPrice: (item: OrderItem) => number;
   273→  getItemQuantityFromCart: (item: MenuItem) => number;

   278→// --- Slice ---

   280→export const createCartSlice: StateCreator<POSSliceAll, [], [], CartSlice> = (set, get) => ({
   281→  activeOrders: [],
   282→  cartId: null,
   283→
   284→  initializeCart: async () => {
   285→    set({ cartId: uuidv4() });
   286→  },
   287→
   288→  addToOrder: async (item: OrderItem) => {
   289→    try {
   290→      if (!get().validateQuantity(item.quantity)) {
   291→        throw new CartError(`Quantity must be between ${MIN_QUANTITY} and ${MAX_QUANTITY}`);
   292→      }
   293→
   294→      const uniqueId = generateUniqueId(item);
   295→      const existingItemIndex = get().activeOrders.findIndex(orderItem => orderItem.uniqueId === uniqueId);
   296→
   297→      if (existingItemIndex !== -1) {
   298→        const existingItem = get().activeOrders[existingItemIndex];
   299→        const newQuantity = existingItem.quantity + item.quantity;
   300→        const newComment = item.comment !== undefined ? item.comment : existingItem?.comment || '';
   301→
   302→        if (!get().validateQuantity(newQuantity)) {
   303→          throw new CartError(`Cannot add item. Total quantity would exceed ${MAX_QUANTITY}`);
   304→        }
   305→
   306→        const newOrders = [...get().activeOrders];
   307→        newOrders[existingItemIndex] = {
   308→          ...existingItem,
   309→          quantity: newQuantity,
   310→          comment: newComment,
   311→        };
   312→
   313→        set({ activeOrders: newOrders });
   314→      } else {
   315→        const newOrders = [...get().activeOrders, { ...item, uniqueId }];
   316→        set({ activeOrders: newOrders });
   317→      }
   318→    } catch (error) {
   319→      if (error instanceof CartError) {
   320→        set({ error: error.message });
   321→      } else {
   322→        set({ error: 'Failed to add item to cart' });
   323→      }
   324→    }
   325→  },
   326→
   327→  removeFromOrder: async (uniqueId: string) => {
   328→    try {
   329→      const newOrders = get().activeOrders.filter(item => item.uniqueId !== uniqueId);
   330→      set({ activeOrders: newOrders });
   331→    } catch {
   332→      set({ error: 'Failed to remove item from cart' });
   333→    }
   334→  },
   335→
   336→  updateQuantity: async (uniqueId: string, quantity: number) => {
   337→    try {
   338→      if (!get().validateQuantity(quantity)) {
   339→        throw new CartError(`Quantity must be between ${MIN_QUANTITY} and ${MAX_QUANTITY}`);
   340→      }
   341→
   342→      const newOrders = get().activeOrders.map(item =>
   343→        item.uniqueId === uniqueId ? { ...item, quantity } : item
   344→      );
   345→      set({ activeOrders: newOrders });
   346→    } catch (error) {
   347→      if (error instanceof CartError) {
   348→        set({ error: error.message });
   349→      } else {
   350→        set({ error: 'Failed to update quantity' });
   351→      }
   352→    }
   353→  },
   354→
   355→  clearOrder: async () => {
   356→    try {
   357→      set({ activeOrders: [] });
   358→    } catch {
   359→      set({ error: 'Failed to clear cart' });
   360→    }
   361→  },
   362→
   363→  getCartTotals: (): CartTotals => {
   364→    const items = get().activeOrders;
   365→    const itemCount = items.reduce((sum, item) => sum + item.quantity, 0);
   366→
   367→    const subtotal = items.reduce((sum, item) => {
   368→      const itemPrice = calculateItemPrice(item);
   369→      return sum + (itemPrice * item.quantity);
   370→    }, 0);
   371→
   372→    const tax = items.reduce((sum, item) => {
   373→      const itemPrice = calculateItemPrice(item);
   374→      const taxRate = item.tax_rate || 0;
   375→      return sum + (itemPrice * item.quantity * (taxRate / 100));
   376→    }, 0);
   377→
   378→    return { subtotal, tax, total: subtotal + tax, itemCount };
   379→  },
   380→
   381→  itemExistsInCart: (uniqueId: string): boolean => {
   382→    return get().activeOrders.some(item => item.uniqueId === uniqueId);
   383→  },
   384→
   385→  validateQuantity: (quantity: number): boolean => {
   386→    return !isNaN(quantity) && quantity >= MIN_QUANTITY && quantity <= MAX_QUANTITY;
   387→  },
   388→
   389→  getItemPrice: (item: OrderItem): number => {
   390→    return calculateItemPrice(item);
   391→  },
   392→
   393→  getItemQuantityFromCart: (item: MenuItem): number => {
   394→    const uniqueId = generateUniqueId(item as OrderItem);
   395→    const cartItem = get().activeOrders.find(orderItem => orderItem.uniqueId === uniqueId);
   396→    return cartItem?.quantity || 0;
   397→  },
   398→});

   511→export const generateUniqueId = (item: OrderItem): string => {
   512→  const variantId = item.selectedVariant?.id || 'default';
   513→  const addonIds = item.selectedAddons?.map(addon => addon.id).sort().join('-') || 'no-addons';
   514→  return `${item.id}-${variantId}-${addonIds}`;
   515→};

   517→export const calculateItemPrice = (item: OrderItem): number => {
   518→  const basePrice = item.selectedVariant?.price || item.price;
   519→  const addonsTotal = item.selectedAddons?.reduce((sum, addon) => sum + addon.price, 0) || 0;
   520→  return basePrice + addonsTotal;
