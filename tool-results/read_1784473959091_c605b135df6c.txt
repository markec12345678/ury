   231→   231→    return function(...args) {

   597→   597→    calculateTimeRemaining(targetTime) {
   598→   598→      if (!targetTime || !targetTime.includes(":")) return '— : —';
   599→   599→      const currentTime = new Date();
   600→   600→      // R39-FIX: Guard against malformed time strings with fewer than 3 parts.
   601→   601→      // Previously, "HH:MM" without seconds would set targetSeconds=undefined,
   602→   602→      // which Date() treats as NaN → Invalid Date → NaN propagation.
   603→   603→      const parts = targetTime.split(":");
   604→   604→      const targetHours = parseInt(parts[0], 10);
   605→   605→      const targetMinutes = parseInt(parts[1], 10);
   606→   606→      const targetSeconds = parseInt(parts[2] || '0', 10);
   607→   607→      if (isNaN(targetHours) || isNaN(targetMinutes) || isNaN(targetSeconds)) return '— : —';
   608→   608→      let targetDate = new Date(
   609→   609→        currentTime.getFullYear(),
   610→   610→        currentTime.getMonth(),
   611→   611→        currentTime.getDate(),
   612→   612→        targetHours,
   613→   613→        targetMinutes,
   614→   614→        targetSeconds
   615→   615→      );
   616→   616→      // Handle cross-midnight KOTs: if target is in the future, it was created yesterday
   617→   617→      if (targetDate > currentTime) {
   618→   618→        targetDate = new Date(targetDate.getTime() - 86400000);
   619→   619→      }
   620→   620→
   621→   621→      const timeDifference = Math.max(0, currentTime - targetDate);
   622→   622→      const hoursRemaining = Math.floor(timeDifference / 3600000);
   623→   623→      const minutesRemaining = Math.floor((timeDifference % 3600000) / 60000);
   624→   624→
   625→   625→      return `${hoursRemaining} : ${String(minutesRemaining).padStart(2, '0')}`;
   626→   626→    },
