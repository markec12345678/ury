    20→def _get_order_invoice_doc(table=None, invoiceNo=None, order_type=None, is_payment=None):
    21→    """Internal: returns POS Invoice document object for server-side manipulation.
    22→    Used by sync_order and make_invoice which need the doc, not a dict."""
    23→    # Verify user has access to the table's room (BE-R36-017)
    24→    if table:
    25→        from ury.ury_pos.api import _get_user_branch_rooms
    26→        table_room = frappe.db.get_value("URY Table", table, "restaurant_room")
    27→        if table_room:
    28→            user_rooms = [r.room for r in _get_user_branch_rooms() if r.room]
    29→            if user_rooms and table_room not in user_rooms:
    30→                frappe.throw(_("You do not have access to this table's room"), frappe.PermissionError)
    31→
    32→    if table:
    33→        # Lock the table row to prevent concurrent invoice creation (BE-R36-001)
    34→        frappe.db.sql(
    35→            "SELECT name FROM `tabURY Table` WHERE name = %s FOR UPDATE",
    36→            (table,), as_dict=True
    37→        )
    38→        
    39→        if is_payment == "Payments":
    40→            invoice_name = frappe.get_value(
    41→                "POS Invoice", dict(restaurant_table=table, docstatus=0, name=invoiceNo)
    42→            )
    43→            
    44→        else:
    45→            if invoiceNo:
    46→                invoice_name = frappe.get_value(
    47→                    "POS Invoice",
    48→                    dict(restaurant_table=table, docstatus=0, name=invoiceNo),
    49→                )
    50→               
    51→            else:
    52→                invoice_name = frappe.get_value(
    53→                    "POS Invoice",
    54→                    dict(restaurant_table=table, docstatus=0, invoice_printed=0),
    55→                )
    56→                
    57→        branch, menu_name, restaurant = get_restaurant_and_menu_name(table)
    58→
    59→        if invoice_name:
    60→            invoice = frappe.get_doc("POS Invoice", invoice_name)
    61→
    62→        else:
    63→            invoice = frappe.new_doc("POS Invoice")
    64→
    65→            invoice.naming_series = frappe.db.get_value(
    66→                "URY Restaurant", restaurant, "invoice_series_prefix"
    67→            )
    68→
    69→            invoice.is_pos = 1
    70→            invoice.update_stock = 1
    71→            invoice.restaurant = restaurant
    72→            invoice.branch = branch
    73→
    74→            is_take_away = frappe.db.get_value("URY Table", table, "is_take_away")
    75→            if is_take_away == 1:
    76→                invoice.order_type = "Take Away"
    77→            else:
    78→                invoice.order_type= "Dine In"
    79→
    80→        invoice.taxes_and_charges = frappe.db.get_value(
    81→            "URY Restaurant", restaurant, "default_tax_template"
    82→        )
    83→
    84→        invoice.selling_price_list = frappe.db.get_value(
    85→            "Price List", dict(restaurant_menu=menu_name, enabled=1)
    86→        )
    87→
    88→    else:
    89→
    90→        invoice_name = frappe.get_value(
    91→                "POS Invoice", dict(docstatus=0, name=invoiceNo)
    92→            )
    93→            
    94→        if invoice_name:
    95→            invoice = frappe.get_doc("POS Invoice", invoice_name)
    96→            # R44-FIX: Validate that the loaded invoice belongs to the user's branch
    97→            user_branch = _get_user_branch()
    98→            if invoice.branch and invoice.branch != user_branch:
    99→                frappe.throw(_("You do not have access to invoices from another branch"), frappe.PermissionError)
   100→            
   101→
   102→        else:
   103→            invoice = frappe.new_doc("POS Invoice")
   104→            invoice.is_pos = 1
   105→            invoice.update_stock = 1
   106→        
   107→        branch = _get_user_branch()
   108→        restaurant = frappe.db.get_value("URY Restaurant", {"branch": branch}, "name")
   109→
   110→        menu=get_menu_name(order_type)
   111→
   112→        if (order_type == "Aggregators" and frappe.db.get_value("Branch", branch, "custom_no_taxes") == 0) or order_type != "Aggregators":
   113→            invoice.taxes_and_charges = frappe.db.get_value("URY Restaurant", restaurant, "default_tax_template")
   114→        
   115→        invoice.selling_price_list = frappe.db.get_value(
   116→            "Price List", dict(restaurant_menu=menu, enabled=1)
   117→        )
   118→        # R43-FIX: Set branch on non-table invoices to prevent branchless invoices
   119→        invoice.branch = branch
   120→
   121→    return invoice

   167→@frappe.whitelist()
   168→def sync_order(
   169→    items,
   170→    cashier,
   171→    owner,
   172→    mode_of_payment,
   173→    customer,
   174→    no_of_pax,
   175→    last_invoice,
   176→    waiter,
   177→    pos_profile,
   178→    last_modified_time=None,
   179→    table=None,
   180→    invoice=None,
   181→    comments=None,
   182→    order_type=None,
   183→    aggregator_id=None,
   184→    room=None
   185→):
   186→    frappe.only_for("Restaurant Manager", "Restaurant User", "Cashier")
   187→    # Validate pos_profile belongs to user's branch (BE-R36-002)
   188→    pos_profile_branch = frappe.db.get_value("POS Profile", pos_profile, "branch")
   189→    user_branch = _get_user_branch()
   190→    if pos_profile_branch != user_branch:
   191→        frappe.throw(_("POS Profile does not belong to your branch"), frappe.PermissionError)
   192→    user_role = frappe.get_roles()
   193→    billing_roles = frappe.get_all(
   194→        "POS Profile Role",
   195→        filters={"parent": pos_profile},
   196→        fields=["role"],
   197→        pluck="role",
   198→    )
   199→    billing_user = bool(set(user_role).intersection(billing_roles))
   200→
   201→    # Check if the last invoice was already billed
   202→    if (
   203→        last_invoice
   204→        and frappe.db.get_value("POS Invoice", last_invoice, "invoice_printed") == 1
   205→        and (not billing_user)
   206→    ):
   207→        frappe.msgprint(
   208→            title=_("Invoice Already Billed"),
   209→            indicator="red",
   210→            msg=_("This order has already been billed. Please reload the page."),
   211→        )
   212→        return {"status": "Failure"}
   213→
   214→    # R41-FIX: Use _get_order_invoice_doc to get the document object, not a dict.
   215→    # The public get_order_invoice returns a dict for frontend safety (BE-R36-003),
   216→    # but sync_order needs the document to call .save() etc.
   217→    invoice = _get_order_invoice_doc(table, invoice, order_type)
   218→
   219→    # R41-FIX: Validate last_invoice belongs to user's branch
   220→    if last_invoice:
   221→        last_inv_branch = frappe.db.get_value("POS Invoice", last_invoice, "branch")
   222→        if last_inv_branch and last_inv_branch != user_branch:
   223→            frappe.throw(_("Invoice does not belong to your branch"), frappe.PermissionError)
   224→
   225→    if last_invoice and last_modified_time:
   226→        lastModifiedTime = invoice.modified
   227→        from datetime import datetime
   228→
   229→        if isinstance(last_modified_time, str):
   230→            try:
   231→                last_modified_time = datetime.strptime(
   232→                    last_modified_time, "%Y-%m-%d %H:%M:%S.%f"
   233→                )
   234→            except ValueError:
   235→                last_modified_time = datetime.strptime(
   236→                    last_modified_time, "%Y-%m-%d %H:%M:%S"
   237→                )
   238→        if isinstance(lastModifiedTime, str):
   239→            try:
   240→                lastModifiedTime = datetime.strptime(
   241→                    lastModifiedTime, "%Y-%m-%d %H:%M:%S.%f"
   242→                )
   243→            except ValueError:
   244→                lastModifiedTime = datetime.strptime(
   245→                    lastModifiedTime, "%Y-%m-%d %H:%M:%S"
   246→                )
   247→        if lastModifiedTime != last_modified_time:
   248→            frappe.msgprint(
   249→                title=_("Order has been modified"),
   250→                indicator="red",
   251→                msg=_("This order has been modified. Please reload the page to retrieve the latest edits."),
   252→            )
   253→            return {"status": "Failure"}
   254→    else:
   255→        if invoice.name and invoice.invoice_printed == 0 and not billing_user:
   256→            frappe.msgprint(
   257→                title=_("Table Occupied"),
   258→                indicator="red",
   259→                msg=_("{0} is already occupied. Please refresh the page.").format(table),
   260→            )
   261→            return {"status": "Failure"}
   262→
   263→    if not customer:
   264→        frappe.throw(_("Please enter valid customer details"))
   265→    else:
   266→        invoice.customer = customer
   267→
   268→    if order_type:
   269→        invoice.order_type = order_type
   270→
   271→    invoice.mobile_number = frappe.db.get_value("Customer", customer, "mobile_number")
   272→    if comments:
   273→        invoice.custom_comments = comments
   274→    invoice.no_of_pax = no_of_pax
   275→    invoice.pos_profile = pos_profile
   276→    # H-01: Validate cashier and waiter are active users
   277→    for role_user in [("Cashier", cashier), ("Waiter", waiter)]:
   278→        if role_user[1] and not frappe.db.exists("User", {"name": role_user[1], "enabled": 1}):
   279→            frappe.throw(_("Invalid {0}: {1}").format(role_user[0], role_user[1]))
   280→
   281→    invoice.cashier = cashier
   282→    invoice.waiter = waiter
   283→    invoice.custom_aggregator_id = aggregator_id
   284→    invoice.custom_restaurant_room =room
   285→    invoice.restaurant_table = table
   286→    
   287→    if order_type == "Aggregators":
   288→        price_list = frappe.db.get_value("Aggregator Settings",{"customer": customer, "parent": invoice.branch, "parenttype": "Branch"},"price_list",)
   289→        
   290→        if not price_list:
   291→            frappe.throw(_("Price list for customer {0} in branch {1} not found in Aggregator Settings.").format(customer, invoice.branch))
   292→    else:
   293→        price_list = invoice.selling_price_list
   294→
   295→    # dummy payment
   296→    # C-02: Validate mode_of_payment against POS Profile's allowed payment methods
   297→    allowed_mops = frappe.get_all("POS Profile Payment", filters={"parent": pos_profile}, pluck="mode_of_payment")
   298→    if mode_of_payment not in allowed_mops:
   299→        frappe.throw(_("Invalid mode of payment: {0}").format(mode_of_payment))
   300→
   301→    if invoice.invoice_created == 0:
   302→        invoice.append(
   303→            "payments",
   304→            dict(mode_of_payment=mode_of_payment, amount=invoice.grand_total),
   305→        )
   306→        invoice.invoice_created = 1
   307→
   308→    past_item = []
   309→    for item in invoice.items:
   310→        previous_item = {
   311→            "item_code": item.item_code,
   312→            "item_name": item.item_name,
   313→            "qty": item.qty,
   314→            "comments": "",
   315→        }
   316→        past_item.append(previous_item)
   317→        
   318→
   319→    # Conditional checking for 'items' type:
   320→    # - 'ury': JSON passed, hence using isinstance
   321→    # - 'ury_pos': Already formatted list, hence using else
   322→    if isinstance(items, str):
   323→        try:
   324→            items = json.loads(items)
   325→        except json.JSONDecodeError:
   326→            frappe.throw(_("Invalid items data"), frappe.ValidationError)
   327→    invoice.items = []
   328→    
   329→    menu = frappe.db.get_value("URY Menu", {"branch": invoice.branch}, "name")
   330→
   331→    # Batch-fetch courses, prices, and cost_center to avoid N+1 queries
   332→    item_codes = [d.get("item") for d in items]
   333→    item_codes_unique = list(set(item_codes))
   334→
   335→    courses = {}
   336→    if item_codes_unique:
   337→        course_rows = frappe.db.get_all(
   338→            "URY Menu Item",
   339→            filters={"item": ("in", item_codes_unique), "parent": menu},
   340→            fields=["item", "course"],
   341→        )
   342→        courses = {r.item: r.course for r in course_rows}
   343→
   344→    prices = {}
   345→    if item_codes_unique:
   346→        price_rows = frappe.db.get_all(
   347→            "Item Price",
   348→            filters={"item_code": ("in", item_codes_unique), "price_list": price_list},
   349→            fields=["item_code", "price_list_rate"],
   350→        )
   351→        prices = {r.item_code: r.price_list_rate for r in price_rows}
   352→
   353→    cost_center = frappe.db.get_value("POS Profile", pos_profile, "cost_center")
   354→
   355→    for d in items:
   356→        course = courses.get(d.get("item"))
   357→        rate = prices.get(d.get("item"))
   358→        if not rate:
   359→            frappe.throw(_("No item price found for Item: {0} in Price List: {1}. Please check the price list settings.").format(d.get("item"), price_list))
   360→
   361→        invoice.append(
   362→            "items",
   363→            dict(
   364→                item_code=d.get("item"),
   365→                item_name=d.get("item_name"),
   366→                qty=d.get("qty"),
   367→                **({"custom_course": course} if course else {}),
   368→                comment=d.get("comment"),
   369→                rate=rate,
   370→                price_list_rate=rate,
   371→                base_price_list_rate=rate,
   372→                cost_center=cost_center,
   373→            ),
   374→        )
   375→
   376→    try:
   377→        invoice.save()
   378→    except Exception as e:
   379→        frappe.log_error(frappe.get_traceback(), "Invoice Save Error")
   380→        frappe.throw(_("An error occurred. Please check the error log."))
   381→
   382→
   383→    try:
   384→        kot_execute(invoice.name, customer, table, items, past_item, comments)
   385→
   386→    except Exception as e:
   387→        # Log the KOT error but also notify the user — kitchen won't see this order
   388→        error_msg = f"KOT Creation Failed: {str(e)}"
   389→        frappe.log_error(f"{error_msg}\n{frappe.get_traceback()}", "KOT Error")
   390→        frappe.msgprint(
   391→            title=_("KOT Creation Failed"),
   392→            indicator="orange",
   393→            msg=_("The order was saved but the KOT was not sent to the kitchen. Please check the error log or retry."),
   394→        )
   395→
   396→    # table status
   397→    if invoice.invoice_printed == 0:
   398→        frappe.db.set_value(
   399→            "URY Table", table, {"occupied": 1, "latest_invoice_time": invoice.creation}
   400→        )
   401→
   402→    invoice.db_set("owner", frappe.session.user)
   403→    return {
   404→        "name": invoice.name,
   405→        "customer": invoice.customer,
   406→        "grand_total": invoice.grand_total,
   407→        "modified": invoice.modified,
   408→        "invoice_printed": invoice.invoice_printed,
   409→        "restaurant_table": invoice.restaurant_table,
   410→    }

   433→@frappe.whitelist()
   434→def get_restaurant_and_menu_name(table):
   435→    frappe.only_for("Restaurant Manager", "Restaurant User", "Cashier")
   436→    if not table:
   437→        frappe.throw(_("Please select a table"))
   438→
   439→    result = frappe.get_value(
   440→        "URY Table",
   441→        table,
   442→        ["restaurant", "branch", "restaurant_room"],
   443→    )
   444→    if not result:
   445→        frappe.throw(_("URY Table {0} not found").format(table))
   446→    restaurant, branch, room = result
   447→    room_wise_menu = frappe.db.get_value(
   448→        "URY Restaurant",
   449→        restaurant,
   450→        "room_wise_menu",
   451→    )
   452→
   453→    if not room_wise_menu:
   454→        menu = frappe.db.get_value("URY Restaurant", restaurant, "active_menu")
   455→    else:
   456→        menu = frappe.db.get_value(
   457→            "Menu for Room",
   458→            {"parent": restaurant, "room": room},
   459→            "menu",
   460→        )
   461→
   462→    if not menu:
   463→        frappe.throw(
   464→            _("Please set an active menu for Restaurant {0}").format(restaurant)
   465→        )
   466→
   467→    return branch, menu, restaurant

   495→@frappe.whitelist()
   496→def pos_opening_check():
   497→    frappe.only_for("Restaurant Manager", "Restaurant User", "Cashier")
   498→    user = frappe.session.user
   499→    # Handle the administrator case differently
   500→    if user == "Administrator":
   501→        return {
   502→            "opening_exists": False,  # Assuming no POS opening entry is needed for Administrator
   503→            "cashier": None,
   504→            "pos_profile": None,
   505→        }
   506→    
   507→    details = getBranchRoom()
   508→    if not details:
   509→        frappe.throw(_("No room/branch configuration found for the current user."))
   510→    room = details[0].get('name')
   511→    branch = details[0].get('branch')
   512→    
   513→    pos_opening_list = frappe.db.sql("""
   514→        SELECT DISTINCT `tabPOS Opening Entry`.name 
   515→        FROM `tabPOS Opening Entry`
   516→        INNER JOIN `tabMultiple Rooms` 
   517→        ON `tabMultiple Rooms`.parent = `tabPOS Opening Entry`.name
   518→        WHERE `tabPOS Opening Entry`.branch = %s
   519→        AND `tabPOS Opening Entry`.status = 'Open'
   520→        AND `tabPOS Opening Entry`.docstatus = 1
   521→        AND `tabMultiple Rooms`.room = %s
   522→    """, (branch, room), as_dict=True)
   523→    
   524→    
   525→    result = {
   526→        "opening_exists": len(pos_opening_list) > 0,
   527→        "cashier": None,
   528→        "pos_profile": None,
   529→    }
   530→
   531→    if result["opening_exists"]:
   532→        opening_vals = frappe.db.get_value(
   533→            "POS Opening Entry", pos_opening_list[0].name,
   534→            ["user", "pos_profile"],
   535→            as_dict=True,
   536→        )
   537→        result["cashier"] = opening_vals.user

   725→@frappe.whitelist()
   726→def make_invoice(customer, payments, cashier, pos_profile, additionalDiscount=None, table=None, invoice=None):
   727→    frappe.only_for("Restaurant Manager", "Restaurant User", "Cashier")
   728→
   729→    # R39-FIX: Validate POS Profile belongs to user's branch
   730→    pos_profile_branch = frappe.db.get_value("POS Profile", pos_profile, "branch")
   731→    user_branch = _get_user_branch()
   732→    if pos_profile_branch != user_branch:
   733→        frappe.throw(_("POS Profile does not belong to your branch"), frappe.PermissionError)
   734→
   735→    # Validate additional discount
   736→    if additionalDiscount is not None:
   737→        additionalDiscount = flt(additionalDiscount)
   738→        if additionalDiscount < 0 or additionalDiscount > 100:
   739→            frappe.throw(_("Additional discount must be between 0 and 100"))
   740→
   741→    # Check if discounts are enabled for this POS Profile (BE-R36-006)
   742→    if additionalDiscount:
   743→        enable_discount = frappe.db.get_value("POS Profile", pos_profile, "custom_enable_discount")
   744→        if not enable_discount:
   745→            frappe.throw(_("Discounts are not enabled for this POS Profile"), frappe.PermissionError)
   746→
   747→    # Validate payments
   748→    if isinstance(payments, str):
   749→        try:
   750→            payments = json.loads(payments)
   751→        except json.JSONDecodeError:
   752→            frappe.throw(_("Invalid payments data"), frappe.ValidationError)
   753→    if not payments:
   754→        frappe.throw(_("At least one payment is required"))
   755→    # H-03: Validate each payment's mode_of_payment against POS Profile's configured payment methods
   756→    allowed_mops = frappe.get_all("POS Profile Payment", filters={"parent": pos_profile}, pluck="mode_of_payment")
   757→    for p in payments:
   758→        amount = flt(p.get("amount", 0))
   759→        if amount < 0:
   760→            frappe.throw(_("Payment amount cannot be negative"))
   761→        mop = p.get("mode_of_payment")
   762→        if mop not in allowed_mops:
   763→            frappe.throw(_("Invalid mode of payment: {0}").format(mop))
   764→
   765→    # R44-FIX: At least one of invoice or table must be provided — otherwise
   766→    # we would create an orphaned invoice with no way to reference it.
   767→    if not invoice and not table:
   768→        frappe.throw(_("Invoice or table is required to make a payment"), frappe.ValidationError)
   769→
   770→    # R43-FIX: Guard against None invoice — get_value with None returns None/error
   771→    order_type = None
   772→    if invoice:
   773→        order_type = frappe.get_value("POS Invoice", invoice, "order_type")
   774→    # R41-FIX: Use _get_order_invoice_doc to get the document object, not a dict.
   775→    # make_invoice needs to call .save() and .submit() on the document.
   776→    invoice = _get_order_invoice_doc(table, invoice, order_type, "Payments")
   777→
   778→    if table:
   779→        _, _, restaurant = get_restaurant_and_menu_name(table)
   780→        invoice.restaurant = restaurant
   781→    elif not invoice.restaurant:
   782→        restaurant = frappe.db.get_value("URY Restaurant", {"branch": invoice.branch}, "name")
   783→        if restaurant:
   784→            invoice.restaurant = restaurant
   785→
   786→    invoice.customer = customer
   787→    invoice.pos_profile = pos_profile
   788→    invoice.additional_discount_percentage=additionalDiscount
   789→    invoice.calculate_taxes_and_totals()
   790→
   791→    # H-07: Validate that payment amounts match the grand total (prevent negative invoice from discount)
   792→    total_payments = flt(sum(flt(p.get("amount", 0)) for p in payments))
   793→    if flt(total_payments - invoice.grand_total, 2) < 0:
   794→        frappe.throw(_("Total payment amount ({0}) is less than grand total ({1}). Discount may have created a negative balance.").format(total_payments, invoice.grand_total))
   795→
   796→    invoice.payments = []
   797→
   798→    for d in payments:
   799→        invoice.append(
   800→            "payments", dict(mode_of_payment=d["mode_of_payment"], amount=d["amount"])
   801→        )
   802→
   803→    # invoice.owner = owner
   804→    invoice.save()
   805→    try:
   806→        invoice.submit()
   807→    except Exception as e:
   808→        frappe.log_error(frappe.get_traceback(), "Invoice Submit Error")
   809→        frappe.throw(_("An error occurred. Please check the error log."))
