   816→   816→   816→  computed: {
   817→   817→   817→    sortedKotItems() {
   818→   818→   818→      // R37-FIX: Pure computed — no side effects on reactive kot objects.
   819→   819→   819→      // Uses instance-level _sortedItemsCache Map (non-reactive) instead of
   820→   820→   820→      // mutating kot._sortedItems / kot._sortKey which caused reactivity issues
   821→   821→   821→      // and stale cache after Object.assign overwrites kot_items in socket handler.

   823→   823→   823→        const items = kot.kot_items || [];
   824→   824→   824→        const key = `${kot.name}:${items.length}:${items.map(i => i.name).join(',')}`;

   830→   830→   830→        this._sortedItemsCache.set(kot.name, { key, items: sorted });
