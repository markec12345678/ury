   238→// R41-FIX: Add cancel() method to debounce return value so pending timers

   240→function debounce(fn, delay) {
   241→    let timer = null;
   242→    const debounced = function(...args) {
   243→        clearTimeout(timer);
   244→        timer = setTimeout(() => fn.apply(this, args), delay);
   245→    };
   246→    debounced.cancel = () => { clearTimeout(timer); timer = null; };
   247→    return debounced;
   248→}

   818→    calculateTimeRemaining(targetTime) {
   819→      if (!targetTime || !targetTime.includes(":")) return '— : —';

   826→      const targetMinutes = parseInt(parts[1], 10);

   829→      let targetDate = new Date(
   830→        currentTime.getFullYear(),

   849→    // Without this, a transient server error leaves the KDS showing stale
   850→    // data indefinitely — the only recovery was a manual page reload.
   851→    fetchKOTWithRetry(maxRetries = 3, initialDelay = 2000) {
   852→      // R41-FIX: Clear any pending retry timer before starting a new retry chain.
   853→      // Without this, calling fetchKOTWithRetry() while a previous retry timer is
   854→      // pending would leave the old timer running, potentially causing duplicate
   855→      // fetchKOT calls when both timers fire.
   856→      if (this._fetchRetryTimer) { clearTimeout(this._fetchRetryTimer); this._fetchRetryTimer = null; }
   857→      // R42-FIX: Bump generation counter so any still-running retry chain from
   858→      // a previous fetchKOTWithRetry call knows it's been superseded. Without
   859→      // this, clearing the timer orphans the old promise — its resolve/reject
   860→      // callbacks are never called, leaking the promise and its closures.
   861→      const generation = ++this._fetchGeneration;
   862→      const attempt = (retriesLeft, delay) => {
   863→        // If a newer fetchKOTWithRetry has started, abort this chain
   864→        if (generation !== this._fetchGeneration) {
   865→          return Promise.reject(new Error('Superseded by newer fetchKOTWithRetry call'));
   866→        }
   867→        return this.fetchKOT().catch(err => {
   868→          if (retriesLeft <= 0 || !this._isMounted || generation !== this._fetchGeneration) throw err;
   869→          if (import.meta.env?.DEV) console.warn(`fetchKOT failed, retrying in ${delay}ms...`, err);
   870→          return new Promise((resolve, reject) => {
   871→            this._fetchRetryTimer = setTimeout(() => {
   872→              if (!this._isMounted || generation !== this._fetchGeneration) { reject(err); return; }
   873→              attempt(retriesLeft - 1, Math.min(delay * 2, 30000)).then(resolve).catch(reject);
   874→            }, delay);
   875→          });
   876→        });
   877→      };
   878→      return attempt(maxRetries, initialDelay);
   879→    },

  1438→          }
  1439→        });
  1440→      } else {
  1441→        // R48-FIX (M3): Restore focus to the element that was active before modal opened
  1442→        if (this._preModalFocus && typeof this._preModalFocus.focus === 'function') {
  1443→          this.$nextTick(() => {
  1444→            this._preModalFocus.focus();
  1445→            this._preModalFocus = null;
  1446→          });
  1447→        }
  1448→      }
  1449→    },
  1450→    // R41-FIX: Vue reuses component instances when navigating between routes
  1451→    // that use the same component (Home). Without this watcher, navigating
  1452→    // from /station/kitchen to /station/bar would NOT re-mount the KOT
  1453→    // component, so this.production would still be "kitchen" and the KDS
  1454→    // would filter for the wrong station.
  1455→    '$route.params.production'(newVal) {
  1456→      // R42-FIX: decodeURIComponent can throw URIError on malformed input
  1457→      let decoded;
  1458→      try {
  1459→        decoded = decodeURIComponent(newVal || '');
  1460→      } catch (e) {
  1461→        if (import.meta.env?.DEV) console.warn('Failed to decode production param:', newVal, e);
  1462→        decoded = newVal || '';
  1463→      }
  1464→      if (decoded === this.production) return;
  1465→      this.production = decoded;
  1466→      // Re-register socket handler on the new channel
  1467→      if (this.kot_channel && this._socket && this.socketHandler) {
  1468→        this._socket.off(this.kot_channel, this.socketHandler);
  1469→      }
  1470→      // R48-FIX (M1): Abort any in-flight fetch for the old station before switching.
  1471→      if (this._fetchAbortController) {
  1472→        this._fetchAbortController.abort();
  1473→        this._fetchAbortController = null;
  1474→      }
  1475→      // R46-FIX (H1): Invalidate stale fetch promise so a fresh one is created
  1476→      // for the new station. Without this, fetchKOT() returns the in-flight
  1477→      // promise for the OLD station, resulting in wrong data + wrong socket channel.
  1478→      // R47-FIX (H2): Increment fetch generation so the old fetch's .then() discards results
  1479→      this._fetchId++;
  1480→      this._fetchInProgress = null;
  1481→      // fetchKOT will set the new kot_channel and re-register the handler
  1482→      // R45-FIX: Handle fetch failure on station change — previously .catch(() => {})
  1483→      // silently swallowed errors, leaving the user with an empty KDS and no feedback.
  1484→      this.fetchKOTWithRetry().catch((error) => {
  1485→        this._handleFetchError(error, "Failed to load station data. Click Refresh.");
  1486→      });
  1487→    },
  1488→  },
  1489→};
