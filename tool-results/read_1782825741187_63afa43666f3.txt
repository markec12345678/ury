   131→   131→function loadUIPrefs(): Partial<UIPreferences> {
   132→   132→  if (typeof window === 'undefined') return {};
   133→   133→  try {
   134→   134→    const raw = localStorage.getItem(UI_PREFS_KEY);
   135→   135→    if (!raw) return {};
   136→   136→    return JSON.parse(raw);
   137→   137→  } catch {
   138→   138→    return {};
   139→   139→  }
   140→   140→}
   141→   141→
   142→   142→function saveUIPrefs(prefs: Partial<UIPreferences>): void {
   143→   143→  if (typeof window === 'undefined') return;
   144→   144→  try {
   145→   145→    localStorage.setItem(UI_PREFS_KEY, JSON.stringify(prefs));
   146→   146→  } catch {
   147→   147→    // ignore storage errors
   148→   148→  }
   149→   149→}
