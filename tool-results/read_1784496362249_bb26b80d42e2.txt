   708→   708→    calculateTimeRemaining(targetTime) {
   709→   709→      if (!targetTime || !targetTime.includes(":")) return '— : —';
   710→   710→      const currentTime = new Date();
   711→   711→      // R39-FIX: Guard against malformed time strings with fewer than 3 parts.
   712→   712→      // Previously, "HH:MM" without seconds would set targetSeconds=undefined,
   713→   713→      // which Date() treats as NaN → Invalid Date → NaN propagation.
   714→   714→      const parts = targetTime.split(":");
   715→   715→      const targetHours = parseInt(parts[0], 10);
   716→   716→      const targetMinutes = parseInt(parts[1], 10);
   717→   717→      const targetSeconds = parseInt(parts[2] || '0', 10);
   718→   718→      if (isNaN(targetHours) || isNaN(targetMinutes) || isNaN(targetSeconds)) return '— : —';
   719→   719→      let targetDate = new Date(
   720→   720→        currentTime.getFullYear(),
   721→   721→        currentTime.getMonth(),
   722→   722→        currentTime.getDate(),
   723→   723→        targetHours,
   724→   724→        targetMinutes,
   725→   725→        targetSeconds
   726→   726→      );
   727→   727→      // Handle cross-midnight KOTs: if target is in the future, it was created yesterday
   728→   728→      if (targetDate > currentTime) {
   729→   729→        targetDate = new Date(targetDate.getTime() - 86400000);
   730→   730→      }
   731→   731→
   732→   732→      const timeDifference = Math.max(0, currentTime - targetDate);
   733→   733→      const hoursRemaining = Math.floor(timeDifference / 3600000);
   734→   734→      const minutesRemaining = Math.floor((timeDifference % 3600000) / 60000);
   735→   735→
   736→   736→      return `${hoursRemaining} : ${String(minutesRemaining).padStart(2, '0')}`;
   737→   737→    },
   738→   738→    // R39-FIX: Retry wrapper for fetchKOT with exponential backoff.
   739→   739→    // Without this, a transient server error leaves the KDS showing stale
   740→   740→    // data indefinitely — the only recovery was a manual page reload.
   741→   741→    fetchKOTWithRetry(maxRetries = 3, initialDelay = 2000) {
   742→   742→      // R41-FIX: Clear any pending retry timer before starting a new retry chain.
   743→   743→      // Without this, calling fetchKOTWithRetry() while a previous retry timer is
   744→   744→      // pending would leave the old timer running, potentially causing duplicate
   745→   745→      // fetchKOT calls when both timers fire.
   746→   746→      if (this._fetchRetryTimer) { clearTimeout(this._fetchRetryTimer); this._fetchRetryTimer = null; }
   747→   747→      // R42-FIX: Bump generation counter so any still-running retry chain from
   748→   748→      // a previous fetchKOTWithRetry call knows it's been superseded. Without
   749→   749→      // this, clearing the timer orphans the old promise — its resolve/reject
   750→   750→      // callbacks are never called, leaking the promise and its closures.
   751→   751→      const generation = ++this._fetchGeneration;
   752→   752→      const attempt = (retriesLeft, delay) => {
   753→   753→        // If a newer fetchKOTWithRetry has started, abort this chain
   754→   754→        if (generation !== this._fetchGeneration) {
   755→   755→          return Promise.reject(new Error('Superseded by newer fetchKOTWithRetry call'));
   756→   756→        }
   757→   757→        return this.fetchKOT().catch(err => {
   758→   758→          if (retriesLeft <= 0 || !this._isMounted || generation !== this._fetchGeneration) throw err;
   759→   759→          if (import.meta.env?.DEV) console.warn(`fetchKOT failed, retrying in ${delay}ms...`, err);
   760→   760→          return new Promise((resolve, reject) => {
   761→   761→            this._fetchRetryTimer = setTimeout(() => {
   762→   762→              if (!this._isMounted || generation !== this._fetchGeneration) { reject(err); return; }
   763→   763→              attempt(retriesLeft - 1, Math.min(delay * 2, 30000)).then(resolve).catch(reject);
   764→   764→            }, delay);
   765→   765→          });
   766→   766→        });
   767→   767→      };
   768→   768→      return attempt(maxRetries, initialDelay);
   769→   769→    },
