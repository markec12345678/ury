   190→   190→    user_branch = _get_user_branch()
   191→   191→    if pos_profile_branch != user_branch:
   192→   192→        frappe.throw(_("POS Profile does not belong to your branch"), frappe.PermissionError)
   193→   193→    user_role = frappe.get_roles()
   194→   194→    billing_roles = frappe.get_all(
   195→   195→        "POS Profile Role",
   196→   196→        filters={"parent": pos_profile},
   197→   197→        fields=["role"],
   198→   198→        pluck="role",
   199→   199→    )
   200→   200→    billing_user = bool(set(user_role).intersection(billing_roles))
   201→   201→
   202→   202→    # Check if the last invoice was already billed

   204→   204→        last_invoice
   205→   205→        and frappe.db.get_value("POS Invoice", last_invoice, "invoice_printed") == 1
   206→   206→        and (not billing_user)

   211→   211→            msg=_("This order has already been billed. Please reload the page."),
   212→   212→        )
   213→   213→        return {"status": "Failure"}
   214→   214→
   215→   215→    # R41-FIX: Use _get_order_invoice_doc to get the document object, not a dict.
   216→   216→    # The public get_order_invoice returns a dict for frontend safety (BE-R36-003),
   217→   217→    # but sync_order needs the document to call .save() etc.
   218→   218→    invoice = _get_order_invoice_doc(table, invoice, order_type)
   219→   219→
   220→   220→    # R41-FIX: Validate last_invoice belongs to user's branch
   221→   221→    if last_invoice:
   222→   222→        last_inv_branch = frappe.db.get_value("POS Invoice", last_invoice, "branch")
   223→   223→        if last_inv_branch and last_inv_branch != user_branch:
   224→   224→            frappe.throw(_("Invoice does not belong to your branch"), frappe.PermissionError)
   225→   225→
   226→   226→    if last_invoice and last_modified_time:
   227→   227→        lastModifiedTime = invoice.modified
   228→   228→        if isinstance(last_modified_time, str):
   229→   229→            try:
   230→   230→                last_modified_time = datetime.strptime(
   231→   231→                    last_modified_time, "%Y-%m-%d %H:%M:%S.%f"
   232→   232→                )
   233→   233→            except ValueError:
   234→   234→                last_modified_time = datetime.strptime(
   235→   235→                    last_modified_time, "%Y-%m-%d %H:%M:%S"
   236→   236→                )
   237→   237→        if isinstance(lastModifiedTime, str):
   238→   238→            try:
   239→   239→                lastModifiedTime = datetime.strptime(
   240→   240→                    lastModifiedTime, "%Y-%m-%d %H:%M:%S.%f"
   241→   241→                )
   242→   242→            except ValueError:
   243→   243→                lastModifiedTime = datetime.strptime(
   244→   244→                    lastModifiedTime, "%Y-%m-%d %H:%M:%S"
   245→   245→                )
   246→   246→        if lastModifiedTime != last_modified_time:
   247→   247→            frappe.msgprint(
   248→   248→                title=_("Order has been modified"),
   249→   249→                indicator="red",
   250→   250→                msg=_("This order has been modified. Please reload the page to retrieve the latest edits."),
   251→   251→            )
   252→   252→            return {"status": "Failure"}
   253→   253→    elif last_invoice and not last_modified_time:
   254→   254→        # R49-FIX: When last_invoice is provided but last_modified_time is missing,
   255→   255→        # require the client to provide it — otherwise the optimistic concurrency
   256→   256→        # check is bypassed entirely, allowing silent overwrites.
   257→   257→        frappe.msgprint(
   258→   258→            title=_("Stale Data"),
   259→   259→            indicator="red",
   260→   260→            msg=_("Please reload the page to retrieve the latest edits before saving."),
   261→   261→        )
   262→   262→        return {"status": "Failure"}
   263→   263→    else:
   264→   264→        if invoice.name and invoice.invoice_printed == 0 and not billing_user:
   265→   265→            frappe.msgprint(
   266→   266→                title=_("Table Occupied"),
   267→   267→                indicator="red",
   268→   268→                msg=_("{0} is already occupied. Please refresh the page.").format(table),
   269→   269→            )
   270→   270→            return {"status": "Failure"}
   271→   271→
   272→   272→    if not customer:
   273→   273→        frappe.throw(_("Please enter valid customer details"))
   274→   274→    else:
   275→   275→        invoice.customer = customer
   276→   276→
   277→   277→    if order_type:
   278→   278→        invoice.order_type = order_type
   279→   279→
   280→   280→    invoice.mobile_number = frappe.db.get_value("Customer", customer, "mobile_number")
   281→   281→    if comments:
   282→   282→        invoice.custom_comments = comments
   283→   283→    invoice.no_of_pax = no_of_pax
   284→   284→    invoice.pos_profile = pos_profile
   285→   285→    # H-01: Validate cashier and waiter are active users

   287→   287→        if role_user[1] and not frappe.db.exists("User", {"name": role_user[1], "enabled": 1}):
   288→   288→            frappe.throw(_("Invalid {0}: {1}").format(role_user[0], role_user[1]))
   289→   289→
   290→   290→    invoice.cashier = cashier
   291→   291→    invoice.waiter = waiter
   292→   292→    invoice.custom_aggregator_id = aggregator_id
   293→   293→    invoice.custom_restaurant_room =room
   294→   294→    invoice.restaurant_table = table
   295→   295→    
   296→   296→    if order_type == "Aggregators":
   297→   297→        price_list = frappe.db.get_value("Aggregator Settings",{"customer": customer, "parent": invoice.branch, "parenttype": "Branch"},"price_list",)
   298→   298→        
   299→   299→        if not price_list:
   300→   300→            frappe.throw(_("Price list for customer {0} in branch {1} not found in Aggregator Settings.").format(customer, invoice.branch))
   301→   301→    else:
   302→   302→        price_list = invoice.selling_price_list
   303→   303→
   304→   304→    # dummy payment
   305→   305→    # C-02: Validate mode_of_payment against POS Profile's allowed payment methods

   307→   307→    if mode_of_payment not in allowed_mops:
   308→   308→        frappe.throw(_("Invalid mode of payment: {0}").format(mode_of_payment))
   309→   309→
   310→   310→    if invoice.invoice_created == 0:
   311→   311→        invoice.append(
   312→   312→            "payments",
   313→   313→            dict(mode_of_payment=mode_of_payment, amount=invoice.grand_total),
   314→   314→        )
   315→   315→        invoice.invoice_created = 1
   316→   316→
   317→   317→    past_item = []

   376→   376→            "items",

   380→   380→                qty=qty,
   381→   381→                **({"custom_course": course} if course else {}),
   382→   382→                comment=d.get("comment"),
   383→   383→                rate=rate,
   384→   384→                price_list_rate=rate,
   385→   385→                base_price_list_rate=rate,
   386→   386→                cost_center=cost_center,
   387→   387→            ),
   388→   388→        )
   389→   389→
   390→   390→    try:

   392→   392→    except Exception as e:

   394→   394→        frappe.throw(_("An error occurred. Please check the error log."))
   395→   395→
   396→   396→
   397→   397→    try:
   398→   398→        kot_execute(invoice.name, customer, table, items, past_item, comments)
   399→   399→
   400→   400→    except Exception as e:
   401→   401→        # Log the KOT error but also notify the user — kitchen won't see this order

   403→   403→        frappe.log_error(f"{error_msg}\n{frappe.get_traceback()}", "KOT Error")

   405→   405→            title=_("KOT Creation Failed"),

   408→   408→        )
   409→   409→
   410→   410→    # table status
   411→   411→    if invoice.invoice_printed == 0:
   412→   412→        # R47-FIX: Use update_modified=False for auxiliary status updates

   414→   414→            "URY Table", table, {"occupied": 1, "latest_invoice_time": invoice.creation},
   415→   415→            update_modified=False
   416→   416→        )
   417→   417→
   418→   418→    invoice.db_set("owner", frappe.session.user)
   419→   419→    return {
   420→   420→        "name": invoice.name,
   421→   421→        "customer": invoice.customer,
   422→   422→        "grand_total": invoice.grand_total,
   423→   423→        "modified": invoice.modified,
   424→   424→        "invoice_printed": invoice.invoice_printed,
   425→   425→        "restaurant_table": invoice.restaurant_table,
   426→   426→    }
